1//! Exports the authoritative protocol types as a deterministic schema bundle.
2
3use schemars::schema_for;
4use scriberig_protocol::{
5    CapabilitiesResponse, Event, InspectEntityRequest, InspectEntityResponse, Operation,
6    ProtocolError, VersionExchange,
7};
8use std::fs;
9use std::path::{Path, PathBuf};
10
11/// Writes all public protocol schemas to the requested output directory.
12///
13/// # Returns
14/// An empty result after every schema file is written.
15///
16/// # Errors
17/// Returns an error when the output directory or schema cannot be written.
18fn main() -> Result<(), Box<dyn std::error::Error>> {
19    let output = PathBuf::from("target/scriberig-protocol-schema");
20    fs::create_dir_all(&output)?;
21    write_schema::<CapabilitiesResponse>(&output, "capabilities")?;
22    write_schema::<Event>(&output, "event")?;
23    write_schema::<InspectEntityRequest>(&output, "inspect-request")?;
24    write_schema::<InspectEntityResponse>(&output, "inspect-response")?;
25    write_schema::<Operation>(&output, "operation")?;
26    write_schema::<ProtocolError>(&output, "error")?;
27    write_schema::<VersionExchange>(&output, "version")?;
28    Ok(())
29}
30
31/// Serializes one schema with stable pretty JSON formatting.
32///
33/// # Parameters
34/// - `output`: Destination directory.
35/// - `name`: Stable schema file stem.
36///
37/// # Returns
38/// An empty result after the schema is written.
39///
40/// # Errors
41/// Returns an error when serialization or file writing fails.
42fn write_schema<T: schemars::JsonSchema>(
43    output: &Path,
44    name: &str,
45) -> Result<(), Box<dyn std::error::Error>> {
46    let schema = schema_for!(T);
47    let path = output.join(format!("{name}.json"));
48    fs::write(path, serde_json::to_vec_pretty(&schema)?)?;
49    Ok(())
50}

11/// Binds an upstream provider/model and approved tools to one `ScribeRig` branch.
12///
13/// The provider, model, compiled context, and tools are exact inputs for one
14/// branch-local run. No upstream agent or session identity becomes canonical.
15///
16/// # Public members
17/// - `branch`: `ScribeRig` branch identity for result correlation.
18/// - `provider`: Upstream provider selected for this invocation.
19/// - `model`: Exact upstream model selected for this invocation.
20/// - `request_secret`: Optional opaque credential for protected provider calls.
21/// - `bearer`: Whether a protected credential uses bearer authorization.
22/// - `applied_settings`: Optional sink for sanitized final request settings.
23/// - `thinking_level`: Upstream thinking level selected for this invocation.
24/// - `context`: `ScribeRig`-compiled context for this branch.
25/// - `tools`: Unique semantic adapters built from this branch's capability-
26///   filtered tool catalog.
27///
28/// # Private members
29/// None.
30pub struct UpstreamBranchInvocation {
31    /// `ScribeRig` branch identity for result correlation.
32    pub branch: EntityId,
33    /// Upstream provider selected for this invocation.
34    pub provider: Arc<dyn Provider>,
35    /// Exact upstream model selected for this invocation.
36    pub model: Model,
37    /// Optional opaque credential for protected provider calls.
38    pub request_secret: Option<RequestSecret>,
39    /// Whether a protected credential uses bearer authorization.
40    pub bearer: bool,
41    /// Optional sink for typed, sanitized final request settings.
42    pub applied_settings: Option<AppliedRequestSettingsCallback>,
43    /// Safe `ScribeRig` provider profile used for request provenance.
44    pub provider_profile: Option<String>,
45    /// Upstream thinking level selected for this invocation.
46    pub thinking_level: rpi_ai::ThinkingLevel,
47    /// `ScribeRig`-compiled context for this branch.
48    pub context: CompiledContext,
49    /// Branch-approved `ScribeRig` semantic tools only. The concrete type makes
50    /// upstream plugin adapters unrepresentable here until an isolated host
51    /// can mediate and constrain their process capabilities.
52    pub tools: Vec<Arc<SemanticAgentTool>>,
53}

55/// Compact output from one upstream invocation on a `ScribeRig` branch.
56///
57/// The upstream transcript is discarded after loop mechanics finish.
58///
59/// # Public members
60/// - `branch`: `ScribeRig` branch identity supplied by the caller.
61/// - `output`: Final assistant text projected for `ScribeRig` persistence.
62/// - `model_requests`: Bounded provider-neutral request projections.
63///
64/// # Private members
65/// None.
66#[derive(Clone, Debug, Eq, PartialEq)]
67pub struct UpstreamBranchResult {
68    /// `ScribeRig` branch identity supplied by the caller.
69    pub branch: EntityId,
70    /// Final assistant text projected for `ScribeRig` persistence.
71    pub output: String,
72    /// Bounded request projections captured before serialization.
73    pub model_requests: Vec<scriberig_session::ModelRequestProjection>,
74}

76/// Failure to execute one branch through upstream `rpi-agent`.
77#[derive(Clone, Debug, Eq, PartialEq)]
78pub enum UpstreamBranchError {
79    /// The upstream agent loop returned an ordinary failure.
80    Agent(String),
81    /// The upstream provider rejected authentication or authorization.
82    Authentication(String),
83    /// The branch tool set is mis-scoped or contains an invalid name.
84    InvalidToolSet,
85    /// The branch cancellation token was cancelled.
86    Cancelled,
87    /// The upstream loop ended without a final assistant message.
88    MissingFinalAssistant,
89}
90
91/// Failure paired with any safe requests captured before settlement failed.
92///
93/// # Public members
94/// - `error`: Safe upstream execution failure.
95/// - `model_requests`: Requests that were prepared before the failure.
96///
97/// # Private members
98/// None.
99#[derive(Clone, Debug, Eq, PartialEq)]
100pub struct UpstreamBranchFailure {
101    /// Safe upstream execution failure.
102    pub error: UpstreamBranchError,
103    /// Requests prepared before the failure.
104    pub model_requests: Vec<scriberig_session::ModelRequestProjection>,
105}

107impl std::fmt::Display for UpstreamBranchError {
108    /// Formats a safe branch invocation failure.
109    ///
110    /// # Parameters
111    /// - `self`: Failure to format.
112    /// - `formatter`: Destination formatter.
113    ///
114    /// # Returns
115    /// Formatting success or the formatter error.
116    ///
117    /// # Errors
118    /// Returns the formatter error when output cannot be written.
119    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
120        match self {
121            Self::Agent(message) => write!(formatter, "upstream agent: {message}"),
122            Self::Authentication(message) => {
123                write!(formatter, "upstream authentication: {message}")
124            }
125            Self::InvalidToolSet => formatter.write_str("branch tool set has invalid tool names"),
126            Self::Cancelled => formatter.write_str("branch invocation cancelled"),
127            Self::MissingFinalAssistant => {
128                formatter.write_str("upstream agent returned no final assistant message")
129            }
130        }
131    }
132}

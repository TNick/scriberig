1//! Captures the upstream provider's safe applied-settings report.
2
3use rpi_ai::{AppliedRequestSettings, AppliedRequestSettingsCallback};
4
5use super::CapturedRequestSink;
6
7/// Records an upstream applied-settings report after validating model identity.
8///
9/// # Parameters
10/// - `settings`: Upstream report about transformed provider request settings.
11/// - `request_index`: Captured request updated by the report.
12/// - `captured_requests`: Sink for bounded request provenance.
13/// - `expected_provider`: Provider identity bound to the invocation.
14/// - `expected_model`: Model identity bound to the invocation.
15/// - `expected_api`: API identifier bound to the invocation.
16/// - `forwarded_settings`: Optional host observer for safe reports.
17pub(super) fn capture_applied_settings(
18    settings: AppliedRequestSettings,
19    request_index: Option<usize>,
20    captured_requests: &CapturedRequestSink,
21    expected_provider: &str,
22    expected_model: &str,
23    expected_api: &str,
24    forwarded_settings: Option<&AppliedRequestSettingsCallback>,
25) {
26    if settings.schema_version != "rpi-applied-request-settings-v1"
27        || settings.provider_id != expected_provider
28        || settings.model_id != expected_model
29        || settings.api_id != expected_api
30        || (!settings.complete && settings.omitted_categories.is_empty())
31    {
32        return;
33    }
34    if let (Some(index), Ok(mut requests)) = (request_index, captured_requests.lock())
35        && let Some(request) = requests.get_mut(index)
36        && let Ok(value) = serde_json::to_value(&settings)
37    {
38        request.applied_settings_fingerprint = applied_settings_fingerprint(&value);
39        request.request_fingerprint = request_fingerprint_with_applied_settings(
40            request.request_fingerprint.as_deref(),
41            &value,
42        );
43        request.applied_settings = Some(value);
44    }
45    if let Some(callback) = forwarded_settings {
46        callback(settings);
47    }
48}

50/// Fingerprints one serialized upstream report without interpreting its fields.
51///
52/// # Parameters
53/// - `settings`: Sanitized upstream report captured after provider preparation.
54///
55/// # Returns
56/// A versioned BLAKE3 digest, or `None` if serialization fails.
57fn applied_settings_fingerprint(settings: &serde_json::Value) -> Option<String> {
58    let material = serde_json::to_vec(&("scriberig-applied-settings-v1", settings)).ok()?;
59    Some(blake3::hash(&material).to_hex().to_string())
60}

62/// Adds the upstream applied-settings report to the request identity.
63///
64/// The report may be partial, so this fingerprint identifies captured
65/// provenance and does not claim full provider-request equivalence.
66///
67/// # Parameters
68/// - `request_fingerprint`: Existing `ScribeRig` input and model fingerprint.
69/// - `settings`: Safe report captured at the provider preparation boundary.
70///
71/// # Returns
72/// A versioned digest combining the base fingerprint and report, or `None`
73/// when the base fingerprint is unavailable or serialization fails.
74fn request_fingerprint_with_applied_settings(
75    request_fingerprint: Option<&str>,
76    settings: &serde_json::Value,
77) -> Option<String> {
78    let request_fingerprint = request_fingerprint?;
79    let material = serde_json::to_vec(&(
80        "scriberig-request-with-applied-settings-v1",
81        request_fingerprint,
82        settings,
83    ))
84    .ok()?;
85    Some(blake3::hash(&material).to_hex().to_string())
86}

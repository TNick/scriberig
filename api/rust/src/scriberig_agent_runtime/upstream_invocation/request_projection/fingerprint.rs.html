8/// Borrows values that define request and semantic context identity.
9///
10/// # Public members
11/// None; this internal helper is not part of the crate API.
12///
13/// # Private members
14/// - `model`: Exact upstream model selected for the invocation.
15/// - `thinking_level`: Upstream thinking level selected for the invocation.
16/// - `messages`: Provider-visible message inputs.
17/// - `tools`: Upstream tool schemas supplied to the provider.
18/// - `provider_profile`: Safe `ScribeRig` provider profile, when selected.
19/// - `profile`: Exact immutable `ScribeRig` profile revision.
20/// - `active_components`: Exact active context component references.
21/// - `context_fingerprint`: Fingerprint of compiled `ScribeRig` context.
22pub(super) struct RequestFingerprintInput<'a> {
23    /// Exact upstream model selected for this invocation.
24    pub(super) model: &'a Model,
25    /// Upstream thinking level selected for this invocation.
26    pub(super) thinking_level: &'a str,
27    /// Ordered safe provider-visible message parts.
28    pub(super) messages: &'a [ModelRequestDetailMessage],
29    /// Complete safe upstream tool definitions supplied to the provider.
30    pub(super) tools: &'a [ModelRequestDetailTool],
31    /// Safe `ScribeRig` provider profile identifier, if selected.
32    pub(super) provider_profile: Option<&'a str>,
33    /// Exact immutable `ScribeRig` profile revision.
34    pub(super) profile: Option<&'a ModelContextRevisionRef>,
35    /// Exact immutable references for active context components.
36    pub(super) active_components: &'a [ModelContextRevisionRef],
37    /// Fingerprint of the compiled `ScribeRig` context.
38    pub(super) context_fingerprint: Option<&'a str>,
39}

41/// Folds the upstream-owned execution fingerprint into request identity.
42///
43/// The upstream digest covers safe typed model metadata, including
44/// compatibility flags and output limits. It omits endpoint, headers,
45/// sampling values, provider options, and thinking-map labels, and reports
46/// incomplete coverage. This adds that digest without defining a `ScribeRig`
47/// property schema.
48///
49/// # Parameters
50/// - `request_fingerprint`: Existing request and model-config digest.
51/// - `model`: Exact upstream model selected for this invocation.
52///
53/// # Returns
54/// A combined digest, or `None` when inputs cannot be serialized.
55pub(super) fn request_fingerprint_with_model_execution(
56    request_fingerprint: Option<&str>,
57    model: &Model,
58) -> Option<String> {
59    let request_fingerprint = request_fingerprint?;
60    let execution_fingerprint = rpi_model_runtime::model_execution_fingerprint(model);
61    let material = serde_json::to_vec(&(
62        "scriberig-request-with-model-execution-v1",
63        request_fingerprint,
64        &execution_fingerprint,
65    ))
66    .ok()?;
67    Some(blake3::hash(&material).to_hex().to_string())
68}

70/// Fingerprints normalized request inputs and semantic context.
71///
72/// This digest may be extended with upstream model configuration and applied
73/// settings. Omitted secrets, opaque properties, and other fields mean it is
74/// not a complete provider request-equivalence fingerprint.
75///
76/// # Parameters
77/// - `input`: Exact model, request content, and semantic context data.
78///
79/// # Returns
80/// The hex-encoded base request fingerprint.
81pub(super) fn request_fingerprint_for(input: &RequestFingerprintInput<'_>) -> String {
82    let profile_identity = input
83        .profile
84        .map(|reference| (&reference.identity, &reference.revision));
85    let component_identity = input
86        .active_components
87        .iter()
88        .map(|reference| (&reference.identity, &reference.revision))
89        .collect::<Vec<_>>();
90    let material = serde_json::to_vec(&(
91        "scriberig-request-v3",
92        Some(input.model.provider.as_str()),
93        input.provider_profile,
94        &input.model.id,
95        Some(input.thinking_level),
96        ("auto", None::<&str>),
97        &None::<serde_json::Value>,
98        None::<u32>,
99        None::<u32>,
100        profile_identity,
101        component_identity,
102        input.context_fingerprint,
103        input.messages,
104        input.tools,
105    ))
106    .expect("request fingerprint inputs are serializable");
107    blake3::hash(&material).to_hex().to_string()
108}

3use serde::Serialize;
4
5use crate::provider::{MessageRole, ModelRequest, ToolChoice};
6/// Serializes a provider-neutral request into the compatible wire subset.
7#[derive(Serialize)]
8pub(super) struct WireRequest {
9    model: String,
10    messages: Vec<WireMessage>,
11    #[serde(skip_serializing_if = "Vec::is_empty")]
12    tools: Vec<WireTool>,
13    #[serde(skip_serializing_if = "Option::is_none")]
14    tool_choice: Option<serde_json::Value>,
15    #[serde(skip_serializing_if = "Option::is_none")]
16    response_format: Option<serde_json::Value>,
17    #[serde(skip_serializing_if = "Option::is_none")]
18    temperature: Option<f32>,
19    #[serde(skip_serializing_if = "Option::is_none")]
20    max_tokens: Option<u32>,
21    #[serde(skip_serializing_if = "is_false")]
22    stream: bool,
23    #[serde(flatten)]
24    provider_parameters: std::collections::BTreeMap<String, serde_json::Value>,
25}
26
27impl WireRequest {
28    /// Converts a request using both global and model-specific translations.
29    ///
30    /// - `request`: Provider-neutral request.
31    /// - `stream`: Whether the provider should stream SSE fragments.
32    /// - `thinking_parameters`: Adapter-wide translations by level id.
33    /// - `model_thinking_parameters`: Translations by model and level id.
34    ///
35    /// OpenAI-compatible request fields.
36    /// # Returns
37    /// The operation result.
38    pub(super) fn from_request_for_models(
39        request: &ModelRequest,
40        stream: bool,
41        thinking_parameters: &std::collections::BTreeMap<
42            String,
43            std::collections::BTreeMap<String, serde_json::Value>,
44        >,
45        model_thinking_parameters: &std::collections::BTreeMap<
46            (String, String),
47            std::collections::BTreeMap<String, serde_json::Value>,
48        >,
49    ) -> Self {
50        let tool_choice = match &request.tool_choice {
51            ToolChoice::Auto => Some(serde_json::json!("auto")),
52            ToolChoice::None => Some(serde_json::json!("none")),
53            ToolChoice::Required(name) => Some(serde_json::json!({
54                "type": "function",
55                "function": {"name": name},
56            })),
57        };
58        let response_format = request.response_schema.as_ref().map(|schema| {
59            serde_json::json!({
60                "type": "json_schema",
61                "json_schema": {"name": "scriberig_output", "schema": schema},
62            })
63        });
64        Self {
65            model: request.model.clone(),
66            messages: request
67                .messages
68                .iter()
69                .map(WireMessage::from_message)
70                .collect(),
71            tools: request
72                .tools
73                .iter()
74                .map(WireTool::from_definition)
75                .collect(),
76            tool_choice,
77            response_format,
78            temperature: request.temperature,
79            max_tokens: request.max_tokens,
80            stream,
81            provider_parameters: request
82                .thinking_level
83                .as_ref()
84                .and_then(|level| {
85                    model_thinking_parameters
86                        .get(&(request.model.clone(), level.clone()))
87                        .or_else(|| thinking_parameters.get(level))
88                })
89                .cloned()
90                .unwrap_or_default(),
91        }
92    }
93}

95/// Omits the default false streaming flag from ordinary requests.
96#[allow(clippy::trivially_copy_pass_by_ref)]
97///
98/// - `value`: Streaming flag to inspect.
99///
100/// `true` when the flag is false.
101/// # Returns
102/// The operation result.
103fn is_false(value: &bool) -> bool {
104    !*value
105}

107/// Serializes one provider-neutral message.
108#[derive(Serialize)]
109struct WireMessage {
110    role: &'static str,
111    content: String,
112    #[serde(skip_serializing_if = "Option::is_none")]
113    tool_call_id: Option<String>,
114}
115
116impl WireMessage {
117    /// Converts a semantic message role to its wire spelling.
118    ///
119    /// - `message`: Provider-neutral message.
120    ///
121    /// Wire-compatible message fields.
122    /// # Returns
123    /// The operation result.
124    fn from_message(message: &crate::provider::ModelMessage) -> Self {
125        let role = match message.role {
126            MessageRole::System => "system",
127            MessageRole::User => "user",
128            MessageRole::Assistant => "assistant",
129            MessageRole::Tool => "tool",
130        };
131        Self {
132            role,
133            content: message.content.clone(),
134            tool_call_id: message.tool_call_id.clone(),
135        }
136    }
137}

139#[derive(Serialize)]
140/// Represents one tool descriptor in an OpenAI-compatible request.
141struct WireTool {
142    r#type: &'static str,
143    function: WireFunction,
144}
145
146#[derive(Serialize)]
147/// Represents a callable function descriptor in a model request.
148struct WireFunction {
149    name: String,
150    description: String,
151    parameters: serde_json::Value,
152}
153
154impl WireTool {
155    ///
156    /// - `definition`: Trusted provider-neutral tool definition.
157    ///
158    /// Wire-compatible function tool fields.
159    /// # Returns
160    /// The operation result.
161    fn from_definition(definition: &crate::provider::ToolDefinition) -> Self {
162        Self {
163            r#type: "function",
164            function: WireFunction {
165                name: definition.name.clone(),
166                description: definition.description.clone(),
167                parameters: provider_schema(&definition.parameters),
168            },
169        }
170    }
171}

173/// Removes Scriberig-only sensitivity annotations from provider schemas.
174///
175/// - `schema`: Trusted schema used for local presentation redaction.
176///
177/// Provider-compatible schema without the private `x-sensitive` extension.
178/// # Returns
179/// The operation result.
180fn provider_schema(schema: &serde_json::Value) -> serde_json::Value {
181    match schema {
182        serde_json::Value::Object(object) => {
183            let mut filtered = serde_json::Map::new();
184            for (key, value) in object {
185                if key != "x-sensitive" {
186                    filtered.insert(key.clone(), provider_schema(value));
187                }
188            }
189            serde_json::Value::Object(filtered)
190        }
191        serde_json::Value::Array(items) => {
192            serde_json::Value::Array(items.iter().map(provider_schema).collect())
193        }
194        value => value.clone(),
195    }

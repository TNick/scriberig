58/// Structured failure returned by the agent orchestration boundary.
59#[derive(Clone, Debug, Eq, PartialEq)]
60pub enum AgentError {
61    /// Upstream provider or agent execution failed.
62    Provider(String),
63    /// The upstream provider rejected authentication or authorization.
64    Authentication(String),
65    /// The caller requested cancellation.
66    Cancelled,
67}
68
69impl AgentError {
70    /// Reports whether this failure is an upstream authentication failure.
71    ///
72    /// # Returns
73    /// `true` for [`AgentError::Authentication`].
74    #[must_use]
75    pub fn is_authentication(&self) -> bool {
76        matches!(self, Self::Authentication(_))
77    }
78}
79
80impl std::fmt::Display for AgentError {
81    /// Formats a safe agent-runtime failure.
82    ///
83    /// - `self`: Agent error.
84    /// - `formatter`: Destination formatter.
85    ///
86    /// Formatting success or formatter failure.
87    ///
88    /// Returns the formatter error when output cannot be written.
89    /// # Returns
90    /// The operation result.
91    /// # Errors
92    /// Returns an error when the operation fails.
93    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
94        match self {
95            Self::Provider(message) => write!(formatter, "agent provider: {message}"),
96            Self::Authentication(message) => {
97                write!(formatter, "agent authentication: {message}")
98            }
99            Self::Cancelled => formatter.write_str("agent operation cancelled"),
100        }
101    }
102}
103
104impl std::error::Error for AgentError {}

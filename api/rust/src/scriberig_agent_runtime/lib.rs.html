77/// Executes a trusted child branch on behalf of the parent runtime.
78pub trait ChildExecutor {
79    /// Runs one child with an explicitly bounded cancellation token.
80    ///
81    /// # Parameters
82    /// - `context`: Parent trusted application context.
83    /// - `scope`: Optional durable parent branch identity.
84    /// - `request`: Validated child purpose and prompt.
85    /// - `cancelled`: Shared cancellation flag for the execution tree.
86    ///
87    /// # Returns
88    /// A compact child result suitable for the parent model context.
89    ///
90    /// # Errors
91    /// Returns a safe child execution failure.
92    fn execute_child(
93        &mut self,
94        context: &ApplicationContext,
95        scope: Option<ChildExecutionScope>,
96        request: ChildRequest,
97        cancelled: &AtomicBool,
98    ) -> Result<ChildResult, String>;
99}

101/// One model decision returned to the runtime.
102#[derive(Clone, Debug, Eq, PartialEq)]
103pub enum ModelDecision {
104    /// Ask an application-backed tool to execute.
105    Tool(ToolInput),
106    /// Ask a named application control tool to execute.
107    NamedTool {
108        /// Registered provider-facing tool name.
109        name: String,
110        /// Validated JSON arguments supplied by the provider.
111        arguments: serde_json::Value,
112        /// Schema-redacted arguments safe for detailed presentation.
113        presentation_arguments: serde_json::Value,
114        /// Optional trusted compact label for human presentation.
115        presentation_label: Option<String>,
116    },
117    /// Delegate focused work to a nested branch.
118    Child(ChildRequest),
119    /// Finish with provider-produced text.
120    Final(String),
121}

123/// Provider boundary used by the integrated runtime.
124pub trait ModelProvider {
125    /// Produces the next decision from the current semantic conversation.
126    ///
127    /// # Parameters
128    /// - `messages`: Immutable conversation and tool observations.
129    ///
130    /// # Returns
131    /// The next tool request or final response.
132    ///
133    /// # Errors
134    /// Returns a provider-specific failure.
135    fn decide(&mut self, messages: &[AgentMessage]) -> Result<ModelDecision, String>;
136
137    /// Returns and clears captured safe request projections since the last call.
138    ///
139    /// # Returns
140    /// Provider-neutral requests prepared for completed invocations.
141    fn take_request_projections(&mut self) -> Vec<scriberig_session::ModelRequestProjection> {
142        Vec::new()
143    }
144
145    /// Produces a decision while forwarding provider-neutral text fragments.
146    ///
147    /// # Parameters
148    /// - `messages`: Immutable conversation and tool observations.
149    /// - `on_text`: Callback receiving bounded assistant-text fragments.
150    ///
151    /// # Returns
152    /// The next tool request or final response.
153    ///
154    /// # Errors
155    /// Returns the same provider failure as [`Self::decide`].
156    fn decide_with_output(
157        &mut self,
158        messages: &[AgentMessage],
159        on_text: &mut dyn FnMut(&str),
160    ) -> Result<ModelDecision, String> {
161        let decision = self.decide(messages)?;
162        if let ModelDecision::Final(output) = &decision {
163            on_text(output);
164        }
165        Ok(decision)
166    }
167}

169/// Deterministic provider used by local one-shot composition and tests.
170pub struct EchoProvider;
171
172impl ModelProvider for EchoProvider {
173    /// Returns the latest user prompt as a completed response.
174    ///
175    /// # Parameters
176    /// - `messages`: Conversation messages available to the provider.
177    ///
178    /// # Returns
179    /// A final response containing the latest user prompt.
180    ///
181    /// # Errors
182    /// Returns an error when no user prompt is available.
183    fn decide(&mut self, messages: &[AgentMessage]) -> Result<ModelDecision, String> {
184        let Some(AgentMessage::User(prompt)) = messages.first() else {
185            return Err("agent prompt is missing".to_owned());
186        };
187        Ok(ModelDecision::Final(prompt.clone()))
188    }
189}

191/// Tool implementation that prevents unconfigured providers from mutating state.
192pub struct UnconfiguredTool;
193
194impl Tool for UnconfiguredTool {
195    /// Identifies the guarded fallback tool.
196    ///
197    /// # Returns
198    /// The stable fallback tool name.
199    fn name(&self) -> &'static str {
200        "unconfigured"
201    }
202
203    /// Rejects tool execution until a concrete application tool is composed.
204    ///
205    /// # Parameters
206    /// - `_context`: Trusted application context.
207    /// - `_input`: Structured tool input to reject.
208    ///
209    /// # Returns
210    /// Never returns a successful tool result.
211    ///
212    /// # Errors
213    /// Always returns an unsupported-operation tool error.
214    fn invoke(
215        &self,
216        _context: &ApplicationContext,
217        _input: ToolInput,
218    ) -> Result<ToolResult, ToolError> {
219        Err(ToolError::Application {
220            category: "unsupported_operation".to_owned(),
221            message: "no agent application tool is configured".to_owned(),
222        })
223    }
224}

240/// Executes a deterministic local prompt through agent orchestration.
241///
242/// # Parameters

246/// Provider output and lifecycle messages.
247///
248/// # Errors
249/// Returns the structured runtime failure.

272/// Structured failure returned by the agent orchestration boundary.
273#[derive(Clone, Debug, Eq, PartialEq)]
274pub enum AgentError {
275    /// The model provider failed to produce a decision.
276    Provider(String),
277    /// A tool returned a structured semantic failure.
278    Tool(ToolError),
279    /// The configured step limit was reached.
280    StepLimit {
281        /// Configured maximum number of provider decisions.
282        limit: usize,
283    },
284    /// A non-step runtime budget was exhausted.
285    BudgetExceeded {
286        /// Stable budget category.
287        kind: &'static str,
288        /// Configured maximum that was reached.
289        limit: usize,
290    },
291    /// The caller requested cancellation.
292    Cancelled,
293}
294
295impl std::fmt::Display for AgentError {
296    /// Formats a safe agent-runtime failure.
297    ///
298    /// # Parameters
299    /// - `self`: Agent error.
300    /// - `formatter`: Destination formatter.
301    ///
302    /// # Returns
303    /// Formatting success or formatter failure.
304    ///
305    /// # Errors
306    /// Returns the formatter error when output cannot be written.
307    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
308        match self {
309            Self::Provider(message) => write!(formatter, "agent provider: {message}"),
310            Self::Tool(error) => write!(formatter, "agent tool: {error:?}"),
311            Self::StepLimit { limit } => write!(formatter, "agent step limit reached: {limit}"),
312            Self::BudgetExceeded { kind, limit } => {
313                write!(formatter, "agent {kind} budget exceeded: {limit}")
314            }
315            Self::Cancelled => formatter.write_str("agent operation cancelled"),
316        }
317    }
318}

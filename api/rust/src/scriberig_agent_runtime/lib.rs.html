26/// One model-facing message retained by the runtime loop.

28pub enum AgentMessage {

68/// Executes a trusted child branch on behalf of the parent runtime.
69pub trait ChildExecutor {

71    ///
72    /// # Parameters
73    /// - `context`: Parent trusted application context.
74    /// - `scope`: Optional durable parent branch identity.
75    /// - `request`: Validated child purpose and prompt.
76    /// - `cancelled`: Shared cancellation flag for the execution tree.
77    ///
78    /// # Returns
79    /// A compact child result suitable for the parent model context.

92/// One model decision returned to the runtime.
93#[derive(Clone, Debug, Eq, PartialEq)]
94pub enum ModelDecision {
95    /// Ask an application-backed tool to execute.
96    Tool(ToolInput),
97    /// Ask a named application control tool to execute.
98    NamedTool {
99        /// Registered provider-facing tool name.
100        name: String,
101        /// Validated JSON arguments supplied by the provider.
102        arguments: serde_json::Value,
103    },
104    /// Delegate focused work to a nested branch.
105    Child(ChildRequest),
106    /// Finish with provider-produced text.
107    Final(String),
108}

110/// Provider boundary used by the integrated runtime.
111pub trait ModelProvider {
112    /// Produces the next decision from the current semantic conversation.
113    ///
114    /// # Parameters
115    /// - `messages`: Immutable conversation and tool observations.
116    ///
117    /// # Returns
118    /// The next tool request or final response.
119    ///
120    /// # Errors
121    /// Returns a provider-specific failure.
122    fn decide(&mut self, messages: &[AgentMessage]) -> Result<ModelDecision, String>;
123
124    /// Produces a decision while forwarding provider-neutral text fragments.
125    ///
126    /// # Parameters
127    /// - `messages`: Immutable conversation and tool observations.
128    /// - `on_text`: Callback receiving bounded assistant-text fragments.
129    ///
130    /// # Returns
131    /// The next tool request or final response.
132    ///
133    /// # Errors
134    /// Returns the same provider failure as [`Self::decide`].
135    fn decide_with_output(
136        &mut self,
137        messages: &[AgentMessage],
138        on_text: &mut dyn FnMut(&str),
139    ) -> Result<ModelDecision, String> {
140        let decision = self.decide(messages)?;
141        if let ModelDecision::Final(output) = &decision {
142            on_text(output);
143        }
144        Ok(decision)
145    }
146}

148/// Deterministic provider used by local one-shot composition and tests.
149pub struct EchoProvider;
150
151impl ModelProvider for EchoProvider {
152    /// Returns the latest user prompt as a completed response.
153    ///
154    /// # Parameters
155    /// - `messages`: Conversation messages available to the provider.
156    ///
157    /// # Returns
158    /// A final response containing the latest user prompt.
159    ///
160    /// # Errors
161    /// Returns an error when no user prompt is available.
162    fn decide(&mut self, messages: &[AgentMessage]) -> Result<ModelDecision, String> {
163        let Some(AgentMessage::User(prompt)) = messages.first() else {
164            return Err("agent prompt is missing".to_owned());
165        };
166        Ok(ModelDecision::Final(prompt.clone()))
167    }
168}

170/// Tool implementation that prevents unconfigured providers from mutating state.
171pub struct UnconfiguredTool;
172
173impl Tool for UnconfiguredTool {
174    /// Identifies the guarded fallback tool.
175    ///
176    /// # Returns
177    /// The stable fallback tool name.
178    fn name(&self) -> &'static str {
179        "unconfigured"
180    }
181
182    /// Rejects tool execution until a concrete application tool is composed.
183    ///
184    /// # Parameters
185    /// - `_context`: Trusted application context.
186    /// - `_input`: Structured tool input to reject.
187    ///
188    /// # Returns
189    /// Never returns a successful tool result.
190    ///
191    /// # Errors
192    /// Always returns an unsupported-operation tool error.
193    fn invoke(
194        &self,
195        _context: &ApplicationContext,
196        _input: ToolInput,
197    ) -> Result<ToolResult, ToolError> {
198        Err(ToolError::Application {
199            category: "unsupported_operation".to_owned(),
200            message: "no agent application tool is configured".to_owned(),
201        })
202    }
203}

219/// Executes a deterministic local prompt through agent orchestration.
220///
221/// # Parameters
222/// - `prompt`: Prompt accepted by the agent product mode.
223///
224/// # Returns
225/// Provider output and lifecycle messages.
226///
227/// # Errors
228/// Returns the structured runtime failure.
229pub fn run_local_agent(prompt: impl Into<String>) -> Result<AgentResult, AgentError> {
230    AgentRuntime::new(EchoProvider, UnconfiguredTool, 1).run(
231        &ApplicationContext::empty(),
232        prompt,
233        &AtomicBool::new(false),
234    )

251/// Structured failure returned by the agent orchestration boundary.
252#[derive(Clone, Debug, Eq, PartialEq)]
253pub enum AgentError {
254    /// The model provider failed to produce a decision.
255    Provider(String),
256    /// A tool returned a structured semantic failure.
257    Tool(ToolError),
258    /// The configured step limit was reached.
259    StepLimit {
260        /// Configured maximum number of provider decisions.
261        limit: usize,
262    },
263    /// A non-step runtime budget was exhausted.
264    BudgetExceeded {
265        /// Stable budget category.
266        kind: &'static str,
267        /// Configured maximum that was reached.
268        limit: usize,
269    },
270    /// The caller requested cancellation.
271    Cancelled,
272}
273
274impl std::fmt::Display for AgentError {
275    /// Formats a safe agent-runtime failure.
276    ///
277    /// # Parameters
278    /// - `self`: Agent error.
279    /// - `formatter`: Destination formatter.
280    ///
281    /// # Returns
282    /// Formatting success or formatter failure.
283    ///
284    /// # Errors
285    /// Returns the formatter error when output cannot be written.
286    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
287        match self {
288            Self::Provider(message) => write!(formatter, "agent provider: {message}"),
289            Self::Tool(error) => write!(formatter, "agent tool: {error:?}"),
290            Self::StepLimit { limit } => write!(formatter, "agent step limit reached: {limit}"),
291            Self::BudgetExceeded { kind, limit } => {
292                write!(formatter, "agent {kind} budget exceeded: {limit}")
293            }
294            Self::Cancelled => formatter.write_str("agent operation cancelled"),
295        }
296    }
297}

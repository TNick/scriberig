8/// Declares model-provider behavior available to the runtime.
9#[allow(clippy::struct_excessive_bools)]
10#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
11pub struct ProviderCapabilities {
12    /// Whether response streaming is supported.
13    pub streaming: bool,
14    /// Whether structured tool calls are supported.
15    pub tool_calls: bool,
16    /// Whether multiple tool calls may be returned together.
17    pub parallel_tool_calls: bool,
18    /// Whether structured output schemas are supported.
19    pub structured_output: bool,
20    /// Whether token usage is reported.
21    pub usage_reporting: bool,
22    /// Whether provider request identifiers are returned.
23    pub provider_request_id: bool,
24    /// Whether provider model fingerprints are returned.
25    pub provider_model_fingerprint: bool,
26}

28/// Selects known deviations from the OpenAI-compatible request family.
29#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
30pub enum CompatibilityProfile {
31    /// OpenAI-compatible `/chat/completions` semantics.
32    #[default]
33    ChatCompletions,
34}

36/// Selects an indirect source for a provider credential.
37#[derive(Clone, Debug, Eq, PartialEq)]
38pub enum CredentialSource {
39    /// Read the credential from the named environment variable.
40    Environment(String),
41    /// Send no credential header.
42    None,
43}

45/// Runtime provider configuration without an embedded secret value.
46#[derive(Clone, Debug, Eq, PartialEq)]
47pub struct ProviderConfig {
48    /// Safe profile identifier used in diagnostics.
49    pub profile_id: String,
50    /// Provider-compatible chat completion endpoint base URL.
51    pub base_url: String,
52    /// Provider model identifier.
53    pub model: String,
54    /// Indirect credential source.
55    pub credential_source: CredentialSource,
56    /// Maximum time allowed for one request.
57    pub request_timeout: Duration,
58    /// Maximum time allowed to establish a connection.
59    pub connect_timeout: Duration,
60    /// Whether streaming is enabled by default.
61    pub stream_default: bool,
62    /// Known wire compatibility profile.
63    pub compatibility_profile: CompatibilityProfile,
64}
65
66impl ProviderConfig {
67    /// Creates a configuration with conservative request timeouts.
68    ///
69    /// # Parameters
70    /// - `profile_id`: Safe profile display identifier.
71    /// - `base_url`: Provider endpoint base URL.
72    /// - `model`: Provider model identifier.
73    /// - `credential_source`: Indirect credential source.
74    ///
75    /// # Returns
76    /// A provider configuration with streaming disabled.
77    #[must_use]
78    pub fn new(
79        profile_id: impl Into<String>,
80        base_url: impl Into<String>,
81        model: impl Into<String>,
82        credential_source: CredentialSource,
83    ) -> Self {
84        Self {
85            profile_id: profile_id.into(),
86            base_url: base_url.into(),
87            model: model.into(),
88            credential_source,
89            request_timeout: Duration::from_secs(60),
90            connect_timeout: Duration::from_secs(10),
91            stream_default: false,
92            compatibility_profile: CompatibilityProfile::default(),
93        }
94    }
95
96    /// Resolves the configured credential at request time.
97    ///
98    /// # Returns
99    /// A secret wrapper when the configured source contains a value.
100    ///
101    /// # Errors
102    /// Returns a configuration error when the environment source is missing
103    /// or empty.
104    pub fn resolve_credential(&self) -> Result<Option<ResolvedCredential>, ProviderError> {
105        match &self.credential_source {
106            CredentialSource::Environment(name) => {
107                let value = std::env::var(name).map_err(|_| {
108                    ProviderError::Configuration(format!(
109                        "credential environment variable {name} is not set"
110                    ))
111                })?;
112                if value.is_empty() {
113                    return Err(ProviderError::Configuration(format!(
114                        "credential environment variable {name} is empty"
115                    )));
116                }
117                Ok(Some(ResolvedCredential(value)))
118            }
119            CredentialSource::None => Ok(None),
120        }
121    }
122}

124/// Opaque credential value resolved only inside the provider adapter.
125pub struct ResolvedCredential(String);
126
127impl ResolvedCredential {
128    /// Returns the secret for constructing an outbound authorization header.
129    ///
130    /// # Returns
131    /// The resolved credential value.
132    #[must_use]
133    pub fn as_str(&self) -> &str {
134        &self.0
135    }
136}
137
138impl fmt::Debug for ResolvedCredential {
139    /// Redacts the credential in diagnostics.
140    ///
141    /// # Parameters
142    /// - `self`: Credential to redact.
143    /// - `formatter`: Destination formatter.
144    ///
145    /// # Returns
146    /// Formatting success or the formatter error.
147    ///
148    /// # Errors
149    /// Returns the formatter error when output cannot be written.
150    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
151        formatter.write_str("ResolvedCredential(REDACTED)")
152    }
153}

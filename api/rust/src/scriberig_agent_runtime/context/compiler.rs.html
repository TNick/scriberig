14/// Deterministic context selection limits.
15#[derive(Clone, Copy, Debug, Eq, PartialEq)]
16pub struct ContextPolicy {
17    /// Maximum estimated characters in compiled context.
18    pub max_chars: usize,
19}
20
21impl Default for ContextPolicy {
22    /// Creates a bounded default context policy.
23    ///
24    /// The default character budget.
25    /// # Returns
26    /// The operation result.
27    fn default() -> Self {
28        Self { max_chars: 16_000 }
29    }
30}

32/// Provider-neutral compiled context and exact dependency summary.
33#[derive(Clone, Debug, Eq, PartialEq)]
34pub struct CompiledContext {
35    /// Exact immutable profile revision used by compilation.
36    pub profile: AgentProfileRevisionRef,
37    /// Active exact component revisions in compilation order.
38    pub active_components: Vec<ContextComponentRevisionRef>,
39    /// Canonical compiled messages, including the current instruction.
40    ///
41    /// Adapters that also receive the current instruction as a runtime
42    /// message must use `provider_context` to avoid sending it twice.
43    pub messages: Vec<ModelMessage>,
44    /// Exact semantic dependency identities used by compilation.
45    pub dependencies: Vec<String>,
46    /// Handles for details omitted by the budget.
47    pub omitted_handles: Vec<String>,
48    /// Canonical representation used by request identity.
49    pub canonical: String,
50    /// Provider-visible compiled context without the current user instruction.
51    pub provider_context: String,
52}

54/// Compiles branch context using deterministic priority and ordering rules.
55pub struct ContextCompiler {
56    /// Bounded selection policy.
57    policy: ContextPolicy,
58    /// Exact profile descriptor used to compile branch instructions.
59    profile: AgentProfileDescriptor,
60}
61
62impl ContextCompiler {
63    /// Creates a compiler with the supplied budget policy.
64    ///
65    /// - `policy`: Deterministic context selection policy.
66    ///
67    /// A configured context compiler.
68    ///
69    /// # Returns
70    /// A compiler using the supplied policy and bootstrap profile.
71    #[must_use]
72    pub fn new(policy: ContextPolicy) -> Self {
73        Self {
74            policy,
75            profile: bootstrap_default_agent_profile(),
76        }
77    }
78
79    /// Replaces the bootstrap profile with an exact resolved profile.
80    ///
81    /// # Parameters
82    /// - `self`: Context compiler being configured.
83    /// - `profile`: Exact profile descriptor for the invocation.
84    ///
85    /// The compiler configured to use the supplied profile.
86    ///
87    /// # Returns
88    /// The configured context compiler.
89    #[must_use]
90    pub fn with_profile(mut self, profile: AgentProfileDescriptor) -> Self {
91        self.profile = profile;
92        self
93    }
94
95    /// Compiles required branch inputs and bounded current-branch activity.
96    ///
97    /// - `context`: Explicit branch execution inputs.
98    ///
99    /// Canonical provider-neutral context and dependency manifest.
100    #[must_use]
101    /// # Returns
102    /// The operation result.
103    pub fn compile(&self, context: &BranchExecutionContext) -> CompiledContext {
104        let mut messages = Vec::new();
105        let mut dependencies = Vec::new();
106        let mut canonical = String::new();
107        let active_components =
108            self.append_required_context(context, &mut messages, &mut dependencies, &mut canonical);
109        let mut provider_context = canonical.clone();
110        push_message(
111            &mut messages,
112            &mut canonical,
113            "instruction",
114            &context.instruction,
115        );
116        let omitted_handles = self.append_activities(
117            context,
118            &mut messages,
119            &mut dependencies,
120            &mut canonical,
121            &mut provider_context,
122        );
123        CompiledContext {
124            profile: self.profile.reference.clone(),
125            active_components: active_components
126                .iter()
127                .map(|component| component.reference.clone())
128                .collect(),
129            messages,
130            dependencies,
131            omitted_handles,
132            canonical,
133            provider_context,
134        }
135    }
136
137    /// Appends branch scope, profile, and selected durable results.
138    ///
139    /// # Parameters
140    /// - `context`: Explicit branch execution inputs.
141    /// - `messages`: Compiled provider messages.
142    /// - `dependencies`: Exact semantic dependency identities.
143    /// - `canonical`: Canonical request identity text.
144    ///
145    /// Active profile components in deterministic activation order.
146    ///
147    /// # Returns
148    /// Active components used to compile the required context.
149    fn append_required_context(
150        &self,
151        context: &BranchExecutionContext,
152        messages: &mut Vec<ModelMessage>,
153        dependencies: &mut Vec<String>,
154        canonical: &mut String,
155    ) -> Vec<ResolvedContextComponent> {
156        dependencies.extend([
157            format!("session:{}", context.session),
158            format!("branch:{}", context.branch),
159        ]);
160        let active_components =
161            resolve_active_context_components(&self.profile, &context.branch_role);
162        let profile_ref = &self.profile.reference;
163        dependencies.push(format!(
164            "agent-profile:{}@{}",
165            profile_ref.identity, profile_ref.revision
166        ));
167        let _ = writeln!(
168            canonical,
169            "agent-profile-ref:{}@{}",
170            profile_ref.identity, profile_ref.revision
171        );
172        append_components(messages, canonical, dependencies, &active_components);
173        if let Some(anchor) = &context.continuation_anchor {
174            dependencies.push(format!("continuation-anchor:{anchor}"));
175            push_message(messages, canonical, "continuation_anchor", anchor);
176        }
177        push_message(messages, canonical, "purpose", &context.purpose);
178        if let Some(summary) = &context.parent_summary {
179            push_message(messages, canonical, "parent", summary);
180            dependencies.push(format!("parent-summary:{summary}"));
181        }
182        append_selected_results(
183            messages,
184            canonical,
185            dependencies,
186            &context.ancestor_results,
187            "ancestor_result",
188            "ancestor-result",
189        );
190        append_selected_results(
191            messages,
192            canonical,
193            dependencies,
194            &context.dependency_results,
195            "dependency_result",
196            "dependency-result",
197        );
198        if let Some(scope) = &context.project_scope {
199            let orientation = format!(
200                "project={} snapshot={} root={} selection={}",
201                scope.project,
202                scope.snapshot,
203                scope.root,
204                scope.selection.as_deref().unwrap_or("none")
205            );
206            push_message(messages, canonical, "semantic_project_scope", &orientation);
207            dependencies.extend([
208                format!("project:{}", scope.project),
209                format!("snapshot:{}", scope.snapshot),
210                format!("root:{}", scope.root),
211            ]);
212            if let Some(selection) = &scope.selection {
213                dependencies.push(format!("selection:{selection}"));
214            }
215        }
216        active_components
217    }

219    /// Adds budgeted branch activity to canonical and provider context.
220    ///
221    /// # Parameters
222    /// - `context`: Explicit branch execution inputs.
223    /// - `messages`: Compiled provider messages.
224    /// - `dependencies`: Exact semantic dependency identities.
225    /// - `canonical`: Canonical request identity text.
226    /// - `provider_context`: Provider context excluding the current instruction.
227    ///
228    /// Handles for activities omitted by the context budget.
229    ///
230    /// # Returns
231    /// Handles belonging to activities that exceeded the budget.
232    fn append_activities(
233        &self,
234        context: &BranchExecutionContext,
235        messages: &mut Vec<ModelMessage>,
236        dependencies: &mut Vec<String>,
237        canonical: &mut String,
238        provider_context: &mut String,
239    ) -> Vec<String> {
240        let mut activities = context.activities.clone();
241        activities.sort_by(|left, right| {
242            left.ordinal
243                .cmp(&right.ordinal)
244                .then_with(|| left.kind.cmp(&right.kind))
245                .then_with(|| left.content.cmp(&right.content))
246        });
247        let mut used = canonical.len();
248        let mut omitted_handles = Vec::new();
249        for activity in activities {
250            let entry = format!("{}:{}", activity.kind, activity.content);
251            if used + entry.len() > self.policy.max_chars {
252                if let Some(handle) = activity.handle {
253                    omitted_handles.push(handle);
254                }
255                continue;
256            }
257            used += entry.len();
258            push_message(messages, canonical, &activity.kind, &activity.content);
259            let _ = writeln!(provider_context, "{}:{}", activity.kind, activity.content);
260            dependencies.push(format!("activity:{}:{}", activity.kind, activity.ordinal));
261        }
262        omitted_handles
263    }
264}

266/// Adds activated profile components and exact revision provenance.
267///
268/// # Parameters
269/// - `messages`: Provider-neutral messages receiving component content.
270/// - `canonical`: Request identity receiving component refs and content.
271/// - `dependencies`: Exact profile-component refs used by the request.
272/// - `components`: Active components in deterministic order.
273fn append_components(
274    messages: &mut Vec<ModelMessage>,
275    canonical: &mut String,
276    dependencies: &mut Vec<String>,
277    components: &[ResolvedContextComponent],
278) {
279    for component in components {
280        let reference = &component.reference;
281        dependencies.push(format!(
282            "context-component:{}@{}",
283            reference.identity, reference.revision
284        ));
285        let _ = writeln!(
286            canonical,
287            "context-component-ref:{}@{}",
288            reference.identity, reference.revision
289        );
290        push_message(
291            messages,
292            canonical,
293            "system",
294            &format!(
295                "[{}] {}",
296                component_kind_key(component.kind),
297                component.content
298            ),
299        );
300    }
301}

303/// Returns the stable section name used for a typed component kind.
304///
305/// # Parameters
306/// - `kind`: Typed semantic purpose of a component.
307///
308/// The stable provider-neutral section name.
309/// # Returns
310/// The section name for the component kind.
311fn component_kind_key(kind: ContextComponentKind) -> &'static str {
312    match kind {
313        ContextComponentKind::CoreInvariant => "core_invariant",
314        ContextComponentKind::BehaviorPolicy => "behavior_policy",
315        ContextComponentKind::DomainGuidance => "domain_guidance",
316        ContextComponentKind::ToolGuidance => "tool_guidance",
317        ContextComponentKind::SkillTrigger => "skill_trigger",
318        ContextComponentKind::ExecutionLesson => "execution_lesson",
319    }
320}

322/// Adds selected durable results and their exact semantic handles.
323///
324/// - `messages`: Provider messages receiving compact result context.
325/// - `canonical`: Canonical request identity receiving serialized content.
326/// - `dependencies`: Dependency manifest receiving exact result handles.
327/// - `results`: Selected durable results to include.
328/// - `role`: Provider message role for this result class.
329/// - `dependency_kind`: Manifest prefix identifying the result class.
330fn append_selected_results(
331    messages: &mut Vec<ModelMessage>,
332    canonical: &mut String,
333    dependencies: &mut Vec<String>,
334    results: &[CompactResultContext],
335    role: &str,
336    dependency_kind: &str,
337) {
338    let mut ordered = results.to_vec();
339    ordered.sort_by(|left, right| left.handle.cmp(&right.handle));
340    for result in ordered {
341        let rendered_result = format_result_context(&result);
342        push_message(messages, canonical, role, &rendered_result);
343        dependencies.push(format!("{dependency_kind}:{}", result.handle));
344        dependencies.extend(
345            result
346                .semantic_handles
347                .iter()
348                .map(|handle| format!("semantic-handle:{handle}")),
349        );
350    }
351}

353/// Formats a compact durable result without activity or transcript detail.
354///
355/// - `result`: Selected result summary.
356///
357/// Stable provider-neutral result content.
358/// # Returns
359/// The operation result.
360fn format_result_context(result: &CompactResultContext) -> String {
361    let mut handles = result.semantic_handles.clone();
362    handles.sort();
363    format!(
364        "{} revision={}: {}; handles={}",
365        result.handle,
366        result.revision.as_deref().unwrap_or("none"),
367        result.summary,
368        handles.join(",")
369    )
370}

372/// Adds one canonical message to the compiled context.
373///
374/// - `messages`: Message list receiving the compiled message.
375/// - `canonical`: Canonical identity text receiving the message.
376/// - `role`: Semantic message role.
377/// - `content`: Message content.
378fn push_message(
379    messages: &mut Vec<ModelMessage>,
380    canonical: &mut String,
381    role: &str,
382    content: &str,
383) {
384    let message = format!("[{role}] {content}");
385    messages.push(if role == "system" {
386        ModelMessage::system(message)
387    } else {
388        ModelMessage::user(message)
389    });
390    let _ = writeln!(canonical, "{role}:{content}");
391}

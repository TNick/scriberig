9/// Classifies how a trusted tool executes.
10#[derive(Clone, Copy, Debug, Eq, PartialEq)]
11pub enum ToolExecutionMode {
12    /// Executes one application operation in the current branch.
13    Atomic,
14    /// Starts a separate reasoning branch.
15    ChildBranch,
16}
17
18/// Shared durable semantic category for one trusted tool.
19pub type ToolCategory = scriberig_session::ToolActivityKind;
20
21/// Complete trusted descriptor exposed to provider request compilation.
22#[derive(Clone, Debug, Eq, PartialEq)]
23pub struct ToolDescriptor {
24    /// Provider-neutral function definition.
25    pub definition: ToolDefinition,
26    /// Trusted provider-neutral semantic category used for exposure ordering.
27    pub category: ToolCategory,
28    /// Bounded result-shape description.
29    pub result_shape: String,
30    /// Optional trusted command-like label for compact invocation display.
31    pub invocation_label: Option<String>,
32    /// Whether the operation can change canonical state.
33    pub state_changing: bool,
34    /// Execution mode selected by the application runtime.
35    pub execution_mode: ToolExecutionMode,
36    /// Capability required before the tool may be exposed or invoked.
37    pub required_capability: Option<String>,
38}
39
40/// Trusted descriptors available to one branch execution.
41#[derive(Clone, Debug, Default)]
42pub struct ToolCatalog {
43    descriptors: Vec<ToolDescriptor>,
44}
45
46impl ToolCatalog {
47    /// Creates a deterministic catalog sorted by stable tool identifier.
48    ///
49    /// - `descriptors`: Trusted application descriptors.
50    ///
51    /// A catalog with deterministic provider exposure order.
52    #[must_use]
53    /// # Returns
54    /// The operation result.
55    pub fn new(mut descriptors: Vec<ToolDescriptor>) -> Self {
56        for descriptor in &mut descriptors {
57            descriptor.invocation_label = descriptor
58                .invocation_label
59                .as_deref()
60                .map(bounded_invocation_label)
61                .filter(|label| !label.is_empty());
62        }
63        descriptors.sort_by(|left, right| {
64            left.category
65                .cmp(&right.category)
66                .then_with(|| left.definition.name.cmp(&right.definition.name))
67        });
68        Self { descriptors }
69    }
70
71    /// Returns provider definitions in deterministic order.
72    ///
73    /// The definitions suitable for a normalized model request.
74    #[must_use]
75    /// # Returns
76    /// The operation result.
77    pub fn definitions(&self) -> Vec<ToolDefinition> {
78        self.descriptors
79            .iter()
80            .map(|descriptor| descriptor.definition.clone())
81            .collect()
82    }
83
84    /// Returns descriptors authorized by one branch capability context.
85    ///
86    /// - `context`: Capabilities granted to the current branch.
87    ///
88    /// Authorized descriptors in stable catalog order.
89    #[must_use]
90    /// # Returns
91    /// The operation result.
92    pub fn descriptors_for(&self, context: &CapabilityContext) -> Vec<ToolDescriptor> {
93        self.descriptors
94            .iter()
95            .filter(|descriptor| {
96                descriptor
97                    .required_capability
98                    .as_deref()
99                    .is_some_and(|required| capability_granted(context, required))
100                    || (descriptor.category != ToolCategory::OpaqueFallback
101                        && descriptor.required_capability.is_none())
102            })
103            .cloned()
104            .collect()
105    }
106
107    /// Validates a provider-selected call against trusted descriptors.
108    ///
109    /// - `call`: Normalized provider tool call.
110    ///
111    /// The trusted descriptor for the call.
112    ///
113    /// Returns a safe validation error for unknown tools or invalid arguments.
114    /// # Returns
115    /// The operation result.
116    /// # Errors
117    /// Returns an error when the operation fails.
118    pub fn validate(&self, call: &ToolCall) -> Result<&ToolDescriptor, ToolValidationError> {
119        let descriptor = self
120            .descriptors
121            .iter()
122            .find(|descriptor| descriptor.definition.name == call.name)
123            .ok_or_else(|| ToolValidationError::UnknownTool(call.name.clone()))?;
124        if !call.arguments.is_object() {
125            return Err(ToolValidationError::ArgumentsNotObject {
126                tool: call.name.clone(),
127            });
128        }
129        if call.name == "inspect"
130            && call
131                .arguments
132                .get("entity")
133                .and_then(serde_json::Value::as_str)
134                .is_none()
135        {
136            return Err(ToolValidationError::MissingArgument {
137                tool: call.name.clone(),
138                argument: "entity".to_owned(),
139            });
140        }
141        if call.name == "delegate"
142            && (call
143                .arguments
144                .get("purpose")
145                .and_then(serde_json::Value::as_str)
146                .is_none()
147                || call
148                    .arguments
149                    .get("prompt")
150                    .and_then(serde_json::Value::as_str)
151                    .is_none()
152                || call
153                    .arguments
154                    .get("capabilities")
155                    .and_then(serde_json::Value::as_array)
156                    .is_none()
157                || call
158                    .arguments
159                    .get("expected_result")
160                    .and_then(serde_json::Value::as_str)
161                    .is_none())
162        {
163            return Err(ToolValidationError::MissingArgument {
164                tool: call.name.clone(),
165                argument: "purpose, prompt, capabilities, and expected_result".to_owned(),
166            });
167        }
168        Ok(descriptor)
169    }
170
171    /// Validates a provider call and its required branch capability.
172    ///
173    /// - `call`: Normalized provider tool call.
174    /// - `context`: Capabilities granted to the current branch.
175    ///
176    /// The trusted descriptor when validation and authorization succeed.
177    ///
178    /// # Returns
179    /// The descriptor authorized for this branch.
180    ///
181    /// # Errors
182    /// Returns a validation or capability-denial error.
183    pub fn validate_for_context<'a>(
184        &'a self,
185        call: &ToolCall,
186        context: &CapabilityContext,
187    ) -> Result<&'a ToolDescriptor, ToolValidationError> {
188        let descriptor = self.validate(call)?;
189        if let Some(required) = descriptor.required_capability.as_deref()
190            && !capability_granted(context, required)
191        {
192            return Err(ToolValidationError::CapabilityDenied(required.to_owned()));
193        }
194        Ok(descriptor)
195    }
196}

198/// Sanitizes and bounds one trusted invocation label for client presentation.
199///
200/// - `label`: Descriptor-provided human-readable label.
201///
202/// At most 128 non-control Unicode scalar values.
203/// # Returns
204/// The operation result.
205fn bounded_invocation_label(label: &str) -> String {
206    label
207        .chars()
208        .filter(|character| !character.is_control())
209        .take(128)
210        .collect()
211}
212
213/// Checks both named grants and scoped application grants.
214///
215/// - `context`: Capability context to inspect.
216/// - `required`: Stable required capability name.
217///
218/// `true` when the capability is granted.
219/// # Returns
220/// The operation result.
221fn capability_granted(context: &CapabilityContext, required: &str) -> bool {
222    if context
223        .granted
224        .iter()
225        .any(|capability| capability.as_str() == required)
226    {
227        return true;
228    }
229    let action = match required {
230        "inspect" | "inspect.entity" => CapabilityAction::Inspect,
231        name => CapabilityAction::Custom(name.to_owned()),
232    };
233    context.permits(&action, &CapabilityScope::Project, None, None)
234}

236/// Safe error returned when a model call fails trusted validation.
237#[derive(Clone, Debug, Eq, PartialEq)]
238pub enum ToolValidationError {
239    /// Provider selected an unregistered tool identifier.
240    UnknownTool(String),
241    /// Provider supplied a non-object argument value.
242    ArgumentsNotObject {
243        /// Tool being invoked.
244        tool: String,
245    },
246    /// Provider omitted a required argument.
247    MissingArgument {
248        /// Tool being invoked.
249        tool: String,
250        /// Required argument name.
251        argument: String,
252    },
253    /// The branch lacks the capability required by a trusted descriptor.
254    CapabilityDenied(String),
255}
256
257impl fmt::Display for ToolValidationError {
258    /// Formats a safe tool validation error.
259    ///
260    /// - `self`: Validation failure.
261    /// - `formatter`: Destination formatter.
262    ///
263    /// Formatting success or the formatter error.
264    ///
265    /// Returns the formatter error when output cannot be written.
266    /// # Returns
267    /// The operation result.
268    /// # Errors
269    /// Returns an error when the operation fails.
270    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
271        match self {
272            Self::UnknownTool(tool) => write!(formatter, "unknown trusted tool: {tool}"),
273            Self::ArgumentsNotObject { tool } => {
274                write!(formatter, "tool {tool} arguments must be an object")
275            }
276            Self::MissingArgument { tool, argument } => {
277                write!(formatter, "tool {tool} requires argument {argument}")
278            }
279            Self::CapabilityDenied(capability) => {
280                write!(formatter, "required capability denied: {capability}")
281            }
282        }
283    }
284}

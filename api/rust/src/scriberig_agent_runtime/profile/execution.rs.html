11/// Resolves a profile preference through the shared upstream-backed resolver.
12///
13/// Invocation, plan-node, branch, session, and project selections retain their
14/// existing precedence. When none supplies a model, the exact profile's
15/// preference is considered before the general user-profile fallback.
16///
17/// - `profile`: Exact `ScribeRig` profile descriptor for this execution.
18/// - `sources`: Existing semantic execution assignments.
19/// - `upstream_models`: Host-owned upstream catalog snapshot.
20/// - `runtime_fallback`: Model selected by host configuration when unassigned.
21///
22/// Provider/model IDs and the upstream-resolved thinking level.
23/// # Returns
24/// The effective execution selection resolved against the host catalog.
25/// # Errors
26/// Returns an application error for unavailable model or thinking IDs.
27pub fn resolve_profile_execution_selection(
28    profile: &AgentProfileDescriptor,
29    sources: ExecutionSelectionSources<'_>,
30    upstream_models: &[rpi_ai::Model],
31    runtime_fallback: &str,
32) -> Result<(String, String, Option<rpi_ai::ThinkingLevel>), ApplicationError> {
33    let selection = profile_execution_selection(profile);
34    resolve_execution_selection_with_profile_preference(
35        sources,
36        selection.as_ref(),
37        upstream_models,
38        runtime_fallback,
39    )
40}

42/// Resolves a plan node with its exact profile preference as a lower default.
43///
44/// The app resolver preserves a node's explicit assignment and all narrower
45/// execution scopes before consulting the profile preference.
46///
47/// - `profile`: Exact `ScribeRig` profile descriptor for this execution.
48/// - `node`: Durable plan node with an optional explicit assignment.
49/// - `sources`: Existing branch, session, project, and user selections.
50/// - `upstream_models`: Host-owned upstream catalog snapshot.
51/// - `runtime_fallback`: Model selected by host configuration when unassigned.
52///
53/// The exact upstream model and resolved assignment for this worker.
54/// # Returns
55/// The plan-node execution resolved against the host catalog.
56/// # Errors
57/// Returns an application error for unavailable model or thinking IDs.
58pub fn resolve_profile_plan_node_execution(
59    profile: &AgentProfileDescriptor,
60    node: &PlanNode,
61    sources: ExecutionSelectionSources<'_>,
62    upstream_models: &[rpi_ai::Model],
63    runtime_fallback: &str,
64) -> Result<ResolvedPlanExecution, ApplicationError> {
65    let selection = profile_execution_selection(profile);
66    resolve_plan_node_execution_with_profile_preference(
67        node,
68        sources,
69        selection.as_ref(),
70        upstream_models,
71        runtime_fallback,
72    )
73}

75/// Projects only public profile selection identifiers into app resolution.
76///
77/// The temporary value carries no profile revision, credentials, provider
78/// configuration, or model metadata. The exact profile revision remains part
79/// of context compilation and request provenance.
80///
81/// - `profile`: Profile whose preferred execution selection is projected.
82///
83/// A temporary session-selection shape for shared app resolution, if selected.
84/// # Returns
85/// `None` when the profile has no execution preference.
86fn profile_execution_selection(
87    profile: &AgentProfileDescriptor,
88) -> Option<SessionExecutionSelection> {
89    profile
90        .preferred_execution
91        .as_ref()
92        .map(|preference| SessionExecutionSelection {
93            revision: 0,
94            provider_id: Some(preference.provider_id.clone()),
95            model_id: Some(preference.model_id.clone()),
96            thinking_level: preference.thinking_level.clone(),
97            history: Vec::new(),
98        })
99}

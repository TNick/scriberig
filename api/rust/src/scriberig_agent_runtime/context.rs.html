11/// One bounded semantic activity item available to a branch compiler.
12#[derive(Clone, Debug, Eq, PartialEq)]
13pub struct ContextActivity {
14    /// Deterministic branch-local ordering value.
15    pub ordinal: u64,
16    /// Semantic activity category.
17    pub kind: String,
18    /// Human/model-readable compact content.
19    pub content: String,
20    /// Stable handle that can reopen omitted detail.
21    pub handle: Option<String>,
22}
23
24impl ContextActivity {
25    /// Creates an activity without a reopenable handle.
26    ///
27    /// # Parameters
28    /// - `ordinal`: Stable activity order.
29    /// - `kind`: Activity category.
30    /// - `content`: Compact activity content.
31    ///
32    /// # Returns
33    /// A context activity item.
34    #[must_use]
35    pub fn new(ordinal: u64, kind: impl Into<String>, content: impl Into<String>) -> Self {
36        Self {
37            ordinal,
38            kind: kind.into(),
39            content: content.into(),
40            handle: None,
41        }
42    }
43
44    /// Attaches a stable handle for omitted detail.
45    ///
46    /// # Parameters
47    /// - `self`: Activity being configured.
48    /// - `handle`: Stable semantic handle.
49    ///
50    /// # Returns
51    /// The activity with its reopenable handle.
52    #[must_use]
53    pub fn with_handle(mut self, handle: impl Into<String>) -> Self {
54        self.handle = Some(handle.into());
55        self
56    }
57}

180/// Provider-neutral compiled context and exact dependency summary.
181#[derive(Clone, Debug, Eq, PartialEq)]
182pub struct CompiledContext {
183    /// Ordered messages supplied to a provider adapter.
184    pub messages: Vec<ModelMessage>,
185    /// Exact semantic dependency identities used by compilation.
186    pub dependencies: Vec<String>,
187    /// Handles for details omitted by the budget.
188    pub omitted_handles: Vec<String>,
189    /// Canonical representation used by request identity.
190    pub canonical: String,
191}

193/// Compiles branch context using deterministic priority and ordering rules.
194pub struct ContextCompiler {
195    /// Bounded selection policy.
196    policy: ContextPolicy,
197}
198
199impl ContextCompiler {
200    /// Creates a compiler with the supplied budget policy.
201    ///
202    /// # Parameters
203    /// - `policy`: Deterministic context selection policy.
204    ///
205    /// # Returns
206    /// A configured context compiler.
207    #[must_use]
208    pub const fn new(policy: ContextPolicy) -> Self {
209        Self { policy }
210    }
211
212    /// Compiles required branch inputs and bounded current-branch activity.
213    ///
214    /// # Parameters
215    /// - `context`: Explicit branch execution inputs.
216    ///
217    /// # Returns
218    /// Canonical provider-neutral context and dependency manifest.
219    #[must_use]
220    pub fn compile(&self, context: &BranchExecutionContext) -> CompiledContext {
221        let mut messages = Vec::new();
222        let mut dependencies = vec![
223            format!("session:{}", context.session),
224            format!("branch:{}", context.branch),
225        ];
226        let mut canonical = String::new();
227        push_message(
228            &mut messages,
229            &mut canonical,
230            "system",
231            "You are operating within one isolated semantic branch.",
232        );
233        push_message(&mut messages, &mut canonical, "purpose", &context.purpose);
234        if let Some(summary) = &context.parent_summary {
235            push_message(&mut messages, &mut canonical, "parent", summary);
236            dependencies.push(format!("parent-summary:{summary}"));
237        }
238        if let Some(scope) = &context.project_scope {
239            let orientation = format!(
240                "project={} snapshot={} root={} selection={}",
241                scope.project,
242                scope.snapshot,
243                scope.root,
244                scope.selection.as_deref().unwrap_or("none")
245            );
246            push_message(
247                &mut messages,
248                &mut canonical,
249                "semantic_project_scope",
250                &orientation,
251            );
252            dependencies.extend([
253                format!("project:{}", scope.project),
254                format!("snapshot:{}", scope.snapshot),
255                format!("root:{}", scope.root),
256            ]);
257            if let Some(selection) = &scope.selection {
258                dependencies.push(format!("selection:{selection}"));
259            }
260        }
261        push_message(
262            &mut messages,
263            &mut canonical,
264            "instruction",
265            &context.instruction,
266        );
267        let mut activities = context.activities.clone();
268        activities.sort_by(|left, right| {
269            left.ordinal
270                .cmp(&right.ordinal)
271                .then_with(|| left.kind.cmp(&right.kind))
272                .then_with(|| left.content.cmp(&right.content))
273        });
274        let required_chars = canonical.len();
275        let mut used = required_chars;
276        let mut omitted_handles = Vec::new();
277        for activity in activities {
278            let entry = format!("{}:{}", activity.kind, activity.content);
279            if used + entry.len() > self.policy.max_chars {
280                if let Some(handle) = activity.handle {
281                    omitted_handles.push(handle);
282                }
283                continue;
284            }
285            used += entry.len();
286            push_message(
287                &mut messages,
288                &mut canonical,
289                &activity.kind,
290                &activity.content,
291            );
292            dependencies.push(format!("activity:{}:{}", activity.kind, activity.ordinal));
293        }
294        CompiledContext {
295            messages,
296            dependencies,
297            omitted_handles,
298            canonical,
299        }
300    }
301}

303/// Adds one canonical message to the compiled context.
304///
305/// # Parameters
306/// - `messages`: Message list receiving the compiled message.
307/// - `canonical`: Canonical identity text receiving the message.
308/// - `role`: Semantic message role.
309/// - `content`: Message content.
310fn push_message(
311    messages: &mut Vec<ModelMessage>,
312    canonical: &mut String,
313    role: &str,
314    content: &str,
315) {
316    let message = format!("[{role}] {content}");
317    messages.push(if role == "system" {
318        ModelMessage::system(message)
319    } else {
320        ModelMessage::user(message)
321    });
322    let _ = writeln!(canonical, "{role}:{content}");
323}

325/// Retains a tool catalog in a stable identity string.
326///
327/// # Parameters
328/// - `tools`: Trusted tool definitions to identify.
329///
330/// # Returns
331/// Deterministic catalog identity text.
332#[must_use]
333pub fn tool_catalog_identity(tools: &[ToolDefinition]) -> String {
334    tools
335        .iter()
336        .map(|tool| {
337            format!(
338                "{}:{}:{}",
339                tool.name,
340                tool.description,
341                serde_json::to_string(&tool.parameters).unwrap_or_default()
342            )
343        })
344        .collect::<Vec<_>>()
345        .join("\n")
346}

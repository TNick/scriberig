7/// One bounded semantic activity item available to a branch compiler.
8#[derive(Clone, Debug, Eq, PartialEq)]
9pub struct ContextActivity {
10    /// Deterministic branch-local ordering value.
11    pub ordinal: u64,
12    /// Semantic activity category.
13    pub kind: String,
14    /// Human/model-readable compact content.
15    pub content: String,
16    /// Stable handle that can reopen omitted detail.
17    pub handle: Option<String>,
18}
19
20impl ContextActivity {
21    /// Creates an activity without a reopenable handle.
22    ///
23    /// # Parameters
24    /// - `ordinal`: Stable activity order.
25    /// - `kind`: Activity category.
26    /// - `content`: Compact activity content.
27    ///
28    /// # Returns
29    /// A context activity item.
30    #[must_use]
31    pub fn new(ordinal: u64, kind: impl Into<String>, content: impl Into<String>) -> Self {
32        Self {
33            ordinal,
34            kind: kind.into(),
35            content: content.into(),
36            handle: None,
37        }
38    }
39
40    /// Attaches a stable handle for omitted detail.
41    ///
42    /// # Parameters
43    /// - `self`: Activity being configured.
44    /// - `handle`: Stable semantic handle.
45    ///
46    /// # Returns
47    /// The activity with its reopenable handle.
48    #[must_use]
49    pub fn with_handle(mut self, handle: impl Into<String>) -> Self {
50        self.handle = Some(handle.into());
51        self
52    }
53}

146/// Provider-neutral compiled context and exact dependency summary.
147#[derive(Clone, Debug, Eq, PartialEq)]
148pub struct CompiledContext {
149    /// Ordered messages supplied to a provider adapter.
150    pub messages: Vec<ModelMessage>,
151    /// Exact semantic dependency identities used by compilation.
152    pub dependencies: Vec<String>,
153    /// Handles for details omitted by the budget.
154    pub omitted_handles: Vec<String>,
155    /// Canonical representation used by request identity.
156    pub canonical: String,
157}

159/// Compiles branch context using deterministic priority and ordering rules.
160pub struct ContextCompiler {
161    /// Bounded selection policy.
162    policy: ContextPolicy,
163}
164
165impl ContextCompiler {
166    /// Creates a compiler with the supplied budget policy.
167    ///
168    /// # Parameters
169    /// - `policy`: Deterministic context selection policy.
170    ///
171    /// # Returns
172    /// A configured context compiler.
173    #[must_use]
174    pub const fn new(policy: ContextPolicy) -> Self {
175        Self { policy }
176    }
177
178    /// Compiles required branch inputs and bounded current-branch activity.
179    ///
180    /// # Parameters
181    /// - `context`: Explicit branch execution inputs.
182    ///
183    /// # Returns
184    /// Canonical provider-neutral context and dependency manifest.
185    #[must_use]
186    pub fn compile(&self, context: &BranchExecutionContext) -> CompiledContext {
187        let mut messages = Vec::new();
188        let mut dependencies = vec![
189            format!("session:{}", context.session),
190            format!("branch:{}", context.branch),
191        ];
192        let mut canonical = String::new();
193        push_message(
194            &mut messages,
195            &mut canonical,
196            "system",
197            "You are operating within one isolated semantic branch.",
198        );
199        push_message(&mut messages, &mut canonical, "purpose", &context.purpose);
200        if let Some(summary) = &context.parent_summary {
201            push_message(&mut messages, &mut canonical, "parent", summary);
202            dependencies.push(format!("parent-summary:{summary}"));
203        }
204        push_message(
205            &mut messages,
206            &mut canonical,
207            "instruction",
208            &context.instruction,
209        );
210        let mut activities = context.activities.clone();
211        activities.sort_by(|left, right| {
212            left.ordinal
213                .cmp(&right.ordinal)
214                .then_with(|| left.kind.cmp(&right.kind))
215                .then_with(|| left.content.cmp(&right.content))
216        });
217        let required_chars = canonical.len();
218        let mut used = required_chars;
219        let mut omitted_handles = Vec::new();
220        for activity in activities {
221            let entry = format!("{}:{}", activity.kind, activity.content);
222            if used + entry.len() > self.policy.max_chars {
223                if let Some(handle) = activity.handle {
224                    omitted_handles.push(handle);
225                }
226                continue;
227            }
228            used += entry.len();
229            push_message(
230                &mut messages,
231                &mut canonical,
232                &activity.kind,
233                &activity.content,
234            );
235            dependencies.push(format!("activity:{}:{}", activity.kind, activity.ordinal));
236        }
237        CompiledContext {
238            messages,
239            dependencies,
240            omitted_handles,
241            canonical,
242        }
243    }
244}

246/// Inputs that contribute to exact model request identity.
247#[derive(Clone, Debug, Eq, PartialEq)]
248pub struct RequestFingerprintInput {
249    /// Fingerprint schema version.
250    pub schema_version: String,
251    /// Provider profile or compatibility identity.
252    pub profile: String,
253    /// Provider model identity.
254    pub model: String,
255    /// Provider capability descriptor.
256    pub capabilities: ProviderCapabilities,
257    /// Canonical compiled context.
258    pub compiled_context: String,
259    /// Canonical tool catalog identity.
260    pub tools: String,
261    /// Secret deliberately excluded from identity material.
262    credential: Option<String>,
263}
264
265impl RequestFingerprintInput {
266    /// Creates a fingerprint input with default capability metadata.
267    ///
268    /// # Parameters
269    /// - `profile`: Provider profile identity.
270    /// - `model`: Provider model identity.
271    /// - `compiled_context`: Canonical context.
272    /// - `tools`: Canonical tool catalog identity.
273    ///
274    /// # Returns
275    /// Fingerprint inputs with secret-free identity material.
276    #[must_use]
277    pub fn new(
278        profile: impl Into<String>,
279        model: impl Into<String>,
280        compiled_context: impl Into<String>,
281        tools: impl Into<String>,
282    ) -> Self {
283        Self {
284            schema_version: "model-request-v1".to_owned(),
285            profile: profile.into(),
286            model: model.into(),
287            capabilities: ProviderCapabilities::default(),
288            compiled_context: compiled_context.into(),
289            tools: tools.into(),
290            credential: None,
291        }
292    }
293
294    /// Replaces the model identity for a fingerprint comparison.
295    ///
296    /// # Parameters
297    /// - `self`: Fingerprint input.
298    /// - `model`: New model identity.
299    ///
300    /// # Returns
301    /// Inputs using the new model identity.
302    #[must_use]
303    pub fn with_model(mut self, model: impl Into<String>) -> Self {
304        self.model = model.into();
305        self
306    }
307
308    /// Adds a secret only to prove it is excluded from identity material.
309    ///
310    /// # Parameters
311    /// - `self`: Fingerprint input.
312    /// - `credential`: Secret value never hashed or serialized.
313    ///
314    /// # Returns
315    /// Inputs retaining the secret outside canonical material.
316    #[must_use]
317    pub fn with_credential(mut self, credential: impl Into<String>) -> Self {
318        self.credential = Some(credential.into());
319        self
320    }
321
322    /// Computes the canonical content fingerprint.
323    ///
324    /// # Returns
325    /// Lowercase BLAKE3 digest over secret-free request identity material.
326    #[must_use]
327    pub fn fingerprint(&self) -> String {
328        let material = format!(
329            "{}\n{}\n{}\n{:?}\n{}\n{}",
330            self.schema_version,
331            self.profile,
332            self.model,
333            self.capabilities,
334            self.compiled_context,
335            self.tools
336        );
337        blake3::hash(material.as_bytes()).to_hex().to_string()
338    }
339}

341/// Adds one canonical message to the compiled context.
342///
343/// # Parameters
344/// - `messages`: Message list receiving the compiled message.
345/// - `canonical`: Canonical identity text receiving the message.
346/// - `role`: Semantic message role.
347/// - `content`: Message content.
348fn push_message(
349    messages: &mut Vec<ModelMessage>,
350    canonical: &mut String,
351    role: &str,
352    content: &str,
353) {
354    let message = format!("[{role}] {content}");
355    messages.push(if role == "system" {
356        ModelMessage::system(message)
357    } else {
358        ModelMessage::user(message)
359    });
360    let _ = writeln!(canonical, "{role}:{content}");
361}

363/// Retains a tool catalog in a stable identity string.
364///
365/// # Parameters
366/// - `tools`: Trusted tool definitions to identify.
367///
368/// # Returns
369/// Deterministic catalog identity text.
370#[must_use]
371pub fn tool_catalog_identity(tools: &[ToolDefinition]) -> String {
372    tools
373        .iter()
374        .map(|tool| {
375            format!(
376                "{}:{}:{}",
377                tool.name,
378                tool.description,
379                serde_json::to_string(&tool.parameters).unwrap_or_default()
380            )
381        })
382        .collect::<Vec<_>>()
383        .join("\n")
384}

12/// Resolves a protected credential for one already-resolved upstream model.
13///
14/// Upstream requests the host-selected source through its protected resolver.
15/// The request-scoped host resolver remains bound to the exact model endpoint,
16/// and authorization or source errors stop resolution rather than becoming a
17/// credential miss.
18///
19/// # Parameters
20/// - `credentials`: `ScribeRig` credential authority for this invocation.
21/// - `model`: Exact upstream model selected for the provider request.
22///
23/// # Returns
24/// An opaque upstream request secret, or `None` when the authority explicitly
25/// reports that no authentication is configured.
26///
27/// # Errors
28/// Returns the safe classified credential error when authorization or lookup
29/// fails.
30pub fn request_secret_for_model(
31    credentials: &dyn CredentialProvider,
32    model: &Model,
33) -> Result<Option<RequestSecret>, SecretError> {
34    let resolver = authorized_resolver(credentials, model);
35    let protected = resolve_host_selected_credential_with_policy(false, &resolver)
36        .map_err(map_upstream_resolution_error)?;
37    let Some(secret) = protected.material else {
38        return Ok(None);
39    };
40    let material = SecretValueMaterial(secret);
41    Ok(Some(RequestSecret::new(Arc::new(material))))
42}

44/// Resolves a safe upstream credential-source classification for readiness.
45///
46/// This uses the same request-scoped host resolver as provider execution and
47/// drops the protected material before returning the classification.
48///
49/// # Parameters
50/// - `credentials`: `ScribeRig` credential authority for this invocation.
51/// - `model`: Exact upstream model selected for the provider request.
52///
53/// # Returns
54/// Upstream's host-selected, catalog-header, or missing credential source.
55///
56/// # Errors
57/// Returns the safe classified credential error when authorization or lookup
58/// fails.
59pub fn credential_source_for_model(
60    credentials: &dyn CredentialProvider,
61    model: &Model,
62) -> Result<CredentialSource, SecretError> {
63    let resolver = authorized_resolver(credentials, model);
64    let protected = resolve_host_selected_credential_with_policy(false, &resolver)
65        .map_err(map_upstream_resolution_error)?;
66    Ok(rpi_model_runtime::readiness_from_catalog(
67        &model.provider,
68        protected.source,
69        std::slice::from_ref(model),
70    ))
71}

73/// Binds `ScribeRig` authorization to the exact selected upstream endpoint.
74///
75/// # Parameters
76/// - `credentials`: Authorized `ScribeRig` credential source.
77/// - `model`: Exact upstream model used to identify the destination endpoint.
78///
79/// # Returns
80/// A resolver bound to the selected model endpoint.
81fn authorized_resolver<'a>(
82    credentials: &'a dyn CredentialProvider,
83    model: &Model,
84) -> AuthorizedSecretResolver<'a> {
85    AuthorizedSecretResolver {
86        credentials,
87        endpoint: model.base_url.clone(),
88    }
89}

91/// Resolves only the host-selected source for one authorized destination.
92///
93/// # Public members
94/// None.
95///
96/// # Private members
97/// - `credentials`: `ScribeRig`'s destination-authorizing credential provider.
98/// - `endpoint`: Exact upstream model endpoint for this request.
99struct AuthorizedSecretResolver<'a> {
100    /// `ScribeRig` authority that rechecks destination policy before reading.
101    credentials: &'a dyn CredentialProvider,
102    /// Exact endpoint copied from the selected upstream model.
103    endpoint: String,
104}
105
106impl ProtectedCredentialResolver<SecretValue> for AuthorizedSecretResolver<'_> {
107    /// Resolves a legacy source request as a safe miss.
108    ///
109    /// The selected host credential is available only through the error-bearing
110    /// `resolve_with_policy` method.
111    ///
112    /// # Parameters
113    /// - `_source`: Non-secret source descriptor requested by upstream.
114    ///
115    /// # Returns
116    /// No legacy source material.
117    fn resolve(
118        &self,
119        _source: CredentialRequestSource,
120    ) -> Option<HostCredentialMaterial<SecretValue>> {
121        None
122    }
123
124    /// Resolves the single host-selected reference with fail-closed errors.
125    ///
126    /// # Parameters
127    /// - `source`: Non-secret source descriptor requested by upstream.
128    ///
129    /// # Returns
130    /// The authorized secret material, or a miss when no credential is set.
131    ///
132    /// # Errors
133    /// Returns a safe denial or resolver failure without source fallthrough.
134    fn resolve_with_policy(
135        &self,
136        source: CredentialRequestSource,
137    ) -> Result<Option<HostCredentialMaterial<SecretValue>>, CredentialResolutionError> {
138        if source != CredentialRequestSource::HostSelected {
139            return Ok(None);
140        }
141        let secret = self
142            .credentials
143            .credential(&self.endpoint)
144            .map_err(map_resolution_error)?;
145        Ok(secret.map(|material| HostCredentialMaterial {
146            material,
147            source: CredentialSource::HostSelected,
148        }))
149    }
150}

152/// Maps host errors into upstream's non-secret fail-closed classifications.
153///
154/// # Parameters
155/// - `error`: Safe `ScribeRig` credential error.
156///
157/// # Returns
158/// An upstream denial or generic resolver failure.
159fn map_resolution_error(error: SecretError) -> CredentialResolutionError {
160    match error {
161        SecretError::Denied | SecretError::Invalid => CredentialResolutionError::AccessDenied,
162        SecretError::Missing
163        | SecretError::Locked
164        | SecretError::Unavailable
165        | SecretError::Unsupported
166        | SecretError::Conflict
167        | SecretError::Partial => CredentialResolutionError::ResolverFailure,
168    }
169}

171/// Restores a safe `ScribeRig` error from upstream's non-secret failure class.
172///
173/// # Parameters
174/// - `error`: Upstream credential resolution failure.
175///
176/// # Returns
177/// A safe host denial or availability failure.
178fn map_upstream_resolution_error(error: CredentialResolutionError) -> SecretError {
179    match error {
180        CredentialResolutionError::AccessDenied => SecretError::Denied,
181        _ => SecretError::Unavailable,
182    }
183}

185/// Keeps a `ScribeRig` zeroizing secret behind upstream's synchronous secret API.
186///
187/// # Public members
188/// None.
189///
190/// # Private members
191/// - `0`: Zeroizing `ScribeRig` credential buffer.
192struct SecretValueMaterial(SecretValue);
193
194impl SecretMaterial for SecretValueMaterial {
195    /// Borrows the secret only for the duration of the request consumer.
196    ///
197    /// # Parameters
198    /// - `consumer`: Synchronous upstream request operation receiving the
199    ///   borrowed credential.
200    fn with_secret(&self, consumer: &mut dyn FnMut(&str)) {
201        consumer(self.0.expose());
202    }

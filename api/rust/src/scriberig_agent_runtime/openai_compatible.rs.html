23/// OpenAI-compatible chat-completions provider.
24#[derive(Clone)]
25pub struct OpenAiCompatibleProvider {
26    /// Runtime provider configuration.
27    config: ProviderConfig,
28    /// HTTP client with bounded connection behavior.
29    client: reqwest::blocking::Client,
30    /// Completion transport used by the normalized request path.
31    transport: Arc<dyn CompletionTransport>,
32    /// Declared capabilities for this compatibility adapter.
33    capabilities: ProviderCapabilities,
34    /// Adapter-local translation from public level ids to wire fields.
35    thinking_parameters:
36        std::collections::BTreeMap<String, std::collections::BTreeMap<String, serde_json::Value>>,
37    /// Model-specific translation from public levels to wire fields.
38    model_thinking_parameters: std::collections::BTreeMap<
39        (String, String),
40        std::collections::BTreeMap<String, serde_json::Value>,
41    >,
42}

44impl OpenAiCompatibleProvider {
45    /// Creates an adapter with the configured endpoint and timeouts.
46    ///
47    /// # Parameters
48    /// - `config`: Provider configuration without an embedded secret.
49    ///
50    /// # Returns
51    /// A configured OpenAI-compatible adapter.
52    ///
53    /// # Errors
54    /// Returns a configuration error when the HTTP client cannot be built.
55    pub fn new(config: ProviderConfig) -> Result<Self, ProviderError> {
56        scriberig_secrets::validate_public_endpoint(&config.base_url).map_err(|_| {
57            ProviderError::Configuration("invalid provider endpoint descriptor".to_owned())
58        })?;
59        let client = reqwest::blocking::Client::builder()
60            .redirect(reqwest::redirect::Policy::none())
61            .connect_timeout(config.connect_timeout)
62            .no_proxy()
63            .http1_only()
64            .user_agent("scriberig-agent-runtime/0.1")
65            .build()
66            .map_err(|_| {
67                ProviderError::Configuration("provider client could not be initialized".to_owned())
68            })?;
69        let transport = Arc::new(ReqwestCompletionTransport::new(client.clone()));
70        Ok(Self {
71            config,
72            client,
73            transport,
74            capabilities: ProviderCapabilities {
75                tool_calls: true,
76                streaming: true,
77                usage_reporting: true,
78                provider_request_id: true,
79                provider_model_fingerprint: true,
80                ..ProviderCapabilities::default()
81            },
82            thinking_parameters: std::collections::BTreeMap::new(),
83            model_thinking_parameters: std::collections::BTreeMap::new(),
84        })
85    }
86
87    /// Configures this adapter's translation for one advertised thinking id.
88    ///
89    /// # Parameters
90    /// - `self`: OpenAI-compatible provider adapter.
91    /// - `level`: Provider-neutral thinking capability id.
92    /// - `parameters`: Provider-specific request fields for this adapter.
93    ///
94    /// # Returns
95    /// The adapter with the translation installed.
96    #[must_use]
97    pub fn with_thinking_parameters(
98        mut self,
99        level: impl Into<String>,
100        parameters: std::collections::BTreeMap<String, serde_json::Value>,
101    ) -> Self {
102        self.thinking_parameters.insert(level.into(), parameters);
103        self
104    }
105
106    /// Configures a model-specific thinking-level translation.
107    ///
108    /// # Parameters
109    /// - `self`: OpenAI-compatible provider adapter.
110    /// - `model`: Provider-native model identifier.
111    /// - `level`: Model-advertised thinking capability id.
112    /// - `parameters`: Provider-specific request fields for this model level.
113    ///
114    /// # Returns
115    /// The adapter with the model-specific translation installed.
116    #[must_use]
117    pub fn with_model_thinking_parameters(
118        mut self,
119        model: impl Into<String>,
120        level: impl Into<String>,
121        parameters: std::collections::BTreeMap<String, serde_json::Value>,
122    ) -> Self {
123        self.model_thinking_parameters
124            .insert((model.into(), level.into()), parameters);
125        self
126    }
127
128    /// Replaces the completion transport for deterministic adapter tests.
129    ///
130    /// # Parameters
131    /// - `config`: Provider configuration without an embedded secret.
132    /// - `transport`: In-process completion transport fixture.
133    ///
134    /// # Returns
135    /// A configured provider using the supplied transport.
136    ///
137    /// # Errors
138    /// Returns a configuration error when the backing HTTP client cannot be built.
139    #[cfg(test)]
140    fn with_transport(
141        config: ProviderConfig,
142        transport: Arc<dyn CompletionTransport>,
143    ) -> Result<Self, ProviderError> {
144        let mut provider = Self::new(config)?;
145        provider.transport = transport;
146        Ok(provider)
147    }
148
149    /// Returns the adapter's declared model capabilities.
150    ///
151    /// # Returns
152    /// A copy of the capability descriptor.
153    #[must_use]
154    pub const fn capabilities(&self) -> ProviderCapabilities {
155        self.capabilities
156    }
157
158    /// Returns the configured endpoint with the chat-completions suffix.
159    ///
160    /// # Returns
161    /// The endpoint URL used for requests.
162    fn endpoint(&self) -> String {
163        format!(
164            "{}/chat/completions",
165            self.config.base_url.trim_end_matches('/')
166        )
167    }

170impl ProviderAdapter for OpenAiCompatibleProvider {
171    /// Sends a normalized request and parses one normalized response.
172    ///
173    /// # Parameters
174    /// - `request`: Provider-neutral model request.
175    ///
176    /// # Returns
177    /// A provider-neutral model response.
178    ///
179    /// # Errors
180    /// Returns a classified transport, HTTP, configuration, or parse failure.
181    fn complete(&self, request: &ModelRequest) -> Result<ModelResponse, ProviderError> {
182        let credential = self.config.resolve_credential()?;
183        let payload = WireRequest::from_request_for_models(
184            request,
185            false,
186            &self.thinking_parameters,
187            &self.model_thinking_parameters,
188        );
189        let response = self.transport.send(
190            &self.endpoint(),
191            &payload,
192            credential.as_ref().map(ResolvedCredential::as_str),
193            self.config.request_timeout,
194        )?;
195        let request_id = response.request_id.filter(|id| {
196            credential
197                .as_ref()
198                .is_none_or(|secret| !id.contains(secret.as_str()))
199        });
200        let status = response.status;
201        if !status.is_success() {
202            return Err(classify_status(status, request_id));
203        }
204        let wire: WireResponse = serde_json::from_slice(&response.body).map_err(|_| {
205            ProviderError::MalformedResponse("invalid provider response".to_owned())
206        })?;
207        let mut normalized = wire.into_response(request_id)?;
208        crate::credential_redaction::redact_response(
209            &mut normalized,
210            credential.as_ref().map(ResolvedCredential::as_str),
211        );
212        Ok(normalized)
213    }
214
215    /// Completes through SSE while forwarding assistant text fragments.
216    ///
217    /// # Parameters
218    /// - `request`: Provider-neutral request to execute.
219    /// - `on_text`: Callback receiving provider-neutral text fragments.
220    ///
221    /// # Returns
222    /// The assembled normalized response.
223    ///
224    /// # Errors
225    /// Returns the classified provider failure.
226    fn complete_streaming(
227        &self,
228        request: &ModelRequest,
229        on_text: &mut dyn FnMut(&str),
230    ) -> Result<ModelResponse, ProviderError> {
231        self.complete_streaming_with(request, on_text)
232    }
233}

235impl WireResponse {
236    /// Normalizes a compatible response without exposing wire DTOs.
237    ///
238    /// # Parameters
239    /// - `self`: Wire response to normalize.
240    /// - `request_id`: Safe provider request identifier.
241    ///
242    /// # Returns
243    /// Normalized provider response.
244    ///
245    /// # Errors
246    /// Returns malformed-response when choices or arguments are invalid.
247    fn into_response(self, request_id: Option<String>) -> Result<ModelResponse, ProviderError> {
248        let choice = self.choices.into_iter().next().ok_or_else(|| {
249            ProviderError::MalformedResponse("response has no choices".to_owned())
250        })?;
251        let tool_calls = choice
252            .message
253            .tool_calls
254            .unwrap_or_default()
255            .into_iter()
256            .map(|call| {
257                let arguments = serde_json::from_str(&call.function.arguments).map_err(|_| {
258                    ProviderError::MalformedResponse("tool call arguments are not JSON".to_owned())
259                })?;
260                Ok(ToolCall {
261                    id: call.id,
262                    name: call.function.name,
263                    arguments,
264                })
265            })
266            .collect::<Result<Vec<_>, ProviderError>>()?;
267        let finish_reason = match choice.finish_reason.as_deref() {
268            Some("tool_calls") => FinishReason::ToolCall,
269            Some("length") => FinishReason::Length,
270            Some("content_filter") => FinishReason::Refusal,
271            _ => FinishReason::Stop,
272        };
273        Ok(ModelResponse {
274            text: choice.message.content,
275            tool_calls,
276            finish_reason,
277            usage: self.usage.map(|usage| Usage {
278                prompt_tokens: usage.input,
279                completion_tokens: usage.output,
280                total_tokens: usage.total,
281            }),
282            metadata: ProviderMetadata {
283                request_id,
284                model_fingerprint: self.model,
285            },
286        })
287    }

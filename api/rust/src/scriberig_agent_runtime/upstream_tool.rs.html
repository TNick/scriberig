19const MAX_TOOL_RESULT_BYTES: usize = 16_384;

21/// Future returned by a `ScribeRig` semantic tool executor.
22pub type SemanticToolFuture =
23    Pin<Box<dyn Future<Output = Result<rpi_agent::AgentToolResult, UpstreamAgentError>> + Send>>;

25/// Executes one pre-authorized `ScribeRig` semantic tool through `rpi-agent`.
26///
27/// Tool exposure and capability filtering remain `ScribeRig`-owned. Construct
28/// instances only from the branch-visible tool catalog.
29///
30/// # Public members
31/// None.
32///
33/// # Private members
34/// - `schema`: Upstream provider-facing tool schema.
35/// - `label`: Safe label used by upstream tool events and presentation.
36/// - `branch`: Canonical `ScribeRig` branch that owns this tool adapter.
37/// - `catalog`: Trusted catalog used for final invocation authorization.
38/// - `capabilities`: Capability grants snapshotted for this branch.
39/// - `execute`: `ScribeRig` semantic operation executor.
40pub struct SemanticAgentTool {
41    schema: Tool,
42    label: String,
43    branch: EntityId,
44    catalog: ToolCatalog,
45    capabilities: CapabilityContext,
46    execute: Arc<SemanticToolExecutor>,
47}

49/// Callback that runs a validated semantic invocation.

52        serde_json::Value,
53        CancellationToken,

57    + Sync;

59impl SemanticAgentTool {
60    /// Creates an upstream tool adapter from a `ScribeRig`-approved definition.
61    ///
62    /// The definition must be visible in the supplied branch catalog and
63    /// capability snapshot. The executor must still authorize the concrete
64    /// application operation before performing it. It must return only a safe,
65    /// redacted semantic projection; this adapter bounds text and removes
66    /// structured details but cannot inspect operation-specific secrets.
67    /// For branch-producing tools, it must persist the `ScribeRig` branch result
68    /// and return a compact projection containing its stable `ScribeRig` handle.
69    ///
70    /// # Parameters
71    /// - `definition`: Provider-facing schema approved for this branch.
72    /// - `label`: Bounded display label for tool activity.
73    /// - `branch`: Canonical `ScribeRig` branch that owns this adapter.
74    /// - `catalog`: Trusted catalog containing the definition.
75    /// - `capabilities`: Capability grants for this branch.
76    /// - `execute`: `ScribeRig` operation callback.
77    ///
78    /// # Returns
79    /// An adapter implementing the upstream `AgentTool` contract.
80    ///
81    /// # Errors
82    /// Returns an error when the tool definition is not visible to the branch.
83    pub fn new<F, Fut>(
84        definition: &Tool,
85        label: impl Into<String>,
86        branch: EntityId,
87        catalog: ToolCatalog,
88        capabilities: CapabilityContext,
89        execute: F,
90    ) -> Result<Self, String>
91    where
92        F: Fn(
93                String,
94                serde_json::Value,
95                CancellationToken,
96                Arc<dyn Fn(rpi_agent::ToolResultPartial) + Send + Sync>,
97            ) -> Fut
98            + Send
99            + Sync
100            + 'static,
101        Fut: Future<Output = Result<rpi_agent::AgentToolResult, UpstreamAgentError>>
102            + Send
103            + 'static,
104    {
105        if !catalog
106            .descriptors_for(&capabilities)
107            .iter()
108            .any(|descriptor| descriptor.definition == *definition)
109        {
110            return Err("tool is not visible to this branch".to_owned());
111        }
112
113        Ok(Self {
114            schema: definition.clone(),
115            label: bounded_label(&label.into()),
116            branch,
117            catalog,
118            capabilities,
119            execute: Arc::new(move |id, params, signal, on_update| {
120                Box::pin(execute(id, params, signal, on_update))
121            }),
122        })
123    }
124
125    /// Returns the canonical `ScribeRig` branch owning this adapter.
126    ///
127    /// # Returns
128    /// The `ScribeRig` branch ID supplied at construction.
129    pub(crate) fn branch(&self) -> EntityId {
130        self.branch
131    }
132}

134#[async_trait]
135impl AgentTool for SemanticAgentTool {
136    /// Returns the provider-facing schema selected by `ScribeRig`.
137    ///
138    /// # Returns
139    /// The selected provider-facing schema.
140    fn schema(&self) -> &Tool {
141        &self.schema
142    }
143
144    /// Returns the safe activity label supplied by `ScribeRig`.
145    ///
146    /// # Returns
147    /// The safe activity label.
148    fn label(&self) -> &str {
149        &self.label
150    }
151
152    /// Serializes semantic operations until branch execution is concurrency-safe.
153    ///
154    /// # Returns
155    /// The sequential tool execution mode.
156    fn execution_mode(&self) -> ToolExecutionMode {
157        ToolExecutionMode::Sequential
158    }
159
160    /// Authorizes and executes the callback after upstream schema validation.
161    ///
162    /// # Parameters
163    /// - `tool_call_id`: Upstream correlation identity for this invocation.
164    /// - `params`: Arguments validated against `schema` by `rpi-agent`.
165    /// - `signal`: Cancellation token scoped to this tool execution.
166    /// - `on_update`: Callback for bounded partial semantic results.
167    ///
168    /// # Returns
169    /// A bounded provider-visible projection of the semantic tool result.
170    ///
171    /// # Errors
172    /// Returns the semantic executor's safe failure.
173    async fn execute(
174        &self,
175        tool_call_id: &str,
176        params: serde_json::Value,
177        signal: CancellationToken,
178        on_update: Arc<dyn Fn(rpi_agent::ToolResultPartial) + Send + Sync>,
179    ) -> Result<rpi_agent::AgentToolResult, UpstreamAgentError> {
180        self.catalog
181            .authorize_for_context(&self.schema.name, &self.capabilities)
182            .map_err(|_| UpstreamAgentError::tool("tool is not authorized for this branch"))?;
183
184        let on_update = Arc::new(move |update| on_update(bound_tool_result(update)));
185        let result = (self.execute)(tool_call_id.to_owned(), params, signal, on_update)
186            .await
187            .map_err(|error| {
188                if error.is_abort() {
189                    error
190                } else {
191                    UpstreamAgentError::tool("semantic tool execution failed")
192                }
193            })?;
194        Ok(bound_tool_result(result))
195    }
196}

198/// Bounds tool labels before they enter upstream activity events.
199///
200/// # Parameters
201/// - `label`: Trusted activity label supplied by the host.
202///
203/// # Returns
204/// A control-free label of at most 128 Unicode scalar values.
205fn bounded_label(label: &str) -> String {
206    label
207        .chars()
208        .filter(|character| !character.is_control())
209        .take(128)
210        .collect()
211}

213/// Removes non-provider details and bounds the model-visible tool result.
214///
215/// Images, structured details, and dynamic tool additions are excluded so a
216/// semantic executor cannot enlarge the branch's advertised tool set or
217/// return an unbounded payload through upstream agent state.
218///
219/// # Parameters
220/// - `result`: Semantic executor result to project.
221///
222/// # Returns
223/// A text-only result bounded to `MAX_TOOL_RESULT_BYTES`.
224///
225/// # Errors
226/// This projection cannot fail.
227fn bound_tool_result(mut result: AgentToolResult) -> AgentToolResult {
228    let mut text = String::with_capacity(MAX_TOOL_RESULT_BYTES);
229    'blocks: for content in result.content {
230        let TextContentOrImage::Text(content) = content else {
231            continue;
232        };
233        for character in content.text.chars() {
234            let width = character.len_utf8();
235            if text.len() + width > MAX_TOOL_RESULT_BYTES {
236                break 'blocks;
237            }
238            text.push(character);
239        }
240    }
241    result.content = vec![TextContentOrImage::text(text)];
242    result.details = serde_json::Value::Null;
243    result.added_tool_names.clear();
244    result
245}

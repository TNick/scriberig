17    /// # Parameters

34impl SnapshotStore for SqliteStore {
35    /// - `snapshot`: Operation input.
36    /// - `entity`: Operation input.
37    ///
38    /// The requested storage result.
39    ///
40    /// Returns a backend error when SQLite rejects the operation.
41    /// Resolves an entity in a stored snapshot.
42    /// # Returns
43    /// The operation result.
44    /// # Errors
45    /// Returns an error when the operation fails.
46    fn resolve(
47        &self,
48        snapshot: SnapshotId,
49        entity: EntityId,
50    ) -> Result<Option<RevisionId>, SnapshotError> {
51        let connection = self
52            .connection()
53            .map_err(|error| SnapshotError::Backend(format!("{error:?}")))?;
54        if snapshot == scriberig_store::Snapshot::empty().id() {
55            return Ok(None);
56        }
57        validate_snapshot_root(&connection, snapshot)?;
58        let revision: Option<Vec<u8>> = connection
59            .query_row(
60                "SELECT revision FROM snapshot_entries WHERE snapshot_id=? AND entity=?",
61                params![snapshot.as_bytes().as_slice(), entity.as_bytes().as_slice()],
62                |row| row.get(0),
63            )
64            .optional()
65            .map_err(|error| SnapshotError::Backend(format!("{error:?}")))?;
66        if let Some(revision) = revision {
67            let bytes: [u8; 32] = revision.try_into().map_err(|_| {
68                SnapshotError::Backend("invalid stored revision identifier length".to_owned())
69            })?;
70            return Ok(Some(RevisionId::new(bytes)));
71        }
72        let exists: bool = connection
73            .query_row(
74                "SELECT EXISTS(SELECT 1 FROM snapshots WHERE id=?)",
75                params![snapshot.as_bytes().as_slice()],
76                |row| row.get(0),
77            )
78            .map_err(|error| SnapshotError::Backend(format!("{error:?}")))?;
79        if exists {
80            Ok(None)
81        } else {
82            Err(SnapshotError::UnknownSnapshot(snapshot))
83        }
84    }
85    /// - `base`: Operation input.
86    /// - `changes`: Operation input.
87    ///
88    /// The requested storage result.
89    ///
90    /// Returns a backend error when SQLite rejects the operation.
91    /// Applies changes and persists the resulting immutable snapshot.
92    /// # Returns
93    /// The operation result.
94    /// # Errors
95    /// Returns an error when the operation fails.
96    fn apply_changes(
97        &self,
98        base: SnapshotId,
99        changes: &[SnapshotChange],
100    ) -> Result<SnapshotId, SnapshotError> {
101        let next = self.snapshot(base)?.apply(changes.iter().copied());
102        self.store_snapshot_with_parent(&next, base)?;
103        Ok(next.id())
104    }
105    /// - `snapshot`: Operation input.
106    ///
107    /// The requested storage result.
108    ///
109    /// Returns a backend error when SQLite rejects the operation.
110    /// Iterates a stored snapshot in deterministic order.
111    /// # Returns
112    /// The operation result.
113    /// # Errors
114    /// Returns an error when the operation fails.
115    fn iter(&self, snapshot: SnapshotId) -> Result<Vec<(EntityId, RevisionId)>, SnapshotError> {
116        Ok(self.snapshot(snapshot)?.iter().collect())
117    }
118    /// - `from`: Operation input.
119    /// - `to`: Operation input.
120    ///
121    /// The requested storage result.
122    ///
123    /// Returns a backend error when SQLite rejects the operation.
124    /// Computes exact mapping changes between stored snapshots.
125    /// # Returns
126    /// The operation result.
127    /// # Errors
128    /// Returns an error when the operation fails.
129    fn diff_snapshots(
130        &self,
131        from: SnapshotId,
132        to: SnapshotId,
133    ) -> Result<Vec<SnapshotChange>, SnapshotError> {
134        Ok(self.snapshot(from)?.diff_changes(&self.snapshot(to)?))
135    }

137    /// Computes semantic revision transitions from one read of each snapshot.
138    ///
139    /// - `from`: Starting snapshot identity.
140    /// - `to`: Ending snapshot identity.
141    ///
142    /// Deterministically ordered additions, removals, and revision changes.
143    ///
144    /// Returns an error when either snapshot is unavailable or unreadable.
145    /// # Returns
146    /// The operation result.
147    /// # Errors
148    /// Returns an error when the operation fails.
149    fn diff_snapshot_revisions(
150        &self,
151        from: SnapshotId,
152        to: SnapshotId,
153    ) -> Result<Vec<SnapshotDiff>, SnapshotError> {
154        Ok(self.snapshot(from)?.diff(&self.snapshot(to)?))
155    }

157    /// Returns direct immutable predecessors retained for one snapshot.
158    ///
159    /// - `snapshot`: Snapshot whose direct parents are requested.
160    ///
161    /// Direct parents in deterministic order.
162    ///
163    /// Returns an error when the snapshot or parent records cannot be read.
164    /// # Returns
165    /// The operation result.
166    /// # Errors
167    /// Returns an error when the operation fails.
168    fn parents(&self, snapshot: SnapshotId) -> Result<Vec<SnapshotId>, SnapshotError> {
169        if snapshot == scriberig_store::Snapshot::empty().id() {
170            return Ok(Vec::new());
171        }
172        let connection = self
173            .connection()
174            .map_err(|error| SnapshotError::Backend(format!("{error:?}")))?;
175        let exists: bool = connection
176            .query_row(
177                "SELECT EXISTS(SELECT 1 FROM snapshots WHERE id=?)",
178                params![snapshot.as_bytes().as_slice()],
179                |row| row.get(0),
180            )
181            .map_err(|error| SnapshotError::Backend(error.to_string()))?;
182        if !exists {
183            return Err(SnapshotError::UnknownSnapshot(snapshot));
184        }
185        let mut statement = connection
186            .prepare(
187                "SELECT parent_id FROM snapshot_parents
188                 WHERE snapshot_id=? ORDER BY parent_id",
189            )
190            .map_err(|error| SnapshotError::Backend(error.to_string()))?;
191        let rows = statement
192            .query_map(params![snapshot.as_bytes().as_slice()], |row| {
193                let bytes: Vec<u8> = row.get(0)?;
194                if bytes.len() != 32 {
195                    return Err(rusqlite::Error::InvalidQuery);
196                }
197                let mut value = [0; 32];
198                value.copy_from_slice(&bytes);
199                Ok(SnapshotId::from_bytes(value))
200            })
201            .map_err(|error| SnapshotError::Backend(error.to_string()))?;
202        rows.map(|row| row.map_err(|error| SnapshotError::Backend(error.to_string())))
203            .collect()
204    }

206    /// Reads the durable accepted change associated with an exact result snapshot.
207    ///
208    /// - `snapshot`: Exact immutable result snapshot.
209    ///
210    /// The accepted project source-change identity when one produced this snapshot.
211    ///
212    /// Returns a backend error when the receipt cannot be read.
213    /// # Returns
214    /// The operation result.
215    /// # Errors
216    /// Returns an error when the operation fails.
217    fn change_provenance(
218        &self,
219        snapshot: SnapshotId,
220    ) -> Result<Option<SnapshotChangeProvenance>, SnapshotError> {
221        let connection = self
222            .connection()
223            .map_err(|error| SnapshotError::Backend(format!("{error:?}")))?;
224        let change_id = connection
225            .query_row(
226                "SELECT candidate_id FROM project_effects
227                 WHERE successor_snapshot=? AND state='applied' ORDER BY candidate_id LIMIT 1",
228                params![snapshot.as_bytes().as_slice()],
229                |row| row.get::<_, String>(0),
230            )
231            .optional()
232            .map_err(|error| SnapshotError::Backend(error.to_string()))?;
233        Ok(change_id.map(|change_id| SnapshotChangeProvenance { change_id }))
234    }

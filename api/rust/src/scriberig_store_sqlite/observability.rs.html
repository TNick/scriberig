1//! Implements SQLite storage observability and derived-index maintenance.
2
3use scriberig_store::{IndexStatus, RootStatus, StoreObservability, StoreStats};
4
5use crate::SqliteStore;
6
7impl StoreObservability for SqliteStore {
8    /// Returns authoritative row counts from SQLite.
9    ///
10    /// # Returns
11    /// Available SQLite storage statistics.
12    ///
13    /// # Errors
14    /// Returns a backend error when statistics cannot be read.
15    fn stats(&self) -> Result<StoreStats, String> {
16        let connection = self
17            .connection
18            .lock()
19            .map_err(|_| "lock poisoned".to_owned())?;
20        let count = |table: &str| {
21            connection
22                .query_row(&format!("SELECT COUNT(*) FROM {table}"), [], |row| {
23                    row.get::<_, i64>(0).and_then(|value| {
24                        u64::try_from(value).map_err(|_| rusqlite::Error::InvalidQuery)
25                    })
26                })
27                .map_err(|error| error.to_string())
28        };
29        Ok(StoreStats {
30            revisions: Some(count("revisions")?),
31            snapshots: Some(count("snapshots")?),
32            named_roots: Some(count("roots")?),
33            ..StoreStats::default()
34        })
35    }
36
37    /// Lists SQLite roots in deterministic name order.
38    ///
39    /// # Returns
40    /// Root status records ordered by root name.
41    ///
42    /// # Errors
43    /// Returns a backend error when roots cannot be read.
44    fn root_statuses(&self) -> Result<Vec<RootStatus>, String> {
45        let connection = self
46            .connection
47            .lock()
48            .map_err(|_| "lock poisoned".to_owned())?;
49        let mut statement = connection
50            .prepare("SELECT name, snapshot_id FROM roots ORDER BY name")
51            .map_err(|error| error.to_string())?;
52        let rows = statement
53            .query_map([], |row| {
54                let name: String = row.get(0)?;
55                let bytes: Vec<u8> = row.get(1)?;
56                Ok((name, bytes))
57            })
58            .map_err(|error| error.to_string())?;
59        let mut statuses = Vec::new();
60        for row in rows {
61            let (name, bytes) = row.map_err(|error| error.to_string())?;
62            if bytes.len() != 32 {
63                return Err("invalid root snapshot identifier".to_owned());
64            }
65            let mut id = [0; 32];
66            id.copy_from_slice(&bytes);
67            statuses.push(RootStatus {
68                name: scriberig_core::RootName::new(name).map_err(|error| format!("{error:?}"))?,
69                snapshot: Some(scriberig_core::SnapshotId::new(id)),
70            });
71        }
72        Ok(statuses)
73    }
74
75    /// Rebuilds SQLite derived indexes.
76    ///
77    /// # Returns
78    /// A consistent index status after rebuilding.
79    ///
80    /// # Errors
81    /// Returns a backend error when rebuilding fails.
82    fn rebuild_indexes(&self) -> Result<IndexStatus, String> {
83        self.rebuild_derived_indexes()
84            .map_err(|error| error.to_string())?;
85        Ok(IndexStatus { consistent: true })
86    }
87}

1//! Implements SQLite storage observability and derived-index maintenance.
2
3use scriberig_store::{
4    IndexStatus, ProtectedRoots, RootStatus, StoreObservability, StoreOperationError, StoreStats,
5};

7use crate::SqliteStore;

9impl StoreObservability for SqliteStore {
10    /// Returns authoritative row counts from SQLite.
11    ///
12    /// # Returns
13    /// Available SQLite storage statistics.
14    ///
15    /// # Errors
16    /// Returns a backend error when statistics cannot be read.
17    fn stats(&self) -> Result<StoreStats, StoreOperationError> {
18        let connection = self
19            .connection
20            .lock()
21            .map_err(|_| StoreOperationError::backend("lock poisoned"))?;
22        let count = |table: &str| {
23            connection
24                .query_row(&format!("SELECT COUNT(*) FROM {table}"), [], |row| {
25                    row.get::<_, i64>(0).and_then(|value| {
26                        u64::try_from(value).map_err(|_| rusqlite::Error::InvalidQuery)
27                    })
28                })
29                .map_err(|error| StoreOperationError::backend(error.to_string()))
30        };
31        Ok(StoreStats {
32            revisions: Some(count("revisions")?),
33            snapshots: Some(count("snapshots")?),
34            named_roots: Some(count("roots")?),
35            ..StoreStats::default()
36        })
37    }
38
39    /// Returns counts annotated with SQLite reachability.
40    ///
41    /// # Parameters
42    /// - `protected`: Snapshot roots used for reachability counts.
43    ///
44    /// # Returns
45    /// Authoritative counts and marked-object counts.
46    ///
47    /// # Errors
48    /// Returns a backend error when integrity inspection cannot complete.
49    fn stats_for(&self, protected: &ProtectedRoots) -> Result<StoreStats, StoreOperationError> {
50        let mut stats = self.stats()?;
51        let report = self
52            .integrity_check(protected)
53            .map_err(|error| StoreOperationError::backend(error.to_string()))?;
54        let total = stats.revisions.unwrap_or(0) + stats.snapshots.unwrap_or(0);
55        let unreachable =
56            (report.unreachable.revisions.len() + report.unreachable.snapshots.len()) as u64;
57        stats.reachable_objects = Some(total.saturating_sub(unreachable));
58        stats.unreachable_objects = Some(unreachable);
59        Ok(stats)
60    }
61
62    /// Lists SQLite roots in deterministic name order.
63    ///
64    /// # Returns
65    /// Root status records ordered by root name.
66    ///
67    /// # Errors
68    /// Returns a backend error when roots cannot be read.
69    fn root_statuses(&self) -> Result<Vec<RootStatus>, StoreOperationError> {
70        let connection = self
71            .connection
72            .lock()
73            .map_err(|_| StoreOperationError::backend("lock poisoned"))?;
74        let mut statement = connection
75            .prepare("SELECT name, snapshot_id FROM roots ORDER BY name")
76            .map_err(|error| StoreOperationError::backend(error.to_string()))?;
77        let rows = statement
78            .query_map([], |row| {
79                let name: String = row.get(0)?;
80                let bytes: Vec<u8> = row.get(1)?;
81                Ok((name, bytes))
82            })
83            .map_err(|error| StoreOperationError::backend(error.to_string()))?;
84        let mut statuses = Vec::new();
85        for row in rows {
86            let (name, bytes) =
87                row.map_err(|error| StoreOperationError::backend(error.to_string()))?;
88            if bytes.len() != 32 {
89                return Err(StoreOperationError {
90                    category: scriberig_store::StoreErrorCategory::InvalidEncoding,
91                    message: "invalid root snapshot identifier".to_owned(),
92                });
93            }
94            let mut id = [0; 32];
95            id.copy_from_slice(&bytes);
96            statuses.push(RootStatus {
97                name: scriberig_core::RootName::new(name).map_err(|error| StoreOperationError {
98                    category: scriberig_store::StoreErrorCategory::InvalidEncoding,
99                    message: format!("{error:?}"),
100                })?,
101                snapshot: Some(scriberig_core::SnapshotId::new(id)),
102            });
103        }
104        Ok(statuses)
105    }
106
107    /// Rebuilds SQLite derived indexes.
108    ///
109    /// # Returns
110    /// A consistent index status after rebuilding.
111    ///
112    /// # Errors
113    /// Returns a backend error when rebuilding fails.
114    fn rebuild_indexes(&self) -> Result<IndexStatus, StoreOperationError> {
115        self.rebuild_derived_indexes()
116            .map_err(|error| StoreOperationError::backend(error.to_string()))?;
117        Ok(IndexStatus { consistent: true })
118    }
119}

8/// Current lifecycle state of an onboarded project.
9#[derive(Clone, Debug, Eq, PartialEq)]
10pub enum ProjectStatus {
11    /// The project has been registered but not imported.
12    Registered,
13    /// A scan has identified files needing review.
14    ScanReady,
15    /// The project has a promoted semantic snapshot.
16    Imported,
17    /// The most recent scan or import reported a failure.
18    Failed,
19}
20
21impl ProjectStatus {
22    /// Converts the status into its durable representation.
23    ///
24    /// The stable database status value.
25    /// # Returns
26    /// The operation result.
27    fn as_str(&self) -> &'static str {
28        match self {
29            Self::Registered => "registered",
30            Self::ScanReady => "scan_ready",
31            Self::Imported => "imported",
32            Self::Failed => "failed",
33        }
34    }
35
36    /// Parses a durable project status.
37    ///
38    /// - `value`: Stored status value.
39    ///
40    /// The recognized status, defaulting to registered for unknown values.
41    /// # Returns
42    /// The operation result.
43    fn parse(value: &str) -> Self {
44        match value {
45            "scan_ready" => Self::ScanReady,
46            "imported" => Self::Imported,
47            "failed" => Self::Failed,
48            _ => Self::Registered,
49        }
50    }
51}

53/// Describes one project stored in a database.
54#[derive(Clone, Debug, Eq, PartialEq)]
55pub struct ProjectRecord {
56    /// Stable project identity.
57    pub id: ProjectId,
58    /// Human-readable project name.
59    pub name: String,
60    /// Current canonical filesystem path.
61    pub path: PathBuf,
62    /// Current onboarding state.
63    pub status: ProjectStatus,
64    /// Current promoted snapshot, when imported.
65    pub current_snapshot: Option<SnapshotId>,
66}
67
68impl SqliteStore {
69    /// Lists every project in deterministic name/path order.
70    ///
71    /// Project records stored in this database.
72    ///
73    /// Returns a SQLite query error.
74    /// # Returns
75    /// The operation result.
76    /// # Errors
77    /// Returns an error when the operation fails.
78    pub fn list_projects(&self) -> rusqlite::Result<Vec<ProjectRecord>> {
79        let connection = self
80            .connection
81            .lock()
82            .map_err(|_| rusqlite::Error::InvalidQuery)?;
83        let mut statement = connection.prepare(
84            "SELECT id,name,path,status,current_snapshot FROM projects ORDER BY name,path",
85        )?;
86        let rows = statement.query_map([], project_from_row)?;
87        rows.collect()
88    }
89
90    /// Registers a project or returns the existing project at the path.
91    ///
92    /// - `name`: Human-readable project name.
93    /// - `path`: Canonical filesystem path.
94    ///
95    /// The existing or newly-created project record.
96    ///
97    /// Returns a SQLite error when the catalog cannot be updated.
98    /// # Returns
99    /// The operation result.
100    /// # Errors
101    /// Returns an error when the operation fails.
102    pub fn register_project(
103        &self,
104        name: impl Into<String>,
105        path: impl AsRef<Path>,
106    ) -> rusqlite::Result<ProjectRecord> {
107        let path = normalize_path(path.as_ref());
108        if let Some(project) = self.project_by_path(&path)? {
109            return Ok(project);
110        }
111        let project = ProjectRecord {
112            id: ProjectId::fresh(),
113            name: name.into(),
114            path,
115            status: ProjectStatus::Registered,
116            current_snapshot: None,
117        };
118        let connection = self
119            .connection
120            .lock()
121            .map_err(|_| rusqlite::Error::InvalidQuery)?;
122        connection.execute(
123            "INSERT INTO projects(id,name,path,created_at,updated_at,status)
124             VALUES (?,?,?,datetime('now'),datetime('now'),?)",
125            params![
126                project.id.as_bytes().as_slice(),
127                project.name,
128                project.path.to_string_lossy().as_ref(),
129                project.status.as_str(),
130            ],
131        )?;
132        Ok(project)
133    }

135    /// Finds a project by stable identity.
136    ///
137    /// - `id`: Project identity to find.
138    ///
139    /// The matching project, when present.
140    ///
141    /// Returns a SQLite query error.
142    /// # Returns
143    /// The operation result.
144    /// # Errors
145    /// Returns an error when the operation fails.
146    pub fn project(&self, id: ProjectId) -> rusqlite::Result<Option<ProjectRecord>> {
147        let connection = self
148            .connection
149            .lock()
150            .map_err(|_| rusqlite::Error::InvalidQuery)?;
151        connection
152            .query_row(
153                "SELECT id,name,path,status,current_snapshot FROM projects WHERE id=?",
154                params![id.as_bytes().as_slice()],
155                project_from_row,
156            )
157            .optional()
158    }

160    /// Updates a project path while preserving its stable identity.
161    ///
162    /// - `id`: Project identity to update.
163    /// - `path`: New canonical filesystem path.
164    ///
165    /// Unit after the path is updated.
166    ///
167    /// Returns a SQLite update or uniqueness error.
168    /// # Returns
169    /// The operation result.
170    /// # Errors
171    /// Returns an error when the operation fails.
172    pub fn relink_project(&self, id: ProjectId, path: impl AsRef<Path>) -> rusqlite::Result<()> {
173        let path = normalize_path(path.as_ref());
174        let connection = self
175            .connection
176            .lock()
177            .map_err(|_| rusqlite::Error::InvalidQuery)?;
178        connection.execute(
179            "UPDATE projects SET path=?,updated_at=datetime('now') WHERE id=?",
180            params![path.to_string_lossy().as_ref(), id.as_bytes().as_slice()],
181        )?;
182        Ok(())
183    }
184
185    /// Sets the lifecycle state and current snapshot for a project.
186    ///
187    /// - `id`: Project identity to update.
188    /// - `status`: New lifecycle state.
189    /// - `snapshot`: Current promoted snapshot, when available.
190    ///
191    /// Unit after the update is stored.
192    ///
193    /// Returns a SQLite update error.
194    /// # Returns
195    /// The operation result.
196    /// # Errors
197    /// Returns an error when the operation fails.
198    pub fn set_project_status(
199        &self,
200        id: ProjectId,
201        status: ProjectStatus,
202        snapshot: Option<SnapshotId>,
203    ) -> rusqlite::Result<()> {
204        let connection = self
205            .connection
206            .lock()
207            .map_err(|_| rusqlite::Error::InvalidQuery)?;
208        connection.execute(
209            "UPDATE projects SET status=?,current_snapshot=?,updated_at=datetime('now') WHERE id=?",
210            params![
211                status.as_str(),
212                snapshot.map(|value| value.as_bytes().to_vec()),
213                id.as_bytes().as_slice(),
214            ],
215        )?;
216        Ok(())
217    }

219    /// Finds a project by normalized path.
220    ///
221    /// - `path`: Normalized path to find.
222    ///
223    /// The matching project, when present.
224    ///
225    /// Returns a SQLite query error.
226    /// # Returns
227    /// The operation result.
228    /// # Errors
229    /// Returns an error when the operation fails.
230    fn project_by_path(&self, path: &Path) -> rusqlite::Result<Option<ProjectRecord>> {
231        let connection = self
232            .connection
233            .lock()
234            .map_err(|_| rusqlite::Error::InvalidQuery)?;
235        connection
236            .query_row(
237                "SELECT id,name,path,status,current_snapshot FROM projects WHERE path=?",
238                params![path.to_string_lossy().as_ref()],
239                project_from_row,
240            )
241            .optional()
242    }

245/// Normalizes a path without requiring it to exist yet.
246///
247/// - `path`: Path to normalize.
248///
249/// The canonical path when available, otherwise the supplied path.
250/// # Returns
251/// The operation result.
252fn normalize_path(path: &Path) -> PathBuf {
253    std::fs::canonicalize(path).unwrap_or_else(|_| path.to_path_buf())
254}

256/// Decodes one SQLite project row.
257///
258/// - `row`: SQLite row containing project columns.
259///
260/// A decoded project record.
261///
262/// Returns a conversion error for malformed stored identifiers.
263/// # Returns
264/// The operation result.
265/// # Errors
266/// Returns an error when the operation fails.
267fn project_from_row(row: &rusqlite::Row<'_>) -> rusqlite::Result<ProjectRecord> {
268    let id: Vec<u8> = row.get(0)?;
269    let path: String = row.get(2)?;
270    let status: String = row.get(3)?;
271    let snapshot: Option<Vec<u8>> = row.get(4)?;
272    let id = id.try_into().map(ProjectId::from_bytes).map_err(|_| {
273        rusqlite::Error::FromSqlConversionFailure(
274            0,
275            rusqlite::types::Type::Blob,
276            Box::new(std::io::Error::other("invalid project id")),
277        )
278    })?;
279    let current_snapshot = snapshot.and_then(|bytes| bytes.try_into().ok().map(SnapshotId::new));
280    Ok(ProjectRecord {
281        id,
282        name: row.get(1)?,
283        path: PathBuf::from(path),
284        status: ProjectStatus::parse(&status),
285        current_snapshot,
286    })

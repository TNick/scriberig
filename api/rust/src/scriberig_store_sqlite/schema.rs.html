8/// Current SQLite backend schema version.
9pub const CURRENT_SCHEMA_VERSION: u32 = 4;

11const INTEGRITY_FORMAT_VERSION: &str = "1";
12const RECOVERY_MODE: &str = "transactional-root-promotion";

14/// Reports a backend schema that this build cannot safely open.
15#[derive(Debug, Eq, PartialEq)]
16pub struct UnsupportedSchemaVersion {
17    /// Version found in the database.
18    pub found: u32,
19}
20
21impl Display for UnsupportedSchemaVersion {
22    /// Formats the schema version failure.
23    ///
24    /// # Parameters
25    /// - `self`: Schema version error.
26    /// - `formatter`: Formatting destination.
27    ///
28    /// # Returns
29    /// A formatting result.
30    ///
31    /// # Errors
32    /// Returns a formatting error when the destination rejects output.
33    fn fmt(&self, formatter: &mut Formatter<'_>) -> std::fmt::Result {
34        write!(
35            formatter,
36            "unsupported SQLite backend schema version {}",
37            self.found
38        )
39    }
40}
41
42impl Error for UnsupportedSchemaVersion {}

76/// Initializes the schema and verifies its backend version.
77///
78/// # Parameters
79/// - `connection`: SQLite connection to initialize.
80///
81/// # Returns
82/// The detected backend schema version.
83///
84/// # Errors
85/// Returns an SQLite error or an explicit unsupported-version error.
86pub fn initialize(connection: &mut Connection) -> rusqlite::Result<u32> {
87    let transaction = connection.transaction()?;
88    initialize_schema(&transaction)?;
89    transaction.execute(
90        "UPDATE metadata SET value=? WHERE key='schema_version' AND value IN ('1','2','3')",
91        [CURRENT_SCHEMA_VERSION.to_string()],
92    )?;
93    let has_name = transaction
94        .prepare("PRAGMA table_info(project_entities)")?
95        .query_map([], |row| row.get::<_, String>(1))?
96        .collect::<rusqlite::Result<Vec<_>>>()?
97        .into_iter()
98        .any(|column| column == "name");
99    if !has_name {
100        transaction.execute("ALTER TABLE project_entities ADD COLUMN name TEXT", [])?;
101    }
102    let version = metadata_u32(&transaction, "schema_version")?;
103    if version > CURRENT_SCHEMA_VERSION || version == 0 {
104        return Err(rusqlite::Error::ToSqlConversionFailure(Box::new(
105            UnsupportedSchemaVersion { found: version },
106        )));
107    }
108    validate_metadata(
109        &transaction,
110        "integrity_format_version",
111        INTEGRITY_FORMAT_VERSION,
112    )?;
113    validate_metadata(&transaction, "recovery_mode", RECOVERY_MODE)?;
114    transaction.commit()?;
115    Ok(version)
116}

118/// Creates the version-one backend tables and metadata within a transaction.
119///
120/// # Parameters
121/// - `transaction`: Transaction receiving schema creation.
122///
123/// # Returns
124/// Unit after schema creation.
125///
126/// # Errors
127/// Returns the SQLite schema error.
128#[allow(clippy::too_many_lines)]
129fn initialize_schema(transaction: &Transaction<'_>) -> rusqlite::Result<()> {
130    transaction.execute_batch(
131        "CREATE TABLE IF NOT EXISTS metadata (
132             key TEXT PRIMARY KEY,
133             value TEXT NOT NULL
134         );
135         CREATE TABLE IF NOT EXISTS revisions (
136             id BLOB PRIMARY KEY,
137             version INTEGER NOT NULL,
138             object_type TEXT NOT NULL,
139             payload BLOB NOT NULL
140         );
141         CREATE TABLE IF NOT EXISTS snapshots (id BLOB PRIMARY KEY);
142         CREATE TABLE IF NOT EXISTS snapshot_parents (
143             snapshot_id BLOB NOT NULL,
144             parent_id BLOB NOT NULL,
145             PRIMARY KEY(snapshot_id, parent_id)
146         );
147         CREATE TABLE IF NOT EXISTS snapshot_entries (
148             snapshot_id BLOB NOT NULL,
149             entity BLOB NOT NULL,
150             revision BLOB NOT NULL,
151             PRIMARY KEY(snapshot_id, entity)
152         );
153         CREATE TABLE IF NOT EXISTS snapshot_nodes (
154             id BLOB PRIMARY KEY,
155             kind INTEGER NOT NULL,
156             entity BLOB,
157             revision BLOB,
158             left_child BLOB,
159             right_child BLOB,
160             item_count INTEGER NOT NULL
161         );
162         CREATE TABLE IF NOT EXISTS snapshot_node_roots (
163             snapshot_id BLOB PRIMARY KEY,
164             node_id BLOB NOT NULL
165         );
166         CREATE TABLE IF NOT EXISTS roots (
167             name TEXT PRIMARY KEY,
168             snapshot_id BLOB NOT NULL
169         );
170         INSERT OR IGNORE INTO metadata(key, value)
171             VALUES ('schema_version', '4');
172         INSERT OR IGNORE INTO metadata(key, value)
173             VALUES ('integrity_format_version', '1');
174         INSERT OR IGNORE INTO metadata(key, value)
175             VALUES ('recovery_mode', 'transactional-root-promotion');
176         CREATE TABLE IF NOT EXISTS projects (
177             id BLOB PRIMARY KEY,
178             name TEXT NOT NULL,
179             path TEXT NOT NULL UNIQUE,
180             created_at TEXT NOT NULL,
181             updated_at TEXT NOT NULL,
182             current_snapshot BLOB,
183             status TEXT NOT NULL
184         );
185         CREATE INDEX IF NOT EXISTS projects_path_idx ON projects(path);
186         CREATE TABLE IF NOT EXISTS project_files (
187             project_id BLOB NOT NULL,
188             relative_path TEXT NOT NULL,
189             entity BLOB NOT NULL,
190             digest BLOB NOT NULL,
191             bytes BLOB NOT NULL,
192             adapter TEXT NOT NULL,
193             status TEXT NOT NULL,
194             PRIMARY KEY(project_id, relative_path),
195             FOREIGN KEY(project_id) REFERENCES projects(id) ON DELETE CASCADE
196         );
197         CREATE INDEX IF NOT EXISTS project_files_project_idx
198             ON project_files(project_id);
199         CREATE TABLE IF NOT EXISTS project_entities (
200             project_id BLOB NOT NULL,
201             relative_path TEXT NOT NULL,
202             entity BLOB NOT NULL,
203             kind TEXT NOT NULL,
204             name TEXT,
205             PRIMARY KEY(project_id, relative_path, entity),
206             FOREIGN KEY(project_id) REFERENCES projects(id) ON DELETE CASCADE
207         );
208         CREATE INDEX IF NOT EXISTS project_entities_project_idx
209             ON project_entities(project_id);
210         CREATE TABLE IF NOT EXISTS project_candidates (
211             id TEXT PRIMARY KEY,
212             project_id BLOB NOT NULL,
213             base_snapshot BLOB NOT NULL,
214             entity BLOB NOT NULL,
215             relative_path TEXT NOT NULL,
216             old_bytes BLOB NOT NULL,
217             new_bytes BLOB NOT NULL,
218             validation TEXT NOT NULL,
219             diagnostics TEXT NOT NULL,
220             created_at TEXT NOT NULL,
221             FOREIGN KEY(project_id) REFERENCES projects(id) ON DELETE CASCADE
222         );
223         CREATE INDEX IF NOT EXISTS project_candidates_project_idx
224             ON project_candidates(project_id);
225         CREATE TABLE IF NOT EXISTS project_effects (
226             candidate_id TEXT PRIMARY KEY,
227             project_id BLOB NOT NULL,
228             base_snapshot BLOB NOT NULL,
229             successor_snapshot BLOB,
230             relative_path TEXT NOT NULL,
231             state TEXT NOT NULL,
232             diagnostic TEXT,
233             updated_at TEXT NOT NULL,
234             FOREIGN KEY(project_id) REFERENCES projects(id) ON DELETE CASCADE
235         );",
236    )
237}

239/// Reads a numeric metadata value.
240///
241/// # Parameters
242/// - `transaction`: Transaction containing metadata.
243/// - `key`: Metadata key to read.
244///
245/// # Returns
246/// The parsed metadata version.
247///
248/// # Errors
249/// Returns a SQLite query or parse error.
250fn metadata_u32(transaction: &Transaction<'_>, key: &str) -> rusqlite::Result<u32> {
251    transaction.query_row("SELECT value FROM metadata WHERE key=?", [key], |row| {
252        let value: String = row.get(0)?;
253        value
254            .parse::<u32>()
255            .map_err(|_| rusqlite::Error::InvalidQuery)
256    })
257}

259/// Validates a required textual metadata value.
260///
261/// # Parameters
262/// - `transaction`: Transaction containing metadata.
263/// - `key`: Metadata key to validate.
264/// - `expected`: Required value.
265///
266/// # Returns
267/// Unit when the value matches.
268///
269/// # Errors
270/// Returns a SQLite query or metadata compatibility error.
271fn validate_metadata(
272    transaction: &Transaction<'_>,
273    key: &str,
274    expected: &str,
275) -> rusqlite::Result<()> {
276    let found: String =
277        transaction.query_row("SELECT value FROM metadata WHERE key=?", [key], |row| {
278            row.get(0)
279        })?;
280    if found != expected {
281        return Err(rusqlite::Error::ToSqlConversionFailure(Box::new(
282            UnsupportedMetadata {
283                key: key.to_owned(),
284                found,
285            },
286        )));
287    }
288    Ok(())

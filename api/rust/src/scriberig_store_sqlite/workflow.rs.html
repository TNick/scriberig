7/// Durable source candidate data independent of the project service type.
8#[derive(Clone, Debug, Eq, PartialEq)]
9pub struct ProjectCandidateRecord {
10    /// Stable candidate identity.
11    pub id: String,
12    /// Candidate project.
13    pub project: ProjectId,
14    /// Exact candidate base snapshot.
15    pub base_snapshot: SnapshotId,
16    /// Selected semantic entity.
17    pub entity: EntityId,
18    /// Candidate-relative source path.
19    pub path: String,
20    /// Source bytes at candidate creation.
21    pub old: Vec<u8>,
22    /// Proposed source bytes.
23    pub new: Vec<u8>,
24    /// Validation state (`passed` or `failed`).
25    pub validation: String,
26    /// Newline-separated validation diagnostics.
27    pub diagnostics: String,
28    /// Most recent registered validation profile, when the candidate was checked.
29    pub validation_profile: Option<String>,
30    /// Ordered project source effects belonging to this candidate.
31    pub effects: Vec<ProjectSourceEffectRecord>,
32}

34/// One durable source effect in an aggregate project candidate.
35#[derive(Clone, Debug, Eq, PartialEq)]
36pub struct ProjectSourceEffectRecord {
37    /// Project-relative path owned by the candidate.
38    pub path: String,
39    /// Effect kind: `create`, `update`, or `delete`.
40    pub kind: String,
41    /// Exact base bytes, absent for a create.
42    pub old: Option<Vec<u8>>,
43    /// Proposed bytes, absent for a delete.
44    pub new: Option<Vec<u8>>,
45}

47/// Durable state for a candidate's filesystem effect.
48#[derive(Clone, Debug, Eq, PartialEq)]
49pub struct ProjectEffectRecord {
50    /// Candidate whose effect is being tracked.
51    pub candidate: String,
52    /// Effect project.
53    pub project: ProjectId,
54    /// Candidate base snapshot.
55    pub base_snapshot: SnapshotId,
56    /// Promoted snapshot, when canonical application completed.
57    pub successor_snapshot: Option<SnapshotId>,
58    /// Materialized project-relative path.
59    pub path: String,
60    /// Effect state such as `attempted`, `applied`, or `failed`.
61    pub state: String,
62    /// Safe diagnostic for failed or unknown effects.
63    pub diagnostic: Option<String>,
64}

66/// Durable receipt for one project-relative candidate effect.
67#[derive(Clone, Debug, Eq, PartialEq)]
68pub struct ProjectEffectPathRecord {
69    /// Candidate whose effect is tracked.
70    pub candidate: String,
71    /// Project-relative affected path.
72    pub path: String,
73    /// State such as `not_attempted`, `applied`, or `conflict`.
74    pub state: String,
75    /// BLAKE3 digest expected from the exact base, when present.
76    pub expected_digest: Option<[u8; 32]>,
77    /// BLAKE3 digest observed after the effect, when present.
78    pub observed_digest: Option<[u8; 32]>,
79    /// Bounded safe diagnostic.
80    pub diagnostic: Option<String>,
81}

83/// Converts an optional SQLite blob into an exact BLAKE3 digest.
84///
85/// # Parameters
86/// - `bytes`: Stored digest bytes.
87///
88/// # Returns
89/// A digest when the column is populated.
90///
91/// # Errors
92/// Returns a query conversion error for an invalid digest length.
93fn digest_from_column(bytes: Option<Vec<u8>>) -> rusqlite::Result<Option<[u8; 32]>> {
94    bytes
95        .map(|value| value.try_into().map_err(|_| rusqlite::Error::InvalidQuery))
96        .transpose()
97}

99impl SqliteStore {
100    /// Persists the latest durable state for one candidate path effect.
101    ///
102    /// # Parameters
103    /// - `effect`: Per-path receipt to persist.
104    ///
105    /// # Returns
106    /// Unit after the receipt is durable.
107    ///
108    /// # Errors
109    /// Returns a SQLite transaction or digest conversion error.
110    pub fn put_project_effect_path(
111        &self,
112        effect: &ProjectEffectPathRecord,
113    ) -> rusqlite::Result<()> {
114        let connection = self
115            .connection
116            .lock()
117            .map_err(|_| rusqlite::Error::InvalidQuery)?;
118        connection.execute(
119            "INSERT INTO project_effect_paths
120             (candidate_id,relative_path,state,expected_digest,observed_digest,diagnostic,updated_at)
121             VALUES (?,?,?,?,?,?,datetime('now'))
122             ON CONFLICT(candidate_id,relative_path) DO UPDATE SET
123                 state=excluded.state,
124                 expected_digest=excluded.expected_digest,
125                 observed_digest=excluded.observed_digest,
126                 diagnostic=excluded.diagnostic,
127                 updated_at=excluded.updated_at",
128            params![
129                effect.candidate,
130                effect.path,
131                effect.state,
132                effect.expected_digest.map(|value| value.to_vec()),
133                effect.observed_digest.map(|value| value.to_vec()),
134                effect.diagnostic,
135            ],
136        )?;
137        Ok(())
138    }

140    /// Loads path-level receipts for one aggregate candidate.
141    ///
142    /// # Parameters
143    /// - `candidate`: Stable candidate identity.
144    ///
145    /// # Returns
146    /// Deterministically ordered effect receipts.
147    ///
148    /// # Errors
149    /// Returns a SQLite query or digest conversion error.
150    pub fn project_effect_paths(
151        &self,
152        candidate: &str,
153    ) -> rusqlite::Result<Vec<ProjectEffectPathRecord>> {
154        let connection = self
155            .connection
156            .lock()
157            .map_err(|_| rusqlite::Error::InvalidQuery)?;
158        let mut statement = connection.prepare(
159            "SELECT candidate_id,relative_path,state,expected_digest,observed_digest,diagnostic
160             FROM project_effect_paths WHERE candidate_id=? ORDER BY relative_path",
161        )?;
162        statement
163            .query_map([candidate], |row| {
164                Ok(ProjectEffectPathRecord {
165                    candidate: row.get(0)?,
166                    path: row.get(1)?,
167                    state: row.get(2)?,
168                    expected_digest: digest_from_column(row.get(3)?)?,
169                    observed_digest: digest_from_column(row.get(4)?)?,
170                    diagnostic: row.get(5)?,
171                })
172            })?
173            .collect()
174    }

176    /// Stores or replaces a durable project candidate.
177    ///
178    /// # Parameters
179    /// - `candidate`: Candidate data to persist.
180    ///
181    /// # Returns
182    /// Unit after the candidate is durable.
183    ///
184    /// # Errors
185    /// Returns a SQLite transaction error.
186    pub fn put_project_candidate(
187        &self,
188        candidate: &ProjectCandidateRecord,
189    ) -> rusqlite::Result<()> {
190        let connection = self
191            .connection
192            .lock()
193            .map_err(|_| rusqlite::Error::InvalidQuery)?;
194        let transaction = connection.unchecked_transaction()?;
195        transaction.execute(
196            "INSERT INTO project_candidates
197             (id,project_id,base_snapshot,entity,relative_path,old_bytes,new_bytes,
198              validation,diagnostics,validation_profile,created_at)
199             VALUES (?,?,?,?,?,?,?,?,?,?,datetime('now'))
200             ON CONFLICT(id) DO UPDATE SET
201                 validation=excluded.validation,
202                 diagnostics=excluded.diagnostics,
203                 validation_profile=excluded.validation_profile",
204            params![
205                candidate.id,
206                candidate.project.as_bytes().as_slice(),
207                candidate.base_snapshot.as_bytes().as_slice(),
208                candidate.entity.as_bytes().as_slice(),
209                candidate.path,
210                candidate.old,
211                candidate.new,
212                candidate.validation,
213                candidate.diagnostics,
214                candidate.validation_profile,
215            ],
216        )?;
217        transaction.execute(
218            "DELETE FROM project_candidate_effects WHERE candidate_id=?",
219            [&candidate.id],
220        )?;
221        for (ordinal, effect) in candidate.effects.iter().enumerate() {
222            let ordinal = i64::try_from(ordinal).map_err(|_| rusqlite::Error::InvalidQuery)?;
223            transaction.execute(
224                "INSERT INTO project_candidate_effects
225                 (candidate_id,ordinal,relative_path,kind,old_bytes,new_bytes)
226                 VALUES (?,?,?,?,?,?)",
227                params![
228                    candidate.id,
229                    ordinal,
230                    effect.path,
231                    effect.kind,
232                    effect.old,
233                    effect.new
234                ],
235            )?;
236        }
237        transaction.commit()?;
238        Ok(())
239    }

241    /// Loads one durable project candidate by identity.
242    ///
243    /// # Parameters
244    /// - `id`: Candidate identity.
245    ///
246    /// # Returns
247    /// The candidate when it exists.
248    ///
249    /// # Errors
250    /// Returns a SQLite query or identifier conversion error.
251    pub fn project_candidate(&self, id: &str) -> rusqlite::Result<Option<ProjectCandidateRecord>> {
252        let connection = self
253            .connection
254            .lock()
255            .map_err(|_| rusqlite::Error::InvalidQuery)?;
256        let mut candidate = connection
257            .query_row(
258                "SELECT id,project_id,base_snapshot,entity,relative_path,old_bytes,new_bytes,
259                        validation,diagnostics,validation_profile
260                 FROM project_candidates WHERE id=?",
261                [id],
262                |row| {
263                    Ok(ProjectCandidateRecord {
264                        id: row.get(0)?,
265                        project: ProjectId::from_bytes(
266                            row.get::<_, Vec<u8>>(1)?
267                                .try_into()
268                                .map_err(|_| rusqlite::Error::InvalidQuery)?,
269                        ),
270                        base_snapshot: SnapshotId::new(
271                            row.get::<_, Vec<u8>>(2)?
272                                .try_into()
273                                .map_err(|_| rusqlite::Error::InvalidQuery)?,
274                        ),
275                        entity: EntityId::from_bytes(
276                            row.get::<_, Vec<u8>>(3)?
277                                .try_into()
278                                .map_err(|_| rusqlite::Error::InvalidQuery)?,
279                        ),
280                        path: row.get(4)?,
281                        old: row.get(5)?,
282                        new: row.get(6)?,
283                        validation: row.get(7)?,
284                        diagnostics: row.get(8)?,
285                        validation_profile: row.get(9)?,
286                        effects: Vec::new(),
287                    })
288                },
289            )
290            .optional()?;
291        if let Some(candidate) = &mut candidate {
292            let mut statement = connection.prepare(
293                "SELECT relative_path,kind,old_bytes,new_bytes
294                 FROM project_candidate_effects WHERE candidate_id=? ORDER BY ordinal",
295            )?;
296            candidate.effects = statement
297                .query_map([id], |row| {
298                    Ok(ProjectSourceEffectRecord {
299                        path: row.get(0)?,
300                        kind: row.get(1)?,
301                        old: row.get(2)?,
302                        new: row.get(3)?,
303                    })
304                })?
305                .collect::<rusqlite::Result<Vec<_>>>()?;
306        }
307        Ok(candidate)
308    }

310    /// Records the current external materialization effect state.
311    ///
312    /// # Parameters
313    /// - `effect`: Effect receipt to persist.
314    ///
315    /// # Returns
316    /// Unit after the receipt is durable.
317    ///
318    /// # Errors
319    /// Returns a SQLite transaction error.
320    pub fn put_project_effect(&self, effect: &ProjectEffectRecord) -> rusqlite::Result<()> {
321        let connection = self
322            .connection
323            .lock()
324            .map_err(|_| rusqlite::Error::InvalidQuery)?;
325        connection.execute(
326            "INSERT INTO project_effects
327             (candidate_id,project_id,base_snapshot,successor_snapshot,relative_path,
328              state,diagnostic,updated_at)
329             VALUES (?,?,?,?,?,?,?,datetime('now'))
330             ON CONFLICT(candidate_id) DO UPDATE SET
331                 successor_snapshot=excluded.successor_snapshot,
332                 state=excluded.state,
333                 diagnostic=excluded.diagnostic,
334                 updated_at=excluded.updated_at",
335            params![
336                effect.candidate,
337                effect.project.as_bytes().as_slice(),
338                effect.base_snapshot.as_bytes().as_slice(),
339                effect
340                    .successor_snapshot
341                    .map(|snapshot| snapshot.as_bytes().to_vec()),
342                effect.path,
343                effect.state,
344                effect.diagnostic,
345            ],
346        )?;
347        Ok(())
348    }

350    /// Loads the effect receipt for one candidate.
351    ///
352    /// # Parameters
353    /// - `candidate`: Candidate identity.
354    ///
355    /// # Returns
356    /// The durable effect receipt when present.
357    ///
358    /// # Errors
359    /// Returns a SQLite query or identifier conversion error.
360    pub fn project_effect(&self, candidate: &str) -> rusqlite::Result<Option<ProjectEffectRecord>> {
361        let connection = self
362            .connection
363            .lock()
364            .map_err(|_| rusqlite::Error::InvalidQuery)?;
365        connection
366            .query_row(
367                "SELECT candidate_id,project_id,base_snapshot,successor_snapshot,
368                        relative_path,state,diagnostic
369                 FROM project_effects WHERE candidate_id=?",
370                [candidate],
371                |row| {
372                    let successor: Option<Vec<u8>> = row.get(3)?;
373                    Ok(ProjectEffectRecord {
374                        candidate: row.get(0)?,
375                        project: ProjectId::from_bytes(
376                            row.get::<_, Vec<u8>>(1)?
377                                .try_into()
378                                .map_err(|_| rusqlite::Error::InvalidQuery)?,
379                        ),
380                        base_snapshot: SnapshotId::new(
381                            row.get::<_, Vec<u8>>(2)?
382                                .try_into()
383                                .map_err(|_| rusqlite::Error::InvalidQuery)?,
384                        ),
385                        successor_snapshot: successor
386                            .map(|bytes| bytes.try_into().map(SnapshotId::new))
387                            .transpose()
388                            .map_err(|_| rusqlite::Error::InvalidQuery)?,
389                        path: row.get(4)?,
390                        state: row.get(5)?,
391                        diagnostic: row.get(6)?,
392                    })
393                },
394            )
395            .optional()
396    }

1//! Persists project candidates and filesystem effect receipts.
2
3use crate::SqliteStore;
4use rusqlite::{OptionalExtension, params};
5use scriberig_core::{EntityId, ProjectId, SnapshotId};
6
7/// Durable source candidate data independent of the project service type.
8#[derive(Clone, Debug, Eq, PartialEq)]
9pub struct ProjectCandidateRecord {
10    /// Stable candidate identity.
11    pub id: String,
12    /// Candidate project.
13    pub project: ProjectId,
14    /// Exact candidate base snapshot.
15    pub base_snapshot: SnapshotId,
16    /// Selected semantic entity.
17    pub entity: EntityId,
18    /// Candidate-relative source path.
19    pub path: String,
20    /// Source bytes at candidate creation.
21    pub old: Vec<u8>,
22    /// Proposed source bytes.
23    pub new: Vec<u8>,
24    /// Validation state (`passed` or `failed`).
25    pub validation: String,
26    /// Newline-separated validation diagnostics.
27    pub diagnostics: String,
28}
29
30/// Durable state for a candidate's filesystem effect.
31#[derive(Clone, Debug, Eq, PartialEq)]
32pub struct ProjectEffectRecord {
33    /// Candidate whose effect is being tracked.
34    pub candidate: String,
35    /// Effect project.
36    pub project: ProjectId,
37    /// Candidate base snapshot.
38    pub base_snapshot: SnapshotId,
39    /// Promoted snapshot, when canonical application completed.
40    pub successor_snapshot: Option<SnapshotId>,
41    /// Materialized project-relative path.
42    pub path: String,
43    /// Effect state such as `attempted`, `applied`, or `failed`.
44    pub state: String,
45    /// Safe diagnostic for failed or unknown effects.
46    pub diagnostic: Option<String>,
47}
48
49impl SqliteStore {
50    /// Stores or replaces a durable project candidate.
51    ///
52    /// # Parameters
53    /// - `candidate`: Candidate data to persist.
54    ///
55    /// # Returns
56    /// Unit after the candidate is durable.
57    ///
58    /// # Errors
59    /// Returns a SQLite transaction error.
60    pub fn put_project_candidate(
61        &self,
62        candidate: &ProjectCandidateRecord,
63    ) -> rusqlite::Result<()> {
64        let connection = self
65            .connection
66            .lock()
67            .map_err(|_| rusqlite::Error::InvalidQuery)?;
68        connection.execute(
69            "INSERT INTO project_candidates
70             (id,project_id,base_snapshot,entity,relative_path,old_bytes,new_bytes,
71              validation,diagnostics,created_at)
72             VALUES (?,?,?,?,?,?,?,?,?,datetime('now'))
73             ON CONFLICT(id) DO UPDATE SET
74                 validation=excluded.validation,
75                 diagnostics=excluded.diagnostics",
76            params![
77                candidate.id,
78                candidate.project.as_bytes().as_slice(),
79                candidate.base_snapshot.as_bytes().as_slice(),
80                candidate.entity.as_bytes().as_slice(),
81                candidate.path,
82                candidate.old,
83                candidate.new,
84                candidate.validation,
85                candidate.diagnostics,
86            ],
87        )?;
88        Ok(())
89    }
90
91    /// Loads one durable project candidate by identity.
92    ///
93    /// # Parameters
94    /// - `id`: Candidate identity.
95    ///
96    /// # Returns
97    /// The candidate when it exists.
98    ///
99    /// # Errors
100    /// Returns a SQLite query or identifier conversion error.
101    pub fn project_candidate(&self, id: &str) -> rusqlite::Result<Option<ProjectCandidateRecord>> {
102        let connection = self
103            .connection
104            .lock()
105            .map_err(|_| rusqlite::Error::InvalidQuery)?;
106        connection
107            .query_row(
108                "SELECT id,project_id,base_snapshot,entity,relative_path,old_bytes,new_bytes,
109                        validation,diagnostics
110                 FROM project_candidates WHERE id=?",
111                [id],
112                |row| {
113                    Ok(ProjectCandidateRecord {
114                        id: row.get(0)?,
115                        project: ProjectId::from_bytes(
116                            row.get::<_, Vec<u8>>(1)?
117                                .try_into()
118                                .map_err(|_| rusqlite::Error::InvalidQuery)?,
119                        ),
120                        base_snapshot: SnapshotId::new(
121                            row.get::<_, Vec<u8>>(2)?
122                                .try_into()
123                                .map_err(|_| rusqlite::Error::InvalidQuery)?,
124                        ),
125                        entity: EntityId::from_bytes(
126                            row.get::<_, Vec<u8>>(3)?
127                                .try_into()
128                                .map_err(|_| rusqlite::Error::InvalidQuery)?,
129                        ),
130                        path: row.get(4)?,
131                        old: row.get(5)?,
132                        new: row.get(6)?,
133                        validation: row.get(7)?,
134                        diagnostics: row.get(8)?,
135                    })
136                },
137            )
138            .optional()
139    }
140
141    /// Records the current external materialization effect state.
142    ///
143    /// # Parameters
144    /// - `effect`: Effect receipt to persist.
145    ///
146    /// # Returns
147    /// Unit after the receipt is durable.
148    ///
149    /// # Errors
150    /// Returns a SQLite transaction error.
151    pub fn put_project_effect(&self, effect: &ProjectEffectRecord) -> rusqlite::Result<()> {
152        let connection = self
153            .connection
154            .lock()
155            .map_err(|_| rusqlite::Error::InvalidQuery)?;
156        connection.execute(
157            "INSERT INTO project_effects
158             (candidate_id,project_id,base_snapshot,successor_snapshot,relative_path,
159              state,diagnostic,updated_at)
160             VALUES (?,?,?,?,?,?,?,datetime('now'))
161             ON CONFLICT(candidate_id) DO UPDATE SET
162                 successor_snapshot=excluded.successor_snapshot,
163                 state=excluded.state,
164                 diagnostic=excluded.diagnostic,
165                 updated_at=excluded.updated_at",
166            params![
167                effect.candidate,
168                effect.project.as_bytes().as_slice(),
169                effect.base_snapshot.as_bytes().as_slice(),
170                effect
171                    .successor_snapshot
172                    .map(|snapshot| snapshot.as_bytes().to_vec()),
173                effect.path,
174                effect.state,
175                effect.diagnostic,
176            ],
177        )?;
178        Ok(())
179    }
180
181    /// Loads the effect receipt for one candidate.
182    ///
183    /// # Parameters
184    /// - `candidate`: Candidate identity.
185    ///
186    /// # Returns
187    /// The durable effect receipt when present.
188    ///
189    /// # Errors
190    /// Returns a SQLite query or identifier conversion error.
191    pub fn project_effect(&self, candidate: &str) -> rusqlite::Result<Option<ProjectEffectRecord>> {
192        let connection = self
193            .connection
194            .lock()
195            .map_err(|_| rusqlite::Error::InvalidQuery)?;
196        connection
197            .query_row(
198                "SELECT candidate_id,project_id,base_snapshot,successor_snapshot,
199                        relative_path,state,diagnostic
200                 FROM project_effects WHERE candidate_id=?",
201                [candidate],
202                |row| {
203                    let successor: Option<Vec<u8>> = row.get(3)?;
204                    Ok(ProjectEffectRecord {
205                        candidate: row.get(0)?,
206                        project: ProjectId::from_bytes(
207                            row.get::<_, Vec<u8>>(1)?
208                                .try_into()
209                                .map_err(|_| rusqlite::Error::InvalidQuery)?,
210                        ),
211                        base_snapshot: SnapshotId::new(
212                            row.get::<_, Vec<u8>>(2)?
213                                .try_into()
214                                .map_err(|_| rusqlite::Error::InvalidQuery)?,
215                        ),
216                        successor_snapshot: successor
217                            .map(|bytes| bytes.try_into().map(SnapshotId::new))
218                            .transpose()
219                            .map_err(|_| rusqlite::Error::InvalidQuery)?,
220                        path: row.get(4)?,
221                        state: row.get(5)?,
222                        diagnostic: row.get(6)?,
223                    })
224                },
225            )
226            .optional()
227    }

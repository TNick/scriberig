1//! Performs conservative SQLite mark-and-sweep maintenance.
2
3use rusqlite::params;
4use scriberig_store::{IntegrityReport, ProtectedRoots};
5
6use crate::SqliteStore;
7
8impl SqliteStore {
9    /// Computes immutable objects that are not protected by a stable root set.
10    ///
11    /// # Parameters
12    /// - `protected`: Explicit snapshot roots captured for this maintenance run.
13    ///
14    /// # Returns
15    /// Integrity findings and valid unreachable candidates.
16    ///
17    /// # Errors
18    /// Returns an error when canonical storage cannot be inspected.
19    pub fn gc_dry_run(&self, protected: &ProtectedRoots) -> rusqlite::Result<IntegrityReport> {
20        self.integrity_check(protected)
21    }
22
23    /// Deletes valid immutable objects proven unreachable from protected roots.
24    ///
25    /// # Parameters
26    /// - `protected`: Explicit snapshot roots captured for this maintenance run.
27    ///
28    /// # Returns
29    /// The pre-sweep integrity report and candidate set.
30    ///
31    /// # Errors
32    /// Returns an error when inspection or the sweep transaction fails.
33    pub fn gc_sweep(&self, protected: &ProtectedRoots) -> rusqlite::Result<IntegrityReport> {
34        let report = self.integrity_check(protected)?;
35        if report.is_corrupt() {
36            return Ok(report);
37        }
38        let connection = self
39            .connection
40            .lock()
41            .map_err(|_| rusqlite::Error::InvalidQuery)?;
42        let transaction = connection.unchecked_transaction()?;
43        for snapshot_id in &report.unreachable.snapshots {
44            transaction.execute(
45                "DELETE FROM snapshot_entries WHERE snapshot_id=?",
46                params![snapshot_id.as_bytes().as_slice()],
47            )?;
48            transaction.execute(
49                "DELETE FROM snapshot_node_roots WHERE snapshot_id=?",
50                params![snapshot_id.as_bytes().as_slice()],
51            )?;
52            transaction.execute(
53                "DELETE FROM snapshots WHERE id=?",
54                params![snapshot_id.as_bytes().as_slice()],
55            )?;
56        }
57        for revision_id in &report.unreachable.revisions {
58            transaction.execute(
59                "DELETE FROM revisions WHERE id=?",
60                params![revision_id.as_bytes().as_slice()],
61            )?;
62        }
63        for node_id in &report.unreachable.snapshot_nodes {
64            transaction.execute(
65                "DELETE FROM snapshot_nodes WHERE id=?",
66                params![node_id.as_bytes().as_slice()],
67            )?;
68        }
69        transaction.commit()?;
70        Ok(report)
71    }
72}

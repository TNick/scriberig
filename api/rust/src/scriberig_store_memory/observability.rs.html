1//! Provides observability for the in-memory storage backend.
2
3use scriberig_store::{IndexStatus, RootStatus, StoreObservability, StoreStats};
4
5use crate::MemoryStore;
6
7impl StoreObservability for MemoryStore {
8    /// Returns counts from the in-memory authoritative collections.
9    ///
10    /// # Returns
11    /// Available in-memory statistics.
12    ///
13    /// # Errors
14    /// Returns an error when a collection lock is poisoned.
15    fn stats(&self) -> Result<StoreStats, String> {
16        let revisions = self
17            .revisions
18            .revisions
19            .read()
20            .map_err(|_| "revision lock poisoned".to_owned())?
21            .len() as u64;
22        let snapshots = self
23            .snapshots
24            .read()
25            .map_err(|_| "snapshot lock poisoned".to_owned())?
26            .len() as u64;
27        let named_roots = self
28            .roots
29            .read()
30            .map_err(|_| "root lock poisoned".to_owned())?
31            .len() as u64;
32        Ok(StoreStats {
33            revisions: Some(revisions),
34            snapshots: Some(snapshots),
35            named_roots: Some(named_roots),
36            ..StoreStats::default()
37        })
38    }
39
40    /// Lists in-memory roots in deterministic order.
41    ///
42    /// # Returns
43    /// Root status records ordered by root name.
44    ///
45    /// # Errors
46    /// Returns an error when the root lock is poisoned.
47    fn root_statuses(&self) -> Result<Vec<RootStatus>, String> {
48        Ok(self
49            .roots
50            .read()
51            .map_err(|_| "root lock poisoned".to_owned())?
52            .iter()
53            .map(|(name, snapshot)| RootStatus {
54                name: name.clone(),
55                snapshot: Some(*snapshot),
56            })
57            .collect())
58    }
59
60    /// Rebuilds the in-memory derived-index boundary.
61    ///
62    /// # Returns
63    /// A consistent index status.
64    ///
65    /// # Errors
66    /// This backend has no derived indexes and does not fail this operation.
67    fn rebuild_indexes(&self) -> Result<IndexStatus, String> {
68        Ok(IndexStatus { consistent: true })
69    }
70}

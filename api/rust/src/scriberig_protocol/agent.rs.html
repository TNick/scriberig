1//! Defines provider-neutral agent operation protocol values.
2
3use crate::OperationStatusResponse;
4use crate::{EntityReference, SnapshotReference};
5use schemars::JsonSchema;
6use serde::{Deserialize, Serialize};
7
8/// Identifies a session in an agent request.
9#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
10pub struct SessionReference {
11    /// Stable session identity.
12    pub id: String,
13}
14
15/// Identifies a branch in an agent continuation request.
16#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
17pub struct BranchReference {
18    /// Stable branch identity.
19    pub id: String,
20}
21
22/// Identifies the exact semantic scope supplied to an agent.
23#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
24pub struct AgentSelection {
25    /// Project identity when the semantic scope is project-oriented.
26    #[serde(default)]
27    pub project_id: Option<String>,
28    /// Semantic root of the selected project workspace.
29    #[serde(default)]
30    pub root: Option<EntityReference>,
31    /// Selected semantic entity.
32    pub entity: EntityReference,
33    /// Snapshot containing the selected entity.
34    pub snapshot: SnapshotReference,
35}
36
37/// Identifies a candidate change that must use normal migration review.
38#[derive(Clone, Debug, Default, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
39pub struct CandidateChangeReference {
40    /// Migration proposal to review, when one exists.
41    #[serde(default)]
42    pub migration: Option<String>,
43    /// Semantic entity affected by the candidate.
44    #[serde(default)]
45    pub entity: Option<EntityReference>,
46    /// Candidate snapshot produced by the proposal, when known.
47    #[serde(default)]
48    pub candidate_snapshot: Option<SnapshotReference>,
49    /// Safe validation evidence handles associated with the candidate.
50    #[serde(default)]
51    pub validation: Vec<String>,
52}
53
54/// Provider-neutral durable terminal result for one agent operation.
55#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
56pub struct AgentResult {
57    /// Session containing the completed work.
58    pub session: SessionReference,
59    /// Branch receiving the completed work.
60    pub root_branch: BranchReference,
61    /// Provider-neutral final output.
62    pub final_output: String,
63    /// Durable branch-result identity.
64    pub branch_result: String,
65    /// Semantic entities changed by the operation.
66    pub produced_changes: Vec<crate::EntityReference>,
67    /// Validation summaries associated with the result.
68    pub validation_results: Vec<String>,
69    /// Questions left unresolved by the operation.
70    pub unresolved_questions: Vec<String>,
71    /// Candidate changes that require normal review and validation.
72    #[serde(default)]
73    pub candidate_changes: Vec<CandidateChangeReference>,
74}
75
76/// Starts an agent operation at a session root.
77#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
78pub struct AgentStartRequest {
79    /// Session receiving the new root operation.
80    pub session: SessionReference,
81    /// User-authored instruction.
82    pub prompt: String,
83    /// Exact semantic scope for the requested work.
84    #[serde(default)]
85    pub selection: Option<AgentSelection>,
86}
87
88/// Continues an agent operation on an existing session branch.
89#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
90pub struct AgentContinueRequest {
91    /// Session containing the target branch.
92    pub session: SessionReference,
93    /// Existing branch receiving the follow-up.
94    pub branch: BranchReference,
95    /// User-authored follow-up instruction.
96    pub prompt: String,
97    /// Exact semantic scope for the requested work.
98    #[serde(default)]
99    pub selection: Option<AgentSelection>,
100}
101
102/// Returns the shared operation handle accepted for agent execution.
103#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
104pub struct AgentAcceptedResponse {
105    /// Shared application operation status.
106    pub operation: OperationStatusResponse,
107    /// Session receiving the operation.
108    pub session: SessionReference,
109    /// Branch targeted by continuation, when present.
110    pub branch: Option<BranchReference>,
111}

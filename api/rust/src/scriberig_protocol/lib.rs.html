53    /// Verifies a value survives JSON encoding without semantic loss.
54    fn round_trip<T>(value: &T)
55    where
56        T: DeserializeOwned + std::fmt::Debug + PartialEq + Serialize,
57    {
58        let encoded = serde_json::to_string(value).expect("serialize protocol value");
59        let decoded: T = serde_json::from_str(&encoded).expect("deserialize protocol value");
60        assert_eq!(decoded, *value);
61    }

39    /// Verifies a value survives JSON encoding without semantic loss.
40    fn round_trip<T>(value: &T)
41    where
42        T: DeserializeOwned + std::fmt::Debug + PartialEq + Serialize,
43    {
44        let encoded = serde_json::to_string(value).expect("serialize protocol value");
45        let decoded: T = serde_json::from_str(&encoded).expect("deserialize protocol value");
46        assert_eq!(decoded, *value);
47    }

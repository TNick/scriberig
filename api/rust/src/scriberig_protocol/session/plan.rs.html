8/// Durable lifecycle state for one planned work item.
9#[derive(Clone, Copy, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
10pub enum SessionPlanNodeStatus {
11    /// Waiting for explicit prerequisite results.
12    Planned,
13    /// Ready for a worker to claim.
14    Ready,
15    /// Running on one session branch.
16    Running,
17    /// Completed with a durable result.
18    Completed,
19    /// Failed without producing a dependency result.
20    Failed,
21    /// Explicitly abandoned.
22    Abandoned,
23}

37/// Durable planned work located in the session tree.
38#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
39pub struct SessionPlanNode {
40    /// Stable plan-node identity.
41    pub id: String,
42    /// Branch or plan node that structurally owns this node.
43    pub parent: String,
44    /// Human-readable work title.
45    pub title: String,
46    /// Focused work instruction.
47    pub purpose: String,
48    /// Durable scheduling state.
49    pub status: SessionPlanNodeStatus,
50    /// Exact immutable node revision, when supplied by the shared view.
51    #[serde(default)]
52    pub exact_revision: Option<String>,
53    /// Derived effective status including required descendants and gates.
54    #[serde(default)]
55    pub effective_status: Option<SessionPlanNodeStatus>,
56    /// Whether this node's subtree contains ignored work.
57    #[serde(default)]
58    pub contains_ignored: bool,
59    /// Whether this node's subtree contains abandoned work.
60    #[serde(default)]
61    pub contains_abandoned: bool,
62    /// Typed reason for direct abandonment.
63    #[serde(default)]
64    pub abandon_reason: Option<SessionPlanAbandonReason>,
65    /// Branch executing this work, when started.
66    #[serde(default)]
67    pub branch: Option<String>,
68    /// Explicit result satisfying downstream dependencies.
69    #[serde(default)]
70    pub result: Option<String>,
71    /// Replacement node identity when this node was superseded.
72    #[serde(default)]
73    pub superseded_by: Option<String>,
74    /// Exact result contract required by downstream work.
75    #[serde(default)]
76    pub expected_result: Option<SessionExpectedResult>,
77    /// Selected semantic handles for this work.
78    #[serde(default)]
79    pub semantic_scope: Vec<String>,
80    /// Independent safe model-library assignment.
81    #[serde(default)]
82    pub assignment: Option<SessionModelAssignment>,
83    /// Exclusive worker claim when the node is running.
84    #[serde(default)]
85    pub claim: Option<SessionWorkerClaim>,
86    /// Stable identity of the node's current summary.
87    #[serde(default)]
88    pub summary_id: Option<String>,
89    /// Bounded concise summary preview.
90    #[serde(default)]
91    pub summary_preview: Option<String>,
92    /// Prerequisite node identities currently blocking this node.
93    #[serde(default)]
94    pub dependency_blockers: Vec<String>,
95    /// Entry gates by evaluated state.
96    #[serde(default)]
97    pub entry_gates: SessionPlanGateCounts,
98    /// Completion gates by evaluated state.
99    #[serde(default)]
100    pub completion_gates: SessionPlanGateCounts,
101    /// Node and gate evidence-association count.
102    #[serde(default)]
103    pub evidence_count: u32,
104    /// Whether an explicit retry is currently available.
105    #[serde(default)]
106    pub can_retry: bool,
107    /// Exact reference to the most recent failed result.
108    #[serde(default)]
109    pub last_failure: Option<SessionPlanEvidenceReference>,
110}

10/// Counts direct plan nodes by their one lifecycle vocabulary.
11#[derive(Clone, Copy, Debug, Default, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
12pub struct SessionPlanStatusCounts {
13    /// Nodes waiting for dependencies or gates.
14    pub planned: u32,
15    /// Nodes ready for a worker claim.
16    pub ready: u32,
17    /// Nodes with active worker claims.
18    pub running: u32,
19    /// Nodes completed with accepted results.
20    pub completed: u32,
21    /// Nodes with a failed attempt.
22    pub failed: u32,
23    /// Explicitly abandoned nodes.
24    pub abandoned: u32,
25}

66/// Identifies the typed source category of evidence.
67#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
68pub enum SessionPlanEvidenceKind {
69    /// Durable successful or failed branch result.
70    BranchResult,
71    /// Validation result with profile and pass/fail state.
72    ValidationResult,
73    /// Resolved or unresolved semantic decision.
74    Decision,
75    /// Human approval under a named policy.
76    HumanApproval,
77    /// Candidate or migration proposal.
78    Candidate,
79    /// Artifact or external observation.
80    Artifact,
81    /// Structured execution failure result.
82    Failure,
83    /// Another registered evidence class.
84    Other(String),
85}

199/// Describes one supported plan-gate requirement.
200#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
201pub enum SessionPlanGateRequirement {
202    /// Requires supporting evidence of the requested kind.
203    EvidencePresent {
204        /// Evidence category required by this gate.
205        kind: SessionPlanEvidenceKind,
206    },
207    /// Requires passing validation for an optional profile.
208    ValidationPassed {
209        /// Required validation profile, when constrained.
210        profile: Option<String>,
211    },
212    /// Requires one exact decision to be resolved.
213    DecisionResolved {
214        /// Stable decision entity identity.
215        decision: String,
216    },
217    /// Requires approval under a policy and optional role.
218    HumanApproval {
219        /// Approval policy identity.
220        policy: String,
221        /// Required approver role, when constrained.
222        role: Option<String>,
223    },
224    /// Requires an exact semantic reference to satisfy a typed predicate.
225    SemanticCondition {
226        /// Entity/revision/snapshot reference to check.
227        reference: SessionSemanticReference,
228        /// Supported non-executable predicate.
229        predicate: SessionPlanSemanticPredicate,
230    },
231    /// Requires one structural child to complete.
232    RequiredChildCompleted {
233        /// Stable child node identity.
234        child: String,
235    },
236}

308/// Plan projection with structural nodes and a separate dependency DAG.
309#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
310pub struct SessionPlan {
311    /// Stable plan identity.
312    pub id: String,
313    /// Current durable plan revision.
314    pub revision: u64,
315    /// Exact immutable Plan revision identity.
316    #[serde(default)]
317    pub exact_revision: Option<String>,
318    /// Exact universal snapshot represented by this projection.
319    #[serde(default)]
320    pub snapshot: Option<String>,
321    /// Durable plan objective.
322    pub goal: String,
323    /// Current plan lifecycle state.
324    pub status: SessionPlanStatus,
325    /// Exact session continuation where this plan was created.
326    #[serde(default)]
327    pub anchor: Option<SessionContinuationAnchor>,
328    /// Root branch from which planning began.
329    pub root_branch: String,
330    /// Durable work nodes.
331    pub nodes: Vec<SessionPlanNode>,
332    /// Independent result-based dependency edges.
333    pub dependencies: Vec<SessionPlanDependency>,
334    /// Structural root nodes in stable membership order.
335    #[serde(default)]
336    pub root_nodes: Vec<String>,
337    /// Direct status counts across all plan nodes.
338    #[serde(default)]
339    pub status_counts: SessionPlanStatusCounts,
340    /// Current typed node lifecycle requirements.
341    #[serde(default)]
342    pub gates: Vec<SessionPlanGate>,
343    /// Current exact evidence associations.
344    #[serde(default)]
345    pub evidence: Vec<SessionPlanEvidenceAssociation>,
346    /// Current concise node summaries and exact sources.
347    #[serde(default)]
348    pub summaries: Vec<SessionPlanSummary>,
349    /// Branch authorized to coordinate automatic plan advancement.
350    #[serde(default)]
351    pub coordinator: Option<String>,
352    /// Stable plan-node identity currently being coordinated.
353    #[serde(default)]
354    pub coordinator_cursor: Option<String>,
355    /// Unique coordinator operation claim.
356    #[serde(default)]
357    pub coordinator_claim: Option<SessionCoordinatorClaim>,
358    /// Immutable earlier plan revisions.
359    #[serde(default)]
360    pub history: Vec<SessionPlanRevision>,

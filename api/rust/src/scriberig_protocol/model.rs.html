6/// Creates a collision-free presentation id for one provider/model pair.
7///
8/// The provider and model ids remain the authoritative selection fields. This
9/// encoded value exists only for consumers that require a single row key.
10///
11/// # Parameters
12/// - `provider_id`: Exact upstream provider identity.
13/// - `model_id`: Provider-scoped upstream model identity.
14///
15/// # Returns
16/// A versioned `rpi1` id using each identity's UTF-8 byte length.
17#[must_use]
18pub fn model_row_id(provider_id: &str, model_id: &str) -> String {
19    format!(
20        "rpi1:{}:{}{}:{}",
21        provider_id.len(),
22        provider_id,
23        model_id.len(),
24        model_id
25    )
26}

38/// Carries typed pricing metadata from the upstream safe snapshot.
39#[derive(Clone, Debug, Deserialize, JsonSchema, PartialEq, Serialize)]
40pub struct ModelCostSnapshot {
41    /// Cost per input token unit.
42    pub input: f64,
43    /// Cost per output token unit.
44    pub output: f64,
45    /// Cost per cached input token unit read.
46    pub cache_read: f64,
47    /// Cost per cached input token unit written.
48    pub cache_write: f64,
49    /// Optional pricing tiers for larger inputs.
50    #[serde(default, skip_serializing_if = "Vec::is_empty")]
51    pub tiers: Vec<ModelCostTierSnapshot>,
52}

107/// Reports credential readiness without exposing credential source values.
108#[derive(Clone, Copy, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
109#[serde(rename_all = "snake_case")]
110pub enum ProviderCredentialReadiness {
111    /// The upstream provider declaration allows unauthenticated requests.
112    NotRequired,
113    /// A host-authorized credential is available for the configured endpoint.
114    Available,
115    /// No usable credential is configured for this provider.
116    Missing,
117    /// Host authorization denied or failed to resolve the configured secret.
118    Denied,
119    /// The configured credential source is not supported by this host path.
120    Unsupported,
121    /// Upstream catalog configuration supplies an authentication header.
122    CatalogHeader,
123}

125/// Associates a provider identity with its safe credential readiness state.
126#[derive(Clone, Debug, Deserialize, Eq, JsonSchema, PartialEq, Serialize)]
127pub struct ProviderReadiness {
128    /// Upstream provider identity.
129    pub provider_id: String,
130    /// Provider-native model identity whose endpoint was checked.
131    pub model_id: String,
132    /// Safe credential status derived from upstream and host authorization.
133    pub credential: ProviderCredentialReadiness,
134}

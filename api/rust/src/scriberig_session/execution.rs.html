5/// Session-scoped model configuration for future invocations.
6#[derive(Clone, Debug, Default, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
7pub struct SessionExecutionSelection {
8    /// Monotonic mutation revision, independent from event cursors.
9    pub revision: u64,
10    /// Selected curated model-profile identity.
11    pub model_profile: Option<String>,
12    /// Selected provider-neutral model identity.
13    pub model: Option<String>,
14    /// Selected model-advertised thinking-level identity.
15    pub thinking_level: Option<String>,
16}
17
18impl SessionExecutionSelection {
19    /// Replaces the future-invocation selection using optimistic concurrency.
20    ///
21    /// # Parameters
22    /// - `self`: Current session selection.
23    /// - `expected_revision`: Revision observed by the mutating client.
24    /// - `model_profile`: Profile identity, when a model is selected.
25    /// - `model`: Provider-neutral model identity, when selected.
26    /// - `thinking_level`: Supported thinking level, when selected.
27    ///
28    /// # Returns
29    /// The updated selection with its next mutation revision.
30    ///
31    /// # Errors
32    /// Returns a conflict for stale revisions or inconsistent absent values.
33    pub fn replace(
34        &mut self,
35        expected_revision: u64,
36        model_profile: Option<String>,
37        model: Option<String>,
38        thinking_level: Option<String>,
39    ) -> Result<(), SelectionError> {
40        if expected_revision != self.revision {
41            return Err(SelectionError::RevisionConflict {
42                expected: expected_revision,
43                actual: self.revision,
44            });
45        }
46        if model_profile.is_some() != model.is_some()
47            || (model.is_none() && thinking_level.is_some())
48        {
49            return Err(SelectionError::InconsistentSelection);
50        }
51        self.revision = self
52            .revision
53            .checked_add(1)
54            .ok_or(SelectionError::RevisionExhausted)?;
55        self.model_profile = model_profile;
56        self.model = model;
57        self.thinking_level = thinking_level;
58        Ok(())
59    }
60}

62/// Describes optimistic-concurrency failures for execution selection.
63#[derive(Clone, Copy, Debug, Eq, PartialEq)]
64pub enum SelectionError {
65    /// The caller's observed revision is stale.
66    RevisionConflict { expected: u64, actual: u64 },
67    /// Model and profile must be set or cleared together.
68    InconsistentSelection,
69    /// The selection revision has no representable successor.
70    RevisionExhausted,
71}

73/// Describes how a model result interacted with the derived-result cache.
74#[derive(Clone, Copy, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
75pub enum CacheDisposition {
76    /// No cache lookup was performed.
77    NotUsed,
78    /// A matching durable result was reused.
79    Hit,
80    /// A lookup was performed but did not produce a reusable result.
81    Miss,
82}

84/// Records non-secret provenance for one model execution.
85#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
86pub struct ModelInvocation {
87    /// Stable invocation identity.
88    pub id: EntityId,
89    /// Branch owning the execution.
90    pub branch: EntityId,
91    /// Explicit activity order within the branch.
92    pub ordinal: u64,
93    /// Stable agent profile reference.
94    pub agent_profile: Option<String>,
95    /// Identity of the context supplied to the model.
96    pub context_fingerprint: Option<String>,
97    /// Safe display identifier for the provider profile.
98    pub provider_profile: Option<String>,
99    /// Safe model identifier.
100    pub model: Option<String>,
101    /// Curated model profile selected for this invocation.
102    #[serde(default)]
103    pub model_profile: Option<String>,
104    /// Provider-native model identifier used by the adapter.
105    #[serde(default)]
106    pub provider_model_id: Option<String>,
107    /// Model-advertised thinking level resolved for this invocation.
108    #[serde(default)]
109    pub thinking_level: Option<String>,
110    /// Fingerprint of the canonical request, never the request secret.
111    pub request_fingerprint: Option<String>,
112    /// Cache behavior for this invocation.
113    pub cache: CacheDisposition,
114    /// Result reference when execution completed.
115    pub output: Option<EntityId>,
116    /// Bounded provider-neutral request captured for later inspection.
117    #[serde(default)]
118    pub request: Option<ModelRequestProjection>,
119}

175/// Identifies one ordered execution activity in a branch.
176#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
177pub enum ExecutionActivity {
178    /// Human input recorded on a branch turn.
179    HumanInput(EntityId),
180    /// Model invocation activity.
181    ModelInvocation(EntityId),
182    /// Model output activity.
183    ModelOutput(EntityId),
184    /// Tool invocation activity.
185    ToolInvocation(EntityId),
186    /// Tool result activity.
187    ToolResult(EntityId),
188    /// Child branch start or result reference.
189    ChildBranch(EntityId),
190    /// Child branch linked to the tool invocation that spawned it.
191    ChildBranchSpawned {
192        /// Spawned child branch.
193        branch: EntityId,
194        /// Parent branch tool invocation that requested the child.
195        parent_tool_invocation: EntityId,
196    },
197    /// Durable result committed for the branch.
198    BranchResult(EntityId),
199    /// Agent output activity.
200    AgentOutput(EntityId),

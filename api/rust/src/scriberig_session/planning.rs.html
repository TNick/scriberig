5/// Represents lifecycle state for a durable plan node.
6#[derive(Clone, Copy, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
7pub enum PlanNodeStatus {
8    /// Work is waiting for required dependency results.
9    Planned,
10    /// All declared dependency results are available.
11    Ready,
12    /// A worker currently owns the node's branch.
13    Running,
14    /// Work completed and produced an explicit result.
15    Completed,
16    /// Work failed without satisfying downstream dependencies.
17    Failed,
18    /// Work was explicitly cancelled or abandoned.
19    Abandoned,
20}

22/// Represents the lifecycle of a durable plan.
23#[derive(Clone, Copy, Debug, Default, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
24pub enum PlanStatus {
25    /// Plan is being authored and cannot auto advance.
26    #[default]
27    Draft,
28    /// Valid plan is ready to start.
29    Planned,
30    /// Plan coordinator may dispatch eligible work.
31    Running,
32    /// Automatic dispatch is paused.
33    Paused,
34    /// All required nodes completed successfully.
35    Completed,
36    /// Plan ended after an unrecoverable failure.
37    Failed,
38    /// Plan was explicitly stopped.
39    Cancelled,
40}

42/// Identifies the separate structural parent relation of a plan node.
43#[derive(Clone, Copy, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
44pub enum PlanNodeParent {
45    /// Node is structurally under a session branch.
46    Branch(EntityId),
47    /// Node is structurally under another plan node.
48    Node(EntityId),
49}

60/// Selects an upstream provider and model for one plan node.
61///
62/// Provider credentials are deliberately excluded. The provider and model IDs
63/// are resolved by upstream runtime machinery at execution.
64#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
65pub struct ModelAssignment {
66    /// Upstream provider identity.
67    pub provider_id: String,
68    /// Upstream provider-scoped model identity.
69    pub model_id: String,
70    /// Optional advertised thinking-level identity.
71    pub thinking_level: Option<String>,
72}

74/// Describes one durable planned unit located in a session tree.
75#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
76pub struct PlanNode {
77    /// Stable plan-node identity.
78    pub id: EntityId,
79    /// Branch or parent plan node that owns this structural child.
80    pub parent: Option<PlanNodeParent>,
81    /// Human-readable work title.
82    pub title: String,
83    /// Focused work instruction.
84    pub purpose: String,
85    /// Required result contract for dependency fulfillment.
86    #[serde(default)]
87    pub expected_result: Option<ExpectedResult>,
88    /// Exact semantic handles limiting or informing the work.
89    #[serde(default)]
90    pub semantic_scope: Vec<String>,
91    /// Current durable scheduling state.
92    pub status: PlanNodeStatus,
93    /// Executing branch once the planned work has started.
94    pub branch: Option<EntityId>,
95    /// Explicit result that can fulfill dependent work.
96    pub result: Option<EntityId>,
97    /// Replacement node identity when this work was explicitly superseded.
98    #[serde(default)]
99    pub superseded_by: Option<EntityId>,
100    /// Explicit safe reason when the node was abandoned.
101    #[serde(default)]
102    pub abandon_reason: Option<PlanAbandonReason>,
103    /// Exact failure evidence for the most recent failed attempt.
104    #[serde(default)]
105    pub last_failure: Option<PlanEvidenceReference>,
106    /// Independent execution configuration for this work.
107    pub assignment: Option<ModelAssignment>,
108    /// Active worker ownership when this node is running.
109    #[serde(default)]
110    pub claim: Option<WorkerClaim>,
111    /// Last worker claim generation, retained across release/recovery.
112    #[serde(default)]
113    pub claim_generation: u64,
114}

148/// Declares that one node requires another node's explicit durable result.

159/// Holds plan nodes and a separate acyclic dependency graph.
160#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
161pub struct Plan {
162    /// Stable plan identity.
163    pub id: EntityId,
164    /// Exact canonical snapshot from which this projection was loaded.
165    #[serde(skip)]
166    pub source_snapshot: Option<SnapshotId>,
167    /// Owning session identity.
168    #[serde(default)]
169    pub session: Option<EntityId>,
170    /// Exact conversation location where the plan was created.
171    #[serde(default)]
172    pub anchor: Option<crate::ContinuationAnchor>,
173    /// Durable plan goal.
174    #[serde(default)]
175    pub goal: String,
176    /// Mutation revision, independent of activity sequence.
177    #[serde(default)]
178    pub revision: u64,
179    /// Plan lifecycle.
180    #[serde(default)]
181    pub status: PlanStatus,
182    /// Root session branch from which the plan was created.
183    pub root_branch: EntityId,
184    /// Durable plan nodes in stable insertion order.
185    pub nodes: Vec<PlanNode>,
186    /// Dependency edges, separate from structural parent relationships.
187    pub dependencies: Vec<PlanDependency>,
188    /// Current node summaries stored as independent semantic objects.
189    #[serde(default)]
190    pub summaries: Vec<PlanNodeSummary>,
191    /// Exact evidence associations for nodes and gates.
192    #[serde(default)]
193    pub evidence: Vec<PlanEvidenceAssociation>,
194    /// Independent typed requirements constraining node entry/completion.
195    #[serde(default)]
196    pub gates: Vec<PlanGate>,
197    /// Branch allowed to coordinate automatic plan advancement.
198    pub coordinator: Option<EntityId>,
199    /// Stable node identity where the coordinator is currently focused.
200    pub coordinator_cursor: Option<EntityId>,
201    /// The one active auto-advancing runtime claim, if any.
202    #[serde(default)]
203    pub coordinator_claim: Option<CoordinatorClaim>,
204    /// Last coordinator claim generation, retained across release/recovery.
205    #[serde(default)]
206    pub coordinator_generation: u64,
207    /// Plan node identities at the top of the structural plan tree.
208    #[serde(default)]
209    pub root_nodes: Vec<EntityId>,
210    /// Immutable inspection history of earlier plan revisions.
211    #[serde(default)]
212    pub history: Vec<PlanRevision>,
213}

242/// Read-only execution attempt reconstructed from immutable plan history.
243///
244/// # Public members
245/// - `branch`: Branch used for this node attempt.
246/// - `claim_generation`: Monotonic node-local attempt generation.
247/// - `claim`: Worker claim associated with the attempt, when retained.
248/// - `status`: Latest recorded lifecycle outcome for the attempt.
249/// - `result`: Accepted result reference, when the attempt completed.
250/// - `failure`: Exact failed result reference, when a failure was recorded.
251///
252/// # Private members
253/// None.
254#[derive(Clone, Debug, Eq, PartialEq)]
255pub struct PlanAttemptView {
256    /// Branch used for this node attempt.
257    pub branch: EntityId,
258    /// Monotonic node-local attempt generation.
259    pub claim_generation: u64,
260    /// Worker claim associated with the attempt, when retained.
261    pub claim: Option<EntityId>,
262    /// Latest recorded lifecycle outcome for the attempt.
263    pub status: PlanNodeStatus,
264    /// Accepted result reference, when the attempt completed.
265    pub result: Option<EntityId>,
266    /// Exact failed result reference, when a failure was recorded.
267    pub failure: Option<PlanEvidenceReference>,
268}

270/// Reports invalid plan graph operations.
271#[derive(Clone, Copy, Debug, Eq, PartialEq)]
272pub enum PlanError {
273    /// The plan node identity already exists.
274    DuplicateNode,
275    /// A referenced plan node does not exist.
276    MissingNode,
277    /// A prerequisite edge would create a dependency cycle.
278    DependencyCycle,
279    /// A node cannot depend on itself.
280    SelfDependency,
281    /// A result does not match the prerequisite's durable result.
282    ResultMismatch,
283    /// A plan node is not in a state that permits the requested operation.
284    InvalidTransition,
285    /// Plan is terminal and cannot accept lifecycle changes.
286    TerminalPlan,
287    /// Plan revision has no representable successor.
288    RevisionExhausted,
289    /// Structural parent relations contain a cycle.
290    StructuralCycle,
291    /// The derived rollup index does not match the requested base snapshot.
292    StaleRollupIndex,
293}

15/// Defines the durable session repository boundary.
16pub mod repository {
17    use std::collections::BTreeMap;

19    use crate::{Plan, PlanSemanticAttempt, PlanSemanticNodeDetails, PlanView, Session};

22    /// Reports durable execution-selection update failures.
23    #[derive(Clone, Debug, Eq, PartialEq)]
24    pub enum SelectionStoreError {
25        /// The stored selection has moved past the caller's observed revision.
26        RevisionConflict { expected: u64, actual: u64 },
27        /// The repository could not perform an atomic update.
28        Persistence(String),
29    }
30
31    /// Reports compare-and-swap plan update failures.
32    #[derive(Clone, Debug, Eq, PartialEq)]
33    pub enum PlanStoreError {
34        /// The stored plan has moved past the caller's observed revision.
35        RevisionConflict { expected: u64, actual: u64 },
36        /// The repository could not perform an atomic update.
37        Persistence(String),
38    }

40    /// Persists provider-neutral session graphs without depending on a model runtime.
41    pub trait SessionStore: Send + Sync {
42        /// Lists durable session graphs in deterministic identity order.
43        ///
44        /// The list is a snapshot of records visible to the backend when the
45        /// method is called. Callers should retain exact session identities
46        /// and load an individual graph before editing it.
47        ///
48        /// Complete stored session graphs in deterministic order.
49        ///
50        /// Returns a bounded backend diagnostic when records cannot be read.
51        /// # Returns
52        /// The operation result.
53        /// # Errors
54        /// Returns an error when the repository cannot read its records.
55        fn list(&self) -> Result<Vec<Session>, String>;
56
57        /// Loads one exact session graph when it exists.
58        ///
59        /// - `session`: Stable session identity.
60        ///
61        /// The stored session, or `None` when no record exists.
62        ///
63        /// Returns a backend diagnostic that is safe for the application layer.
64        /// # Returns
65        /// The operation result.
66        /// # Errors
67        /// Returns an error when the operation fails.
68        fn load(&self, session: EntityId) -> Result<Option<Session>, String>;
69
70        /// Loads one plan from an exact snapshot in that plan's history.
71        ///
72        /// - `_session`: Owning durable session identity.
73        /// - `_plan`: Stable plan identity.
74        /// - `_snapshot`: Exact semantic snapshot to resolve.
75        ///
76        /// The plan projection or `None` when the snapshot is not in the
77        /// selected plan's retained history.
78        /// # Returns
79        /// The operation result.
80        /// # Errors
81        /// Returns a bounded persistence diagnostic when the snapshot cannot
82        /// be read or decoded.
83        fn load_plan_at_snapshot(
84            &self,
85            _session: EntityId,
86            _plan: EntityId,
87            _snapshot: SnapshotId,
88        ) -> Result<Option<Plan>, String> {
89            Err("exact plan snapshot reads are unsupported".to_owned())
90        }
91
92        /// Loads compact plan metadata at an exact retained snapshot.
93        ///
94        /// - `_session`: Owning durable session identity.
95        /// - `_plan`: Stable plan identity.
96        /// - `_snapshot`: Exact semantic snapshot to resolve.
97        ///
98        /// The compact plan projection or `None` when the snapshot is outside
99        /// the selected plan's retained history.
100        /// # Returns
101        /// The operation result.
102        /// # Errors
103        /// Returns a bounded persistence diagnostic when the view cannot be
104        /// read or assembled.
105        fn load_plan_view_at_snapshot(
106            &self,
107            _session: EntityId,
108            _plan: EntityId,
109            _snapshot: SnapshotId,
110        ) -> Result<Option<PlanView>, String> {
111            Err("compact plan snapshot reads are unsupported".to_owned())
112        }
113
114        /// Loads exact-snapshot node rollups, evaluating canonical state by default.
115        ///
116        /// - `session`: Owning durable session identity.
117        /// - `plan`: Stable plan identity.
118        /// - `snapshot`: Exact semantic snapshot to resolve.
119        ///
120        /// All node rollups, or `None` when the snapshot is outside plan history.
121        /// # Returns
122        /// Snapshot-bound effective status and subtree aggregates.
123        /// # Errors
124        /// Returns a bounded diagnostic when the plan or its rollups cannot be read.
125        fn load_plan_rollups_at_snapshot(
126            &self,
127            session: EntityId,
128            plan: EntityId,
129            snapshot: SnapshotId,
130        ) -> Result<Option<BTreeMap<EntityId, crate::PlanNodeRollup>>, String> {
131            self.load_plan_at_snapshot(session, plan, snapshot)?
132                .map(|value| {
133                    value
134                        .node_rollups()
135                        .map_err(|error| format!("plan rollup evaluation failed: {error:?}"))
136                })
137                .transpose()
138        }
139
140        /// Loads one page of worker attempts for a node at an exact snapshot.
141        ///
142        /// Attempt records are read independently from node evidence and gate
143        /// detail so callers can request history only when needed.
144        ///
145        /// - `_session`: Owning durable session identity.
146        /// - `_plan`: Stable plan identity.
147        /// - `_node`: Stable plan-node identity.
148        /// - `_snapshot`: Exact semantic snapshot to resolve.
149        /// - `_offset`: Number of matching attempts to skip in stable order.
150        /// - `_limit`: Maximum number of attempts to return.
151        ///
152        /// The requested page, or `None` when the snapshot is outside the
153        /// selected plan's retained history.
154        /// # Returns
155        /// The operation result.
156        /// # Errors
157        /// Returns a bounded persistence diagnostic when attempt records cannot
158        /// be read or decoded.
159        fn load_plan_attempts_at_snapshot(
160            &self,
161            _session: EntityId,
162            _plan: EntityId,
163            _node: EntityId,
164            _snapshot: SnapshotId,
165            _offset: usize,
166            _limit: usize,
167        ) -> Result<Option<Vec<PlanSemanticAttempt>>, String> {
168            Err("plan attempt history reads are unsupported".to_owned())
169        }
170
171        /// Loads one page of gate and evidence details for a node.
172        ///
173        /// - `_session`: Owning durable session identity.
174        /// - `_plan`: Stable plan identity.
175        /// - `_node`: Stable plan-node identity.
176        /// - `_snapshot`: Exact semantic snapshot to resolve.
177        /// - `_offset`: Number of records to skip in each category.
178        /// - `_limit`: Maximum records to return in each category.
179        ///
180        /// The requested details, or `None` when the snapshot is outside the
181        /// selected plan's retained history.
182        /// # Returns
183        /// The operation result.
184        /// # Errors
185        /// Returns a persistence diagnostic when records cannot be read.
186        fn load_plan_node_details_at_snapshot(
187            &self,
188            _session: EntityId,
189            _plan: EntityId,
190            _node: EntityId,
191            _snapshot: SnapshotId,
192            _offset: usize,
193            _limit: usize,
194        ) -> Result<Option<PlanSemanticNodeDetails>, String> {
195            Err("plan node detail reads are unsupported".to_owned())
196        }
197
198        /// Removes one exact durable session graph.
199        ///
200        /// - `_session`: Stable session identity to remove.
201        ///
202        /// Unit after the session record is removed.
203        ///
204        /// Returns a backend diagnostic when removal is unavailable or fails.
205        /// # Returns
206        /// The operation result.
207        /// # Errors
208        /// Returns an error when the repository cannot remove the record.
209        fn remove(&self, _session: EntityId) -> Result<(), String> {
210            Err("session removal is unsupported".to_owned())
211        }
212
213        /// Stores one complete session graph atomically.
214        ///
215        /// - `session`: Session graph to persist.
216        ///
217        /// Returns a backend diagnostic when persistence fails.
218        ///
219        /// Unit after the session is persisted.
220        /// # Returns
221        /// The operation result.
222        /// # Errors
223        /// Returns an error when the operation fails.
224        fn save(&self, session: &Session) -> Result<(), String>;
225
226        /// Atomically updates only execution selection at an observed revision.
227        ///
228        /// - `_session`: Session carrying the proposed new selection.
229        /// - `_expected_revision`: Selection revision observed by the caller.
230        ///
231        /// Unit after the selection is committed.
232        ///
233        /// Returns a revision conflict or a repository diagnostic.
234        /// # Returns
235        /// The operation result.
236        /// # Errors
237        /// Returns an error when the operation fails.
238        fn compare_and_swap_execution_selection(
239            &self,
240            _session: &Session,
241            _expected_revision: u64,
242        ) -> Result<(), SelectionStoreError> {
243            Err(SelectionStoreError::Persistence(
244                "atomic session selection updates are unsupported".to_owned(),
245            ))
246        }
247
248        /// Atomically replaces one plan at its observed revision.
249        ///
250        /// The repository must preserve unrelated session and branch updates.
251        /// New branch records are merged and linked to their parents. Existing
252        /// branches are updated only when a prior active worker claim settles;
253        /// that result-bearing branch and the plan revision commit together.
254        ///
255        /// - `_session`: Snapshot containing the proposed plan.
256        /// - `_plan`: Stable plan identity being replaced.
257        /// - `_expected_revision`: Plan revision observed by the caller.
258        ///
259        /// Returns a revision conflict or a repository diagnostic.
260        /// # Returns
261        /// The operation result.
262        /// # Errors
263        /// Returns an error when the operation fails.
264        fn compare_and_swap_plan(
265            &self,
266            _session: &Session,
267            _plan: EntityId,
268            _expected_revision: u64,
269        ) -> Result<(), PlanStoreError> {
270            Err(PlanStoreError::Persistence(
271                "atomic plan updates are unsupported".to_owned(),
272            ))
273        }
274    }
275}

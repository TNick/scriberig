13/// Counts gates by their deterministic evaluated state.
14///
15/// # Public members
16/// - `pending`: Gates awaiting evidence or child completion.
17/// - `satisfied`: Gates satisfied by current canonical state.
18/// - `failed`: Gates blocked by contradicting or failed evidence.
19/// - `waived`: Gates explicitly waived with retained provenance.
20///
21/// # Private members
22/// None.
23#[derive(Clone, Copy, Debug, Default, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
24pub struct PlanGateCounts {
25    /// Gates awaiting evidence or child completion.
26    pub pending: u32,
27    /// Gates satisfied by current canonical state.
28    pub satisfied: u32,
29    /// Gates blocked by contradicting or failed evidence.
30    pub failed: u32,
31    /// Gates explicitly waived with retained provenance.
32    pub waived: u32,
33}

154impl Plan {
155    /// Builds a compact shared plan projection from one exact content revision.
156    ///
157    /// # Parameters
158    /// - `self`: Canonical plan projection.
159    /// - `snapshot`: Exact universal snapshot, when known by the caller.
160    ///
161    /// A bounded plan view without per-node evidence/history details.
162    ///
163    /// Returns a serialization error when a canonical revision cannot be encoded.
164    /// # Returns
165    /// The operation result.
166    /// # Errors
167    /// Returns an error when the operation fails.
168    pub fn plan_view(&self, snapshot: Option<SnapshotId>) -> Result<PlanView, PlanSemanticError> {
169        let semantic = PlanSemanticSnapshot::from_plan(self)?;
170        let plan_revision = semantic
171            .objects
172            .iter()
173            .find(|object| object.entity == self.id)
174            .ok_or_else(|| invalid_view("plan revision is missing"))?;
175        let mut status_counts = PlanNodeStatusCounts::default();
176        for node in &self.nodes {
177            increment_count(&mut status_counts, node.status);
178        }
179        Ok(PlanView {
180            id: self.id,
181            revision: plan_revision.revision,
182            snapshot: snapshot.or(self.source_snapshot),
183            mutation_revision: self.revision,
184            goal: self.goal.clone(),
185            status: self.status,
186            coordinator: self.coordinator,
187            coordinator_cursor: self.coordinator_cursor,
188            root_nodes: self.root_nodes.clone(),
189            node_count: count_u32(self.nodes.len()),
190            status_counts,
191        })
192    }
193
194    /// Builds one detail projection using exact current results and rollups.
195    ///
196    /// # Parameters
197    /// - `self`: Canonical plan projection.
198    /// - `node_id`: Stable `PlanNode` identity to inspect.
199    /// - `snapshot`: Exact universal snapshot, when known by the caller.
200    /// - `results`: Branch results visible in the same session snapshot, when available.
201    /// - `rollups`: Derived rollups computed for the same canonical state.
202    ///
203    /// A node view with bounded summary and gate/evidence counts.
204    ///
205    /// Returns an error for unknown nodes, invalid structure, or encoding failure.
206    /// # Returns
207    /// The operation result.
208    /// # Errors
209    /// Returns an error when the operation fails.
210    pub fn node_view(
211        &self,
212        node_id: EntityId,
213        snapshot: Option<SnapshotId>,
214        results: Option<&[crate::BranchResult]>,
215        rollups: &BTreeMap<EntityId, PlanNodeRollup>,
216    ) -> Result<PlanNodeView, PlanSemanticError> {
217        let node = self
218            .nodes
219            .iter()
220            .find(|candidate| candidate.id == node_id)
221            .ok_or_else(|| invalid_view("plan node is missing"))?;
222        let semantic = PlanSemanticSnapshot::from_plan(self)?;
223        let revision = semantic
224            .objects
225            .iter()
226            .find(|object| object.entity == node_id)
227            .ok_or_else(|| invalid_view("plan node revision is missing"))?
228            .revision;
229        let rollup = rollups
230            .get(&node_id)
231            .ok_or_else(|| invalid_view("plan node rollup is missing"))?;
232        let incoming = self
233            .dependencies
234            .iter()
235            .filter(|edge| edge.dependent == node_id)
236            .collect::<Vec<_>>();
237        let dependency_blockers = incoming
238            .iter()
239            .filter(|edge| match results {
240                Some(results) => !self.dependency_satisfied(edge, results),
241                None => edge.fulfilled_by.is_none(),
242            })
243            .map(|edge| edge.prerequisite)
244            .collect();
245        let entry_gates = self.gate_counts(node_id, PlanGatePhase::Entry);
246        let completion_gates = self.gate_counts(node_id, PlanGatePhase::Completion);
247        let summary = self.summaries.iter().find(|item| item.node == node_id);
248        let evidence_count = self
249            .evidence
250            .iter()
251            .filter(|association| match association.owner {
252                super::PlanEvidenceOwner::Node(owner) => owner == node_id,
253                super::PlanEvidenceOwner::Gate(gate) => self
254                    .gates
255                    .iter()
256                    .any(|item| item.id == gate && item.owner == node_id),
257            })
258            .count();
259        Ok(PlanNodeView {
260            id: node.id,
261            revision,
262            snapshot: snapshot.or(self.source_snapshot),
263            plan: self.id,
264            parent: node.parent,
265            title: node.title.clone(),
266            purpose: node.purpose.clone(),
267            direct_status: node.status,
268            effective_status: rollup.effective_status,
269            contains_ignored: rollup.contains_ignored,
270            contains_abandoned: rollup.contains_abandoned,
271            summary_id: summary.map(|item| item.id),
272            summary_preview: summary.map(|item| truncate_summary(&item.text)),
273            branch: node.branch,
274            result: node.result,
275            assignment: node.assignment.clone(),
276            dependency_blockers,
277            entry_gates,
278            completion_gates,
279            evidence_count: count_u32(evidence_count),
280            can_retry: node.status == PlanNodeStatus::Failed
281                && node.claim.is_none()
282                && node.branch.is_some()
283                && matches!(self.status, PlanStatus::Running | PlanStatus::Paused)
284                && results.is_some_and(|results| self.dependencies_satisfied(node_id, results))
285                && self.node_gates_satisfied(node_id, PlanGatePhase::Entry),
286            last_failure: node.last_failure.clone(),
287        })
288    }

290    /// Counts gates attached to a node by their current evaluated state.
291    ///
292    /// # Parameters
293    /// - `self`: Canonical plan with gate definitions and evidence.
294    /// - `node`: Stable owner node identity.
295    /// - `phase`: Entry or completion gate phase.
296    ///
297    /// Stable counts by gate result.
298    /// # Returns
299    /// Counts of gates in each evaluated state.
300    fn gate_counts(&self, node: EntityId, phase: PlanGatePhase) -> PlanGateCounts {
301        let mut counts = PlanGateCounts::default();
302        for evaluation in self.evaluate_node_gates(node, phase) {
303            match evaluation.status {
304                PlanGateStatus::Pending => counts.pending = counts.pending.saturating_add(1),
305                PlanGateStatus::Satisfied => {
306                    counts.satisfied = counts.satisfied.saturating_add(1);
307                }
308                PlanGateStatus::Failed => counts.failed = counts.failed.saturating_add(1),
309                PlanGateStatus::Waived => counts.waived = counts.waived.saturating_add(1),
310            }
311        }
312        counts
313    }
314}

316/// Counts one direct node state without overflowing the compact view.
317///
318/// # Parameters
319/// - `counts`: Direct status accumulator.
320/// - `status`: Canonical `PlanNode` lifecycle status.
321fn increment_count(counts: &mut PlanNodeStatusCounts, status: PlanNodeStatus) {
322    let count = match status {
323        PlanNodeStatus::Planned => &mut counts.planned,
324        PlanNodeStatus::Ready => &mut counts.ready,
325        PlanNodeStatus::Running => &mut counts.running,
326        PlanNodeStatus::Completed => &mut counts.completed,
327        PlanNodeStatus::Failed => &mut counts.failed,
328        PlanNodeStatus::Abandoned => &mut counts.abandoned,
329    };
330    *count = count.saturating_add(1);
331}

333/// Converts an unbounded collection length to a bounded view count.
334///
335/// # Parameters
336/// - `count`: Collection size to convert.
337///
338/// The value clamped to the public `u32` range.
339/// # Returns
340/// The bounded count.
341fn count_u32(count: usize) -> u32 {
342    u32::try_from(count).unwrap_or(u32::MAX)
343}

345/// Bounds a summary preview at a Unicode character boundary.
346///
347/// # Parameters
348/// - `summary`: Full safe summary text.
349///
350/// At most 240 Unicode scalar values with an ellipsis when truncated.
351/// # Returns
352/// The bounded summary preview.
353fn truncate_summary(summary: &str) -> String {
354    let mut chars = summary.chars();
355    let preview = chars.by_ref().take(240).collect::<String>();
356    if chars.next().is_some() {
357        format!("{preview}…")
358    } else {
359        preview
360    }
361}

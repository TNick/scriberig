1//! Implements one responsibility of durable plan lifecycle behavior.

3use super::{EntityId, Plan, PlanError, PlanNodeStatus, PlanStatus};

5impl Plan {
6    /// Completes a running node with its committed durable branch result.
7    ///
8    /// This operation settles only the node claim. Plan-wide completion is
9    /// evaluated separately against a complete session result snapshot.
10    ///
11    /// - `self`: Plan containing the node.
12    /// - `node`: Running node to complete.
13    /// - `result`: Durable result committed on the executing branch.
14    ///
15    /// Returns `ResultMismatch` if the durable result does not satisfy the
16    /// node contract.
17    /// # Parameters
18    /// - `claim_id`: Input for this operation.
19    /// # Returns
20    /// The operation result.
21    /// # Errors
22    /// Returns an error when the operation fails.
23    pub fn complete_node(
24        &mut self,
25        node: EntityId,
26        claim_id: EntityId,
27        result: &crate::BranchResult,
28    ) -> Result<(), PlanError> {
29        self.settle_workers(&[(node, claim_id, result.clone())], &[], &[])
30    }
31
32    /// Settles a concurrent worker batch as one durable plan revision.
33    ///
34    /// Successful results fulfill explicit dependency edges. Failed workers
35    /// release their claims while leaving dependents blocked. The plan becomes
36    /// complete in the same revision when every required result is present.
37    ///
38    /// - `self`: Plan containing the active worker claims.
39    /// - `completed`: Exact `(node, claim, result)` completion records.
40    /// - `failed`: Exact `(node, claim)` failed worker claims.
41    /// - `available_results`: Durable branch results used for plan completion.
42    ///
43    /// Returns a plan error without partial mutation for stale claims, result
44    /// mismatches, duplicate settlement entries, or revision exhaustion.
45    /// # Returns
46    /// The operation result.
47    /// # Errors
48    /// Returns an error when the operation fails.
49    pub fn settle_workers(
50        &mut self,
51        completed: &[(EntityId, EntityId, crate::BranchResult)],
52        failed: &[(EntityId, EntityId)],
53        available_results: &[crate::BranchResult],
54    ) -> Result<(), PlanError> {
55        self.ensure_mutable()?;
56        if completed.is_empty() && failed.is_empty() {
57            return Err(PlanError::InvalidTransition);
58        }
59        let mut proposed = self.clone();
60        let mut settled_nodes = std::collections::BTreeSet::new();
61        for (node_id, claim_id, result) in completed {
62            if !settled_nodes.insert(*node_id) {
63                return Err(PlanError::InvalidTransition);
64            }
65            let target = proposed
66                .nodes
67                .iter_mut()
68                .find(|node| node.id == *node_id)
69                .ok_or(PlanError::MissingNode)?;
70            if target.status != PlanNodeStatus::Running
71                || target.branch != Some(result.branch)
72                || target.claim.as_ref().map(|claim| claim.id) != Some(*claim_id)
73                || result.status != crate::BranchStatus::Completed
74            {
75                return Err(PlanError::InvalidTransition);
76            }
77            if target.expected_result.as_ref().is_some_and(|contract| {
78                result.result_kind.as_deref() != Some(contract.kind.as_str())
79                    || (contract.requires_handle
80                        && result.result_handles.is_empty()
81                        && result.references.is_empty())
82            }) {
83                return Err(PlanError::ResultMismatch);
84            }
85            target.result = Some(result.id);
86            target.status = PlanNodeStatus::Completed;
87            target.claim = None;
88            for edge in proposed
89                .dependencies
90                .iter_mut()
91                .filter(|edge| edge.prerequisite == *node_id)
92            {
93                edge.fulfilled_by = Some(result.id);
94            }
95        }
96        for (node_id, claim_id) in failed {
97            if !settled_nodes.insert(*node_id) {
98                return Err(PlanError::InvalidTransition);
99            }
100            let target = proposed
101                .nodes
102                .iter_mut()
103                .find(|node| node.id == *node_id)
104                .ok_or(PlanError::MissingNode)?;
105            if target.status != PlanNodeStatus::Running
106                || target.claim.as_ref().map(|claim| claim.id) != Some(*claim_id)
107            {
108                return Err(PlanError::InvalidTransition);
109            }
110            target.status = PlanNodeStatus::Failed;
111            target.claim = None;
112        }
113        proposed.refresh_readiness();
114        if proposed.all_nodes_satisfied(available_results) {
115            proposed.status = PlanStatus::Completed;
116            proposed.coordinator_claim = None;
117        }
118        proposed.bump_revision()?;
119        *self = proposed;
120        Ok(())
121    }
122
123    /// Fails an active worker claim without satisfying dependent nodes.
124    ///
125    /// - `self`: Plan containing the running node.
126    /// - `node`: Stable plan-node identity.
127    /// - `claim`: Exact active worker claim identity.
128    ///
129    /// Returns `InvalidTransition` when the node or claim is not active.
130    /// # Returns
131    /// The operation result.
132    /// # Errors
133    /// Returns an error when the operation fails.
134    pub fn fail_node(&mut self, node: EntityId, claim: EntityId) -> Result<(), PlanError> {
135        self.settle_workers(&[], &[(node, claim)], &[])
136    }

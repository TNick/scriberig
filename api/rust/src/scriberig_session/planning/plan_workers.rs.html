5impl Plan {
6    /// Lists worker branches settled between two consecutive plan revisions.
7    ///
8    /// Only prior active claims that now have an exact successful result or
9    /// failure reference are returned. The repository uses these identities
10    /// to commit branch outcomes with the semantic plan revision.
11    ///
12    /// - `self`: Proposed plan after worker settlement.
13    /// - `previous`: Exact plan revision containing the active claims.
14    ///
15    /// Branch and result identities for newly settled workers.
16    ///
17    /// # Returns
18    /// Settled `(branch, result)` pairs in stable plan-node order.
19    #[must_use]
20    pub fn settlement_result_refs_since(&self, previous: &Plan) -> Vec<(EntityId, EntityId)> {
21        if self.id != previous.id {
22            return Vec::new();
23        }
24        previous
25            .nodes
26            .iter()
27            .filter_map(|prior| {
28                if prior.status != PlanNodeStatus::Running {
29                    return None;
30                }
31                let claim = prior.claim.as_ref()?;
32                let settled = self.nodes.iter().find(|node| node.id == prior.id)?;
33                if settled.branch != Some(claim.branch) {
34                    return None;
35                }
36                let result = match settled.status {
37                    PlanNodeStatus::Completed => settled.result,
38                    PlanNodeStatus::Failed => settled
39                        .last_failure
40                        .as_ref()
41                        .filter(|failure| failure.kind == crate::PlanEvidenceKind::Failure)
42                        .map(|failure| failure.reference.entity),
43                    _ => None,
44                }?;
45                Some((claim.branch, result))
46            })
47            .collect()
48    }

50    /// Completes a running node with its committed durable branch result.
51    ///
52    /// This operation settles only the node claim. Plan-wide completion is
53    /// evaluated separately against a complete session result snapshot.
54    ///
55    /// - `self`: Plan containing the node.
56    /// - `node`: Running node to complete.
57    /// - `result`: Durable result committed on the executing branch.
58    ///
59    /// Returns `ResultMismatch` if the durable result does not satisfy the
60    /// node contract.
61    /// # Parameters
62    /// - `claim_id`: Input for this operation.
63    /// # Returns
64    /// The operation result.
65    /// # Errors
66    /// Returns an error when the operation fails.
67    pub fn complete_node(
68        &mut self,
69        node: EntityId,
70        claim_id: EntityId,
71        result: &crate::BranchResult,
72    ) -> Result<(), PlanError> {
73        self.settle_workers(&[(node, claim_id, result.clone())], &[], &[])
74    }

76    /// Settles a concurrent worker batch as one durable plan revision.
77    ///
78    /// Successful results fulfill explicit dependency edges. Failed workers
79    /// release their claims while leaving dependents blocked. The plan becomes
80    /// complete in the same revision when every required result is present.
81    ///
82    /// - `self`: Plan containing the active worker claims.
83    /// - `completed`: Exact `(node, claim, result)` completion records.
84    /// - `failed`: Exact `(node, claim)` failed worker claims.
85    /// - `available_results`: Durable branch results used for plan completion.
86    ///
87    /// Returns a plan error without partial mutation for stale claims, result
88    /// mismatches, duplicate settlement entries, or revision exhaustion.
89    /// # Returns
90    /// The operation result.
91    /// # Errors
92    /// Returns an error when the operation fails.
93    pub fn settle_workers(
94        &mut self,
95        completed: &[(EntityId, EntityId, crate::BranchResult)],
96        failed: &[(EntityId, EntityId, Option<crate::BranchResult>)],
97        available_results: &[crate::BranchResult],
98    ) -> Result<(), PlanError> {
99        self.ensure_mutable()?;
100        if completed.is_empty() && failed.is_empty() {
101            return Err(PlanError::InvalidTransition);
102        }
103        let mut proposed = self.clone();
104        let mut settled_nodes = std::collections::BTreeSet::new();
105        for (node_id, claim_id, result) in completed {
106            if !settled_nodes.insert(*node_id) {
107                return Err(PlanError::InvalidTransition);
108            }
109            let target = proposed
110                .nodes
111                .iter_mut()
112                .find(|node| node.id == *node_id)
113                .ok_or(PlanError::MissingNode)?;
114            if target.status != PlanNodeStatus::Running
115                || target.branch != Some(result.branch)
116                || target.claim.as_ref().map(|claim| claim.id) != Some(*claim_id)
117                || result.status != crate::BranchStatus::Completed
118            {
119                return Err(PlanError::InvalidTransition);
120            }
121            if target.expected_result.as_ref().is_some_and(|contract| {
122                result.result_kind.as_deref() != Some(contract.kind.as_str())
123                    || (contract.requires_handle
124                        && result.result_handles.is_empty()
125                        && result.references.is_empty())
126            }) {
127                return Err(PlanError::ResultMismatch);
128            }
129            target.result = Some(result.id);
130            target.status = PlanNodeStatus::Completed;
131            target.claim = None;
132            for edge in proposed
133                .dependencies
134                .iter_mut()
135                .filter(|edge| edge.prerequisite == *node_id)
136            {
137                edge.fulfilled_by = Some(result.id);
138            }
139        }
140        for (node_id, claim_id, failure_result) in failed {
141            if !settled_nodes.insert(*node_id) {
142                return Err(PlanError::InvalidTransition);
143            }
144            let target = proposed
145                .nodes
146                .iter_mut()
147                .find(|node| node.id == *node_id)
148                .ok_or(PlanError::MissingNode)?;
149            if target.status != PlanNodeStatus::Running
150                || target.claim.as_ref().map(|claim| claim.id) != Some(*claim_id)
151                || failure_result.as_ref().is_some_and(|result| {
152                    Some(result.branch) != target.branch
153                        || result.status == crate::BranchStatus::Completed
154                })
155            {
156                return Err(PlanError::InvalidTransition);
157            }
158            target.status = PlanNodeStatus::Failed;
159            target.claim = None;
160            if let Some(result) = failure_result {
161                target.last_failure = Some(crate::PlanEvidenceReference {
162                    kind: crate::PlanEvidenceKind::Failure,
163                    reference: crate::PlanSemanticReference {
164                        entity: result.id,
165                        revision: None,
166                        snapshot: None,
167                    },
168                    attributes: std::collections::BTreeMap::from([
169                        ("branch".to_owned(), result.branch.to_string()),
170                        ("status".to_owned(), "failed".to_owned()),
171                    ]),
172                });
173            }
174        }
175        proposed.refresh_readiness();
176        if proposed.all_nodes_satisfied(available_results) {
177            proposed.status = PlanStatus::Completed;
178            proposed.coordinator_claim = None;
179        }
180        proposed.bump_revision()?;
181        *self = proposed;
182        Ok(())
183    }

185    /// Fails an active worker claim without satisfying dependent nodes.
186    ///
187    /// - `self`: Plan containing the running node.
188    /// - `node`: Stable plan-node identity.
189    /// - `claim`: Exact active worker claim identity.
190    ///
191    /// Returns `InvalidTransition` when the node or claim is not active.
192    /// # Returns
193    /// The operation result.
194    /// # Errors
195    /// Returns an error when the operation fails.
196    pub fn fail_node(&mut self, node: EntityId, claim: EntityId) -> Result<(), PlanError> {
197        self.settle_workers(&[], &[(node, claim, None)], &[])
198    }

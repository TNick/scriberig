14/// Stages affected nodes and ancestor values, then commits only sparse changes.
15///
16/// # Parameters
17/// - `index`: Previously valid snapshot-bound index.
18/// - `plan`: New canonical plan projection.
19/// - `expected_base`: Snapshot from which the change was derived.
20/// - `new_snapshot`: Exact snapshot represented after the change.
21/// - `changed_nodes`: Nodes whose canonical state or owned requirements changed.
22///
23/// Node identities whose derived rollup value changed.
24///
25/// # Returns
26/// Changed derived rollups when the sparse update is valid and committed.
27///
28/// # Errors
29/// Returns a plan error for stale input, missing nodes, or structural cycles.
30pub(super) fn apply_change(
31    index: &mut PlanNodeRollupIndex,
32    plan: &Plan,
33    expected_base: SnapshotId,
34    new_snapshot: SnapshotId,
35    changed_nodes: &[EntityId],
36) -> Result<Vec<EntityId>, PlanError> {
37    if index.source_snapshot != expected_base || index.evaluator_version != 1 {
38        return Err(PlanError::StaleRollupIndex);
39    }
40
41    let changed = changed_nodes.iter().copied().collect::<BTreeSet<_>>();
42    if changed.len() != changed_nodes.len() {
43        return Err(PlanError::InvalidTransition);
44    }
45    let new_node_count = changed
46        .iter()
47        .filter(|node| !index.parents.contains_key(node))
48        .count();
49    if plan.nodes.len() != index.parents.len() + new_node_count {
50        return Err(PlanError::StaleRollupIndex);
51    }
52
53    let mut parent_updates = BTreeMap::new();
54    let mut children_updates = BTreeMap::new();
55    let mut affected_parents = BTreeSet::new();
56    for node_id in &changed {
57        let node = plan
58            .nodes
59            .iter()
60            .find(|node| node.id == *node_id)
61            .ok_or(PlanError::StaleRollupIndex)?;
62        let new_parent = super::node_parent(node.parent);
63        let old_parent = index.parents.get(node_id).copied();
64        if old_parent.flatten() != new_parent {
65            if let Some(parent) = old_parent.flatten() {
66                children_mut(index, &mut children_updates, parent).remove(node_id);
67                affected_parents.insert(parent);
68            }
69            if let Some(parent) = new_parent {
70                if !index.parents.contains_key(&parent) && !changed.contains(&parent) {
71                    return Err(PlanError::MissingNode);
72                }
73                children_mut(index, &mut children_updates, parent).insert(*node_id);
74                affected_parents.insert(parent);
75            }
76        }
77        parent_updates.insert(*node_id, new_parent);
78        children_updates
79            .entry(*node_id)
80            .or_insert_with(|| index.children.get(node_id).cloned().unwrap_or_default());
81    }
82
83    let mut depths = BTreeMap::new();
84    for node in changed.iter().chain(affected_parents.iter()) {
85        depths.insert(*node, node_depth(*node, &parent_updates, &index.parents)?);
86    }
87    let mut changed_order = changed.into_iter().collect::<Vec<_>>();
88    changed_order.sort_by_key(|node| std::cmp::Reverse(depths[node]));
89    let mut parent_order = affected_parents.into_iter().collect::<Vec<_>>();
90    parent_order.sort_by_key(|node| std::cmp::Reverse(depths[node]));
91
92    let mut rollup_updates = BTreeMap::new();
93    for node in changed_order {
94        propagate(
95            index,
96            plan,
97            node,
98            false,
99            &parent_updates,
100            &children_updates,
101            &mut rollup_updates,
102        )?;
103    }
104    for node in parent_order {
105        propagate(
106            index,
107            plan,
108            node,
109            true,
110            &parent_updates,
111            &children_updates,
112            &mut rollup_updates,
113        )?;
114    }
115
116    let changed_rollups = rollup_updates
117        .iter()
118        .filter_map(|(node, rollup)| (index.rollups.get(node) != Some(rollup)).then_some(*node))
119        .collect::<Vec<_>>();
120    index.parents.extend(parent_updates);
121    index.children.extend(children_updates);
122    index.rollups.extend(rollup_updates);
123    index.source_snapshot = new_snapshot;
124    Ok(changed_rollups)
125}

127/// Returns a staged child set, cloning only the one affected adjacency row.
128///
129/// # Parameters
130/// - `index`: Existing index supplying an unchanged row when present.
131/// - `updates`: Sparse child-set overlay being assembled.
132/// - `parent`: Parent whose adjacency row is requested.
133///
134/// The mutable staged child set for `parent`.
135///
136/// # Returns
137/// The staged row, initialized from the index or as an empty set.
138fn children_mut<'a>(
139    index: &PlanNodeRollupIndex,
140    updates: &'a mut BTreeMap<EntityId, BTreeSet<EntityId>>,
141    parent: EntityId,
142) -> &'a mut BTreeSet<EntityId> {
143    updates
144        .entry(parent)
145        .or_insert_with(|| index.children.get(&parent).cloned().unwrap_or_default())
146}

148/// Recomputes one node and stages changed ancestor rollups.
149///
150/// # Parameters
151/// - `index`: Existing values used for unchanged descendants.
152/// - `plan`: Canonical plan projection at the target snapshot.
153/// - `start`: First node to evaluate.
154/// - `force_parent`: Whether to continue after an unchanged first value.
155/// - `parent_updates`: Staged structural-parent overrides.
156/// - `children_updates`: Staged adjacency rows.
157/// - `rollup_updates`: Staged rollup values.
158///
159/// Unit after the changed path is staged through its first unchanged ancestor.
160///
161/// # Returns
162/// Unit when the affected path is valid.
163///
164/// # Errors
165/// Returns a plan error when a required node or child rollup is missing.
166fn propagate(
167    index: &PlanNodeRollupIndex,
168    plan: &Plan,
169    start: EntityId,
170    force_parent: bool,
171    parent_updates: &BTreeMap<EntityId, Option<EntityId>>,
172    children_updates: &BTreeMap<EntityId, BTreeSet<EntityId>>,
173    rollup_updates: &mut BTreeMap<EntityId, PlanNodeRollup>,
174) -> Result<(), PlanError> {
175    let mut current = start;
176    let mut force = force_parent;
177    let mut visited = BTreeSet::new();
178    loop {
179        if !visited.insert(current) {
180            return Err(PlanError::StructuralCycle);
181        }
182        let next = compute(index, plan, current, children_updates, rollup_updates)?;
183        let previous = rollup_updates
184            .get(&current)
185            .or_else(|| index.rollups.get(&current));
186        let changed = previous != Some(&next);
187        rollup_updates.insert(current, next);
188        if !changed && !force {
189            break;
190        }
191        force = false;
192        let Some(parent) = parent_updates
193            .get(&current)
194            .copied()
195            .or_else(|| index.parents.get(&current).copied())
196            .flatten()
197        else {
198            break;
199        };
200        current = parent;
201    }
202    Ok(())
203}

205/// Computes one node using staged child and rollup values.
206///
207/// # Parameters
208/// - `index`: Existing rollups for unchanged descendants.
209/// - `plan`: Canonical plan projection at the target snapshot.
210/// - `node_id`: Node to evaluate.
211/// - `children_updates`: Staged adjacency rows.
212/// - `rollup_updates`: Staged values for changed descendants.
213///
214/// The deterministic rollup for `node_id`.
215///
216/// # Returns
217/// The node rollup when all required child values exist.
218///
219/// # Errors
220/// Returns `MissingNode` or `StaleRollupIndex` when required values are absent.
221fn compute(
222    index: &PlanNodeRollupIndex,
223    plan: &Plan,
224    node_id: EntityId,
225    children_updates: &BTreeMap<EntityId, BTreeSet<EntityId>>,
226    rollup_updates: &BTreeMap<EntityId, PlanNodeRollup>,
227) -> Result<PlanNodeRollup, PlanError> {
228    let node = plan
229        .nodes
230        .iter()
231        .find(|candidate| candidate.id == node_id)
232        .ok_or(PlanError::MissingNode)?;
233    let children = children_updates
234        .get(&node_id)
235        .or_else(|| index.children.get(&node_id));
236    let mut counts = PlanNodeStatusCounts::default();
237    let mut contains_ignored = node.abandon_reason == Some(PlanAbandonReason::Ignored);
238    let mut contains_abandoned = node.status == PlanNodeStatus::Abandoned;
239    if let Some(children) = children {
240        for child in children {
241            let rollup = rollup_updates
242                .get(child)
243                .or_else(|| index.rollups.get(child))
244                .ok_or(PlanError::StaleRollupIndex)?;
245            increment_status_count(&mut counts, rollup.direct_status);
246            contains_ignored |= rollup.contains_ignored;
247            contains_abandoned |= rollup.contains_abandoned;
248        }
249    }
250    let required_gates = plan
251        .gates
252        .iter()
253        .filter(|gate| gate.owner == node_id && gate.phase == PlanGatePhase::Completion)
254        .collect::<Vec<_>>();
255    let unsatisfied = required_gates
256        .iter()
257        .filter(|gate| {
258            plan.evaluate_gate(gate.id).is_none_or(|evaluation| {
259                !matches!(
260                    evaluation.status,
261                    PlanGateStatus::Satisfied | PlanGateStatus::Waived
262                )
263            })
264        })
265        .count();
266    let mut relevant_children = Vec::new();
267    for gate in required_gates {
268        if let super::super::super::PlanGateRequirement::RequiredChildCompleted { child } =
269            &gate.requirement
270            && let Some(rollup) = rollup_updates
271                .get(child)
272                .or_else(|| index.rollups.get(child))
273        {
274            relevant_children.push(rollup);
275        }
276    }
277    Ok(PlanNodeRollup {
278        node: node_id,
279        direct_status: node.status,
280        effective_status: rollup_status(
281            node.status,
282            &relevant_children,
283            u32::try_from(unsatisfied).unwrap_or(u32::MAX),
284        ),
285        contains_ignored,
286        contains_abandoned,
287        child_counts: counts,
288        unsatisfied_completion_gates: u32::try_from(unsatisfied).unwrap_or(u32::MAX),
289    })
290}

292/// Calculates structural depth through the staged parent map.
293///
294/// # Parameters
295/// - `node`: Node whose depth is requested.
296/// - `parent_updates`: Staged parent overrides.
297/// - `parents`: Existing structural-parent lookup.
298///
299/// The number of `PlanNode` ancestors before the branch/root boundary.
300///
301/// # Returns
302/// The node depth in the proposed structure.
303///
304/// # Errors
305/// Returns `StructuralCycle` or `MissingNode` for invalid parent paths.
306fn node_depth(
307    node: EntityId,
308    parent_updates: &BTreeMap<EntityId, Option<EntityId>>,
309    parents: &BTreeMap<EntityId, Option<EntityId>>,
310) -> Result<usize, PlanError> {
311    let mut current = node;
312    let mut depth = 0;
313    let mut visited = BTreeSet::new();
314    while visited.insert(current) {
315        let parent = parent_updates
316            .get(&current)
317            .copied()
318            .or_else(|| parents.get(&current).copied())
319            .ok_or(PlanError::MissingNode)?;
320        let Some(parent) = parent else {
321            return Ok(depth);
322        };
323        current = parent;
324        depth += 1;
325    }
326    Err(PlanError::StructuralCycle)
327}

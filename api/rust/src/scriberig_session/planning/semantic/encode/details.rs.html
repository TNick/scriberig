11/// Adds non-structural semantic records for one plan.
12///
13/// # Parameters
14/// - `plan`: Aggregate plan supplying current semantic details.
15/// - `objects`: Object list receiving encoded records.
16///
17/// # Returns
18/// Unit after every detail record has been encoded.
19///
20/// # Errors
21/// Returns an error when a record cannot be serialized or identified.
22pub(super) fn append_related(
23    plan: &Plan,
24    objects: &mut Vec<PlanSemanticRevision>,
25) -> Result<(), PlanSemanticError> {
26    append_attempts(plan, objects)?;
27    append_dependencies_and_summaries(plan, objects)?;
28    append_gates_evidence_and_history(plan, objects)
29}

31/// Encodes exact attempt history for every current plan node.
32///
33/// # Parameters
34/// - `plan`: Aggregate plan supplying attempt provenance.
35/// - `objects`: Object list receiving encoded attempts.
36///
37/// # Returns
38/// Unit after all attempt records have been encoded.
39///
40/// # Errors
41/// Returns an error when an attempt identity or payload cannot be encoded.
42fn append_attempts(
43    plan: &Plan,
44    objects: &mut Vec<PlanSemanticRevision>,
45) -> Result<(), PlanSemanticError> {
46    for node in &plan.nodes {
47        if let Some(attempts) = plan.attempt_history(node.id) {
48            for attempt in attempts {
49                let identity = AttemptIdentity {
50                    plan: plan.id,
51                    node: node.id,
52                    branch: attempt.branch,
53                    claim_generation: attempt.claim_generation,
54                };
55                let entity = relation_id(ATTEMPT_TYPE, &identity)?;
56                push_object(
57                    objects,
58                    entity,
59                    ATTEMPT_TYPE,
60                    &AttemptPayload {
61                        schema_version: 1,
62                        plan: plan.id,
63                        node: node.id,
64                        branch: attempt.branch,
65                        claim_generation: attempt.claim_generation,
66                        claim: attempt.claim,
67                        status: attempt.status,
68                        result: attempt.result,
69                        failure: &attempt.failure,
70                    },
71                )?;
72            }
73        }
74    }
75    Ok(())
76}

78/// Encodes dependency edges and node summary records.
79///
80/// # Parameters
81/// - `plan`: Aggregate plan supplying dependency and summary state.
82/// - `objects`: Object list receiving encoded records.
83///
84/// # Returns
85/// Unit after dependency and summary records are encoded.
86///
87/// # Errors
88/// Returns an error when a record cannot be serialized or identified.
89fn append_dependencies_and_summaries(
90    plan: &Plan,
91    objects: &mut Vec<PlanSemanticRevision>,
92) -> Result<(), PlanSemanticError> {
93    for dependency in &plan.dependencies {
94        let payload = DependencyPayload {
95            schema_version: 1,
96            plan: plan.id,
97            prerequisite: dependency.prerequisite,
98            dependent: dependency.dependent,
99            fulfilled_by: dependency.fulfilled_by,
100        };
101        push_relation(
102            objects,
103            DEPENDENCY_TYPE,
104            &DependencyIdentity {
105                plan: plan.id,
106                prerequisite: dependency.prerequisite,
107                dependent: dependency.dependent,
108            },
109            &payload,
110        )?;
111    }
112
113    for summary in &plan.summaries {
114        push_object(
115            objects,
116            summary.id,
117            SUMMARY_TYPE,
118            &SummaryPayload {
119                schema_version: 1,
120                plan: plan.id,
121                node: summary.node,
122                text: &summary.text,
123                sources: &summary.sources,
124            },
125        )?;
126        push_relation(
127            objects,
128            SUMMARY_LINK_TYPE,
129            &RelationIdentity {
130                owner: summary.node,
131                target: summary.id,
132            },
133            &SummaryLinkPayload {
134                schema_version: 1,
135                plan: plan.id,
136                node: summary.node,
137                summary: summary.id,
138            },
139        )?;
140    }
141    Ok(())
142}

144/// Encodes typed gates, evidence associations, and mutation history.
145///
146/// # Parameters
147/// - `plan`: Aggregate plan supplying requirements and provenance.
148/// - `objects`: Object list receiving encoded records.
149///
150/// # Returns
151/// Unit after gate, evidence, and history records are encoded.
152///
153/// # Errors
154/// Returns an error when a record cannot be serialized or identified.
155fn append_gates_evidence_and_history(
156    plan: &Plan,
157    objects: &mut Vec<PlanSemanticRevision>,
158) -> Result<(), PlanSemanticError> {
159    for gate in &plan.gates {
160        push_object(
161            objects,
162            gate.id,
163            GATE_TYPE,
164            &GatePayload {
165                schema_version: 1,
166                plan: plan.id,
167                owner: gate.owner,
168                phase: gate.phase,
169                requirement: &gate.requirement,
170                waiver: &gate.waiver,
171            },
172        )?;
173    }
174
175    for association in &plan.evidence {
176        push_object(
177            objects,
178            association.id,
179            EVIDENCE_TYPE,
180            &EvidencePayload {
181                schema_version: 1,
182                plan: plan.id,
183                owner: association.owner,
184                role: association.role,
185                evidence: &association.evidence,
186                note: &association.note,
187                actor: association.actor,
188            },
189        )?;
190    }
191
192    for history in &plan.history {
193        push_relation(
194            objects,
195            HISTORY_TYPE,
196            &HistoryIdentity {
197                plan: plan.id,
198                mutation_revision: history.revision,
199            },
200            &HistoryPayload {
201                schema_version: 1,
202                plan: plan.id,
203                mutation_revision: history.revision,
204                reason: &history.reason,
205            },
206        )?;
207    }
208
209    Ok(())

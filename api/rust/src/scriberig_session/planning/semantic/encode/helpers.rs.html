8/// Encodes one typed semantic object and computes its immutable revision ID.
9///
10/// - `objects`: Destination object set.
11/// - `entity`: Stable entity identity.
12/// - `type_name`: Stable semantic object type.
13/// - `value`: Versioned typed payload.
14///
15/// Unit after the record has been appended.
16/// # Returns
17/// Success after the immutable record is appended.
18/// # Errors
19/// Returns an encoding or identity error.
20pub(in crate::planning::semantic) fn push_object<T: Serialize>(
21    objects: &mut Vec<PlanSemanticRevision>,
22    entity: EntityId,
23    type_name: &str,
24    value: &T,
25) -> Result<(), PlanSemanticError> {
26    let object_type = TypeId::new(type_name)
27        .map_err(|error| PlanSemanticError::InvalidObject(error.to_string()))?;
28    let payload = serde_json::to_vec(value)
29        .map_err(|error| PlanSemanticError::Serialization(format!("{error:?}")))?;
30    let revision = revision_id(EncodingVersion::V1, &object_type, &payload)
31        .map_err(|error| PlanSemanticError::Serialization(format!("{error:?}")))?;
32    objects.push(PlanSemanticRevision {
33        entity,
34        object_type,
35        revision,
36        payload,
37    });
38    Ok(())
39}

41/// Creates a stable relation identity from its immutable key.
42///
43/// - `type_name`: Relation object type.
44/// - `key`: Stable source/target identity tuple.
45///
46/// The relation's stable entity identity.
47/// # Returns
48/// The derived relation entity identity.
49/// # Errors
50/// Returns an encoding error if the key cannot be represented.
51pub(in crate::planning::semantic) fn relation_id<T: Serialize>(
52    type_name: &str,
53    key: &T,
54) -> Result<EntityId, PlanSemanticError> {
55    let object_type = TypeId::new(type_name)
56        .map_err(|error| PlanSemanticError::InvalidObject(error.to_string()))?;
57    let payload = serde_json::to_vec(key)
58        .map_err(|error| PlanSemanticError::Serialization(error.to_string()))?;
59    let revision = revision_id(EncodingVersion::V1, &object_type, &payload)
60        .map_err(|error| PlanSemanticError::Serialization(format!("{error:?}")))?;
61    let mut bytes = [0; 16];
62    bytes.copy_from_slice(&revision.as_bytes()[..16]);
63    Ok(EntityId::from_bytes(bytes))
64}

66/// Appends one relation represented as a stable semantic entity.
67///
68/// - `objects`: Destination object set.
69/// - `type_name`: Stable relation entity type.
70/// - `key`: Relation identity that remains stable across payload changes.
71/// - `payload`: Current relation fields.
72///
73/// Unit after the relation revision is appended.
74/// # Returns
75/// Success after the relation revision is appended.
76/// # Errors
77/// Returns an encoding or identity error.
78pub(in crate::planning::semantic) fn push_relation<K: Serialize, T: Serialize>(
79    objects: &mut Vec<PlanSemanticRevision>,
80    type_name: &str,
81    key: &K,
82    payload: &T,
83) -> Result<(), PlanSemanticError> {
84    let entity = relation_id(type_name, key)?;
85    push_object(objects, entity, type_name, payload)
86}

89///
90/// - `parent`: Structural plan parent.

95pub(in crate::planning::semantic) fn parent_identity(

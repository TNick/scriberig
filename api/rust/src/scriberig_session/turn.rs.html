54/// Identifies an exact durable reply location in a session branch.
55#[derive(
56    Clone, Copy, Debug, Eq, PartialEq, Ord, PartialOrd, serde::Deserialize, serde::Serialize,
57)]
58/// Identifies an exact durable location from which a user may continue.
59pub enum ContinuationAnchor {
60    /// Continue from the current durable end of a branch.
61    BranchHead { branch: EntityId },
62    /// Continue after a turn containing an externally visible assistant output.
63    AssistantOutput { branch: EntityId, turn: EntityId },
64    /// Continue after a user input, whether or not the turn has an answer.
65    HumanInput { branch: EntityId, turn: EntityId },
66    /// Continue from a committed branch result.
67    BranchResult { branch: EntityId, result: EntityId },
68}
69
70/// Resolved location of a continuation anchor in the immutable session snapshot.
71#[derive(Clone, Copy, Debug, Eq, PartialEq)]
72pub struct ResolvedContinuationAnchor {
73    /// Branch containing the anchor.
74    pub branch: EntityId,
75    /// Number of turns included by the anchor.
76    pub turn_count: usize,
77    /// Result identity when the anchor is a branch result.
78    pub result: Option<EntityId>,
79}

81/// Records why and where a continuation branch was created.
82#[derive(Clone, Copy, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
83pub struct ContinuationProvenance {
84    /// Exact durable point used as the branch origin.
85    pub anchor: ContinuationAnchor,
86    /// Product/runtime cause for creating the branch.
87    pub origin: ContinuationOrigin,
88}

90/// Classifies continuation branch creation without changing branch semantics.
91#[derive(Clone, Copy, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
92pub enum ContinuationOrigin {
93    /// A user replied to an existing conversation point.
94    HumanContinuation,
95    /// An agent delegated focused work.
96    AgentDelegation,
97    /// A plan node began execution.
98    PlanExecution,
99    /// A separate attempt was requested.
100    Retry,
101    /// Validation work was split into its own branch.
102    Validation,
103    /// An alternative approach was requested.
104    Alternative,
105    /// Runtime policy created an isolated branch.
106    RuntimePolicy,
107}

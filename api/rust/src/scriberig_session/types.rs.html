1//! Defines session-level entities and cross-domain handles.
2use crate::branch::Branch;
3use crate::execution::SessionExecutionSelection;
4use scriberig_core::{EntityId, Provenance};
5#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
6/// Session-domain structured value.
7pub struct Session {
8    pub id: EntityId,
9    pub branches: Vec<Branch>,
10    pub goals: Vec<Goal>,
11    pub tasks: Vec<Task>,
12    /// Durable selection used as the default for future model invocations.
13    #[serde(default)]
14    pub execution: SessionExecutionSelection,
15}
16
17impl Session {
18    /// Inserts a child branch and updates the parent adjacency list.
19    ///
20    /// # Parameters
21    /// - `self`: Session graph being updated.
22    /// - `id`: Stable child branch identifier.
23    /// - `parent`: Existing parent branch identifier.
24    /// - `title`: Human-readable child title.
25    /// - `purpose`: Focused work assigned to the child.
26    ///
27    /// # Returns
28    /// Unit after the child is linked into the session graph.
29    ///
30    /// # Errors
31    /// Returns `MissingParent` when the parent is absent and `DuplicateBranch`
32    /// when the child identifier is already present.
33    pub fn add_child_branch(
34        &mut self,
35        id: EntityId,
36        parent: EntityId,
37        title: impl Into<String>,
38        purpose: impl Into<String>,
39    ) -> Result<(), crate::SessionError> {
40        if self.branches.iter().any(|branch| branch.id == id) {
41            return Err(crate::SessionError::DuplicateBranch);
42        }
43        let parent_branch = self
44            .branches
45            .iter_mut()
46            .find(|branch| branch.id == parent)
47            .ok_or(crate::SessionError::MissingParent)?;
48        let child = Branch::new_child(id, parent, title, purpose)?;
49        parent_branch.children.push(id);
50        parent_branch.children.sort();
51        self.branches.push(child);
52        self.branches.sort_by_key(|branch| branch.id);
53        Ok(())
54    }
55}

1//! Defines deterministic progressive session views.
2use crate::{

8#[derive(Clone, Debug, Eq, PartialEq)]
9/// Session-domain structured value.
10pub struct BranchDescriptor {
11    pub id: EntityId,
12    pub title: String,
13    pub status: BranchStatus,
14    pub turn_count: usize,
15    pub touched: Vec<EntityId>,
16    pub outcome: Option<String>,
17    pub parent: Option<EntityId>,
18    pub child_count: usize,
19    pub activity_count: usize,
20    pub result: Option<EntityId>,
21}

23/// Provides a compact result for a child branch without loading its activity.
24#[derive(Clone, Debug, Eq, PartialEq)]
25pub struct ChildBranchSummary {
26    /// Child branch identity.
27    pub branch: EntityId,
28    /// Child branch title.
29    pub title: String,
30    /// Child purpose.
31    pub purpose: Option<String>,
32    /// Terminal lifecycle status.
33    pub status: BranchStatus,
34    /// Compact outcome summary.
35    pub outcome: Option<String>,
36    /// Semantic result handles.
37    pub result_handles: Vec<EntityId>,
38}
39#[derive(Clone, Debug, Eq, PartialEq)]
40/// Session-domain structured value.
41pub struct TurnView {
42    pub id: EntityId,
43    pub ordinal: u64,
44    pub has_human_input: bool,
45    pub has_agent_output: bool,
46    pub tool_invocation: Option<EntityId>,
47    pub tool_result: Option<EntityId>,
48}
49#[derive(Clone, Debug, Eq, PartialEq)]
50/// Session-domain structured value.
51pub struct DecisionEntry {
52    pub id: EntityId,
53    pub branch: EntityId,
54    pub text: String,
55}
56/// Session-domain structured value.
57pub struct SessionViews;
58impl SessionViews {
59    #[must_use]
60    /// Executes `branches` for structured session state.
61    ///
62    /// # Parameters
63    /// - `self`: Session value or formatter.
64    /// - `id`: Stable entity identifier when applicable.
65    /// - `title`: Branch title when applicable.
66    /// - `next`: Requested lifecycle state when applicable.
67    /// - `turn`: Turn to append when applicable.
68    /// - `entity`: Cross-domain entity to reference when applicable.
69    /// - `session`: Session snapshot when applicable.
70    /// - `branch`: Branch snapshot when applicable.
71    /// - `decisions`: Decision entries when applicable.
72    /// - `f`: Formatting destination when applicable.
73    ///
74    /// # Returns
75    /// The structured result.
76    ///
77    /// # Errors
78    /// Returns a validation or formatting error when inputs are invalid.
79    pub fn branches(session: &Session) -> Vec<BranchDescriptor> {
80        let mut r = session
81            .branches
82            .iter()
83            .map(Self::branch)
84            .collect::<Vec<_>>();
85        r.sort_by_key(|b| b.id);
86        r
87    }
88    #[must_use]
89    /// Executes `turns` for structured session state.
90    ///
91    /// # Parameters
92    /// - `self`: Session value or formatter.
93    /// - `id`: Stable entity identifier when applicable.
94    /// - `title`: Branch title when applicable.
95    /// - `next`: Requested lifecycle state when applicable.
96    /// - `turn`: Turn to append when applicable.
97    /// - `entity`: Cross-domain entity to reference when applicable.
98    /// - `session`: Session snapshot when applicable.
99    /// - `branch`: Branch snapshot when applicable.
100    /// - `decisions`: Decision entries when applicable.
101    /// - `f`: Formatting destination when applicable.
102    ///
103    /// # Returns
104    /// The structured result.
105    ///
106    /// # Errors
107    /// Returns a validation or formatting error when inputs are invalid.
108    pub fn turns(branch: &Branch) -> Vec<TurnView> {
109        let mut r = branch.turns.iter().map(Self::turn).collect::<Vec<_>>();
110        r.sort_by_key(|t| t.ordinal);
111        r
112    }
113
114    /// Returns compact child results from the parent session graph.
115    ///
116    /// # Parameters
117    /// - `session`: Session graph containing the branches.
118    /// - `parent`: Parent branch whose children are summarized.
119    ///
120    /// # Returns
121    /// Compact summaries sorted by branch identity.
122    #[must_use]
123    pub fn child_summaries(session: &Session, parent: EntityId) -> Vec<ChildBranchSummary> {
124        let mut summaries = session
125            .branches
126            .iter()
127            .filter(|branch| branch.parent == Some(parent))
128            .map(|branch| ChildBranchSummary {
129                branch: branch.id,
130                title: branch.title.clone(),
131                purpose: branch.purpose.clone(),
132                status: branch.status,
133                outcome: branch.outcome.clone(),
134                result_handles: branch
135                    .result
136                    .as_ref()
137                    .map(|result| result.references.clone())
138                    .unwrap_or_default(),
139            })
140            .collect::<Vec<_>>();
141        summaries.sort_by_key(|summary| summary.branch);
142        summaries
143    }
144    #[must_use]
145    /// Executes `decisions` for structured session state.
146    ///
147    /// # Parameters
148    /// - `self`: Session value or formatter.
149    /// - `id`: Stable entity identifier when applicable.
150    /// - `title`: Branch title when applicable.
151    /// - `next`: Requested lifecycle state when applicable.
152    /// - `turn`: Turn to append when applicable.
153    /// - `entity`: Cross-domain entity to reference when applicable.
154    /// - `session`: Session snapshot when applicable.
155    /// - `branch`: Branch snapshot when applicable.
156    /// - `decisions`: Decision entries when applicable.
157    /// - `f`: Formatting destination when applicable.
158    ///
159    /// # Returns
160    /// The structured result.
161    ///
162    /// # Errors
163    /// Returns a validation or formatting error when inputs are invalid.
164    pub fn decisions(decisions: &[Decision]) -> Vec<DecisionEntry> {
165        let mut r = decisions
166            .iter()
167            .map(|d| DecisionEntry {
168                id: d.id,
169                branch: d.branch,
170                text: d.text.clone(),
171            })
172            .collect::<Vec<_>>();
173        r.sort_by_key(|d| d.id);
174        r
175    }
176    /// Builds compact branch metadata.\n    ///\n    /// # Parameters\n    /// - `b`: Branch snapshot.\n    ///\n    /// # Returns\n    /// Compact branch descriptor.
177    fn branch(b: &Branch) -> BranchDescriptor {
178        BranchDescriptor {
179            id: b.id,
180            title: b.title.clone(),
181            status: b.status,
182            turn_count: b.turns.len(),
183            touched: b.touched.clone(),
184            outcome: b.outcome.clone(),
185            parent: b.parent,
186            child_count: b.children.len(),
187            activity_count: b.activity.len(),
188            result: b.result.as_ref().map(|result| result.id),
189        }
190    }
191    /// Builds one turn view.\n    ///\n    /// # Parameters\n    /// - `t`: Turn snapshot.\n    ///\n    /// # Returns\n    /// Compact turn view.
192    fn turn(t: &Turn) -> TurnView {
193        TurnView {
194            id: t.id,
195            ordinal: t.ordinal,
196            has_human_input: t.human.is_some(),
197            has_agent_output: t.agent.is_some(),
198            tool_invocation: t.tool_invocation,
199            tool_result: t.tool_result,
200        }
201    }
202}

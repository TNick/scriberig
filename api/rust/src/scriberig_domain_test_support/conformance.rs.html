1//! Implements readable assertions for the shared domain invariants.
2
3use crate::{CapabilitySet, ConformanceFixture};
4
5/// Runs all applicable domain conformance assertions for one fixture.
6///
7/// # Parameters
8/// - `fixture`: Domain-owned fixture implementing the generic contract.
9///
10/// # Panics
11/// Panics with the domain name, capability, and expected invariant when a
12/// fixture violates an applicable invariant.
13pub fn assert_conformance<F: ConformanceFixture>(fixture: &F) {
14    let domain = fixture.domain();
15    let canonical = fixture.canonical_inspection();
16    let specialized = fixture.domain_inspection();
17    assert_eq!(
18        canonical.entity, specialized.entity,
19        "{domain}/inspection: domain inspection must retain universal entity identity"
20    );
21    assert_eq!(
22        canonical.revision, specialized.revision,
23        "{domain}/inspection: domain inspection must retain exact revision"
24    );
25
26    let unchanged = fixture.unchanged_inspection();
27    assert_eq!(
28        canonical.entity, unchanged.entity,
29        "{domain}/identity: unchanged semantic entity must be reusable"
30    );
31    assert_eq!(
32        canonical.revision, unchanged.revision,
33        "{domain}/revision: unchanged semantic revision must be reusable"
34    );
35
36    let capabilities = fixture.capabilities();
37    assert_optional_capabilities(domain, capabilities, fixture);
38}
39
40/// Checks optional projection and derived-data boundaries only when declared.
41fn assert_optional_capabilities<F: ConformanceFixture>(
42    domain: &str,
43    capabilities: CapabilitySet,
44    fixture: &F,
45) {
46    if capabilities.editable_projection {
47        let projection = fixture.projection().unwrap_or_else(|| {
48            panic!(
49                "{domain}/projection: fixture declares editable projection but supplies no evidence"
50            )
51        });
52        if projection.semantic_change {
53            assert!(
54                projection.shared_change,
55                "{domain}/projection: semantic reconciliation must produce a shared change object"
56            );
57        } else {
58            assert!(
59                projection.unchanged_identity_reused,
60                "{domain}/projection: formatting-only change must reuse identity and revision"
61            );
62        }
63    }
64
65    if capabilities.derived_analysis {
66        let derived = fixture.derived().unwrap_or_else(|| {
67            panic!("{domain}/derived: fixture declares derived analysis but supplies no evidence")
68        });
69        assert_eq!(
70            derived.input.entity_id,
71            fixture.canonical_inspection().entity,
72            "{domain}/derived: provenance must name the canonical source entity"
73        );
74        assert_eq!(
75            derived.input.revision_id,
76            fixture.canonical_inspection().revision,
77            "{domain}/derived: provenance must name the exact source revision"
78        );
79        assert!(
80            derived.provenance.inputs.contains(&derived.input),
81            "{domain}/derived: provenance must retain the exact source reference"
82        );
83        assert!(
84            derived.canonical_preserved,
85            "{domain}/derived: discarding derived data must preserve canonical state"
86        );
87        assert!(
88            derived.generated_summary_is_derived,
89            "{domain}/derived: generated summaries must remain derived"
90        );
91    }
92}

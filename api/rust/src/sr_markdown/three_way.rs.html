1//! Merges Markdown semantic states using an explicit base/current/external triplet.
2
3use crate::{MarkdownBlock, MarkdownDocument, MarkdownStyle, render_document};
4use scriberig_core::SnapshotId;

6/// The semantic operation involved in a Markdown three-way conflict.
7#[derive(Clone, Copy, Debug, Eq, PartialEq)]
8pub enum MarkdownConflictKind {
9    /// Both sides changed one base block differently.
10    SameBlockChanged,
11    /// One side removed a block while the other changed it.
12    DeleteModify,
13    /// Collection ordering or section structure cannot be reconciled safely.
14    Structural,
15    /// The external candidate was already ambiguous before merging.
16    Ambiguous,
17}
18
19/// Structured Markdown conflict retaining all three semantic values.
20#[derive(Clone, Debug, Eq, PartialEq)]
21pub struct MarkdownConflict {
22    /// Conflict operation kind.
23    pub kind: MarkdownConflictKind,
24    /// Top-level semantic path where the conflict occurred.
25    pub path: Vec<usize>,
26    /// Base block, when it exists.
27    pub base: Option<MarkdownBlock>,
28    /// Current canonical block, when it exists.
29    pub current: Option<MarkdownBlock>,
30    /// External candidate block, when it exists.
31    pub external: Option<MarkdownBlock>,
32}
33
34/// Result of explicit three-way Markdown reconciliation.
35#[derive(Clone, Debug, Eq, PartialEq)]
36pub enum ThreeWayOutcome {
37    /// The external representation changed only formatting.
38    FormattingOnly {
39        /// Current canonical semantic document to rerender.
40        document: MarkdownDocument,
41    },
42    /// Semantic changes merged without conflict.
43    Merged {
44        /// Combined semantic document.
45        document: MarkdownDocument,
46    },
47    /// Same semantic result was produced independently by both sides.
48    Converged {
49        /// Shared resulting document.
50        document: MarkdownDocument,
51    },
52    /// One or more semantic conflicts require resolution.
53    Conflict {
54        /// Structured conflict details.
55        conflicts: Vec<MarkdownConflict>,
56    },
57    /// The synchronization base could not be reconstructed.
58    MissingSynchronizationBase,
59}
60
61/// Result of a merge followed by deterministic rerender preparation.
62#[derive(Clone, Debug, Eq, PartialEq)]
63pub struct MarkdownSynchronizationResult {
64    /// Semantic merge result requiring application-layer handling.
65    pub outcome: ThreeWayOutcome,
66    /// Rerendered bytes when the merge is safe to write.
67    pub rendered: Option<Vec<u8>>,
68    /// Snapshot that becomes the synchronization base after a successful write.
69    pub synchronization_base: Option<SnapshotId>,
70    /// Whether preserving the external observation and retrying is safe.
71    pub retryable: bool,
72}

74/// Merges an external Markdown candidate with current canonical state.
75///
76/// # Parameters
77/// - `base`: Last synchronized semantic document, when retrievable.
78/// - `current`: Current canonical semantic document.
79/// - `external`: Parsed external semantic candidate.
80///
81/// # Returns
82/// A formatting-only result, merged result, convergence, structured conflict,
83/// or explicit missing-base outcome.
84#[must_use]
85pub fn three_way_merge(
86    base: Option<&MarkdownDocument>,
87    current: &MarkdownDocument,
88    external: &MarkdownDocument,
89) -> ThreeWayOutcome {
90    let Some(base) = base else {
91        return ThreeWayOutcome::MissingSynchronizationBase;
92    };
93    if external == base {
94        return ThreeWayOutcome::FormattingOnly {
95            document: current.clone(),
96        };
97    }
98    if current == base {
99        return ThreeWayOutcome::Merged {
100            document: external.clone(),
101        };
102    }
103    if current == external {
104        return ThreeWayOutcome::Converged {
105            document: current.clone(),
106        };
107    }
108    let conflicts = detect_conflicts(base, current, external);
109    if !conflicts.is_empty() {
110        return ThreeWayOutcome::Conflict { conflicts };
111    }
112    let mut merged = current.blocks.clone();
113    insert_external_only(&mut merged, base, current, external);
114    ThreeWayOutcome::Merged {
115        document: MarkdownDocument::new(merged),
116    }
117}

119/// Merges and prepares a Markdown projection for atomic rerender/write.
120///
121/// # Parameters
122/// - `base_snapshot`: Last synchronized snapshot, when retrievable.
123/// - `base`: Last synchronized semantic document, when retrievable.
124/// - `current_snapshot`: Current canonical snapshot.
125/// - `current`: Current canonical semantic document.
126/// - `external`: Parsed external candidate.
127/// - `style`: Exact deterministic Markdown style.
128/// - `result_snapshot`: Snapshot to record after a successful merge promotion.
129///
130/// # Returns
131/// Merge outcome, optional rendered bytes, and a deferred synchronization base.
132///
133/// # Errors
134/// This preparation function does not perform I/O and does not return errors.
135#[must_use]
136pub fn synchronize(
137    base_snapshot: Option<SnapshotId>,
138    base: Option<&MarkdownDocument>,
139    current_snapshot: SnapshotId,
140    current: &MarkdownDocument,
141    external: &MarkdownDocument,
142    style: &MarkdownStyle,
143    result_snapshot: SnapshotId,
144) -> MarkdownSynchronizationResult {
145    let outcome = three_way_merge(base, current, external);
146    let safe_document = match &outcome {
147        ThreeWayOutcome::FormattingOnly { document }
148        | ThreeWayOutcome::Merged { document }
149        | ThreeWayOutcome::Converged { document } => Some(document),
150        ThreeWayOutcome::Conflict { .. } | ThreeWayOutcome::MissingSynchronizationBase => None,
151    };
152    let rendered = safe_document.map(|document| render_document(document, style).into_bytes());
153    let synchronization_base = if rendered.is_some() {
154        Some(result_snapshot)
155    } else {
156        base_snapshot.or(Some(current_snapshot))
157    };
158    MarkdownSynchronizationResult {
159        retryable: rendered.is_none(),
160        outcome,
161        rendered,
162        synchronization_base,
163    }
164}

166/// Detects incompatible changes to the same base block.
167///
168/// # Parameters
169/// - `base`: Last synchronized document.
170/// - `current`: Current canonical document.
171/// - `external`: External candidate document.
172///
173/// # Returns
174/// Structured conflicts for incompatible changes.
175fn detect_conflicts(
176    base: &MarkdownDocument,
177    current: &MarkdownDocument,
178    external: &MarkdownDocument,
179) -> Vec<MarkdownConflict> {
180    let mut conflicts = Vec::new();
181    for (index, base_block) in base.blocks.iter().enumerate() {
182        let current_block = current.blocks.get(index);
183        let external_block = external.blocks.get(index);
184        let current_changed = current_block != Some(base_block);
185        let external_changed = external_block != Some(base_block);
186        if current_changed && external_changed && current_block != external_block {
187            let kind = if current_block.is_none() || external_block.is_none() {
188                MarkdownConflictKind::DeleteModify
189            } else {
190                MarkdownConflictKind::SameBlockChanged
191            };
192            conflicts.push(MarkdownConflict {
193                kind,
194                path: vec![index],
195                base: Some(base_block.clone()),
196                current: current_block.cloned(),
197                external: external_block.cloned(),
198            });
199        }
200    }
201    conflicts
202}

204/// Inserts external blocks that are not present in the current state.
205///
206/// # Parameters
207/// - `merged`: Destination merged block sequence.
208/// - `base`: Last synchronized document.
209/// - `current`: Current canonical document.
210/// - `external`: External candidate document.
211fn insert_external_only(
212    merged: &mut Vec<MarkdownBlock>,
213    base: &MarkdownDocument,
214    current: &MarkdownDocument,
215    external: &MarkdownDocument,
216) {
217    for (index, block) in external.blocks.iter().enumerate() {
218        if base.blocks.contains(block) || current.blocks.contains(block) {
219            continue;
220        }
221        let position = index.min(merged.len());
222        merged.insert(position, block.clone());
223    }
224}

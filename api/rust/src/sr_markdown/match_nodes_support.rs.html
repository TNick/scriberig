7/// Checks whether an unchanged neighboring candidate anchors a base node.
8/// - `crate`: Argument named `crate`.
9///
10/// # Returns
11/// The value produced by the declaration.
12pub(crate) fn nearby_anchor(
13    base: &[CanonicalNode],
14    results: &[NodeMatch],
15    candidates: &[(Vec<usize>, MarkdownBlock)],
16    path: &[usize],
17    node: &CanonicalNode,
18) -> bool {
19    let Some(index) = path.last().copied() else {
20        return false;
21    };
22    let parent_path = &path[..path.len().saturating_sub(1)];
23    candidates.iter().any(|(candidate_path, candidate_block)| {
24        let Some(anchor_index) = candidate_path.last().copied() else {
25            return false;
26        };
27        if candidate_path == path
28            || candidate_path[..candidate_path.len().saturating_sub(1)] != *parent_path
29            || index.abs_diff(anchor_index) != 1
30        {
31            return false;
32        }
33        let Some(NodeMatch::Matched { existing, .. }) = results.iter().find(|item| {
34            matches!(item, NodeMatch::Matched { candidate_path: path, .. } if path == candidate_path)
35        }) else {
36            return false;
37        };
38        let Some(anchor) = base.iter().find(|candidate| candidate.id == *existing) else {
39            return false;
40        };
41        anchor.parent == node.parent
42            && anchor.position.abs_diff(node.position) <= 1
43            && anchor.block == *candidate_block
44    })
45}

47/// Computes conservative word-overlap similarity for same-kind text blocks.
48/// - `crate`: Argument named `crate`.
49///
50/// # Returns
51/// The value produced by the declaration.
52pub(crate) fn text_similarity(left: &MarkdownBlock, right: &MarkdownBlock) -> (usize, usize) {
53    let left = block_text(left);
54    let right = block_text(right);
55    if left.is_empty() || right.is_empty() {
56        return (0, 1);
57    }
58    let left_words: std::collections::BTreeSet<_> = left.split_whitespace().collect();
59    let right_words: std::collections::BTreeSet<_> = right.split_whitespace().collect();
60    let shared = left_words.intersection(&right_words).count();
61    let union = left_words.union(&right_words).count();
62    (shared, union)
63}

65/// Extracts comparable author text from a Markdown block.
66/// - `block`: Argument named `block`.
67///
68/// # Returns
69/// The value produced by the declaration.
70fn block_text(block: &MarkdownBlock) -> String {
71    match block {
72        MarkdownBlock::Heading { text, .. }
73        | MarkdownBlock::Paragraph(text)
74        | MarkdownBlock::CodeBlock { text, .. } => text.clone(),
75        MarkdownBlock::Section(section) => section.heading.clone(),
76        MarkdownBlock::Link { label, target } => format!("{label} {target}"),
77        MarkdownBlock::Image { alt, target } => format!("{alt} {target}"),
78        MarkdownBlock::List { items, .. } => items
79            .iter()
80            .flat_map(|item| item.blocks.iter())
81            .map(block_text)
82            .collect::<Vec<_>>()
83            .join(" "),
84        MarkdownBlock::QuoteBlock(blocks) => {
85            blocks.iter().map(block_text).collect::<Vec<_>>().join(" ")
86        }
87        MarkdownBlock::Table { headers, rows } => headers
88            .iter()
89            .chain(rows.iter().flat_map(|row| row.iter()))
90            .cloned()
91            .collect::<Vec<_>>()
92            .join(" "),
93    }
94}

96/// Detects a likely insertion before a later unchanged section heading.
97/// - `crate`: Argument named `crate`.
98///
99/// # Returns
100/// The value produced by the declaration.
101pub(crate) fn section_precedes_heading_anchor(
102    base: &[CanonicalNode],
103    candidates: &[(Vec<usize>, MarkdownBlock)],
104    path: &[usize],
105    block: &MarkdownBlock,
106) -> bool {
107    let MarkdownBlock::Section(candidate_section) = block else {
108        return false;
109    };
110    let has_base_heading = base.iter().any(|node| {
111        matches!(
112            &node.block,
113            MarkdownBlock::Section(section)
114                if section.level == candidate_section.level
115                    && section.heading == candidate_section.heading
116        )
117    });
118    if has_base_heading {
119        return false;
120    }
121    base.iter().any(|node| {
122        let MarkdownBlock::Section(base_section) = &node.block else {
123            return false;
124        };
125        base_section.level == candidate_section.level
126            && node.position >= path.last().copied().unwrap_or(0)
127            && candidates.iter().any(|(candidate_path, candidate)| {
128                candidate_path != path
129                    && matches!(
130                        candidate,
131                        MarkdownBlock::Section(section)
132                            if section.level == base_section.level
133                                && section.heading == base_section.heading
134                    )
135            })
136    })
137}

139/// Checks whether a canonical node belongs to the candidate's known parent.
140/// - `crate`: Argument named `crate`.
141///
142/// # Returns
143/// The value produced by the declaration.
144pub(crate) fn parent_matches(
145    node_parent: Option<EntityId>,
146    mapped_parent: Option<EntityId>,
147    path: &[usize],
148    root_context: bool,
149) -> bool {
150    if path.len() == 1 {
151        !root_context || node_parent.is_none()
152    } else {
153        mapped_parent.is_some_and(|parent| node_parent == Some(parent))
154    }
155}

157/// Resolves the known parent context for one candidate path.
158/// - `crate`: Argument named `crate`.
159///
160/// # Returns
161/// The value produced by the declaration.
162pub(crate) fn candidate_context(path: &[usize], results: &[NodeMatch]) -> (Option<EntityId>, bool) {
163    if path.len() <= 1 {
164        return (None, false);
165    }
166    let parent_path = &path[..path.len() - 1];
167    let mapped_parent = results.iter().find_map(|item| match item {
168        NodeMatch::Matched {
169            candidate_path,
170            existing,
171            ..
172        } if candidate_path == parent_path => Some(*existing),
173        _ => None,
174    });
175    let parent_is_new = results.iter().any(|item| match item {
176        NodeMatch::New { candidate_path } | NodeMatch::Ambiguous { candidate_path, .. } => {
177            parent_path.starts_with(candidate_path)
178        }
179        NodeMatch::Matched { .. } => false,
180    });
181    (mapped_parent, parent_is_new)
182}

184/// Flattens nested semantic blocks into stable structural paths.
185/// - `crate`: Argument named `crate`.
186///
187/// # Returns
188/// The value produced by the declaration.
189pub(crate) fn flatten(document: &MarkdownDocument) -> Vec<(Vec<usize>, MarkdownBlock)> {
190    let mut output = Vec::new();
191    for (index, block) in document.blocks.iter().enumerate() {
192        flatten_block(block, &[index], &mut output);
193    }
194    output
195}

197/// Adds a block and its nested sections to the flattened representation.
198/// - `block`: Argument named `block`.
199/// - `path`: Argument named `path`.
200/// - `output`: Argument named `output`.
201/// - `MarkdownBlock`: Argument named `MarkdownBlock`.
202fn flatten_block(
203    block: &MarkdownBlock,
204    path: &[usize],
205    output: &mut Vec<(Vec<usize>, MarkdownBlock)>,
206) {
207    output.push((path.to_owned(), block.clone()));
208    if let MarkdownBlock::Section(section) = block {
209        for (index, child) in section.blocks.iter().enumerate() {
210            let mut child_path = path.to_owned();
211            child_path.push(index);
212            flatten_block(child, &child_path, output);
213        }
214    }
215}

217/// Reports whether two blocks have the same semantic kind.
218/// - `crate`: Argument named `crate`.
219///
220/// # Returns
221/// The value produced by the declaration.
222pub(crate) fn same_kind(left: &MarkdownBlock, right: &MarkdownBlock) -> bool {
223    std::mem::discriminant(left) == std::mem::discriminant(right)
224}

226/// Reports whether two blocks share enough stable context for a changed edit.
227/// - `crate`: Argument named `crate`.
228///
229/// # Returns
230/// The value produced by the declaration.
231pub(crate) fn same_context(left: &MarkdownBlock, right: &MarkdownBlock) -> bool {
232    match (left, right) {
233        (MarkdownBlock::Section(left), MarkdownBlock::Section(right)) => left.level == right.level,
234        _ => same_kind(left, right),
235    }

1//! Performs deterministic, context-aware matching of Markdown nodes.

5use crate::match_nodes_support::{

11/// A canonical node participating in one projected semantic surface.
12#[derive(Clone, Debug, Eq, PartialEq)]
13pub struct CanonicalNode {
14    /// Stable canonical identity.
15    pub id: EntityId,
16    /// Parent collection identity, when the node is nested.
17    pub parent: Option<EntityId>,
18    /// Semantic block represented by the node.
19    pub block: MarkdownBlock,
20    /// Semantic order within its parent collection.
21    pub position: usize,
22}
23
24/// Identifies why a candidate was matched to a canonical node.
25#[derive(Clone, Copy, Debug, Eq, PartialEq)]
26pub enum MatchEvidence {
27    /// Same structural path and exact semantic content.
28    ExactStructural,
29    /// Exact content with a unique local candidate.
30    ExactContent,
31    /// Exact domain-significant section heading or title.
32    DomainName,
33    /// Same kind at an unchanged structural position.
34    StructuralContext,
35    /// Same sibling collection position with compatible surrounding context.
36    SiblingAlignment,
37    /// Same-kind node retained by an unchanged neighboring anchor.
38    NearbyAnchor,
39    /// Same-kind node selected by a unique conservative text score.
40    TextSimilarity,
41}
42
43/// One candidate matching result.
44#[derive(Clone, Debug, Eq, PartialEq)]
45pub enum NodeMatch {
46    /// Candidate maps to an existing canonical identity.
47    Matched {
48        /// Candidate path in the projected tree.
49        candidate_path: Vec<usize>,
50        /// Existing canonical node.
51        existing: EntityId,
52        /// Matching evidence.
53        evidence: MatchEvidence,
54    },
55    /// Candidate has no safe predecessor.
56    New {
57        /// Candidate path in the projected tree.
58        candidate_path: Vec<usize>,
59    },
60    /// More than one predecessor remains equally plausible.
61    Ambiguous {
62        /// Candidate path in the projected tree.
63        candidate_path: Vec<usize>,
64        /// Possible canonical identities.
65        possible: Vec<EntityId>,
66    },
67}
68
69/// Complete matching evidence, including canonical nodes absent from a candidate.
70#[derive(Clone, Debug, Eq, PartialEq)]
71pub struct MatchReport {
72    /// Result for every candidate node.
73    pub matches: Vec<NodeMatch>,
74    /// Canonical nodes not represented by a candidate.
75    pub unmatched: Vec<EntityId>,
76}
77
78/// Matches a candidate document against canonical nodes in deterministic phases.
79///
80/// # Parameters
81/// - `base`: Canonical nodes in projected structural order.
82/// - `candidate`: Parsed candidate document.
83///
84/// # Returns
85/// Match evidence, explicit ambiguity, and unmatched canonical identities.
86#[must_use]
87pub fn match_nodes(base: &[CanonicalNode], candidate: &MarkdownDocument) -> MatchReport {
88    let candidates = flatten(candidate);
89    let mut used = vec![false; base.len()];
90    let mut matches = Vec::new();
91    for (path, block) in &candidates {
92        matches.push(match_candidate(
93            base,
94            &mut used,
95            &matches,
96            &candidates,
97            path,
98            block,
99        ));
100    }
101    MatchReport {
102        matches,
103        unmatched: base
104            .iter()
105            .enumerate()
106            .filter(|(index, _)| !used[*index])
107            .map(|(_, node)| node.id)
108            .collect(),
109    }
110}

112/// Matches one candidate block against available canonical context.
113#[allow(clippy::too_many_lines)]
114fn match_candidate(
115    base: &[CanonicalNode],
116    used: &mut [bool],
117    results: &[NodeMatch],
118    candidates: &[(Vec<usize>, MarkdownBlock)],
119    path: &[usize],
120    block: &MarkdownBlock,
121) -> NodeMatch {
122    let (mapped_parent, parent_is_new) = candidate_context(path, results);
123    let root_context = base
124        .iter()
125        .any(|node| node.parent.is_none() && same_kind(&node.block, block));
126    let eligible = |index: usize, node: &CanonicalNode| {
127        !used[index]
128            && !parent_is_new
129            && parent_matches(node.parent, mapped_parent, path, root_context)
130    };
131    let exact_content: Vec<_> = base
132        .iter()
133        .enumerate()
134        .filter(|(index, node)| eligible(*index, node) && node.block == *block)
135        .map(|(index, _)| index)
136        .collect();
137    if exact_content.len() > 1 {
138        return ambiguous(path, base, &exact_content);
139    }
140    let exact_path: Vec<_> = base
141        .iter()
142        .enumerate()
143        .filter(|(index, node)| {
144            eligible(*index, node)
145                && node.position == path.last().copied().unwrap_or(0)
146                && node.block == *block
147        })
148        .map(|(index, _)| index)
149        .collect();
150    if exact_path.len() == 1 {
151        return matched(
152            path,
153            base,
154            used,
155            exact_path[0],
156            MatchEvidence::ExactStructural,
157        );
158    }
159    if exact_content.len() == 1 {
160        let evidence = if matches!(block, MarkdownBlock::Section(_)) {
161            MatchEvidence::DomainName
162        } else {
163            MatchEvidence::ExactContent
164        };
165        return matched(path, base, used, exact_content[0], evidence);
166    }
167    let contextual: Vec<_> = base
168        .iter()
169        .enumerate()
170        .filter(|(index, node)| {
171            eligible(*index, node)
172                && node.position == path.last().copied().unwrap_or(0)
173                && same_context(&node.block, block)
174                && !section_precedes_heading_anchor(base, candidates, path, block)
175        })
176        .map(|(index, _)| index)
177        .collect();
178    if contextual.len() == 1 {
179        return matched(
180            path,
181            base,
182            used,
183            contextual[0],
184            MatchEvidence::SiblingAlignment,
185        );
186    }
187    let nearby: Vec<_> = base
188        .iter()
189        .enumerate()
190        .filter(|(index, node)| {
191            eligible(*index, node)
192                && same_context(&node.block, block)
193                && nearby_anchor(base, results, candidates, path, node)
194        })
195        .map(|(index, _)| index)
196        .collect();
197    if nearby.len() == 1 {
198        return matched(path, base, used, nearby[0], MatchEvidence::NearbyAnchor);
199    }
200    if nearby.len() > 1 {
201        return ambiguous(path, base, &nearby);
202    }
203    let same_kind: Vec<_> = base
204        .iter()
205        .enumerate()
206        .filter(|(index, node)| {
207            eligible(*index, node)
208                && same_context(&node.block, block)
209                && !section_precedes_heading_anchor(base, candidates, path, block)
210        })
211        .map(|(index, _)| index)
212        .collect();
213    let scored: Vec<_> = same_kind
214        .iter()
215        .map(|index| (*index, text_similarity(&base[*index].block, block)))
216        .filter(|(_, (shared, union))| shared.saturating_mul(2) >= *union)
217        .collect();
218    if let Some((index, (shared, union))) = scored.first().copied()
219        && scored.get(1).is_none_or(|(_, (next_shared, next_union))| {
220            shared.saturating_mul(*next_union) > next_shared.saturating_mul(union)
221        })
222    {
223        return matched(path, base, used, index, MatchEvidence::TextSimilarity);
224    }
225    match same_kind.as_slice() {
226        [index] => matched(path, base, used, *index, MatchEvidence::StructuralContext),
227        [] => NodeMatch::New {
228            candidate_path: path.to_owned(),
229        },
230        _ => ambiguous(path, base, &same_kind),
231    }
232}

234/// Creates a matched result and marks its canonical identity as consumed.
235fn matched(
236    path: &[usize],
237    base: &[CanonicalNode],
238    used: &mut [bool],
239    index: usize,
240    evidence: MatchEvidence,
241) -> NodeMatch {
242    used[index] = true;
243    NodeMatch::Matched {
244        candidate_path: path.to_owned(),
245        existing: base[index].id,
246        evidence,
247    }
248}
249
250/// Creates an ambiguity result for equally plausible canonical identities.
251fn ambiguous(path: &[usize], base: &[CanonicalNode], indexes: &[usize]) -> NodeMatch {
252    NodeMatch::Ambiguous {
253        candidate_path: path.to_owned(),
254        possible: indexes.iter().map(|index| base[*index].id).collect(),
255    }

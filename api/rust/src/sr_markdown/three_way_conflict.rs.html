1//! Builds structured Markdown merge conflicts.
2
3use super::{Hunk, MarkdownBlock, MarkdownConflict, MarkdownConflictKind};
4
5/// Builds a structured conflict for overlapping base edits.
6/// - `super`: Parent-module imports used by this helper.
7/// - `path`: Argument named `path`.
8/// - `base`: Argument named `base`.
9/// - `current`: Argument named `current`.
10/// - `external`: Argument named `external`.
11///
12/// # Returns
13/// The operation result.
14pub(super) fn make_conflict(
15    path: &[usize],
16    base: &[MarkdownBlock],
17    current: &Hunk,
18    external: &Hunk,
19) -> MarkdownConflict {
20    let mut conflict_path = path.to_vec();
21    conflict_path.push(current.start);
22    MarkdownConflict {
23        kind: if current.replacement.is_empty() || external.replacement.is_empty() {
24            MarkdownConflictKind::DeleteModify
25        } else if current.start == current.end {
26            MarkdownConflictKind::Structural
27        } else {
28            MarkdownConflictKind::SameBlockChanged
29        },
30        path: conflict_path,
31        base: base.get(current.start).cloned(),
32        current: current.replacement.first().cloned(),
33        external: external.replacement.first().cloned(),
34    }
35}

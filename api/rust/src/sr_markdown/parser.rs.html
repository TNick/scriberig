13/// Parses UTF-8 Markdown into a candidate tree.
14///
15/// # Errors
16/// Returns a structured diagnostic when the input cannot be represented by the
17/// supported semantic model.
18///
19/// # Parameters
20/// - `source`: UTF-8 Markdown source text.
21///
22/// # Returns
23/// A candidate document that retains source length for diagnostics.
24pub fn parse(source: &str) -> Result<CandidateDocument, MarkdownDiagnostic> {
25    let document = parse_document(source)?;
26    Ok(CandidateDocument::with_source_spans(
27        document,
28        source.len(),
29        source_spans(source),
30    ))
31}

33/// Collects parser-owned byte coordinates for supported block constructs.
34///
35/// # Parameters
36/// - `source`: External Markdown source.
37///
38/// # Returns
39/// Source spans suitable for diagnostics and projection coordinates.
40fn source_spans(source: &str) -> Vec<SourceSpan> {
41    Parser::new_ext(source, Options::all())
42        .into_offset_iter()
43        .filter_map(|(event, range)| {
44            let is_supported_block = matches!(
45                event,
46                Event::Start(
47                    Tag::Paragraph
48                        | Tag::Heading { .. }
49                        | Tag::List(_)
50                        | Tag::Item
51                        | Tag::CodeBlock(_)
52                        | Tag::BlockQuote
53                        | Tag::Table(_)
54                        | Tag::TableHead
55                        | Tag::TableRow
56                        | Tag::TableCell
57                )
58            );
59            is_supported_block.then_some(SourceSpan {
60                start: range.start,
61                end: range.end,
62            })
63        })
64        .collect()
65}

67/// Parses UTF-8 Markdown into a semantic document.
68///
69/// # Parameters
70/// - `source`: UTF-8 Markdown source text.
71///
72/// # Returns
73/// A normalized semantic Markdown document.
74///
75/// # Errors
76/// Returns a diagnostic when parser state cannot be represented.
77pub fn parse_document(source: &str) -> Result<MarkdownDocument, MarkdownDiagnostic> {
78    let parser = Parser::new_ext(source, Options::all());
79    let mut blocks = Vec::new();
80    let mut text = String::new();
81    let mut lists: Vec<(bool, Vec<ListItem>)> = Vec::new();
82    let mut table: Option<Vec<Vec<String>>> = None;
83    let mut row = Vec::new();
84    let mut code_language = None;
85    let mut in_quote = false;
86    let mut quote_blocks = Vec::new();
87    let mut text_context = false;
88    let mut inline_link_target = None;
89    let mut inline_image_target = None;
90
91    for event in parser {
92        match event {
93            Event::Start(Tag::Heading { .. }) => {
94                text.clear();
95                text_context = true;
96            }
97            Event::End(TagEnd::Heading(level)) => {
98                blocks.push(MarkdownBlock::Heading {
99                    level: heading_level(level),
100                    text: text.trim().to_owned(),
101                });
102                text_context = false;
103            }
104            Event::Start(Tag::Paragraph) => {
105                text.clear();
106                text_context = true;
107            }
108            Event::End(TagEnd::Paragraph) => {
109                if in_quote {
110                    push_text_block(&mut quote_blocks, &text);
111                } else {
112                    push_text_block(&mut blocks, &text);
113                }
114                text_context = false;
115            }
116            Event::Text(value) => {
117                append_list_text(&mut lists, &value, &mut text);
118            }
119            Event::Code(value) => {
120                append_list_text(&mut lists, &format!("`{value}`"), &mut text);
121            }
122            Event::SoftBreak | Event::HardBreak => text.push('\n'),
123            Event::Start(Tag::List(start)) => lists.push((start.is_some(), Vec::new())),
124            Event::Start(Tag::Item) => {
125                if let Some((_, items)) = lists.last_mut() {
126                    items.push(ListItem::text(String::new()));
127                }
128            }
129            Event::End(TagEnd::Item) => {}
130            Event::End(TagEnd::List(_)) => {
131                if let Some((ordered, items)) = lists.pop() {
132                    let list_block = MarkdownBlock::List { ordered, items };
133                    if let Some((_, parent_items)) = lists.last_mut() {
134                        if let Some(parent) = parent_items.last_mut() {
135                            parent.blocks.push(list_block);
136                        }
137                    } else {
138                        blocks.push(list_block);
139                    }
140                }
141            }
142            Event::Start(Tag::CodeBlock(kind)) => {
143                text.clear();
144                code_language = match kind {
145                    CodeBlockKind::Fenced(value) if !value.is_empty() => Some(value.to_string()),
146                    _ => None,
147                };
148            }
149            Event::End(TagEnd::CodeBlock) => {
150                blocks.push(MarkdownBlock::CodeBlock {
151                    language: code_language.take(),
152                    text: text.trim_end().to_owned(),
153                });
154                text.clear();
155            }
156            Event::Start(Tag::BlockQuote) => {
157                in_quote = true;
158                text.clear();
159                quote_blocks.clear();
160            }
161            Event::End(TagEnd::BlockQuote) => {
162                in_quote = false;
163                if !quote_blocks.is_empty() {
164                    blocks.push(MarkdownBlock::QuoteBlock(std::mem::take(&mut quote_blocks)));
165                }
166                text.clear();
167            }
168            Event::Start(Tag::Link { dest_url, .. }) => {
169                if text_context {
170                    text.push('[');
171                    inline_link_target = Some(dest_url.to_string());
172                } else {
173                    text.clear();
174                    text.push('\u{1}');
175                    text.push_str(&dest_url);
176                    text.push('\n');
177                }
178            }
179            Event::End(TagEnd::Link) => {
180                if let Some(target) = inline_link_target.take() {
181                    text.push_str("](");
182                    text.push_str(&target);
183                    text.push(')');
184                } else if let Some((target, label)) = text
185                    .strip_prefix('\u{1}')
186                    .and_then(|value| value.split_once('\n'))
187                {
188                    blocks.push(MarkdownBlock::Link {
189                        label: label.to_owned(),
190                        target: target.to_owned(),
191                    });
192                    text.clear();
193                }
194            }
195            Event::Start(Tag::Image { dest_url, .. }) => {
196                if text_context {
197                    text.push_str("![");
198                    inline_image_target = Some(dest_url.to_string());
199                } else {
200                    text.clear();
201                    text.push('\u{2}');
202                    text.push_str(&dest_url);
203                    text.push('\n');
204                }
205            }
206            Event::End(TagEnd::Image) => {
207                if let Some(target) = inline_image_target.take() {
208                    text.push_str("](");
209                    text.push_str(&target);
210                    text.push(')');
211                } else if let Some((target, alt)) = text
212                    .strip_prefix('\u{2}')
213                    .and_then(|value| value.split_once('\n'))
214                {
215                    blocks.push(MarkdownBlock::Image {
216                        alt: alt.to_owned(),
217                        target: target.to_owned(),
218                    });
219                    text.clear();
220                }
221            }
222            Event::Start(Tag::Table(_)) => {
223                table = Some(Vec::new());
224                row.clear();
225            }
226            Event::Start(Tag::TableHead | Tag::TableRow) => row.clear(),
227            Event::Start(Tag::TableCell) => text.clear(),
228            Event::End(TagEnd::TableCell) => row.push(text.trim().to_owned()),
229            Event::End(TagEnd::TableHead | TagEnd::TableRow) => {
230                if let Some(rows) = &mut table {
231                    rows.push(std::mem::take(&mut row));
232                }
233            }
234            Event::End(TagEnd::Table) => {
235                let rows = table.take().ok_or_else(|| {
236                    MarkdownDiagnostic::new("table-state", "table ended without rows")
237                })?;
238                let mut rows = rows.into_iter();
239                let headers = rows.next().unwrap_or_default();
240                blocks.push(MarkdownBlock::Table {
241                    headers,
242                    rows: rows.collect(),
243                });
244            }
245            Event::Start(Tag::Emphasis) => text.push('*'),
246            Event::End(TagEnd::Emphasis) => text.push('*'),
247            Event::Start(Tag::Strong) => text.push_str("**"),
248            Event::End(TagEnd::Strong) => text.push_str("**"),
249            Event::Start(Tag::Strikethrough) => text.push_str("~~"),
250            Event::End(TagEnd::Strikethrough) => text.push_str("~~"),
251            Event::Start(Tag::FootnoteDefinition(_))
252            | Event::End(TagEnd::FootnoteDefinition)
253            | Event::FootnoteReference(_)
254            | Event::Html(_)
255            | Event::InlineHtml(_) => {
256                return Err(MarkdownDiagnostic::new(
257                    "unsupported-markdown",
258                    "Markdown contains syntax outside the supported projection model",
259                ));
260            }
261            Event::Rule => {}
262            Event::TaskListMarker(_) => {}
263            _ if in_quote => {}
264            _ => {}
265        }
266    }
267    Ok(group_sections(blocks))
268}

270/// Converts a parser heading stream into nested semantic sections.
271///
272/// # Parameters
273/// - `blocks`: Flat parser blocks in source order.
274///
275/// # Returns
276/// A document with heading hierarchy represented by sections.
277fn group_sections(blocks: Vec<MarkdownBlock>) -> MarkdownDocument {
278    let mut roots = Vec::new();
279    let mut stack: Vec<Section> = Vec::new();
280    for block in blocks {
281        if let MarkdownBlock::Heading { level, text } = block {
282            while stack.last().is_some_and(|section| section.level >= level) {
283                finish_section(&mut roots, &mut stack);
284            }
285            stack.push(Section {
286                level,
287                heading: text,
288                blocks: Vec::new(),
289            });
290        } else if let Some(section) = stack.last_mut() {
291            section.blocks.push(block);
292        } else {
293            roots.push(block);
294        }
295    }
296    while !stack.is_empty() {
297        finish_section(&mut roots, &mut stack);
298    }
299    MarkdownDocument::new(roots)
300}

302/// Moves a completed section into its parent or the document root.
303///
304/// # Parameters
305/// - `roots`: Document blocks receiving completed root sections.
306/// - `stack`: Active section stack.
307fn finish_section(roots: &mut Vec<MarkdownBlock>, stack: &mut Vec<Section>) {
308    let section = stack.pop().expect("section stack is non-empty");
309    let block = MarkdownBlock::Section(section);
310    if let Some(parent) = stack.last_mut() {
311        parent.blocks.push(block);
312    } else {
313        roots.push(block);
314    }
315}

317/// Adds a non-empty paragraph to a block sequence.
318///
319/// # Parameters
320/// - `blocks`: Destination block sequence.
321/// - `text`: Candidate paragraph text.
322fn push_text_block(blocks: &mut Vec<MarkdownBlock>, text: &str) {
323    if !text.trim().is_empty() {
324        blocks.push(MarkdownBlock::Paragraph(text.trim().to_owned()));
325    }
326}

328/// Appends inline text to the active list item or block text buffer.
329/// - `lists`: Argument named `lists`.
330fn append_list_text(lists: &mut [(bool, Vec<ListItem>)], value: &str, text: &mut String) {
331    if let Some((_, items)) = lists.last_mut() {
332        if let Some(ListItem { blocks }) = items.last_mut() {
333            if let Some(MarkdownBlock::Paragraph(paragraph)) = blocks.first_mut() {
334                paragraph.push_str(value);
335                return;
336            }
337        }
338    }
339    text.push_str(value);
340}

342/// Converts pulldown heading levels to a stable numeric depth.
343///
344/// # Parameters
345/// - `level`: Parser heading level.
346///
347/// # Returns
348/// Numeric heading depth from one through six.
349fn heading_level(level: HeadingLevel) -> u8 {
350    match level {
351        HeadingLevel::H1 => 1,
352        HeadingLevel::H2 => 2,
353        HeadingLevel::H3 => 3,
354        HeadingLevel::H4 => 4,
355        HeadingLevel::H5 => 5,
356        HeadingLevel::H6 => 6,
357    }
358}

13/// Parses UTF-8 Markdown into a candidate tree.
14///
15/// # Errors
16/// Returns a structured diagnostic when the input cannot be represented by the
17/// supported semantic model.
18///
19/// # Parameters
20/// - `source`: UTF-8 Markdown source text.
21///
22/// # Returns
23/// A candidate document that retains source length for diagnostics.
24pub fn parse(source: &str) -> Result<CandidateDocument, MarkdownDiagnostic> {
25    let document = parse_document(source)?;
26    Ok(CandidateDocument::new(document, source.len()))
27}

29/// Parses UTF-8 Markdown into a semantic document.
30///
31/// # Parameters
32/// - `source`: UTF-8 Markdown source text.
33///
34/// # Returns
35/// A normalized semantic Markdown document.
36///
37/// # Errors
38/// Returns a diagnostic when parser state cannot be represented.
39pub fn parse_document(source: &str) -> Result<MarkdownDocument, MarkdownDiagnostic> {
40    let parser = Parser::new_ext(source, Options::all());
41    let mut blocks = Vec::new();
42    let mut text = String::new();
43    let mut list: Option<(bool, Vec<ListItem>)> = None;
44    let mut table: Option<Vec<Vec<String>>> = None;
45    let mut row = Vec::new();
46    let mut code_language = None;
47    let mut in_quote = false;
48    let mut quote_blocks = Vec::new();
49
50    for event in parser {
51        match event {
52            Event::Start(Tag::Heading { .. }) => text.clear(),
53            Event::End(TagEnd::Heading(level)) => blocks.push(MarkdownBlock::Heading {
54                level: heading_level(level),
55                text: text.trim().to_owned(),
56            }),
57            Event::Start(Tag::Paragraph) => text.clear(),
58            Event::End(TagEnd::Paragraph) => {
59                if in_quote {
60                    push_text_block(&mut quote_blocks, &text);
61                } else {
62                    push_text_block(&mut blocks, &text);
63                }
64            }
65            Event::Text(value) | Event::Code(value) => text.push_str(&value),
66            Event::SoftBreak | Event::HardBreak => text.push('\n'),
67            Event::Start(Tag::List(start)) => list = Some((start.is_some(), Vec::new())),
68            Event::Start(Tag::Item) => text.clear(),
69            Event::End(TagEnd::Item) => {
70                if let Some((_, items)) = &mut list {
71                    items.push(ListItem::text(text.trim()));
72                }
73            }
74            Event::End(TagEnd::List(_)) => {
75                if let Some((ordered, items)) = list.take() {
76                    blocks.push(MarkdownBlock::List { ordered, items });
77                }
78            }
79            Event::Start(Tag::CodeBlock(kind)) => {
80                text.clear();
81                code_language = match kind {
82                    CodeBlockKind::Fenced(value) if !value.is_empty() => Some(value.to_string()),
83                    _ => None,
84                };
85            }
86            Event::End(TagEnd::CodeBlock) => {
87                blocks.push(MarkdownBlock::CodeBlock {
88                    language: code_language.take(),
89                    text: text.trim_end().to_owned(),
90                });
91                text.clear();
92            }
93            Event::Start(Tag::BlockQuote) => {
94                in_quote = true;
95                text.clear();
96                quote_blocks.clear();
97            }
98            Event::End(TagEnd::BlockQuote) => {
99                in_quote = false;
100                if !quote_blocks.is_empty() {
101                    blocks.push(MarkdownBlock::QuoteBlock(std::mem::take(&mut quote_blocks)));
102                }
103                text.clear();
104            }
105            Event::Start(Tag::Link { dest_url, .. }) => {
106                text.clear();
107                text.push('\u{1}');
108                text.push_str(&dest_url);
109                text.push('\n');
110            }
111            Event::End(TagEnd::Link) => {
112                if let Some((target, label)) = text
113                    .strip_prefix('\u{1}')
114                    .and_then(|value| value.split_once('\n'))
115                {
116                    blocks.push(MarkdownBlock::Link {
117                        label: label.to_owned(),
118                        target: target.to_owned(),
119                    });
120                }
121                text.clear();
122            }
123            Event::Start(Tag::Table(_)) => {
124                table = Some(Vec::new());
125                row.clear();
126            }
127            Event::Start(Tag::TableHead | Tag::TableRow) => row.clear(),
128            Event::Start(Tag::TableCell) => text.clear(),
129            Event::End(TagEnd::TableCell) => row.push(text.trim().to_owned()),
130            Event::End(TagEnd::TableHead | TagEnd::TableRow) => {
131                if let Some(rows) = &mut table {
132                    rows.push(std::mem::take(&mut row));
133                }
134            }
135            Event::End(TagEnd::Table) => {
136                let rows = table.take().ok_or_else(|| {
137                    MarkdownDiagnostic::new("table-state", "table ended without rows")
138                })?;
139                let mut rows = rows.into_iter();
140                let headers = rows.next().unwrap_or_default();
141                blocks.push(MarkdownBlock::Table {
142                    headers,
143                    rows: rows.collect(),
144                });
145            }
146            _ if in_quote => {}
147            _ => {}
148        }
149    }
150    Ok(group_sections(blocks))
151}

153/// Converts a parser heading stream into nested semantic sections.
154///
155/// # Parameters
156/// - `blocks`: Flat parser blocks in source order.
157///
158/// # Returns
159/// A document with heading hierarchy represented by sections.
160fn group_sections(blocks: Vec<MarkdownBlock>) -> MarkdownDocument {
161    let mut roots = Vec::new();
162    let mut stack: Vec<Section> = Vec::new();
163    for block in blocks {
164        if let MarkdownBlock::Heading { level, text } = block {
165            while stack.last().is_some_and(|section| section.level >= level) {
166                finish_section(&mut roots, &mut stack);
167            }
168            stack.push(Section {
169                level,
170                heading: text,
171                blocks: Vec::new(),
172            });
173        } else if let Some(section) = stack.last_mut() {
174            section.blocks.push(block);
175        } else {
176            roots.push(block);
177        }
178    }
179    while !stack.is_empty() {
180        finish_section(&mut roots, &mut stack);
181    }
182    MarkdownDocument::new(roots)
183}

185/// Moves a completed section into its parent or the document root.
186///
187/// # Parameters
188/// - `roots`: Document blocks receiving completed root sections.
189/// - `stack`: Active section stack.
190fn finish_section(roots: &mut Vec<MarkdownBlock>, stack: &mut Vec<Section>) {
191    let section = stack.pop().expect("section stack is non-empty");
192    let block = MarkdownBlock::Section(section);
193    if let Some(parent) = stack.last_mut() {
194        parent.blocks.push(block);
195    } else {
196        roots.push(block);
197    }
198}

200/// Adds a non-empty paragraph to a block sequence.
201///
202/// # Parameters
203/// - `blocks`: Destination block sequence.
204/// - `text`: Candidate paragraph text.
205fn push_text_block(blocks: &mut Vec<MarkdownBlock>, text: &str) {
206    if !text.trim().is_empty() {
207        blocks.push(MarkdownBlock::Paragraph(text.trim().to_owned()));
208    }
209}

211/// Converts pulldown heading levels to a stable numeric depth.
212///
213/// # Parameters
214/// - `level`: Parser heading level.
215///
216/// # Returns
217/// Numeric heading depth from one through six.
218fn heading_level(level: HeadingLevel) -> u8 {
219    match level {
220        HeadingLevel::H1 => 1,
221        HeadingLevel::H2 => 2,
222        HeadingLevel::H3 => 3,
223        HeadingLevel::H4 => 4,
224        HeadingLevel::H5 => 5,
225        HeadingLevel::H6 => 6,
226    }

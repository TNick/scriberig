1//! Defines backend-independent synchronization state transitions.
2use crate::{
3    hash::FileDigest,
4    hash::digest_bytes,
5    io::{FileSystem, IoError},
6    location::ProjectionLocation,
7    observe::Observation,
8    ownership::ProjectionOwnership,
9};
10use sr_projection::{ProjectionRef, ProjectionStatus, SemanticStateRef};
11/// Runtime synchronization state for one projection.
12#[derive(Clone, Debug, Eq, PartialEq)]
13pub struct SyncState {
14    pub projection: ProjectionRef,
15    pub location: ProjectionLocation,
16    pub ownership: ProjectionOwnership,
17    pub base: SemanticStateRef,
18    pub rendered: FileDigest,
19    pub observed: Option<FileDigest>,
20    pub status: ProjectionStatus,
21}
22/// In-memory synchronization state boundary.
23#[derive(Clone, Debug, Default)]
24pub struct MemorySyncStore {
25    states: Vec<SyncState>,
26}
27impl MemorySyncStore {
28    /// Creates or replaces a state record.
29    ///
30    /// # Parameters
31    /// - `state`: State to store.
32    pub fn put(&mut self, state: SyncState) {
33        if let Some(existing) = self
34            .states
35            .iter_mut()
36            .find(|s| s.projection == state.projection)
37        {
38            *existing = state;
39        } else {
40            self.states.push(state);
41        }
42    }
43    /// Finds state by location.
44    ///
45    /// # Parameters
46    /// - `location`: Location to find.
47    ///
48    /// # Returns
49    /// Matching state when present.
50    #[must_use]
51    pub fn find(&self, location: &ProjectionLocation) -> Option<&SyncState> {
52        self.states.iter().find(|s| &s.location == location)
53    }
54    /// Finds mutable state by location.
55    ///
56    /// # Parameters
57    /// - `location`: Location to find.
58    ///
59    /// # Returns
60    /// Mutable matching state when present.
61    pub fn find_mut(&mut self, location: &ProjectionLocation) -> Option<&mut SyncState> {
62        self.states.iter_mut().find(|s| &s.location == location)
63    }
64
65    /// Moves a synchronization record when an editable projection is renamed.
66    ///
67    /// # Parameters
68    /// - `from`: Previous physical location.
69    /// - `to`: New physical location.
70    ///
71    /// # Returns
72    /// `true` when a record was moved.
73    pub fn rename(&mut self, from: &ProjectionLocation, to: ProjectionLocation) -> bool {
74        let Some(state) = self.states.iter_mut().find(|state| &state.location == from) else {
75            return false;
76        };
77        state.location = to;
78        true
79    }
80}
81/// Inspects one observation using exact current bytes and digest state.
82/// # Errors
83/// Returns a filesystem read error.
84pub fn inspect_observation<F: FileSystem>(
85    fs: &F,
86    store: &mut MemorySyncStore,
87    event: &Observation,
88) -> Result<Option<ProjectionStatus>, IoError> {
89    let location = match event {
90        Observation::Created(p)
91        | Observation::Modified(p)
92        | Observation::Removed(p)
93        | Observation::RescanRequired(p) => p,
94        Observation::Renamed { from, to } => {
95            if !store.rename(from, to.clone()) {
96                return Ok(None);
97            }
98            to
99        }
100    };
101    let Some(state) = store.find_mut(location) else {
102        return Ok(None);
103    };
104    if matches!(event, Observation::Removed(_)) {
105        state.status = ProjectionStatus::Missing;
106        return Ok(Some(state.status));
107    }
108    let bytes = fs.read(location.path())?;
109    let digest = digest_bytes(&bytes);
110    state.observed = Some(digest);
111    if digest == state.rendered {
112        state.status = ProjectionStatus::Synchronized;
113    } else {
114        state.status = ProjectionStatus::ExternallyModified;
115    }
116    Ok(Some(state.status))
117}

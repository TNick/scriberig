1//! Defines runtime metadata for a filesystem-backed projection.
2
3use crate::{FileDigest, ProjectionLocation, ProjectionOwnership};
4use sr_projection::{ProjectionRef, ProjectionStatus, RendererVersion, SemanticStateRef, StyleRef};
5
6/// Runtime synchronization state kept outside immutable projection configuration.
7#[derive(Clone, Debug, Eq, PartialEq)]
8pub struct FilesystemProjectionState {
9    /// Projection instance.
10    pub projection: ProjectionRef,
11    /// Physical location.
12    pub location: ProjectionLocation,
13    /// Ownership policy.
14    pub ownership: ProjectionOwnership,
15    /// Last exact synchronized semantic state.
16    pub base: SemanticStateRef,
17    /// Last rendered bytes digest.
18    pub rendered_digest: FileDigest,
19    /// Last observed bytes digest, when observed.
20    pub observed_digest: Option<FileDigest>,
21    /// Current generic lifecycle status.
22    pub status: ProjectionStatus,
23    /// Exact style input used for interpretation.
24    pub style: StyleRef,
25    /// Exact renderer version used for interpretation.
26    pub renderer: RendererVersion,
27}

12/// A parsed JavaScript or TypeScript document.
13#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
14pub struct EcmaDocument {
15    /// Adapter identifier selected from the source path.
16    pub adapter: String,
17    /// Complete ESTree-compatible syntax tree.
18    pub tree: Value,
19    /// Every syntax object in deterministic traversal order.
20    pub nodes: Vec<EcmaNode>,
21}
22
23/// One serializable syntax-tree node.
24#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
25pub struct EcmaNode {
26    /// JSON pointer from the document tree to this node.
27    pub path: String,
28    /// `ESTree` node type.
29    pub kind: String,
30    /// Optional UTF-16 source start offset.
31    pub start: Option<u64>,
32    /// Optional UTF-16 source end offset.
33    pub end: Option<u64>,
34    /// Complete node payload.
35    pub value: Map<String, Value>,
36}
37
38/// A parser failure with source-independent diagnostics.
39#[derive(Clone, Debug, Eq, PartialEq)]
40pub struct EcmaError {
41    /// Whether the parser stopped before producing a valid AST.
42    pub fatal: bool,
43    /// Parser diagnostics in source order.
44    pub diagnostics: Vec<String>,
45}
46
47impl std::fmt::Display for EcmaError {
48    /// Formats the first parser diagnostic for human-facing callers.
49    ///
50    /// # Parameters
51    /// - `self`: Parser failure to format.
52    /// - `formatter`: Formatting destination.
53    ///
54    /// # Returns
55    /// A formatting result.
56    ///
57    /// # Errors
58    /// Returns a formatting error when the destination rejects output.
59    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
60        if self.fatal {
61            formatter.write_str("fatal ECMAScript syntax error")?;
62        } else {
63            formatter.write_str("ECMAScript syntax error")?;
64        }
65        if let Some(message) = self.diagnostics.first() {
66            write!(formatter, ": {message}")?;
67        }
68        Ok(())
69    }
70}
71
72impl std::error::Error for EcmaError {}

74/// Parses JavaScript, JSX, TypeScript, or TSX based on the file path.
75///
76/// # Parameters
77/// - `path`: Source path used to select the ECMAScript language mode.
78/// - `source`: UTF-8 source text.
79///
80/// # Returns
81/// A complete serializable syntax tree and its individual nodes.
82///
83/// # Errors
84/// Returns all parser diagnostics when the source is incomplete or invalid.
85pub fn parse(path: &Path, source: &str) -> Result<EcmaDocument, EcmaError> {
86    let source_type = SourceType::from_path(path).map_err(|error| EcmaError {
87        fatal: true,
88        diagnostics: vec![format!("unsupported ECMAScript path: {error}")],
89    })?;
90    let allocator = Allocator::default();
91    let mut parsed = Parser::new(&allocator, source, source_type).parse();
92    if parsed.fatal_error || !parsed.diagnostics.is_empty() {
93        return Err(EcmaError {
94            fatal: parsed.fatal_error,
95            diagnostics: parsed.diagnostics.iter().map(ToString::to_string).collect(),
96        });
97    }
98    Utf8ToUtf16::new(source).convert_program(&mut parsed.program);
99    let tree = serde_json::from_str(
100        &parsed
101            .program
102            .to_pretty_estree_json(!source_type.is_javascript(), false),
103    )
104    .map_err(|error| EcmaError {
105        fatal: true,
106        diagnostics: vec![format!("failed to encode syntax tree: {error}")],
107    })?;
108    let mut nodes = Vec::new();
109    collect_nodes(&tree, "", &mut nodes);
110    Ok(EcmaDocument {
111        adapter: if source_type.is_javascript() {
112            "javascript".to_owned()
113        } else {
114            "typescript".to_owned()
115        },
116        tree,
117        nodes,
118    })
119}

121/// Collects every `ESTree` object with a type field.
122///
123/// # Parameters
124/// - `value`: Current JSON value.
125/// - `path`: JSON pointer of the current value.
126/// - `nodes`: Destination node list.
127///
128/// # Returns
129/// Unit after deterministic depth-first traversal.
130fn collect_nodes(value: &Value, path: &str, nodes: &mut Vec<EcmaNode>) {
131    match value {
132        Value::Object(object) => {
133            if let Some(Value::String(kind)) = object.get("type") {
134                nodes.push(EcmaNode {
135                    path: path.to_owned(),
136                    kind: kind.clone(),
137                    start: object.get("start").and_then(Value::as_u64),
138                    end: object.get("end").and_then(Value::as_u64),
139                    value: object.clone(),
140                });
141            }
142            for (key, child) in object {
143                collect_nodes(child, &format!("{path}/{key}"), nodes);
144            }
145        }
146        Value::Array(array) => {
147            for (index, child) in array.iter().enumerate() {
148                collect_nodes(child, &format!("{path}/{index}"), nodes);
149            }
150        }
151        Value::Null | Value::Bool(_) | Value::Number(_) | Value::String(_) => {}
152    }
153}

155#[cfg(test)]
156mod tests {
157    use super::*;
158
159    /// Parses JavaScript and exposes a complete navigable syntax tree.
160    #[test]
161    fn parses_javascript_nodes() {
162        let document = parse(
163            Path::new("main.js"),
164            "export function answer() { return 42; }",
165        )
166        .expect("JavaScript parses");
167        assert_eq!(document.adapter, "javascript");
168        assert!(
169            document
170                .nodes
171                .iter()
172                .any(|node| node.kind == "FunctionDeclaration")
173        );
174        assert!(
175            document
176                .nodes
177                .iter()
178                .any(|node| node.kind == "ReturnStatement")
179        );
180    }
181
182    /// Rejects syntax errors so callers can request source repair and retry.
183    #[test]
184    fn reports_syntax_errors() {
185        let error =
186            parse(Path::new("broken.ts"), "const value: = 1;").expect_err("invalid TypeScript");
187        assert_ne!(error.diagnostics, Vec::<String>::new());
188    }

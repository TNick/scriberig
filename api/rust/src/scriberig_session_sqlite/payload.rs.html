7const LEGACY_PAYLOAD_VERSION: u32 = 1;

9/// Retains one exact legacy model selection that cannot be resolved safely.
10#[derive(Clone, Debug, Eq, PartialEq, Deserialize, Serialize)]
11pub struct LegacyExecutionRecord {
12    /// JSON pointer locating the original value in the prior session payload.
13    pub path: String,
14    /// Allowlisted historical selection fields; arbitrary properties are removed.
15    pub value: Value,
16}

18/// Holds a decoded session and any unresolved legacy execution choices.
19pub(crate) struct DecodedSession {
20    /// Current typed session graph.
21    pub session: Session,
22    /// Exact retired selection values retained for operator recovery.
23    pub legacy_execution: Vec<LegacyExecutionRecord>,
24    /// Whether the original payload must be rewritten to remove unknown data.
25    pub requires_rewrite: bool,
26}

28/// Versioned envelope for sessions containing sanitized legacy selections.
29///
30/// # Fields
31/// - `version`: Version of the recovery envelope.
32/// - `session`: Current typed session graph.
33/// - `legacy_execution`: Allowlisted legacy choices retained for recovery.
34#[derive(Deserialize, Serialize)]
35struct VersionedSessionPayload {
36    /// Version of this SQLite-only recovery envelope.
37    version: u32,
38    /// Session graph used by current `ScribeRig` readers.
39    session: Session,
40    /// Legacy selection values that could not be mapped to upstream IDs.
41    legacy_execution: Vec<LegacyExecutionRecord>,
42}
43
44/// Decodes a current payload or imports a known legacy selection shape.
45///
46/// - `payload`: JSON bytes read from `session_records`.
47///
48/// The typed session and any exact legacy choices retained for recovery.
49///
50/// Returns an error when the payload is neither current nor a supported
51/// legacy-selection record.
52///
53/// # Returns
54/// The decoded session and rewrite metadata.
55///
56/// # Errors
57/// Returns a diagnostic when JSON decoding or session normalization fails.
58pub(crate) fn decode(payload: &[u8]) -> Result<DecodedSession, String> {
59    let value: Value = serde_json::from_slice(payload).map_err(|error| error.to_string())?;
60    if value.get("version").and_then(Value::as_u64) == Some(u64::from(LEGACY_PAYLOAD_VERSION)) {
61        let envelope: VersionedSessionPayload =
62            serde_json::from_value(value.clone()).map_err(|error| error.to_string())?;
63        let legacy_execution = sanitize_legacy_records(envelope.legacy_execution);
64        let normalized = normalized_value(&envelope.session, &legacy_execution)?;
65        return Ok(DecodedSession {
66            session: envelope.session,
67            legacy_execution,
68            requires_rewrite: normalized != value,
69        });
70    }
71
72    let mut imported = value.clone();
73    let mut legacy_execution = Vec::new();
74    import_session_selection(&mut imported, &mut legacy_execution);
75    import_plan_assignments(&mut imported, &mut legacy_execution);
76    let session = serde_json::from_value(imported).map_err(|error| error.to_string())?;
77    let normalized = normalized_value(&session, &legacy_execution)?;
78    Ok(DecodedSession {
79        session,
80        legacy_execution,
81        requires_rewrite: normalized != value,
82    })
83}

85/// Encodes a normal session directly or wraps it with unresolved legacy data.
86///
87/// - `session`: Current typed session graph.
88/// - `legacy_execution`: Exact older choices that remain unresolved.
89///
90/// JSON bytes suitable for `session_records.payload`.
91///
92/// Returns an error when serialization fails.
93///
94/// # Returns
95/// The encoded session payload.
96///
97/// # Errors
98/// Returns a diagnostic when JSON serialization fails.
99pub(crate) fn encode(
100    session: &Session,
101    legacy_execution: &[LegacyExecutionRecord],
102) -> Result<Vec<u8>, String> {
103    let mut projection = session.clone();
104    projection.plans.clear();
105    if legacy_execution.is_empty() {
106        return serde_json::to_vec(&projection).map_err(|error| error.to_string());
107    }
108    let envelope = VersionedSessionPayload {
109        version: LEGACY_PAYLOAD_VERSION,
110        session: projection,
111        legacy_execution: legacy_execution.to_vec(),
112    };
113    serde_json::to_vec(&envelope).map_err(|error| error.to_string())
114}

116/// Moves a retired session choice into the recoverable payload extension.
117///
118/// - `session`: Untrusted JSON session object being imported.
119/// - `legacy`: Destination for exact legacy selection objects.
120fn import_session_selection(session: &mut Value, legacy: &mut Vec<LegacyExecutionRecord>) {
121    let Some(execution) = session.get_mut("execution").and_then(Value::as_object_mut) else {
122        return;
123    };
124    let has_current_pair = execution
125        .get("provider_id")
126        .and_then(Value::as_str)
127        .is_some()
128        && execution.get("model_id").and_then(Value::as_str).is_some();
129    let has_legacy_choice = execution
130        .get("model_profile")
131        .is_some_and(|value| !value.is_null())
132        || (!has_current_pair && execution.get("model").is_some_and(|value| !value.is_null()));
133    if !has_legacy_choice {
134        return;
135    }
136
137    if has_current_pair {
138        legacy.push(LegacyExecutionRecord {
139            path: "/execution".to_owned(),
140            value: Value::Object(selected_fields(
141                execution,
142                &["model_profile", "model", "thinking_level"],
143            )),
144        });
145        return;
146    }
147    legacy.push(LegacyExecutionRecord {
148        path: "/execution".to_owned(),
149        value: Value::Object(selected_fields(
150            execution,
151            &["model_profile", "model", "thinking_level", "revision"],
152        )),
153    });
154    let revision = execution
155        .get("revision")
156        .cloned()
157        .unwrap_or_else(|| Value::from(0));
158    *execution = current_empty_selection(revision);
159}

161/// Moves legacy profile/model assignments out of durable plan structures.
162///
163/// - `session`: Untrusted JSON session object being imported.
164/// - `legacy`: Destination for exact legacy assignment objects.
165fn import_plan_assignments(session: &mut Value, legacy: &mut Vec<LegacyExecutionRecord>) {
166    let Some(plans) = session.get_mut("plans").and_then(Value::as_array_mut) else {
167        return;
168    };
169    for (plan_index, plan) in plans.iter_mut().enumerate() {
170        if let Some(nodes) = plan.get_mut("nodes").and_then(Value::as_array_mut) {
171            import_nodes(nodes, &format!("/plans/{plan_index}/nodes"), legacy);
172        }
173        if let Some(history) = plan.get_mut("history").and_then(Value::as_array_mut) {
174            for (revision_index, revision) in history.iter_mut().enumerate() {
175                if let Some(nodes) = revision.get_mut("nodes").and_then(Value::as_array_mut) {
176                    import_nodes(
177                        nodes,
178                        &format!("/plans/{plan_index}/history/{revision_index}/nodes"),
179                        legacy,
180                    );
181                }
182            }
183        }
184    }
185}

187/// Clears legacy assignment objects while recording their exact JSON paths.
188///
189/// - `nodes`: Plan node JSON values.
190/// - `base_path`: JSON pointer to the node array.
191/// - `legacy`: Destination for exact legacy assignment objects.
192fn import_nodes(nodes: &mut [Value], base_path: &str, legacy: &mut Vec<LegacyExecutionRecord>) {
193    for (node_index, node) in nodes.iter_mut().enumerate() {
194        let node_path = format!("{base_path}/{node_index}");
195        import_assignment(node, &format!("{node_path}/assignment"), legacy);
196        if let Some(claim) = node.get_mut("claim") {
197            import_assignment(claim, &format!("{node_path}/claim/assignment"), legacy);
198        }
199    }
200}

202/// Retains and clears one legacy `{profile, model}` assignment.
203///
204/// - `owner`: Node or worker-claim object that contains an assignment.
205/// - `path`: Exact JSON pointer to the assignment field.
206/// - `legacy`: Destination for the exact old assignment object.
207fn import_assignment(owner: &mut Value, path: &str, legacy: &mut Vec<LegacyExecutionRecord>) {
208    let Some(owner) = owner.as_object_mut() else {
209        return;
210    };
211    let Some(assignment) = owner.get("assignment") else {
212        return;
213    };
214    let Some(object) = assignment.as_object() else {
215        return;
216    };
217    let has_current_pair = object.get("provider_id").and_then(Value::as_str).is_some()
218        && object.get("model_id").and_then(Value::as_str).is_some();
219    if has_current_pair && object.contains_key("profile") {
220        legacy.push(LegacyExecutionRecord {
221            path: path.to_owned(),
222            value: Value::Object(selected_fields(
223                object,
224                &["profile", "model", "thinking_level"],
225            )),
226        });
227        return;
228    }
229    if has_current_pair || !object.contains_key("profile") {
230        return;
231    }
232    legacy.push(LegacyExecutionRecord {
233        path: path.to_owned(),
234        value: Value::Object(selected_fields(
235            object,
236            &["profile", "model", "thinking_level"],
237        )),
238    });
239    owner.insert("assignment".to_owned(), Value::Null);
240}

242/// Keeps only fields declared by the historical selection data types.
243///
244/// - `value`: Historical JSON object.
245/// - `allowed`: Exact historical field names safe for selection recovery.
246///
247/// A projection without arbitrary extension or credential-bearing fields.
248///
249/// # Returns
250/// The allowlisted fields with unsafe values omitted.
251fn selected_fields(value: &Map<String, Value>, allowed: &[&str]) -> Map<String, Value> {
252    let mut selected = Map::new();
253    for key in allowed {
254        if let Some(value) = value.get(*key) {
255            let safe_value = if *key == "revision" {
256                value.as_u64().map(Value::from)
257            } else if value.is_null() || value.as_str().is_some() {
258                Some(value.clone())
259            } else {
260                None
261            };
262            if let Some(value) = safe_value {
263                selected.insert((*key).to_owned(), value);
264            }
265        }
266    }
267    selected
268}

270/// Redacts unsupported fields from already-versioned recovery records.
271///
272/// - `records`: Recovery records from a persisted envelope.
273///
274/// Records limited to the known legacy selection schema.
275///
276/// # Returns
277/// Sanitized records with unsupported paths and fields removed.
278fn sanitize_legacy_records(records: Vec<LegacyExecutionRecord>) -> Vec<LegacyExecutionRecord> {
279    records
280        .into_iter()
281        .filter_map(|mut record| {
282            let allowed = if record.path == "/execution" {
283                &["model_profile", "model", "thinking_level", "revision"][..]
284            } else if is_assignment_path(&record.path) {
285                &["profile", "model", "thinking_level"][..]
286            } else {
287                return None;
288            };
289            if let Value::Object(object) = &record.value {
290                record.value = Value::Object(selected_fields(object, allowed));
291            } else {
292                record.value = Value::Object(Map::new());
293            }
294            record
295                .value
296                .as_object()
297                .is_some_and(|object| !object.is_empty())
298                .then_some(record)
299        })
300        .collect()
301}

303/// Serializes the canonical current payload after dropping unknown fields.
304///
305/// - `session`: Typed `ScribeRig` session graph.
306/// - `legacy_execution`: Allowlisted retired choices retained for recovery.
307///
308/// Canonical JSON used to detect raw extension data that must be scrubbed.
309/// # Returns
310/// The normalized payload value.
311/// # Errors
312/// Returns a serialization error when the typed payload cannot be encoded.
313fn normalized_value(
314    session: &Session,
315    legacy_execution: &[LegacyExecutionRecord],
316) -> Result<Value, String> {
317    let mut projection = session.clone();
318    projection.plans.clear();
319    if legacy_execution.is_empty() {
320        serde_json::to_value(projection).map_err(|error| error.to_string())
321    } else {
322        serde_json::to_value(VersionedSessionPayload {
323            version: LEGACY_PAYLOAD_VERSION,
324            session: projection,
325            legacy_execution: legacy_execution.to_vec(),
326        })
327        .map_err(|error| error.to_string())
328    }
329}

331/// Accepts only JSON pointers generated for known plan assignment locations.
332///
333/// - `path`: Candidate recovery path from a versioned payload.
334///
335/// `true` when the path has the expected plan/node/assignment shape.
336///
337/// # Returns
338/// Whether the path points to a supported plan assignment location.
339fn is_assignment_path(path: &str) -> bool {
340    let parts = path.split('/').collect::<Vec<_>>();
341    let is_index =
342        |value: &str| !value.is_empty() && value.bytes().all(|byte| byte.is_ascii_digit());
343    match parts.as_slice() {
344        ["", "plans", plan, "nodes", node, "assignment"] => is_index(plan) && is_index(node),
345        ["", "plans", plan, "nodes", node, "claim", "assignment"] => {
346            is_index(plan) && is_index(node)
347        }
348        [
349            "",
350            "plans",
351            plan,
352            "history",
353            revision,
354            "nodes",
355            node,
356            "assignment",
357        ]
358        | [
359            "",
360            "plans",
361            plan,
362            "history",
363            revision,
364            "nodes",
365            node,
366            "claim",
367            "assignment",
368        ] => is_index(plan) && is_index(revision) && is_index(node),
369        _ => false,
370    }
371}

373/// Builds the typed empty selection used while the old choice is unresolved.
374///
375/// - `revision`: Preserved selection revision from the legacy record.
376///
377/// A current-format session selection with no resolved model choice.
378///
379/// # Returns
380/// The current selection fields with the supplied revision.
381fn current_empty_selection(revision: Value) -> Map<String, Value> {
382    let mut selection = Map::new();
383    selection.insert("revision".to_owned(), revision);
384    selection.insert("provider_id".to_owned(), Value::Null);
385    selection.insert("model_id".to_owned(), Value::Null);
386    selection.insert("thinking_level".to_owned(), Value::Null);
387    selection.insert("history".to_owned(), Value::Array(Vec::new()));
388    selection

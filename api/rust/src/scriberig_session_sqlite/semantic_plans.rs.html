30    /// Replaces aggregate plan projections with the universal-store revisions.
31    ///
32    /// Existing aggregate plans are imported once when their semantic root is
33    /// absent. Later reads always resolve the current plan from that root.
34    ///
35    /// # Parameters
36    /// - `session`: Session projection whose plans are hydrated.
37    /// - `recover_legacy_claims`: Whether claims imported from legacy data are stale.
38    ///
39    /// # Errors
40    /// Returns a persistence diagnostic when migration or semantic loading
41    /// fails.
42    /// # Returns
43    /// Unit after all plan projections are hydrated.
44    pub(super) fn hydrate_semantic_plans(
45        &self,
46        session: &mut Session,
47        recover_legacy_claims: bool,
48    ) -> Result<(), String> {
49        let mut plans = Vec::with_capacity(session.plans.len());
50        for legacy_plan in &session.plans {
51            self.migrate_plan_history(session.id, legacy_plan, recover_legacy_claims)?;
52            plans.push(self.load_semantic_plan(session.id, legacy_plan.id)?);
53        }
54        session.plans = plans;
55        Ok(())
56    }

58    /// Builds a compatibility projection without accepting aggregate updates.
59    ///
60    /// # Parameters
61    /// - `proposed`: Caller session projection to persist.
62    /// - `prior`: Previously persisted session projection, when available.
63    ///
64    /// # Returns
65    /// A projection whose plans are loaded from semantic roots, with prior
66    /// plans retained if the caller supplied a stale session snapshot.
67    ///
68    /// # Errors
69    /// Returns a persistence diagnostic when plan initialization or loading
70    /// fails.
71    /// # Returns
72    /// The session projection with canonical plans loaded.
73    pub(super) fn prepare_session_projection(
74        &self,
75        proposed: &Session,
76        prior: Option<crate::payload::DecodedSession>,
77    ) -> Result<Session, String> {
78        let mut projection = proposed.clone();
79        if let Some(prior) = prior {
80            for old_plan in prior.session.plans {
81                let root = plan_root(proposed.id, old_plan.id)?;
82                if self.semantic_store.read_root(&root).is_none() {
83                    self.migrate_plan_history(proposed.id, &old_plan, false)?;
84                }
85                if !projection.plans.iter().any(|plan| plan.id == old_plan.id) {
86                    projection
87                        .plans
88                        .push(self.load_semantic_plan(proposed.id, old_plan.id)?);
89                }
90            }
91        }
92        self.hydrate_semantic_plans(&mut projection, false)?;
93        Ok(projection)
94    }

96    /// Promotes one plan mutation through a universal root compare-and-swap.
97    ///
98    /// # Parameters
99    /// - `session`: Owning session identity.
100    /// - `proposed`: New plan state to promote.
101    /// - `expected_revision`: Plan-local revision observed by the caller.
102    ///
103    /// # Errors
104    /// Returns a plan revision conflict or bounded persistence diagnostic.
105    /// # Returns
106    /// Unit after the semantic root is promoted.
107    pub(super) fn compare_and_swap_semantic_plan<F>(
108        &self,
109        session: EntityId,
110        proposed: &Plan,
111        expected_revision: u64,
112        commit_related: F,
113    ) -> Result<(), PlanStoreError>
114    where
115        F: FnOnce(&rusqlite::Transaction<'_>, &Plan) -> Result<(), String>,
116    {
117        let root = plan_root(session, proposed.id).map_err(plan_persistence)?;
118        if self.semantic_store.read_root(&root).is_none() {
119            self.migrate_plan_history(session, proposed, false)
120                .map_err(plan_persistence)?;
121        }
122        let current_snapshot = self
123            .semantic_store
124            .read_root(&root)
125            .ok_or_else(|| plan_persistence("plan semantic root disappeared"))?;
126        let current = self
127            .load_plan_at_snapshot(session, proposed.id, current_snapshot)
128            .map_err(plan_persistence)?;
129        if current.revision != expected_revision {
130            return Err(PlanStoreError::RevisionConflict {
131                expected: expected_revision,
132                actual: current.revision,
133            });
134        }
135        let next_revision = expected_revision
136            .checked_add(1)
137            .ok_or_else(|| plan_persistence("plan revision is exhausted"))?;
138        if proposed.revision != next_revision {
139            return Err(plan_persistence("proposed plan has an unexpected revision"));
140        }
141        let write_set =
142            build_plan_write_set(self, current_snapshot, proposed).map_err(plan_persistence)?;
143        let staged_rollups = rollup_update::stage_rollup_update(
144            self,
145            session,
146            &current,
147            proposed,
148            current_snapshot,
149            write_set.successor_snapshot,
150        )
151        .map_err(plan_persistence)?;
152        let staged_nodes = node_index::stage_update(
153            self,
154            session,
155            &current,
156            proposed,
157            current_snapshot,
158            write_set.successor_snapshot,
159            write_set.indexed_nodes.clone(),
160        )
161        .map_err(plan_persistence)?;
162        match self.semantic_store.apply_revisions_and_promote_with(
163            root,
164            Some(current_snapshot),
165            current_snapshot,
166            &write_set.revisions,
167            &write_set.changes,
168            |transaction| {
169                commit_related(transaction, &current)?;
170                node_index::persist_in_transaction(
171                    transaction,
172                    session,
173                    proposed.id,
174                    &staged_nodes,
175                )?;
176                rollup_update::persist_in_transaction(
177                    transaction,
178                    session,
179                    proposed.id,
180                    &staged_rollups,
181                )
182            },
183        ) {
184            Ok(_) => Ok(()),
185            Err(PromotionError::Root(scriberig_store::RootError::Conflict { actual, .. })) => {
186                let actual_revision = actual
187                    .map(|snapshot| self.load_plan_at_snapshot(session, proposed.id, snapshot))
188                    .transpose()
189                    .map_err(plan_persistence)?
190                    .map_or(current.revision, |plan| plan.revision);
191                Err(PlanStoreError::RevisionConflict {
192                    expected: expected_revision,
193                    actual: actual_revision,
194                })
195            }
196            Err(error) => Err(plan_persistence(format!(
197                "plan semantic promotion failed: {error:?}"
198            ))),
199        }
200    }

202    /// Imports missing aggregate revisions as a linear semantic snapshot chain.
203    ///
204    /// # Parameters
205    /// - `session`: Owning session identity.
206    /// - `plan`: Legacy or newly authored plan to migrate.
207    /// - `recover_legacy_claims`: Whether imported claims are stale by policy.
208    ///
209    /// # Errors
210    /// Returns a persistence diagnostic when a universal root promotion fails.
211    /// # Returns
212    /// Unit after all legacy revisions are imported.
213    fn migrate_plan_history(
214        &self,
215        session: EntityId,
216        plan: &Plan,
217        recover_legacy_claims: bool,
218    ) -> Result<(), String> {
219        let root = plan_root(session, plan.id)?;
220        let mut expected = self.semantic_store.read_root(&root);
221        let mut base = expected.unwrap_or_else(|| Snapshot::empty().id());
222        let current_revision = expected
223            .map(|snapshot| self.load_plan_at_snapshot(session, plan.id, snapshot))
224            .transpose()?
225            .map(|stored| stored.revision);
226        let mut recovery_pending = self.legacy_migration_pending(session, plan.id)?;
227        let mut rebuild_indexes = recovery_pending;
228        if recover_legacy_claims && expected.is_none() {
229            self.set_legacy_migration_pending(session, plan.id, true)?;
230            recovery_pending = true;
231        }
232        let mut history = plan.history.clone();
233        history.sort_by_key(|record| record.revision);
234        history.dedup_by_key(|record| record.revision);
235        if history
236            .last()
237            .is_none_or(|record| record.revision != plan.revision)
238        {
239            history.push(plan_revision(plan));
240        }
241        for record in history {
242            if current_revision.is_some_and(|revision| record.revision <= revision) {
243                continue;
244            }
245            let mut historical = if record.revision == plan.revision {
246                plan.clone()
247            } else {
248                plan_at_revision(plan, &record)
249            };
250            historical.history = plan
251                .history
252                .iter()
253                .filter(|known| known.revision <= record.revision)
254                .cloned()
255                .collect();
256            if !historical
257                .history
258                .iter()
259                .any(|known| known.revision == historical.revision)
260            {
261                historical.history.push(record);
262            }
263            let encoded = scriberig_session::PlanSemanticSnapshot::from_plan(&historical)
264                .map_err(|error| format!("plan encoding failed: {error:?}"))?;
265            let (revisions, changes) = semantic_writes(encoded.objects)?;
266            let promoted = self
267                .semantic_store
268                .apply_revisions_and_promote(root.clone(), expected, base, &revisions, &changes)
269                .map_err(|error| format!("plan semantic migration failed: {error:?}"))?;
270            base = promoted;
271            expected = Some(promoted);
272            rebuild_indexes = true;
273        }
274        if recovery_pending
275            && (plan.coordinator_claim.is_some()
276                || plan.nodes.iter().any(|node| {
277                    node.status == scriberig_session::PlanNodeStatus::Running
278                        && node.claim.is_some()
279                }))
280        {
281            let mut recovered = plan.clone();
282            let count = recovered
283                .recover_stale_claims(|_| false)
284                .map_err(|error| format!("legacy plan claim recovery failed: {error:?}"))?;
285            if count > 0 && current_revision.is_none_or(|revision| revision < recovered.revision) {
286                let encoded = scriberig_session::PlanSemanticSnapshot::from_plan(&recovered)
287                    .map_err(|error| format!("recovered plan encoding failed: {error:?}"))?;
288                let (revisions, changes) = semantic_writes(encoded.objects)?;
289                let promoted = self
290                    .semantic_store
291                    .apply_revisions_and_promote(root, expected, base, &revisions, &changes)
292                    .map_err(|error| format!("recovered plan migration failed: {error:?}"))?;
293                expected = Some(promoted);
294                rebuild_indexes = true;
295            }
296        }
297        if expected.is_some() && rebuild_indexes {
298            node_index::rebuild_plan_node_index(self, session, plan.id)?;
299        }
300        if recovery_pending {
301            self.set_legacy_migration_pending(session, plan.id, false)?;
302        }
303        Ok(())
304    }
305}

307/// Names the canonical universal root for one session plan.
308///
309/// # Parameters
310/// - `session`: Owning session identity.
311/// - `plan`: Stable plan identity.
312///
313/// The deterministic plan root name.
314///
315/// # Returns
316/// The stable universal root name.
317///
318/// # Errors
319/// Returns an error if the generated name is invalid.
320fn plan_root(session: EntityId, plan: EntityId) -> Result<RootName, String> {
321    RootName::new(format!("scriberig-plan-{session}-{plan}"))
322        .map_err(|error| format!("invalid plan root name: {error:?}"))
323}

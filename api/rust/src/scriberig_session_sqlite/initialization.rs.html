1//! Initializes the SQLite-backed session repository.
2
3use std::path::Path;
4
5use scriberig_store_sqlite::SqliteStore;
6
7use crate::SqliteSessionStore;
8
9impl SqliteSessionStore {
10    /// Opens or creates a session repository at `path`.
11    ///
12    /// - `path`: SQLite database path.
13    ///
14    /// An initialized session repository.
15    ///
16    /// Returns a SQLite initialization error.
17    /// # Returns
18    /// The operation result.
19    /// # Errors
20    /// Returns an error when the operation fails.
21    pub fn open(path: impl AsRef<Path>) -> rusqlite::Result<Self> {
22        let path = path.as_ref();
23        let semantic_store = SqliteStore::open(path)?;
24        Self::from_semantic_store(semantic_store)
25    }
26
27    /// Creates an isolated in-memory session repository.
28    ///
29    /// An initialized repository.
30    ///
31    /// Returns a SQLite initialization error.
32    /// # Returns
33    /// The operation result.
34    /// # Errors
35    /// Returns an error when the operation fails.
36    pub fn open_in_memory() -> rusqlite::Result<Self> {
37        let semantic_store = SqliteStore::open_in_memory()?;
38        Self::from_semantic_store(semantic_store)
39    }
40
41    /// Initializes the repository schema and universal semantic store.
42    ///
43    /// - `semantic_store`: Universal store for immutable semantic records.
44    ///
45    /// An initialized session repository.
46    ///
47    /// Returns a SQLite schema error.
48    /// # Returns
49    /// The operation result.
50    /// # Errors
51    /// Returns an error when the operation fails.
52    fn from_semantic_store(semantic_store: SqliteStore) -> rusqlite::Result<Self> {
53        let connection = semantic_store.shared_connection();
54        let connection_guard = connection
55            .lock()
56            .map_err(|_| rusqlite::Error::InvalidQuery)?;
57        connection_guard.execute_batch(
58            "CREATE TABLE IF NOT EXISTS session_records (
59                 session_id BLOB PRIMARY KEY,
60                 payload BLOB NOT NULL
61             );
62             CREATE TABLE IF NOT EXISTS plan_node_index (
63                 session_id BLOB NOT NULL,
64                 plan_id BLOB NOT NULL,
65                 node_id BLOB NOT NULL,
66                 parent_node_id BLOB,
67                 parent_branch_id BLOB,
68                 current_revision_id BLOB NOT NULL,
69                 direct_status TEXT NOT NULL,
70                 source_snapshot BLOB NOT NULL,
71                 PRIMARY KEY (session_id, plan_id, node_id)
72             );
73             CREATE INDEX IF NOT EXISTS plan_node_index_parent
74                 ON plan_node_index (session_id, plan_id, parent_node_id, node_id);
75             CREATE TABLE IF NOT EXISTS plan_node_index_meta (
76                 session_id BLOB NOT NULL,
77                 plan_id BLOB NOT NULL,
78                 source_snapshot BLOB NOT NULL,
79                 node_count INTEGER NOT NULL,
80                 PRIMARY KEY (session_id, plan_id)
81             );
82             CREATE TABLE IF NOT EXISTS plan_node_rollup_index (
83                 session_id BLOB NOT NULL,
84                 plan_id BLOB NOT NULL,
85                 node_id BLOB NOT NULL,
86                 effective_status TEXT NOT NULL,
87                 contains_ignored INTEGER NOT NULL,
88                 contains_abandoned INTEGER NOT NULL,
89                 child_planned INTEGER NOT NULL,
90                 child_ready INTEGER NOT NULL,
91                 child_running INTEGER NOT NULL,
92                 child_completed INTEGER NOT NULL,
93                 child_failed INTEGER NOT NULL,
94                 child_abandoned INTEGER NOT NULL,
95                 unsatisfied_completion_gates INTEGER NOT NULL,
96                 evaluator_version INTEGER NOT NULL,
97                 source_snapshot BLOB NOT NULL,
98                 PRIMARY KEY (session_id, plan_id, node_id)
99             );
100             CREATE TABLE IF NOT EXISTS plan_node_rollup_meta (
101                 session_id BLOB NOT NULL,
102                 plan_id BLOB NOT NULL,
103                 source_snapshot BLOB NOT NULL,
104                 evaluator_version INTEGER NOT NULL,
105                 node_count INTEGER NOT NULL,
106                 PRIMARY KEY (session_id, plan_id)
107             );
108             CREATE TABLE IF NOT EXISTS plan_legacy_migration_pending (
109                 session_id BLOB NOT NULL,
110                 plan_id BLOB NOT NULL,
111                 PRIMARY KEY (session_id, plan_id)
112             );",
113        )?;
114        drop(connection_guard);
115        Ok(Self {
116            connection,
117            semantic_store,
118        })
119    }
120}

1//! Persists ordered session-to-plan identity projections for semantic roots.
2
3use rusqlite::{Connection, Transaction, TransactionBehavior, params};
4use scriberig_core::EntityId;
5
6use crate::SqliteSessionStore;
7
8impl SqliteSessionStore {
9    /// Registers a semantic plan root in the owning session's ordered projection.
10    ///
11    /// # Parameters
12    /// - `session`: Owning session identity.
13    /// - `plan`: Stable semantic plan identity.
14    /// - `ordinal`: Plan order in the public session projection.
15    ///
16    /// # Returns
17    /// Unit after the registry row is durable.
18    ///
19    /// # Errors
20    /// Returns a SQLite diagnostic when the registry cannot be updated.
21    pub(crate) fn register_plan(
22        &self,
23        session: EntityId,
24        plan: EntityId,
25        ordinal: usize,
26    ) -> Result<(), String> {
27        let ordinal = i64::try_from(ordinal).map_err(|_| "plan order exceeds SQLite range")?;
28        let mut connection = self
29            .connection
30            .lock()
31            .map_err(|_| "session repository lock poisoned".to_owned())?;
32        let transaction = connection
33            .transaction_with_behavior(TransactionBehavior::Immediate)
34            .map_err(|error| error.to_string())?;
35        write(&transaction, session, plan, ordinal)?;
36        transaction.commit().map_err(|error| error.to_string())
37    }
38
39    /// Loads stable plan identities in session presentation order.
40    ///
41    /// # Parameters
42    /// - `session`: Owning session identity.
43    ///
44    /// # Returns
45    /// Registered plan identities ordered by ordinal and stable ID.
46    ///
47    /// # Errors
48    /// Returns a SQLite diagnostic when the registry cannot be read.
49    pub(crate) fn registered_plans(&self, session: EntityId) -> Result<Vec<EntityId>, String> {
50        let connection = self
51            .connection
52            .lock()
53            .map_err(|_| "session repository lock poisoned".to_owned())?;
54        read(&connection, session)
55    }
56}

58/// Upserts one plan identity inside an existing session transaction.
59///
60/// # Parameters
61/// - `transaction`: Active SQLite transaction.
62/// - `session`: Owning session identity.
63/// - `plan`: Stable semantic plan identity.
64/// - `ordinal`: Plan order in the public session projection.
65///
66/// # Returns
67/// Unit after the plan identity is registered.
68///
69/// # Errors
70/// Returns a SQLite diagnostic when the row cannot be written.
71pub(crate) fn write(
72    transaction: &Transaction<'_>,
73    session: EntityId,
74    plan: EntityId,
75    ordinal: i64,
76) -> Result<(), String> {
77    transaction
78        .execute(
79            "INSERT INTO session_plan_records(session_id, plan_id, ordinal)
80             VALUES (?1, ?2, ?3)
81             ON CONFLICT(session_id, plan_id) DO UPDATE SET ordinal=excluded.ordinal",
82            params![
83                session.as_bytes().as_slice(),
84                plan.as_bytes().as_slice(),
85                ordinal
86            ],
87        )
88        .map_err(|error| error.to_string())?;
89    Ok(())
90}

92/// Registers a plan using its position in the proposed session projection.
93///
94/// # Parameters
95/// - `transaction`: Active SQLite transaction.
96/// - `session`: Owning session identity.
97/// - `plans`: Proposed ordered plan projections.
98/// - `plan`: Stable semantic plan identity.
99///
100/// # Returns
101/// Unit after the plan identity is registered.
102///
103/// # Errors
104/// Returns an error when the plan order exceeds SQLite range or the row cannot
105/// be written.
106pub(crate) fn write_proposed_order(
107    transaction: &Transaction<'_>,
108    session: EntityId,
109    plans: &[scriberig_session::Plan],
110    plan: EntityId,
111) -> Result<(), String> {
112    let mut registered = false;
113    for (ordinal, proposed) in plans.iter().enumerate() {
114        let ordinal = i64::try_from(ordinal).map_err(|_| "plan order exceeds SQLite range")?;
115        write(transaction, session, proposed.id, ordinal)?;
116        registered |= proposed.id == plan;
117    }
118    if !registered {
119        let ordinal = i64::try_from(plans.len()).map_err(|_| "plan order exceeds SQLite range")?;
120        write(transaction, session, plan, ordinal)?;
121    }
122    Ok(())
123}

125/// Reads registered plan IDs from one SQLite connection.
126///
127/// # Parameters
128/// - `connection`: SQLite connection containing the registry.
129/// - `session`: Owning session identity.
130///
131/// # Returns
132/// Plan IDs ordered by their persisted session order.
133///
134/// # Errors
135/// Returns a SQLite diagnostic when a row cannot be decoded.
136fn read(connection: &Connection, session: EntityId) -> Result<Vec<EntityId>, String> {
137    let mut statement = connection
138        .prepare(
139            "SELECT plan_id FROM session_plan_records
140             WHERE session_id=?1 ORDER BY ordinal, plan_id",
141        )
142        .map_err(|error| error.to_string())?;
143    let records = statement
144        .query_map(params![session.as_bytes().as_slice()], |row| {
145            row.get::<_, Vec<u8>>(0)
146        })
147        .map_err(|error| error.to_string())?
148        .collect::<rusqlite::Result<Vec<_>>>()
149        .map_err(|error| error.to_string())?;
150    records
151        .into_iter()
152        .map(|bytes| {
153            let bytes: [u8; 16] = bytes
154                .try_into()
155                .map_err(|_| "stored plan identity has invalid length".to_owned())?;
156            Ok(EntityId::from_bytes(bytes))
157        })
158        .collect()
159}

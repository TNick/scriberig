1//! Converts aggregate plan history to and from semantic revisions.
2
3use scriberig_core::EncodingVersion;
4use scriberig_session::{Plan, PlanRevision};
5use scriberig_store::{Revision, SnapshotChange};
6
7/// Projects the current semantic plan state into its history record shape.
8///
9/// # Parameters
10/// - `plan`: Plan state to retain in the historical view.
11///
12/// The compact historical state for one mutation revision.
13/// # Returns
14/// A history record representing the supplied revision.
15pub(super) fn plan_revision(plan: &Plan) -> PlanRevision {
16    PlanRevision {
17        revision: plan.revision,
18        goal: plan.goal.clone(),
19        status: plan.status,
20        nodes: plan.nodes.clone(),
21        dependencies: plan.dependencies.clone(),
22        summaries: plan.summaries.clone(),
23        evidence: plan.evidence.clone(),
24        gates: plan.gates.clone(),
25        reason: None,
26    }
27}
28
29/// Reconstructs one legacy historical state using current invariant fields.
30///
31/// # Parameters
32/// - `current`: Plan supplying stable owner, anchor, and coordinator fields.
33/// - `revision`: Legacy graph state to restore.
34///
35/// The reconstructed plan state at `revision.revision`.
36/// # Returns
37/// The historical plan projection.
38pub(super) fn plan_at_revision(current: &Plan, revision: &PlanRevision) -> Plan {
39    let mut historical = current.clone();
40    historical.revision = revision.revision;
41    historical.goal.clone_from(&revision.goal);
42    historical.status = revision.status;
43    historical.nodes.clone_from(&revision.nodes);
44    historical.dependencies.clone_from(&revision.dependencies);
45    historical.summaries.clone_from(&revision.summaries);
46    historical.evidence.clone_from(&revision.evidence);
47    historical.gates.clone_from(&revision.gates);
48    historical.root_nodes = historical
49        .nodes
50        .iter()
51        .filter(|node| {
52            matches!(
53                node.parent,
54                None | Some(scriberig_session::PlanNodeParent::Branch(_))
55            )
56        })
57        .map(|node| node.id)
58        .collect();
59    historical
60}

62/// Converts semantic objects into universal revisions and snapshot changes.
63///
64/// # Parameters
65/// - `objects`: Independently revisioned semantic plan objects.
66///
67/// # Returns
68/// Universal revision records and their snapshot mappings.
69///
70/// # Errors
71/// Returns an error when a semantic revision fails identity validation.
72pub(super) fn semantic_writes(
73    objects: Vec<scriberig_session::PlanSemanticRevision>,
74) -> Result<(Vec<Revision>, Vec<SnapshotChange>), String> {
75    let mut revisions = Vec::with_capacity(objects.len());
76    let mut changes = Vec::with_capacity(objects.len());
77    for object in objects {
78        let revision = Revision::from_parts(
79            object.revision,
80            EncodingVersion::V1,
81            object.object_type,
82            object.payload,
83        )
84        .map_err(|error| format!("invalid plan revision: {error:?}"))?;
85        changes.push(SnapshotChange::Set(object.entity, revision.id()));
86        revisions.push(revision);
87    }
88    Ok((revisions, changes))
89}

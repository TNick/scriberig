11/// Holds a validated successor rollup index and its rows to persist.
12///
13/// # Public members
14/// None.
15///
16/// # Private members
17/// - `index`: Complete successor values with sparse in-memory updates applied.
18/// - `changed_rows`: Rollup rows whose derived value changed.
19/// - `replace_all`: Whether the cache was absent or plan membership changed.
20pub(super) struct StagedRollupIndex {
21    index: PlanNodeRollupIndex,
22    changed_rows: Vec<EntityId>,
23    replace_all: bool,
24}

26/// Stages a cache update from a validated exact-snapshot predecessor.
27///
28/// A missing, incomplete, or obsolete cache selects a full rebuild. The
29/// reference evaluator remains the fallback and recovery source.
30///
31/// # Parameters
32/// - `store`: SQLite repository owning the snapshot-bound rollup cache.
33/// - `session`: Owning durable session identity.
34/// - `previous`: Canonical plan before the proposed mutation.
35/// - `proposed`: Canonical plan being promoted.
36/// - `old_snapshot`: Exact base snapshot.
37/// - `new_snapshot`: Exact successor snapshot.
38///
39/// # Returns
40/// A complete staged cache update suitable for the root-promotion transaction.
41///
42/// # Errors
43/// Returns a diagnostic if canonical reference rollup computation fails.
44pub(super) fn stage_rollup_update(
45    store: &SqliteSessionStore,
46    session: EntityId,
47    previous: &Plan,
48    proposed: &Plan,
49    old_snapshot: SnapshotId,
50    new_snapshot: SnapshotId,
51) -> Result<StagedRollupIndex, String> {
52    if node_ids(previous) != node_ids(proposed) {
53        return full_rebuild(proposed, new_snapshot);
54    }
55    let Some(cached) = store
56        .load_plan_rollups_at_snapshot(session, proposed.id, old_snapshot)
57        .ok()
58        .flatten()
59    else {
60        return full_rebuild(proposed, new_snapshot);
61    };
62    let Ok(mut index) = PlanNodeRollupIndex::from_cached_rollups(previous, old_snapshot, cached)
63    else {
64        return full_rebuild(proposed, new_snapshot);
65    };
66    let changed = changed_nodes(previous, proposed);
67    let Ok(changed_rows) = index.apply_change(proposed, old_snapshot, new_snapshot, &changed)
68    else {
69        return full_rebuild(proposed, new_snapshot);
70    };
71    Ok(StagedRollupIndex {
72        index,
73        changed_rows,
74        replace_all: false,
75    })
76}

78/// Persists staged values and snapshot metadata in the root-promotion transaction.
79///
80/// # Parameters
81/// - `transaction`: Transaction promoting the semantic root.
82/// - `session`: Owning durable session identity.
83/// - `plan`: Stable plan identity.
84/// - `staged`: Validated complete successor cache.
85///
86/// # Returns
87/// Unit after changed values and exact-snapshot metadata are written.
88///
89/// # Errors
90/// Returns a SQLite diagnostic if the normalized cache cannot be updated.
91pub(super) fn persist_in_transaction(
92    transaction: &Transaction<'_>,
93    session: EntityId,
94    plan: EntityId,
95    staged: &StagedRollupIndex,
96) -> Result<(), String> {
97    let snapshot = staged.index.source_snapshot();
98    if staged.replace_all {
99        return super::rollup_index::replace_in_transaction(
100            transaction,
101            session,
102            plan,
103            snapshot,
104            staged.index.all(),
105        );
106    }
107    for node in &staged.changed_rows {
108        let rollup = staged
109            .index
110            .get(*node)
111            .ok_or_else(|| "staged rollup row is missing".to_owned())?;
112        super::rollup_index::write_rollup_row(transaction, session, plan, snapshot, rollup)?;
113    }
114    super::rollup_index::write_rollup_metadata(
115        transaction,
116        session,
117        plan,
118        snapshot,
119        staged.index.all().len(),
120    )
121}

123/// Builds a complete replacement from the backend-neutral reference evaluator.
124///
125/// # Parameters
126/// - `plan`: Canonical plan projection at the target snapshot.
127/// - `snapshot`: Exact source snapshot represented by the rebuilt cache.
128///
129/// # Returns
130/// A full-replacement staged index.
131///
132/// # Errors
133/// Returns a diagnostic when the canonical plan cannot be rolled up.
134fn full_rebuild(plan: &Plan, snapshot: SnapshotId) -> Result<StagedRollupIndex, String> {
135    let index = PlanNodeRollupIndex::rebuild(plan, snapshot)
136        .map_err(|error| format!("plan rollup index rebuild failed: {error:?}"))?;
137    let changed_rows = index.all().keys().copied().collect();
138    Ok(StagedRollupIndex {
139        index,
140        changed_rows,
141        replace_all: true,
142    })
143}

145/// Returns stable node identities in one canonical plan.
146///
147/// # Parameters
148/// - `plan`: Plan whose nodes are enumerated.
149///
150/// # Returns
151/// The plan's node identities in sorted order.
152fn node_ids(plan: &Plan) -> BTreeSet<EntityId> {
153    plan.nodes.iter().map(|node| node.id).collect()
154}

156/// Finds node and gate owners whose inputs can change derived rollups.
157///
158/// # Parameters
159/// - `previous`: Canonical plan before the mutation.
160/// - `proposed`: Canonical plan after the mutation.
161///
162/// # Returns
163/// Stable owner node identities requiring incremental evaluation.
164fn changed_nodes(previous: &Plan, proposed: &Plan) -> Vec<EntityId> {
165    let previous_nodes = previous
166        .nodes
167        .iter()
168        .map(|node| (node.id, node))
169        .collect::<BTreeMap<_, _>>();
170    let mut changed = proposed
171        .nodes
172        .iter()
173        .filter(|node| {
174            previous_nodes
175                .get(&node.id)
176                .is_some_and(|old| *old != *node)
177        })
178        .map(|node| node.id)
179        .collect::<BTreeSet<_>>();
180    if previous.gates != proposed.gates || previous.evidence != proposed.evidence {
181        changed.extend(previous.gates.iter().map(|gate| gate.owner));
182        changed.extend(proposed.gates.iter().map(|gate| gate.owner));
183        for evidence in previous.evidence.iter().chain(&proposed.evidence) {
184            match evidence.owner {
185                PlanEvidenceOwner::Node(node) => {
186                    changed.insert(node);
187                }
188                PlanEvidenceOwner::Gate(gate) => {
189                    add_gate_owner(&mut changed, &previous.gates, gate);
190                    add_gate_owner(&mut changed, &proposed.gates, gate);
191                }
192            }
193        }
194    }
195    changed.into_iter().collect()
196}

198/// Adds the owning node of a gate when the gate exists in the supplied plan.
199///
200/// # Parameters
201/// - `nodes`: Staged set of nodes to recompute.
202/// - `gates`: Gate definitions that may contain the association.
203/// - `gate_id`: Stable gate identity referenced by evidence.
204fn add_gate_owner(nodes: &mut BTreeSet<EntityId>, gates: &[PlanGate], gate_id: EntityId) {
205    if let Some(gate) = gates.iter().find(|gate| gate.id == gate_id) {
206        nodes.insert(gate.owner);
207    }

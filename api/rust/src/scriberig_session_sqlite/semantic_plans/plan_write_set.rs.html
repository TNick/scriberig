1//! Encodes one proposed plan snapshot and derives its atomic root changes.
2
3use scriberig_core::{EncodingVersion, SnapshotId};
4use scriberig_session::{Plan, PlanSemanticNodeRef};
5use scriberig_store::{Revision, Snapshot, SnapshotChange, SnapshotStore};
6
7use crate::SqliteSessionStore;
8
9/// Holds semantic revisions, changes, and indexes for one root promotion.
10///
11/// # Public members
12/// None.
13///
14/// # Private members
15/// - `revisions`: Immutable object revisions to insert before promotion.
16/// - `changes`: Exact entity-to-revision changes for the successor snapshot.
17/// - `indexed_nodes`: Revision-bound nodes for the normalized direct index.
18/// - `successor_snapshot`: Snapshot produced by applying `changes`.
19pub(super) struct PlanWriteSet {
20    /// Immutable object revisions to insert before root promotion.
21    pub(super) revisions: Vec<Revision>,
22    /// Exact entity-to-revision changes for the successor snapshot.
23    pub(super) changes: Vec<SnapshotChange>,
24    /// Node references used by the normalized direct-node index.
25    pub(super) indexed_nodes: Vec<PlanSemanticNodeRef>,
26    /// Snapshot produced by applying `changes` to the current snapshot.
27    pub(super) successor_snapshot: SnapshotId,
28}
29
30/// Encodes plan revisions and computes the exact successor root snapshot.
31///
32/// # Parameters
33/// - `store`: SQLite repository containing the universal semantic store.
34/// - `current_snapshot`: Exact base snapshot to compare-and-swap.
35/// - `proposed`: Validated plan projection being promoted.
36///
37/// # Returns
38/// The semantic write set and exact successor snapshot.
39///
40/// # Errors
41/// Returns a persistence diagnostic when encoding or reading the base snapshot fails.
42pub(super) fn build_plan_write_set(
43    store: &SqliteSessionStore,
44    current_snapshot: SnapshotId,
45    proposed: &Plan,
46) -> Result<PlanWriteSet, String> {
47    let encoded = scriberig_session::PlanSemanticSnapshot::from_plan(proposed)
48        .map_err(|error| format!("plan encoding failed: {error:?}"))?;
49    let indexed_nodes = encoded
50        .node_refs()
51        .map_err(|error| format!("plan node index encode failed: {error:?}"))?;
52    let old_entries = store
53        .semantic_store
54        .iter(current_snapshot)
55        .map_err(|error| format!("plan snapshot read failed: {error:?}"))?;
56    let new_entities = encoded
57        .objects
58        .iter()
59        .map(|object| object.entity)
60        .collect::<std::collections::BTreeSet<_>>();
61    let mut revisions = Vec::with_capacity(encoded.objects.len());
62    let mut changes = Vec::with_capacity(encoded.objects.len() + old_entries.len());
63    for object in encoded.objects {
64        let revision = Revision::from_parts(
65            object.revision,
66            EncodingVersion::V1,
67            object.object_type,
68            object.payload,
69        )
70        .map_err(|error| format!("invalid plan revision: {error:?}"))?;
71        changes.push(SnapshotChange::Set(object.entity, revision.id()));
72        revisions.push(revision);
73    }
74    for (entity, _) in &old_entries {
75        if !new_entities.contains(entity) {
76            changes.push(SnapshotChange::Remove(*entity));
77        }
78    }
79    let successor_snapshot = Snapshot::from_entries(old_entries.iter().copied().collect())
80        .apply(changes.iter().copied())
81        .id();
82    Ok(PlanWriteSet {
83        revisions,
84        changes,
85        indexed_nodes,
86        successor_snapshot,
87    })
88}

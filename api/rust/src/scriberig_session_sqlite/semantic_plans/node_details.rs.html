1//! Reads bounded gate and evidence pages from exact plan snapshots.
2
3use scriberig_core::{EntityId, SnapshotId};
4use scriberig_store::{RevisionStore, RootStore, SnapshotStore};
5
6use super::plan_root;
7use crate::SqliteSessionStore;
8
9impl SqliteSessionStore {
10    /// Loads a bounded page of gate and evidence records for one node.
11    ///
12    /// # Parameters
13    /// - `session`: Owning durable session identity.
14    /// - `plan`: Stable plan identity.
15    /// - `node`: Stable plan-node identity.
16    /// - `snapshot`: Exact semantic snapshot to resolve.
17    /// - `offset`: Number of matching records to skip per category.
18    /// - `limit`: Maximum records to return per category.
19    ///
20    /// # Returns
21    /// The requested details, or `None` outside the plan's retained history.
22    ///
23    /// # Errors
24    /// Returns a diagnostic when detail revisions cannot be read or decoded.
25    pub(crate) fn load_plan_node_details_snapshot(
26        &self,
27        session: EntityId,
28        plan: EntityId,
29        node: EntityId,
30        snapshot: SnapshotId,
31        offset: usize,
32        limit: usize,
33    ) -> Result<Option<scriberig_session::PlanSemanticNodeDetails>, String> {
34        let root = plan_root(session, plan)?;
35        let current = self
36            .semantic_store
37            .read_root(&root)
38            .ok_or_else(|| "plan semantic root is absent".to_owned())?;
39        if snapshot != current
40            && !self
41                .semantic_store
42                .ancestors(current)
43                .map_err(|error| format!("plan history read failed: {error:?}"))?
44                .contains(&snapshot)
45        {
46            return Ok(None);
47        }
48        let selected_types = [
49            scriberig_session::schema::SessionEntityType::PlanGate.as_str(),
50            scriberig_session::schema::SessionEntityType::PlanEvidence.as_str(),
51        ];
52        let entries = self
53            .semantic_store
54            .iter(snapshot)
55            .map_err(|error| format!("plan snapshot read failed: {error:?}"))?;
56        let revision_ids = entries
57            .iter()
58            .map(|(_, revision)| *revision)
59            .collect::<Vec<_>>();
60        let object_types = self
61            .semantic_store
62            .get_object_types(&revision_ids)
63            .map_err(|error| format!("plan revision metadata read failed: {error:?}"))?;
64        let selected_entries = entries
65            .into_iter()
66            .filter(|(_, revision)| {
67                object_types
68                    .get(revision)
69                    .is_some_and(|object_type| selected_types.contains(&object_type.as_str()))
70            })
71            .collect::<Vec<_>>();
72        let selected_revision_ids = selected_entries
73            .iter()
74            .map(|(_, revision)| *revision)
75            .collect::<Vec<_>>();
76        let revisions = self
77            .semantic_store
78            .get_many(&selected_revision_ids)
79            .map_err(|error| format!("plan detail read failed: {error:?}"))?;
80        let mut objects = Vec::with_capacity(selected_entries.len());
81        for (entity, revision_id) in selected_entries {
82            let Some(revision) = revisions.get(&revision_id) else {
83                return Err(format!("plan detail revision {revision_id} is missing"));
84            };
85            let Some(object_type) = object_types.get(&revision_id) else {
86                return Err(format!("plan detail revision {revision_id} has no type"));
87            };
88            if revision.object_type() != object_type {
89                return Err(format!(
90                    "plan detail revision {revision_id} has mismatched type"
91                ));
92            }
93            objects.push(scriberig_session::PlanSemanticRevision {
94                entity,
95                object_type: object_type.clone(),
96                revision: revision.id(),
97                payload: revision.payload().to_vec(),
98            });
99        }
100        let semantic = scriberig_session::PlanSemanticSnapshot {
101            plan,
102            snapshot: Some(snapshot),
103            objects,
104        };
105        semantic
106            .node_details(node, offset, limit)
107            .map(Some)
108            .map_err(|error| format!("plan detail decode failed: {error:?}"))
109    }
110}

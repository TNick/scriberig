1//! Defines exact-target entity inspection requests.
2
3use crate::{DetailLevel, Inspection, ProjectionRef, QueryError, TargetRef};
4
5/// Requests inspection of one exact semantic target.
6#[derive(Clone, Debug, Eq, PartialEq)]
7pub struct InspectRequest {
8    /// Snapshot-relative or exact-revision target.
9    pub target: TargetRef,
10    /// Amount of orientation detail requested.
11    pub detail: DetailLevel,
12    /// Optional registered view requested by the caller.
13    pub view: Option<ProjectionRef>,
14}
15
16/// Provides exact-target entity inspection.
17pub trait InspectService {
18    /// Inspects one target without changing its snapshot or revision context.
19    ///
20    /// # Parameters
21    /// - `self`: Read service implementation.
22    /// - `request`: Exact-target inspection request.
23    ///
24    /// # Returns
25    /// Compact or expanded inspection metadata.
26    ///
27    /// # Errors
28    /// Returns a structured query error when the target or view is unavailable.
29    fn inspect(&self, request: &InspectRequest) -> Result<Inspection, QueryError>;
30}

6/// Domain-neutral scalar value used by structural predicates.
7#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
8pub enum QueryValue {
9    /// UTF-8 scalar value.
10    Text(String),
11    /// Signed integer value.
12    Integer(i64),
13    /// Boolean value.
14    Boolean(bool),
15}
16
17/// Direction for typed relation predicates.
18#[derive(Clone, Copy, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
19pub enum QueryDirection {
20    /// Traverse relation targets from the subject.
21    Outgoing,
22    /// Traverse reverse relation targets when supported.
23    Incoming,
24}
25
26/// Declarative structural query expression.
27#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
28pub enum QueryExpr {
29    /// Matches every entity in the scope.
30    All,
31    /// Matches entities with the exact semantic type.
32    TypeIs(ObjectType),
33    /// Matches entities whose property equals a scalar value.
34    PropertyEq {
35        /// Property key to compare.
36        key: String,
37        /// Expected property value.
38        value: QueryValue,
39    },
40    /// Matches entities that expose a property key.
41    PropertyExists {
42        /// Property key to require.
43        key: String,
44    },
45    /// Matches entities related to a target expression.
46    Related {
47        /// Stable semantic relation identifier.
48        relation: String,
49        /// Relation traversal direction.
50        direction: QueryDirection,
51        /// Predicate applied to related entities.
52        target: Box<QueryExpr>,
53    },
54    /// Matches entities contained in a typed collection.
55    MemberOf {
56        /// Stable semantic collection identifier.
57        collection: String,
58    },
59    /// Requires every nested expression to match.
60    And(Vec<QueryExpr>),
61    /// Requires at least one nested expression to match.
62    Or(Vec<QueryExpr>),
63    /// Negates one nested expression.
64    Not(Box<QueryExpr>),
65}

67/// Orders exact structural query results using domain-independent semantics.
68#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
69pub enum QueryOrdering {
70    /// Provider-defined stable ordering.
71    StableDefault,
72    /// Orders by a semantic property key.
73    Property {
74        /// Property key used for ordering.
75        key: String,
76        /// Ordering direction.
77        direction: SortDirection,
78    },
79}
80
81/// Selects ascending or descending property ordering.
82#[derive(Clone, Copy, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
83pub enum SortDirection {
84    /// Lowest values first.
85    Ascending,
86    /// Highest values first.
87    Descending,
88}

90/// Requests exact structural selection within one snapshot-bound scope.
91#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
92pub struct QuerySpec {
93    /// Exact semantic scope to query.
94    pub scope: QueryScope,
95    /// Declarative structural predicate.
96    pub expression: QueryExpr,
97    /// Stable result ordering.
98    pub ordering: QueryOrdering,
99    /// Page bounds and opaque continuation.
100    pub page: PageRequest,
101    /// Detail level for returned rows.
102    pub detail: DetailLevel,
103}
104
105/// One semantic row returned by an exact structural query.
106#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
107pub struct QueryRow {
108    /// Semantic handle reusable by inspect/list/follow operations.
109    pub target: TargetRef,
110    /// Entity type when known.
111    pub entity_type: Option<ObjectType>,
112    /// Optional compact metadata.
113    pub metadata: Option<String>,
114}
115
116/// Returns exact-scope structural query rows.
117#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
118pub struct QueryResult {
119    /// Exact scope used by evaluation.
120    pub scope: QueryScope,
121    /// Bounded semantic rows.
122    pub page: Page<QueryRow>,
123    /// Advisory cost estimate.
124    pub information_cost: Option<InformationCost>,
125}
126
127/// Provides declarative structural query execution.
128pub trait StructuredQueryService {
129    /// Evaluates one query specification without changing its snapshot scope.
130    ///
131    /// # Parameters
132    /// - `self`: Query provider.
133    /// - `specification`: Exact-scope structural query.
134    ///
135    /// # Returns
136    /// A bounded page of semantic rows.
137    ///
138    /// # Errors
139    /// Returns a structured error for unsupported expressions or invalid queries.
140    fn query(&self, specification: &QuerySpec) -> Result<QueryResult, QueryError>;
141}

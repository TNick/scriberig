12/// Converts parser heading levels to stable numeric levels.
13///
14/// - `level`: Parser heading level.
15///
16/// Numeric heading depth from one through six.
17/// # Returns
18/// The operation result.
19fn heading_level(level: HeadingLevel) -> u8 {
20    match level {
21        HeadingLevel::H1 => 1,
22        HeadingLevel::H2 => 2,
23        HeadingLevel::H3 => 3,
24        HeadingLevel::H4 => 4,
25        HeadingLevel::H5 => 5,
26        HeadingLevel::H6 => 6,
27    }
28}
29use pulldown_cmark::{CodeBlockKind, Event, HeadingLevel, Options, Parser, Tag, TagEnd};
30
31/// Parses supported Markdown syntax into semantic blocks.
32///
33/// - `source`: Markdown source text.
34///
35/// A semantic Markdown document preserving block order.
36#[must_use]
37/// # Returns
38/// The operation result.
39pub fn parse(source: &str) -> MarkdownDocument {
40    let parser = Parser::new_ext(source, Options::all());
41    let mut blocks = Vec::new();
42    let mut text = String::new();
43    let mut list: Option<(bool, Vec<String>)> = None;
44    let mut table: Option<Vec<Vec<String>>> = None;
45    let mut row: Vec<String> = Vec::new();
46    for event in parser {
47        match event {
48            Event::Start(Tag::Heading { level, .. }) => {
49                text.clear();
50                text.push(char::from_digit(u32::from(heading_level(level)), 10).unwrap_or('1'));
51                text.push('\0');
52            }
53            Event::End(TagEnd::Heading(_)) => {
54                let value = text.split_off(0);
55                let (level, body) = value.split_once('\0').unwrap_or(("1", value.as_str()));
56                blocks.push(MarkdownBlock::Heading {
57                    level: level.parse().unwrap_or(1),
58                    text: body.trim().to_owned(),
59                });
60            }
61            Event::Start(Tag::Paragraph) => text.clear(),
62            Event::End(TagEnd::Paragraph) => {
63                if !text.trim().is_empty() {
64                    blocks.push(MarkdownBlock::Paragraph(text.trim().to_owned()));
65                }
66            }
67            Event::Text(value) | Event::Code(value) => text.push_str(&value),
68            Event::Start(Tag::List(start)) => list = Some((start.is_some(), Vec::new())),
69            Event::Start(Tag::Item) => text.clear(),
70            Event::End(TagEnd::Item) => {
71                if let Some((_, items)) = &mut list {
72                    items.push(text.trim().to_owned());
73                }
74            }
75            Event::End(TagEnd::List(_)) => {
76                if let Some((ordered, items)) = list.take() {
77                    blocks.push(MarkdownBlock::List { ordered, items });
78                }
79            }
80            Event::Start(Tag::CodeBlock(kind)) => {
81                text.clear();
82                if let CodeBlockKind::Fenced(language) = kind {
83                    text.push_str(&format!("\u{0}{language}\n"));
84                }
85            }
86            Event::End(TagEnd::CodeBlock) => {
87                let value = text.split_off(0);
88                let language = value.strip_prefix('\0').and_then(|rest| {
89                    rest.split_once('\n')
90                        .map(|(name, body)| (name.to_owned(), body.to_owned()))
91                });
92                let (language, body) =
93                    language.map_or((None, value), |(lang, body)| (Some(lang), body));
94                blocks.push(MarkdownBlock::CodeBlock {
95                    language,
96                    text: body.trim_end().to_owned(),
97                });
98            }
99            Event::Start(Tag::BlockQuote) => text.clear(),
100            Event::End(TagEnd::BlockQuote) => {
101                blocks.push(MarkdownBlock::Quote(text.trim().to_owned()));
102            }
103            Event::Start(Tag::Link { dest_url, .. }) => {
104                text.clear();
105                text.push_str(&format!("\u{1}{dest_url}\n"));
106            }
107            Event::End(TagEnd::Link) => {
108                let value = text.split_off(0);
109                if let Some((target, visible)) = value
110                    .strip_prefix('\x01')
111                    .and_then(|rest| rest.split_once('\n'))
112                {
113                    blocks.push(MarkdownBlock::Link {
114                        text: visible.to_owned(),
115                        target: target.to_owned(),
116                    });
117                }
118            }
119            Event::Start(Tag::Image { dest_url, .. }) => {
120                text.clear();
121                text.push_str(&format!("\u{2}{dest_url}\n"));
122            }
123            Event::End(TagEnd::Image) => {
124                let value = text.split_off(0);
125                if let Some((target, alt)) = value
126                    .strip_prefix('\x02')
127                    .and_then(|rest| rest.split_once('\n'))
128                {
129                    blocks.push(MarkdownBlock::Image {
130                        alt: alt.to_owned(),
131                        target: target.to_owned(),
132                    });
133                }
134            }
135            Event::Start(Tag::Table(_)) => {
136                table = Some(Vec::new());
137                row.clear();
138            }
139            Event::End(TagEnd::Table) => {
140                if let Some(rows) = table.take() {
141                    blocks.push(MarkdownBlock::Table(rows));
142                }
143            }
144            Event::Start(Tag::TableHead | Tag::TableRow) => row.clear(),
145            Event::End(TagEnd::TableHead | TagEnd::TableRow) => {
146                if let Some(rows) = &mut table {
147                    rows.push(std::mem::take(&mut row));
148                }
149            }
150            Event::Start(Tag::TableCell) => text.clear(),
151            Event::End(TagEnd::TableCell) => row.push(text.trim().to_owned()),
152            Event::SoftBreak | Event::HardBreak => text.push('\n'),
153            _ => {}
154        }
155    }
156    MarkdownDocument { blocks }
157}

1//! Defines deterministic Markdown block matching.
2
3use crate::mapping::MarkdownDocument;
4
5/// One candidate-to-base block match.
6#[derive(Clone, Debug, Eq, PartialEq)]
7pub struct BlockMatch {
8    /// Candidate block index.
9    pub candidate: usize,
10    /// Base block index.
11    pub base: usize,
12}
13
14/// Matches equal semantic blocks and reports duplicate ambiguity.
15///
16/// - `base`: Existing semantic document.
17/// - `candidate`: Parsed candidate document.
18///
19/// Matches and ambiguous candidate indexes.
20#[must_use]
21/// # Returns
22/// The operation result.
23pub fn match_blocks(
24    base: &MarkdownDocument,
25    candidate: &MarkdownDocument,
26) -> (Vec<BlockMatch>, Vec<usize>) {
27    let mut matches = Vec::new();
28    let mut ambiguous = Vec::new();
29    for (candidate_index, block) in candidate.blocks.iter().enumerate() {
30        let choices: Vec<_> = base
31            .blocks
32            .iter()
33            .enumerate()
34            .filter(|(_, old)| *old == block)
35            .map(|(index, _)| index)
36            .collect();
37        if choices.len() == 1 {
38            matches.push(BlockMatch {
39                candidate: candidate_index,
40                base: choices[0],
41            });
42        } else if choices.len() > 1 {
43            ambiguous.push(candidate_index);
44        }
45    }
46    (matches, ambiguous)
47}

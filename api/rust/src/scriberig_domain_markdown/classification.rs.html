1//! Classifies Markdown information by canonical ownership.
2
3/// Identifies whether a Markdown field contributes to semantic identity.
4#[derive(Clone, Copy, Debug, Eq, PartialEq)]
5pub enum InformationClass {
6    /// Changes to this information belong to canonical state.
7    Canonical,
8    /// This information is formatting detail by default.
9    ProjectionOnly,
10    /// This information can be recomputed from canonical inputs.
11    Derived,
12}
13
14/// Returns the class assigned to one named Markdown information group.
15///
16/// - `name`: Stable information group name.
17///
18/// The canonical ownership class for the group, or `None` when unknown.
19#[must_use]
20/// # Returns
21/// The operation result.
22pub fn classify(name: &str) -> Option<InformationClass> {
23    match name {
24        "structure" | "text" | "links" | "images" | "inline" => Some(InformationClass::Canonical),
25        "indentation" | "line_wrapping" | "blank_lines" | "table_padding" => {
26            Some(InformationClass::ProjectionOnly)
27        }
28        "summary" | "topic_labels" | "outline" | "description" | "search_index" => {
29            Some(InformationClass::Derived)
30        }
31        _ => None,
32    }
33}

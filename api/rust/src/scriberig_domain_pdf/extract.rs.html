3use crate::{
4    PdfDocument, PdfPage, TextRegion,
5    geometry::{PageGeometry, Rectangle},
6    provenance::PdfProvenance,
7};
8
9/// Extracts pages and literal parenthesized text regions from a source artifact.
10///
11/// The baseline extractor intentionally retains derived output alongside the exact source.
12///
13/// - `document`: Imported PDF document to enrich.
14/// - `producer`: Extractor identity.
15/// - `profile`: Extractor profile/version.
16///
17/// A document with ordered pages and derived text regions.
18/// # Returns
19/// The operation result.
20pub fn extract(
21    document: &PdfDocument,
22    producer: impl Into<String>,
23    profile: impl Into<String>,
24) -> PdfDocument {
25    let producer = producer.into();
26    let profile = profile.into();
27    let page_count = document
28        .source
29        .bytes
30        .windows(6)
31        .filter(|w| *w == b"/Type ")
32        .count()
33        .max(1)
34        .try_into()
35        .unwrap_or(u32::MAX);
36    let mut pages = Vec::new();
37    let mut regions = Vec::new();
38    let text = String::from_utf8_lossy(&document.source.bytes);
39    let mut n = 0u32;
40    for part in text.split('(').skip(1) {
41        if let Some(value) = part.split(')').next() {
42            if !value.is_empty() && value.len() < 4096 {
43                n += 1;
44                regions.push(TextRegion {
45                    id: document.child_id("region", n),
46                    page_number: 1,
47                    text: value.to_owned(),
48                    provenance: PdfProvenance {
49                        source_digest: document.source.digest,
50                        page_number: 1,
51                        producer: producer.clone(),
52                        profile: profile.clone(),
53                        region: Some(Rectangle {
54                            x: 0.0,
55                            y: 0.0,
56                            width: f32::from(
57                                u16::try_from(value.len().min(65_535)).unwrap_or(u16::MAX),
58                            ),
59                            height: 12.0,
60                        }),
61                        confidence: None,
62                    },
63                });
64            }
65        }
66    }
67    for page_number in 1..=page_count {
68        pages.push(PdfPage {
69            id: document.child_id("page", page_number),
70            number: page_number,
71            geometry: PageGeometry {
72                width: 595.0,
73                height: 842.0,
74            },
75            regions: regions
76                .iter()
77                .filter(|r| r.page_number == page_number)
78                .map(|r| r.id)
79                .collect(),
80        });
81    }
82    PdfDocument {
83        source: document.source.clone(),
84        pages,
85        regions,
86    }
87}

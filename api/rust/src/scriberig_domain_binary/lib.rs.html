1//! Preserves binary project artifacts as typed metadata documents.
2
3use std::path::Path;
4
5use serde::{Deserialize, Serialize};
6
7/// Metadata for one binary source artifact.
8#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
9pub struct BinaryDocument {
10    /// Project-relative source path.
11    pub path: String,
12    /// Stable media-type label inferred from the path.
13    pub media_type: String,
14    /// Exact number of source bytes.
15    pub byte_length: usize,
16}
17
18/// Creates binary metadata without decoding or modifying source bytes.
19///
20/// # Parameters
21/// - `path`: Source path used for media-type inference.
22/// - `source`: Exact source bytes.
23///
24/// # Returns
25/// Binary metadata retaining path, type, and byte length.
26#[must_use]
27pub fn parse(path: &Path, source: &[u8]) -> BinaryDocument {
28    BinaryDocument {
29        path: path.to_string_lossy().replace('\\', "/"),
30        media_type: media_type(path),
31        byte_length: source.len(),
32    }
33}
34
35/// Infers a stable media type for a binary path.
36///
37/// # Parameters
38/// - `path`: Source path.
39///
40/// # Returns
41/// Media type suitable for indexing and agent context.
42fn media_type(path: &Path) -> String {
43    let extension = path
44        .extension()
45        .and_then(|value| value.to_str())
46        .unwrap_or_default()
47        .to_ascii_lowercase();
48    let media = match extension.as_str() {
49        "png" => "image/png",
50        "jpg" | "jpeg" => "image/jpeg",
51        "gif" => "image/gif",
52        "webp" => "image/webp",
53        "svg" => "image/svg+xml",
54        "ico" => "image/x-icon",
55        "bmp" => "image/bmp",
56        "woff" => "font/woff",
57        "woff2" => "font/woff2",
58        "ttf" => "font/ttf",
59        "otf" => "font/otf",
60        "wasm" => "application/wasm",
61        "zip" => "application/zip",
62        "gz" => "application/gzip",
63        "tar" => "application/x-tar",
64        "7z" => "application/x-7z-compressed",
65        "jar" => "application/java-archive",
66        "pdf" => "application/pdf",
67        "exe" | "dll" => "application/vnd.microsoft.portable-executable",
68        "o" | "a" | "so" => "application/x-object",
69        _ => "application/octet-stream",
70    };
71    media.to_owned()
72}

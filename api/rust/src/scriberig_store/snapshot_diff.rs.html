1//! Implements immutable snapshot comparison helpers.
2
3use super::{Snapshot, SnapshotChange, SnapshotDiff};
4use std::collections::BTreeMap;
5
6impl Snapshot {
7    /// Computes explicit semantic differences in ascending entity order.
8    ///
9    /// - `other`: Snapshot to compare with this snapshot.
10    ///
11    /// Semantic changes between the snapshots.
12    #[must_use]
13    /// # Returns
14    /// The operation result.
15    pub fn diff(&self, other: &Self) -> Vec<SnapshotDiff> {
16        self.diff_semantic(other)
17    }
18
19    /// Computes storage-level changes suitable for applying to another snapshot.
20    ///
21    /// - `other`: Snapshot to compare with this snapshot.
22    ///
23    /// Storage changes that transform this snapshot into the other snapshot.
24    #[must_use]
25    /// # Returns
26    /// The operation result.
27    pub fn diff_changes(&self, other: &Self) -> Vec<SnapshotChange> {
28        self.diff_semantic(other)
29            .into_iter()
30            .map(|change| match change {
31                SnapshotDiff::Added { entity, revision }
32                | SnapshotDiff::Changed {
33                    entity,
34                    to: revision,
35                    ..
36                } => SnapshotChange::Set(entity, revision),
37                SnapshotDiff::Removed { entity, .. } => SnapshotChange::Remove(entity),
38            })
39            .collect()
40    }
41
42    /// Computes explicit semantic differences in ascending entity order.
43    ///
44    /// - `other`: Snapshot to compare with this snapshot.
45    ///
46    /// Semantic changes between the snapshots.
47    #[must_use]
48    /// # Returns
49    /// The operation result.
50    pub fn diff_semantic(&self, other: &Self) -> Vec<SnapshotDiff> {
51        let left: BTreeMap<_, _> = self.iter().collect();
52        let right: BTreeMap<_, _> = other.iter().collect();
53        left.keys()
54            .chain(right.keys())
55            .copied()
56            .collect::<std::collections::BTreeSet<_>>()
57            .into_iter()
58            .filter_map(|entity| match (left.get(&entity), right.get(&entity)) {
59                (None, Some(revision)) => Some(SnapshotDiff::Added {
60                    entity,
61                    revision: *revision,
62                }),
63                (Some(revision), None) => Some(SnapshotDiff::Removed {
64                    entity,
65                    revision: *revision,
66                }),
67                (Some(from), Some(to)) if from != to => Some(SnapshotDiff::Changed {
68                    entity,
69                    from: *from,
70                    to: *to,
71                }),
72                _ => None,
73            })
74            .collect()
75    }
76}

1//! Defines backend-neutral operations over retained immutable snapshots.
2
3use scriberig_core::{EntityId, RevisionId, SnapshotId};
4use std::collections::BTreeSet;
5
6use crate::{SnapshotChange, SnapshotError};
7
8/// Defines backend-neutral snapshot operations.
9pub trait SnapshotStore {
10    /// Resolves an entity in a stored snapshot.
11    ///
12    /// # Parameters
13    /// - `snapshot`: Exact snapshot identity.
14    /// - `entity`: Entity identity to resolve.
15    ///
16    /// # Returns
17    /// The revision mapped to the entity, when present.
18    ///
19    /// # Errors
20    /// Returns an error when the snapshot is unavailable.
21    fn resolve(
22        &self,
23        snapshot: SnapshotId,
24        entity: EntityId,
25    ) -> Result<Option<RevisionId>, SnapshotError>;
26
27    /// Applies storage changes and returns the resulting snapshot.
28    ///
29    /// # Parameters
30    /// - `base`: Existing snapshot identity.
31    /// - `changes`: Exact entity mapping changes to apply.
32    ///
33    /// # Returns
34    /// The immutable resulting snapshot identity.
35    ///
36    /// # Errors
37    /// Returns an error when the base snapshot is unavailable.
38    fn apply_changes(
39        &self,
40        base: SnapshotId,
41        changes: &[SnapshotChange],
42    ) -> Result<SnapshotId, SnapshotError>;
43
44    /// Iterates a stored snapshot deterministically.
45    ///
46    /// # Parameters
47    /// - `snapshot`: Exact snapshot identity.
48    ///
49    /// # Returns
50    /// Entity and revision mappings in ascending entity order.
51    ///
52    /// # Errors
53    /// Returns an error when the snapshot is unavailable.
54    #[allow(clippy::iter_not_returning_iterator)]
55    fn iter(&self, snapshot: SnapshotId) -> Result<Vec<(EntityId, RevisionId)>, SnapshotError>;
56
57    /// Computes exact mapping changes between stored snapshots.
58    ///
59    /// # Parameters
60    /// - `from`: Starting snapshot identity.
61    /// - `to`: Ending snapshot identity.
62    ///
63    /// # Returns
64    /// Deterministically ordered storage changes.
65    ///
66    /// # Errors
67    /// Returns an error when either snapshot is unavailable.
68    fn diff_snapshots(
69        &self,
70        from: SnapshotId,
71        to: SnapshotId,
72    ) -> Result<Vec<SnapshotChange>, SnapshotError>;
73
74    /// Returns the direct immutable predecessors of a snapshot.
75    ///
76    /// # Parameters
77    /// - `snapshot`: Exact snapshot whose parents are requested.
78    ///
79    /// # Returns
80    /// Deterministically ordered direct parent snapshots.
81    ///
82    /// # Errors
83    /// Returns an error when the snapshot is unavailable.
84    fn parents(&self, snapshot: SnapshotId) -> Result<Vec<SnapshotId>, SnapshotError>;
85
86    /// Walks immutable snapshot ancestry without exposing backend details.
87    ///
88    /// # Parameters
89    /// - `snapshot`: Exact snapshot at which to begin the walk.
90    ///
91    /// # Returns
92    /// Ancestors in deterministic breadth-first order, excluding `snapshot`.
93    ///
94    /// # Errors
95    /// Returns an error when the starting snapshot or an ancestor is unavailable.
96    fn ancestors(&self, snapshot: SnapshotId) -> Result<Vec<SnapshotId>, SnapshotError> {
97        let mut visited = BTreeSet::new();
98        let mut pending = self.parents(snapshot)?;
99        let mut result = Vec::new();
100        while let Some(candidate) = pending.pop() {
101            if !visited.insert(candidate) {
102                continue;
103            }
104            result.push(candidate);
105            pending.extend(self.parents(candidate)?);
106        }
107        result.sort();
108        Ok(result)
109    }
110}

5/// Machine-readable category for a storage failure.
6#[derive(Clone, Copy, Debug, Eq, PartialEq)]
7pub enum StoreErrorCategory {
8    /// Requested immutable data was not found.
9    NotFound,
10    /// An identifier or canonical encoding is invalid.
11    InvalidEncoding,
12    /// The backend or encoding version is unsupported.
13    UnsupportedVersion,
14    /// An optimistic root update used a stale value.
15    StaleRoot,
16    /// Persisted canonical data failed integrity checks.
17    Integrity,
18    /// The requested operation is invalid.
19    InvalidOperation,
20    /// The backend could not complete an I/O or transaction operation.
21    Backend,
22}
23
24impl crate::StoreError {
25    /// Returns the backend-neutral category for this revision error.
26    ///
27    /// # Returns
28    /// A stable machine-readable error category.
29    #[must_use]
30    pub const fn category(&self) -> StoreErrorCategory {
31        match self {
32            Self::IntegrityCollision { .. } | Self::RevisionIdentityMismatch { .. } => {
33                StoreErrorCategory::Integrity
34            }
35            Self::InvalidCanonicalEncoding(error) => match error {
36                scriberig_core::EncodingError::UnsupportedVersion { .. } => {
37                    StoreErrorCategory::UnsupportedVersion
38                }
39            },
40            Self::NotFound { .. } => StoreErrorCategory::NotFound,
41            Self::InvalidOperation { .. } => StoreErrorCategory::InvalidOperation,
42            Self::LockPoisoned | Self::Backend(_) => StoreErrorCategory::Backend,
43        }
44    }
45}

47impl crate::RootError {
48    /// Returns the backend-neutral category for this root error.
49    ///
50    /// # Returns
51    /// A stable machine-readable error category.
52    #[must_use]
53    pub const fn category(&self) -> StoreErrorCategory {
54        match self {
55            Self::Conflict { .. } => StoreErrorCategory::StaleRoot,
56            Self::Backend(_) => StoreErrorCategory::Backend,
57        }
58    }
59}

61impl crate::SnapshotError {
62    /// Returns the backend-neutral category for this snapshot error.
63    ///
64    /// # Returns
65    /// A stable machine-readable error category.
66    #[must_use]
67    pub const fn category(&self) -> StoreErrorCategory {
68        match self {
69            Self::UnknownSnapshot(_) => StoreErrorCategory::NotFound,
70            Self::Backend(_) => StoreErrorCategory::Backend,
71        }
72    }

75/// Counts and optional backend storage measurements.
76#[derive(Clone, Debug, Default, Eq, PartialEq)]
77pub struct StoreStats {
78    /// Number of retained immutable revisions.
79    pub revisions: Option<u64>,
80    /// Number of retained snapshots.
81    pub snapshots: Option<u64>,
82    /// Number of named roots.
83    pub named_roots: Option<u64>,
84    /// Backend storage bytes when cheaply available.
85    pub storage_bytes: Option<u64>,
86    /// Number of reachable objects from the last analysis.
87    pub reachable_objects: Option<u64>,
88    /// Number of unreachable candidates from the last analysis.
89    pub unreachable_objects: Option<u64>,
90}

92/// Current status of one named root.

117/// Optional observability contract implemented by storage backends.
118pub trait StoreObservability {
119    /// Returns backend statistics without changing canonical state.
120    ///
121    /// # Returns
122    /// A snapshot of available storage measurements.
123    ///
124    /// # Errors
125    /// Returns a backend error when statistics cannot be read.
126    fn stats(&self) -> Result<StoreStats, String>;
127
128    /// Lists named roots without exposing backend handles.
129    ///
130    /// # Returns
131    /// Root status records in deterministic name order.
132    ///
133    /// # Errors
134    /// Returns a backend error when roots cannot be read.
135    fn root_statuses(&self) -> Result<Vec<RootStatus>, String>;
136
137    /// Rebuilds derived indexes without changing canonical objects.
138    ///
139    /// # Returns
140    /// The resulting index status.
141    ///
142    /// # Errors
143    /// Returns a backend error when rebuild fails.
144    fn rebuild_indexes(&self) -> Result<IndexStatus, String>;
145}

1//! Defines backend-independent immutable revision storage behavior.
2
3use crate::Revision;
4use scriberig_core::{EncodingError, RevisionId, TypeId};
5use std::collections::BTreeMap;
6
7/// Reports a revision storage failure.
8#[derive(Clone, Debug, Eq, PartialEq)]
9pub enum StoreError {
10    /// A stable ID was found with different canonical fields.
11    IntegrityCollision {
12        /// Conflicting revision identity.
13        id: RevisionId,
14    },
15    /// A canonical encoding version or envelope was not supported.
16    InvalidCanonicalEncoding(EncodingError),
17    /// The supplied identity does not match the canonical revision fields.
18    RevisionIdentityMismatch {
19        /// Identity recomputed from canonical fields.
20        expected: RevisionId,
21        /// Identity supplied by the caller.
22        actual: RevisionId,
23    },
24    /// A requested revision was not retained.
25    NotFound {
26        /// Missing revision identity.
27        id: RevisionId,
28    },
29    /// The requested storage operation is invalid in the current state.
30    InvalidOperation {
31        /// Machine-readable operation description.
32        operation: String,
33    },
34    /// A synchronization primitive could not be read or written.
35    LockPoisoned,
36    /// A backend-specific durable storage failure.
37    Backend(String),
38}
39
40/// Stores immutable revisions by their content identity.
41pub trait RevisionStore {
42    /// Inserts an immutable revision idempotently.
43    ///
44    /// - `revision`: Immutable revision to retain.
45    ///
46    /// The revision identity.
47    ///
48    /// Returns storage or integrity errors.
49    /// # Returns
50    /// The operation result.
51    /// # Errors
52    /// Returns an error when the operation fails.
53    fn put(&self, revision: Revision) -> Result<RevisionId, StoreError>;
54    /// Reads an immutable revision by identity.
55    ///
56    /// - `id`: Revision identity.
57    ///
58    /// The retained revision, when present.
59    ///
60    /// Returns a storage synchronization error.
61    /// # Returns
62    /// The operation result.
63    /// # Errors
64    /// Returns an error when the operation fails.
65    fn get(&self, id: RevisionId) -> Result<Option<Revision>, StoreError>;
66
67    /// Reads a batch of immutable revisions.
68    ///
69    /// Backends may use a bulk query. The default preserves compatibility by
70    /// reading revisions individually.
71    ///
72    /// # Parameters
73    /// - `ids`: Immutable revision identities to load.
74    ///
75    /// # Returns
76    /// Retained revisions keyed by identity; missing IDs are omitted.
77    ///
78    /// # Errors
79    /// Returns a storage or integrity error.
80    fn get_many(&self, ids: &[RevisionId]) -> Result<BTreeMap<RevisionId, Revision>, StoreError> {
81        let mut revisions = BTreeMap::new();
82        for id in ids {
83            if let Some(revision) = self.get(*id)? {
84                revisions.insert(*id, revision);
85            }
86        }
87        Ok(revisions)
88    }
89
90    /// Reads only the declared type of an immutable revision.
91    ///
92    /// Backends may use a metadata-only index to avoid reading revision
93    /// payloads. The default preserves compatibility by loading the revision.
94    ///
95    /// # Parameters
96    /// - `id`: Immutable revision identity.
97    ///
98    /// # Returns
99    /// The declared object type, or `None` when the revision is absent.
100    ///
101    /// # Errors
102    /// Returns a storage or metadata validation error.
103    fn get_object_type(&self, id: RevisionId) -> Result<Option<TypeId>, StoreError> {
104        self.get(id)
105            .map(|revision| revision.map(|value| value.object_type().clone()))
106    }
107
108    /// Reads declared types for a batch of revisions without loading payloads.
109    ///
110    /// Backends may use a bulk metadata query. The default preserves
111    /// compatibility by reading metadata one revision at a time.
112    ///
113    /// # Parameters
114    /// - `ids`: Immutable revision identities to inspect.
115    ///
116    /// # Returns
117    /// Types for retained revisions, keyed by revision identity.
118    ///
119    /// # Errors
120    /// Returns a storage or metadata validation error.
121    fn get_object_types(
122        &self,
123        ids: &[RevisionId],
124    ) -> Result<BTreeMap<RevisionId, TypeId>, StoreError> {
125        let mut types = BTreeMap::new();
126        for id in ids {
127            if let Some(object_type) = self.get_object_type(*id)? {
128                types.insert(*id, object_type);
129            }
130        }
131        Ok(types)
132    }
133
134    /// Reads an immutable revision and reports absence as a typed error.
135    ///
136    /// - `id`: Revision identity.
137    ///
138    /// The retained revision.
139    ///
140    /// Returns `NotFound` when the revision is absent or propagates a backend
141    /// error from `get`.
142    /// # Returns
143    /// The operation result.
144    /// # Errors
145    /// Returns an error when the operation fails.
146    fn get_required(&self, id: RevisionId) -> Result<Revision, StoreError> {
147        self.get(id)?.ok_or(StoreError::NotFound { id })
148    }
149    /// Checks whether an immutable revision is retained.
150    ///
151    /// - `id`: Revision identity.
152    ///
153    /// Whether the revision is present.
154    ///
155    /// Returns a storage synchronization error.
156    /// # Returns
157    /// The operation result.
158    /// # Errors
159    /// Returns an error when the operation fails.
160    fn contains(&self, id: RevisionId) -> Result<bool, StoreError>;
161}

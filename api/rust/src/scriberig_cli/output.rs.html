7/// Renders a protocol value as stable pretty JSON.
8///
9/// - `value`: Serializable public response value.
10///
11/// Pretty-printed JSON text.
12///
13/// # Panics
14/// Panics only if a public response type violates its serialization contract.
15#[must_use]
16pub fn render_json<T: serde::Serialize>(value: &T) -> String {
17    serde_json::to_string_pretty(value).expect("public response is serializable")
18}

20/// Renders an inspection result for a selected terminal representation.
21///
22/// - `result`: Shared application inspection result.
23/// - `depth`: Requested trust detail.

38/// Renders one terminal agent result from the shared application result.
39///
40/// - `result`: Durable terminal agent result.
41/// - `operation`: Stable operation identity.
42/// - `format`: Human or structured output representation.
43///
44/// Text intended for standard output.
45#[must_use]
46/// # Returns
47/// The operation result.
48pub fn render_agent_result(
49    result: &scriberig_app::AgentResult,
50    operation: &str,
51    format: crate::RunFormat,
52) -> String {
53    match format {
54        crate::RunFormat::Human => result.final_output.clone(),
55        crate::RunFormat::Json => serde_json::json!({
56            "operation": operation,
57            "session": result.session.to_string(),
58            "root_branch": result.root_branch.to_string(),
59            "final_output": result.final_output,
60            "branch_result": result.branch_result.to_string(),
61            "produced_changes": result
62                .produced_changes
63                .iter()
64                .map(ToString::to_string)
65                .collect::<Vec<_>>(),
66            "validation_results": result.validation_results,
67            "unresolved_questions": result.unresolved_questions,
68            "candidate_changes": result
69                .candidate_changes
70                .iter()
71                .map(|change| {
72                    serde_json::json!({
73                        "migration": change.migration,
74                        "entity": change.entity.map(|value| value.to_string()),
75                        "candidate_snapshot": change
76                            .candidate_snapshot
77                            .map(|value| value.to_string()),
78                        "validation": change.validation,
79                    })
80                })
81                .collect::<Vec<_>>(),
82        })
83        .to_string(),
84    }
85}

87/// Renders non-result diagnostics for the agent command's error stream.
88///
89/// - `result`: Durable terminal agent result.
90/// - `operation`: Stable operation identity.
91///
92/// Human-readable progress and identity diagnostics.
93#[must_use]
94/// # Returns
95/// The operation result.
96pub fn render_agent_diagnostics(result: &scriberig_app::AgentResult, operation: &str) -> String {
97    format!(
98        "operation: {operation}\nsession: {}\nroot branch: {}",
99        result.session, result.root_branch
100    )
101}

103#[cfg(test)]
104mod tests {

111    /// Creates a minimal stable agent result for output tests.
112    ///
113    /// A deterministic terminal result.
114    ///
115    /// This test helper never fails.
116    /// # Returns
117    /// The operation result.
118    /// # Errors
119    /// Returns an error when the operation fails.
120    fn result() -> AgentResult {
121        AgentResult {
122            session: EntityId::from_bytes([1; 16]),
123            root_branch: EntityId::from_bytes([2; 16]),
124            final_output: "answer".to_owned(),
125            branch_result: EntityId::from_bytes([3; 16]),
126            produced_changes: Vec::new(),
127            validation_results: vec!["passed".to_owned()],
128            unresolved_questions: Vec::new(),
129            candidate_changes: Vec::new(),
130        }
131    }
132
133    /// Verifies human output contains only the final semantic answer.
134    #[test]
135    fn human_agent_output_is_final_answer() {
136        assert_eq!(
137            render_agent_result(&result(), "op-1", RunFormat::Human),
138            "answer"
139        );
140    }
141
142    /// Verifies JSON output retains stable semantic handles.
143    #[test]
144    fn json_agent_output_contains_handles() {
145        let output = render_agent_result(&result(), "op-1", RunFormat::Json);
146        assert!(output.contains("\"operation\":\"op-1\""));
147        assert!(output.contains("\"branch_result\""));
148        assert!(output.contains("\"final_output\":\"answer\""));
149    }
150
151    /// Verifies diagnostics are separate from the final result payload.
152    #[test]
153    fn diagnostics_contain_identity_only() {
154        let output = render_agent_diagnostics(&result(), "op-1");
155        assert!(output.contains("operation: op-1"));
156        assert!(output.contains("session:"));
157        assert!(!output.contains("answer"));
158    }

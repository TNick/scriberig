3use clap::{Args, Parser, Subcommand};

5pub mod trust;

7/// Parse the command line accepted by the `srig` executable.
8///
9/// # Public members
10/// None.
11///
12/// # Private members
13/// - `command`: Product mode selected by the caller.
14#[derive(Debug, Parser)]
15#[command(name = "srig", version, about, arg_required_else_help = true)]
16pub struct Cli {
17    /// Product mode selected by the caller.
18    #[command(subcommand)]
19    command: Command,
20}
21
22/// Select a native Scriberig product mode.
23///
24/// # Public members
25/// - `Tui`: Starts the terminal UI mode.
26/// - `Serve`: Starts the API server.
27/// - `Run`: Accepts one prompt in one-shot mode.
28/// - `Agent`: Accepts one prompt in agent mode.
29/// - `Maintenance`: Selects repository or database maintenance.
30///
31/// # Private members
32/// None.
33#[derive(Debug, Subcommand)]
34pub enum Command {
35    /// Start the interactive terminal UI.
36    Tui,
37    /// Start the API server.
38    Serve(ServeArgs),
39    /// Run one prompt in headless one-shot mode.
40    Run(PromptArgs),
41    /// Run one prompt through the integrated agent runtime.
42    Agent(PromptArgs),
43    /// Run repository or database maintenance operations.
44    Maintenance,
45    /// Inspect one entity and its trust metadata.
46    Inspect(InspectArgs),
47}
48
49/// Configure the API server startup stub.
50///
51/// # Public members
52/// - `host`: Interface address that the server should bind.
53/// - `port`: TCP port that the server should bind.
54/// - `db`: SQLite database file for the server.
55///
56/// # Private members
57/// None.
58#[derive(Debug, Args)]
59pub struct ServeArgs {
60    /// Interface address that the server should bind.
61    #[arg(long, default_value = "127.0.0.1")]
62    pub host: String,
63    /// TCP port that the server should bind.
64    #[arg(long, default_value_t = 7337)]
65    pub port: u16,
66    /// SQLite database path used by the local API server.
67    #[arg(long, default_value = "playground/dev-app-runtime/scriberig.sqlite")]
68    pub db: std::path::PathBuf,
69}
70
71/// Store a prompt accepted by a prompt-driven command.
72///
73/// # Public members
74/// - `prompt`: User prompt forwarded by a future runtime implementation.
75///
76/// # Private members
77/// None.
78#[derive(Debug, Args)]
79pub struct PromptArgs {
80    /// User prompt forwarded by a future runtime implementation.
81    pub prompt: String,
82}
83
84/// Configure an entity inspection request.
85#[derive(Debug, Args)]
86pub struct InspectArgs {
87    /// Entity identifier to inspect.
88    pub entity: scriberig_core::EntityId,
89    /// Exact snapshot to inspect when supplied.
90    #[arg(long)]
91    pub snapshot: Option<scriberig_core::SnapshotId>,
92    /// Print provenance handles and deeper trust metadata.
93    #[arg(long, value_enum, default_value_t = trust::InspectDepth::Compact)]
94    pub depth: trust::InspectDepth,
95}

97/// Parse command-line input for the `srig` executable.
98///
99/// # Returns

101#[must_use]
102pub fn parse() -> Command {
103    Cli::parse().command

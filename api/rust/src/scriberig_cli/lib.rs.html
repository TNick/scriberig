9/// Reports a CLI dispatch failure while preserving its process status.
10#[derive(Debug)]
11pub struct DispatchError {
12    /// Human-readable safe failure text.
13    pub message: String,
14    /// Process status associated with the failure.
15    pub exit_code: i32,
16}
17
18/// Dispatches an inspect command through an injected application service.
19///
20/// # Parameters
21/// - `service`: Application query service supplied by composition.
22/// - `context`: Caller and capability context.
23/// - `command`: Parsed product command.
24///
25/// # Returns
26/// Optional output for commands handled by this dispatcher.
27///
28/// # Errors
29/// Returns a structured dispatch failure for unsupported or denied commands.
30pub fn dispatch<S: scriberig_app::QueryService>(
31    service: &S,
32    context: &scriberig_app::ApplicationContext,
33    command: Command,
34) -> Result<Option<String>, DispatchError> {
35    match command {
36        Command::Inspect(args) => {
37            inspect::run(service, context, &args)
38                .map(Some)
39                .map_err(|error| DispatchError {
40                    message: match &error {
41                        trust::TrustCommandError::AuthorizationDenied(denial) => {
42                            trust::render_denial(denial)
43                        }
44                        trust::TrustCommandError::Application(application) => {
45                            application.to_string()
46                        }
47                    },
48                    exit_code: error.exit_code(),
49                })
50        }
51        _ => Err(DispatchError {
52            message: "command is not available in this composition".to_owned(),
53            exit_code: 1,
54        }),
55    }
56}

58/// Parse the command line accepted by the `srig` executable.
59///
60/// # Public members
61/// None.
62///
63/// # Private members
64/// - `command`: Product mode selected by the caller.
65#[derive(Debug, Parser)]
66#[command(name = "srig", version, about, arg_required_else_help = true)]
67pub struct Cli {
68    /// Product mode selected by the caller.
69    #[command(subcommand)]
70    command: Command,
71}
72
73/// Select a native Scriberig product mode.
74///
75/// # Public members
76/// - `Tui`: Starts the terminal UI mode.
77/// - `Serve`: Starts the API server.
78/// - `Run`: Accepts one prompt in one-shot mode.
79/// - `Agent`: Accepts one prompt in agent mode.
80/// - `Maintenance`: Selects repository or database maintenance.
81///
82/// # Private members
83/// None.
84#[derive(Debug, Subcommand)]
85pub enum Command {
86    /// Start the interactive terminal UI.
87    Tui(TuiArgs),
88    /// Start the API server.
89    Serve(ServeArgs),
90    /// Run one prompt in headless one-shot mode.
91    Run(PromptArgs),
92    /// Run one prompt through the integrated agent runtime.
93    Agent(PromptArgs),
94    /// Run repository or database maintenance operations.
95    Maintenance,
96    /// Inspect one entity and its trust metadata.
97    Inspect(InspectArgs),
98}

100/// Configure the API server startup.
101///
102/// # Public members
103/// - `host`: Interface address that the server should bind.
104/// - `port`: TCP port that the server should bind.
105/// - `db`: SQLite database file for the server.
106///
107/// # Private members
108/// None.
109#[derive(Debug, Args)]
110pub struct ServeArgs {
111    /// Interface address that the server should bind.
112    #[arg(long, default_value = "127.0.0.1")]
113    pub host: String,
114    /// TCP port that the server should bind.
115    #[arg(long, default_value_t = 7337)]
116    pub port: u16,
117    /// SQLite database path used by the local API server.
118    #[arg(long, default_value = "playground/dev-app-runtime/scriberig.sqlite")]
119    pub db: std::path::PathBuf,
120}

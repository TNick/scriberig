21/// Reports a CLI dispatch failure while preserving its process status.
22#[derive(Debug)]
23pub struct DispatchError {
24    /// Human-readable safe failure text.
25    pub message: String,
26    /// Process status associated with the failure.
27    pub exit_code: i32,
28}
29
30/// Dispatches an inspect command through an injected application service.
31///
32/// # Parameters
33/// - `service`: Application query service supplied by composition.
34/// - `context`: Caller and capability context.
35/// - `command`: Parsed product command.
36///
37/// # Returns
38/// Optional output for commands handled by this dispatcher.
39///
40/// # Errors
41/// Returns a structured dispatch failure for unsupported or denied commands.
42pub fn dispatch<S: scriberig_app::QueryService>(
43    service: &S,
44    context: &scriberig_app::ApplicationContext,
45    command: Command,
46) -> Result<Option<String>, DispatchError> {
47    match command {
48        Command::Inspect(args) => {
49            inspect::run(service, context, &args)
50                .map(Some)
51                .map_err(|error| DispatchError {
52                    message: match &error {
53                        trust::TrustCommandError::AuthorizationDenied(denial) => {
54                            trust::render_denial(denial)
55                        }
56                        trust::TrustCommandError::Application(application) => {
57                            application.to_string()
58                        }
59                    },
60                    exit_code: error.exit_code(),
61                })
62        }
63        _ => Err(DispatchError {
64            message: "command is not available in this composition".to_owned(),
65            exit_code: 1,
66        }),
67    }
68}

70/// Parse command-line input for the `srig` executable.
71///
72/// # Returns
73/// The selected product mode and its arguments.
74///
75/// # Panics
76/// Panics only if the generated Clap parser cannot deserialize its own matches.
77#[must_use]
78pub fn parse() -> Command {
79    let parser = Cli::command();
80    let matches = parser.get_matches();
81    let mut command = Cli::from_arg_matches(&matches)
82        .expect("Clap matches must deserialize into the CLI model")
83        .command;
84    let explicit = matches
85        .subcommand()
86        .map_or_else(BTreeSet::new, |(name, command)| {
87            if matches!(name, "run" | "agent") {
88                explicit_run_options(command)
89            } else {
90                BTreeSet::new()
91            }
92        });
93    match &mut command {
94        Command::Run(args) | Command::Agent(args) => args.explicit_cli = explicit,
95        _ => {}
96    }
97    command
98}

100/// Collects options supplied directly to a run-like command.
101///
102/// # Parameters
103/// - `matches`: Parsed Clap matches for the selected command.
104///
105/// # Returns
106/// Explicit option identifiers used for configuration precedence.
107fn explicit_run_options(matches: &clap::ArgMatches) -> BTreeSet<String> {
108    [
109        "db",
110        "format",
111        "no_stream",
112        "provider",
113        "model",
114        "base_url",
115        "api_key_env",
116        "credential_ref",
117    ]
118    .into_iter()
119    .filter(|name| matches.value_source(name) == Some(clap::parser::ValueSource::CommandLine))
120    .map(str::to_owned)
121    .collect()
122}

124#[cfg(test)]
125mod tests {
126    use super::{Cli, Command, ConfigCommand, RunFormat, explicit_run_options};
127    use clap::{CommandFactory, Parser};
128
129    /// Verifies headless provider options remain part of the native command.
130    #[test]
131    fn run_parses_provider_configuration_without_secret_value() {
132        let cli = Cli::try_parse_from([
133            "srig",
134            "run",
135            "hello",
136            "--provider",
137            "local",
138            "--model",
139            "model-a",
140            "--base-url",
141            "http://127.0.0.1:9000/v1",
142            "--api-key-env",
143            "PROVIDER_KEY",
144            "--format",
145            "json",
146            "--no-stream",
147        ])
148        .expect("run arguments");
149
150        let Command::Run(args) = cli.command else {
151            panic!("expected run command");
152        };
153        assert_eq!(args.prompt, "hello");
154        assert_eq!(args.provider, "local");
155        assert_eq!(args.model, "model-a");
156        assert_eq!(args.api_key_env.as_deref(), Some("PROVIDER_KEY"));
157        assert!(args.no_stream);
158        assert!(matches!(args.format, RunFormat::Json));
159    }
160
161    /// Verifies the effective configuration inspection command parses.
162    #[test]
163    fn config_show_parses_json_output() {
164        let cli = Cli::try_parse_from(["srig", "config", "show", "--format", "json"])
165            .expect("config show arguments");
166        let Command::Config(args) = cli.command else {
167            panic!("expected config command");
168        };
169        let ConfigCommand::Show(show) = args.command;
170        assert!(matches!(show.format, RunFormat::Json));
171    }
172
173    /// Verifies parser defaults do not masquerade as explicit CLI overrides.
174    #[test]
175    fn tracks_only_run_options_supplied_on_the_command_line() {
176        let parser = Cli::command();
177        let matches = parser
178            .try_get_matches_from(["srig", "run", "hello", "--model", "chosen"])
179            .expect("run matches");
180        let run_matches = matches.subcommand_matches("run").expect("run subcommand");
181
182        let explicit = explicit_run_options(run_matches);
183        assert_eq!(explicit, ["model".to_owned()].into_iter().collect());
184    }

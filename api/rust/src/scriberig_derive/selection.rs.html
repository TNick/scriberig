1//! Selects immutable result records using caller-supplied policy.
2
3use crate::freshness::{FreshnessPolicy, FreshnessState, evaluate_freshness};
4use crate::profile::DerivationProfile;
5use crate::result::{DerivedResultRecord, DerivedResultStatus};
6
7/// Selects a compatible candidate and defines how failures/freshness are handled.
8///
9/// Candidate slice order is the caller's explicit ranking; selection never
10/// mutates records or persists a preferred-result pointer.
11#[derive(Clone, Debug, PartialEq)]
12pub struct SelectionPolicy {
13    /// Required exact profile; `None` accepts any profile.
14    pub profile: Option<DerivationProfile>,
15    /// Include deterministic failed results for inspection.
16    pub include_deterministic_failures: bool,
17    /// Require freshness at the provided evaluation time.
18    pub require_fresh: bool,
19    /// Explicit logical timestamp used for freshness evaluation.
20    pub now: u64,
21}
22
23/// Explains which candidate was chosen or why no candidate qualified.
24#[derive(Clone, Debug, PartialEq)]
25pub enum SelectionOutcome<'a> {
26    /// Candidate was selected according to the requested policy.
27    Selected {
28        /// Exact immutable record selected by candidate ranking.
29        record: &'a DerivedResultRecord,
30        /// Structured reason the candidate satisfied the policy.
31        reason: SelectionReason,
32    },
33    /// No candidate met profile, status, and freshness requirements.
34    NoMatch {
35        /// Structured reason no candidate qualified.
36        reason: SelectionReason,
37    },
38}
39
40/// Provides a structured reason for a selection result.
41#[derive(Clone, Copy, Debug, Eq, Hash, PartialEq)]
42pub enum SelectionReason {
43    /// Candidate matched exact profile and status requirements.
44    ExactProfile,
45    /// No candidate had the required profile.
46    ProfileMismatch,
47    /// Candidates existed but none had a reusable status.
48    StatusNotReusable,
49    /// Candidates existed but none met freshness requirements.
50    FreshnessExpired,
51    /// No candidates were supplied.
52    NoStoredResult,
53}
54
55/// Selects the first eligible immutable candidate without mutating it.
56///
57/// # Parameters
58/// - `candidates`: Existing records in caller-defined ranking order.
59/// - `policy`: Profile, failure, and freshness constraints.
60/// - `freshness`: Freshness policy applied to observations.
61///
62/// # Returns
63/// The selected exact record or a structured no-match explanation.
64#[must_use]
65pub fn select<'a>(
66    candidates: &'a [DerivedResultRecord],
67    policy: &SelectionPolicy,
68    freshness: &FreshnessPolicy,
69) -> SelectionOutcome<'a> {
70    if candidates.is_empty() {
71        return SelectionOutcome::NoMatch {
72            reason: SelectionReason::NoStoredResult,
73        };
74    }
75    let mut profile_seen = false;
76    let mut reusable_seen = false;
77    for record in candidates {
78        if policy
79            .profile
80            .as_ref()
81            .is_some_and(|profile| profile != &record.key().profile)
82        {
83            continue;
84        }
85        profile_seen = true;
86        let reusable = matches!(record.status(), DerivedResultStatus::Succeeded)
87            || (policy.include_deterministic_failures
88                && matches!(record.status(), DerivedResultStatus::FailedDeterministic));
89        if !reusable {
90            continue;
91        }
92        reusable_seen = true;
93        if policy.require_fresh
94            && !matches!(
95                evaluate_freshness(freshness, record.freshness(), policy.now),
96                FreshnessState::Fresh | FreshnessState::NotApplicable
97            )
98        {
99            continue;
100        }
101        return SelectionOutcome::Selected {
102            record,
103            reason: SelectionReason::ExactProfile,
104        };
105    }
106    let reason = if !profile_seen {
107        SelectionReason::ProfileMismatch
108    } else if !reusable_seen {
109        SelectionReason::StatusNotReusable
110    } else {
111        SelectionReason::FreshnessExpired
112    };
113    SelectionOutcome::NoMatch { reason }
114}

1//! Defines retention defaults independently from validity and freshness.
2//!
3//! Retention answers how strongly an artifact should be kept under storage
4//! pressure. It does not establish whether the artifact satisfies a derivation
5//! request. Weak cache references do not pin canonical history; pinned
6//! historical evidence can request strong preservation of exact inputs.
7
8/// Suggests how long registered derivation results should be retained.
9#[derive(Clone, Copy, Debug, Default, Eq, Hash, PartialEq)]
10pub enum RetentionPolicy {
11    /// Results may be discarded and recomputed when storage is needed.
12    #[default]
13    Disposable,
14    /// Retain expensive reusable results when practical.
15    ReusableExpensive,
16    /// Retain results as historical evidence.
17    HistoricalEvidence,
18    /// Keep results pinned until explicitly unpinned by an owner.
19    Pinned,
20}
21
22/// Describes whether a derived result preserves its canonical inputs strongly.
23#[derive(Clone, Copy, Debug, Default, Eq, Hash, PartialEq)]
24pub enum RetentionStrength {
25    /// Cache references do not pin canonical input history.
26    #[default]
27    WeakDependency,
28    /// Historical evidence requests preservation of exact canonical inputs.
29    StrongPinnedDependency,
30}

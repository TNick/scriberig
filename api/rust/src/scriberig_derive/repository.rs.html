1//! Defines a backend-independent immutable derived-result repository contract.
2
3use crate::dependency::{Dependency, DependencySet};
4use crate::event::DerivedResultHandle;
5use crate::key::DerivationKey;
6use crate::result::DerivedResultRecord;
7use scriberig_core::{EntityId, RevisionId};
8
9/// Reports an attempted overwrite of an immutable result identity.
10#[derive(Clone, Debug, Eq, PartialEq)]
11pub enum RepositoryError {
12    /// The handle already identifies a different immutable record.
13    ConflictingResult {
14        /// Immutable handle already associated with different record data.
15        handle: DerivedResultHandle,
16    },
17}
18
19/// Stores and discovers immutable derived-result records.
20///
21/// Implementations may deduplicate identical inserts and must permit distinct
22/// handles for one key. Removal is explicit and applies only to disposable
23/// derived state; it has no canonical snapshot semantics.
24pub trait DerivedRepository {
25    /// Inserts an immutable result without replacing historical records.
26    ///
27    /// # Parameters
28    /// - `self`: Repository to update.
29    /// - `record`: Immutable derivation outcome.
30    ///
31    /// # Returns
32    /// Success when inserted or already present identically.
33    ///
34    /// # Errors
35    /// Returns a conflict if the handle already names different record data.
36    fn insert(&mut self, record: DerivedResultRecord) -> Result<(), RepositoryError>;
37
38    /// Retrieves an exact immutable record by its stable handle.
39    ///
40    /// # Parameters
41    /// - `self`: Repository to inspect.
42    /// - `handle`: Exact result identity.
43    ///
44    /// # Returns
45    /// The historical record when retained.
46    ///
47    /// # Errors
48    /// This lookup is infallible and returns `None` when absent.
49    ///
50    /// # Errors
51    /// This lookup is infallible and returns `None` when the record is absent.
52    fn get(&self, handle: DerivedResultHandle) -> Option<&DerivedResultRecord>;
53
54    /// Finds every retained record for one exact derivation key.
55    ///
56    /// # Parameters
57    /// - `self`: Repository to inspect.
58    /// - `key`: Exact profile and dependency identity.
59    ///
60    /// # Returns
61    /// Matching immutable records.
62    ///
63    /// # Errors
64    /// This lookup is infallible and returns an empty list when no key matches.
65    ///
66    /// # Errors
67    /// This lookup is infallible and returns an empty list when no key matches.
68    fn by_key(&self, key: &DerivationKey) -> Vec<&DerivedResultRecord>;
69
70    /// Finds result handles that declare one exact dependency.
71    ///
72    /// # Parameters
73    /// - `self`: Repository to inspect.
74    /// - `dependency`: Exact dependency to reverse-index.
75    ///
76    /// # Returns
77    /// Stable handles of results depending on the requested value.
78    ///
79    /// # Errors
80    /// This lookup is infallible and returns an empty list when unmatched.
81    fn by_dependency(&self, dependency: &Dependency) -> Vec<DerivedResultHandle>;
82
83    /// Returns the normalized dependencies recorded for one result.
84    ///
85    /// # Parameters
86    /// - `self`: Repository to inspect.
87    /// - `handle`: Exact result identity.
88    ///
89    /// # Returns
90    /// The immutable dependency set when the result is retained.
91    ///
92    /// # Errors
93    /// This lookup is infallible and returns `None` when absent.
94    fn dependencies(&self, handle: DerivedResultHandle) -> Option<&DependencySet>;
95
96    /// Finds results associated with an exact entity revision for discovery.
97    ///
98    /// # Parameters
99    /// - `self`: Repository to inspect.
100    /// - `entity`: Exact entity identity.
101    /// - `revision`: Exact revision identity.
102    ///
103    /// # Returns
104    /// Stable handles associated with the exact entity revision.
105    ///
106    /// # Errors
107    /// This lookup is infallible and returns an empty list when unmatched.
108    fn by_entity_revision(
109        &self,
110        entity: EntityId,
111        revision: RevisionId,
112    ) -> Vec<DerivedResultHandle>;
113
114    /// Removes one record only when its retention category is disposable.
115    ///
116    /// # Parameters
117    /// - `self`: Repository to update.
118    /// - `handle`: Exact result identity.
119    ///
120    /// # Returns
121    /// Whether a disposable record was removed.
122    fn remove_disposable(&mut self, handle: DerivedResultHandle) -> bool;
123}

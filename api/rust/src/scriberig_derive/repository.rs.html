1//! Defines a backend-independent immutable derived-result repository contract.
2
3use crate::event::DerivedResultHandle;
4use crate::key::DerivationKey;
5use crate::result::DerivedResultRecord;

7/// Reports an attempted overwrite of an immutable result identity.
8#[derive(Clone, Debug, Eq, PartialEq)]
9pub enum RepositoryError {
10    /// The handle already identifies a different immutable record.
11    ConflictingResult {
12        /// Immutable handle already associated with different record data.
13        handle: DerivedResultHandle,
14    },
15}
16
17/// Stores and discovers immutable derived-result records.
18///
19/// Implementations may deduplicate identical inserts and must permit distinct
20/// handles for one key. Removal is explicit and applies only to disposable
21/// derived state; it has no canonical snapshot semantics.
22pub trait DerivedRepository {
23    /// Inserts an immutable result without replacing historical records.
24    ///
25    /// # Parameters
26    /// - `self`: Repository to update.
27    /// - `record`: Immutable derivation outcome.
28    ///
29    /// # Returns
30    /// Success when inserted or already present identically.
31    ///
32    /// # Errors
33    /// Returns a conflict if the handle already names different record data.
34    fn insert(&mut self, record: DerivedResultRecord) -> Result<(), RepositoryError>;
35
36    /// Retrieves an exact immutable record by its stable handle.
37    ///
38    /// # Parameters
39    /// - `self`: Repository to inspect.
40    /// - `handle`: Exact result identity.
41    ///
42    /// # Returns
43    /// The historical record when retained.
44    ///
45    /// # Errors
46    /// This lookup is infallible and returns `None` when absent.
47    ///
48    /// # Errors
49    /// This lookup is infallible and returns `None` when the record is absent.
50    fn get(&self, handle: DerivedResultHandle) -> Option<&DerivedResultRecord>;
51
52    /// Finds every retained record for one exact derivation key.
53    ///
54    /// # Parameters
55    /// - `self`: Repository to inspect.
56    /// - `key`: Exact profile and dependency identity.
57    ///
58    /// # Returns
59    /// Matching immutable records.
60    ///
61    /// # Errors
62    /// This lookup is infallible and returns an empty list when no key matches.
63    ///
64    /// # Errors
65    /// This lookup is infallible and returns an empty list when no key matches.
66    fn by_key(&self, key: &DerivationKey) -> Vec<&DerivedResultRecord>;
67
68    /// Removes one record only when its retention category is disposable.
69    ///
70    /// # Parameters
71    /// - `self`: Repository to update.
72    /// - `handle`: Exact result identity.
73    ///
74    /// # Returns
75    /// Whether a disposable record was removed.
76    fn remove_disposable(&mut self, handle: DerivedResultHandle) -> bool;
77}

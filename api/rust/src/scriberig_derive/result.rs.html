11/// Classifies whether a derivation attempt succeeded or can be reused.
12#[derive(Clone, Copy, Debug, Eq, Hash, PartialEq)]
13pub enum DerivedResultStatus {
14    /// A usable result was produced.
15    Succeeded,
16    /// An immutable-input failure may be inspected or reused deterministically.
17    FailedDeterministic,
18    /// A retryable failure should not suppress a later attempt.
19    FailedTransient,
20    /// Work did not complete and should not satisfy a request.
21    Cancelled,
22}
23
24/// Identifies structured failure information without parsing display prose.
25#[derive(Clone, Debug, Eq, Hash, PartialEq)]
26pub enum FailureReason {
27    /// Input did not satisfy derivation requirements.
28    InvalidInput,
29    /// An external dependency was unavailable temporarily.
30    TemporaryUnavailable,
31    /// Work was cancelled before producing a result.
32    Cancelled,
33    /// Adapter-defined machine-readable failure code.
34    Extension(String),
35}

37/// Stores one immutable derivation outcome and its exact inputs.
38///
39/// # Public members
40/// - `new`: Creates a result record.
41/// - `with_freshness`: Attaches an external observation.
42/// - `with_failure`: Attaches a structured failure reason.
43/// - `with_retention`: Sets its retention category and strength.
44/// - `with_payload`: Attaches the derived value.
45/// - `with_provenance`: Attaches producer and input provenance.
46/// - `key`: Returns the exact derivation key.
47/// - `handle`: Returns the immutable result handle.
48/// - `payload`: Returns the optional derived value.
49/// - `provenance`: Returns producer and input provenance.
50/// - `status`: Returns structured attempt status.
51/// - `freshness`: Returns optional observation metadata.
52/// - `failure`: Returns optional structured failure.
53/// - `retention`: Returns retention metadata.
54///
55/// # Private members
56/// - `key`: Exact profile and dependency identity.
57/// - `handle`: Immutable artifact identity.
58/// - `status`: Structured success/failure category.
59/// - `freshness`: Optional source observation data.
60/// - `failure`: Optional machine-readable failure reason.
61/// - `retention`: Storage policy independent of validity.
62/// - `payload`: Optional derived value.
63/// - `provenance`: Producer and input provenance.
64#[derive(Clone, Debug, PartialEq)]
65pub struct DerivedResultRecord {
66    key: DerivationKey,
67    handle: DerivedResultHandle,
68    status: DerivedResultStatus,
69    freshness: Option<FreshnessObservation>,
70    failure: Option<FailureReason>,
71    retention: (RetentionPolicy, RetentionStrength),
72    payload: Option<KnowledgeValue>,
73    provenance: Provenance,
74}
75
76// Construct and inspect immutable derivation outcomes.
77impl DerivedResultRecord {
78    /// Wraps a universal graph artifact with derivation identity and policy.
79    ///
80    /// - `key`: Exact derivation request identity.
81    /// - `handle`: Immutable result identity.
82    /// - `artifact`: Universal derived artifact and its provenance.
83    ///
84    /// A cache record retaining the artifact payload, provenance, and status.
85    #[must_use]
86    /// # Returns
87    /// The operation result.
88    pub fn from_artifact(
89        key: DerivationKey,
90        handle: DerivedResultHandle,
91        artifact: DerivedArtifact,
92    ) -> Self {
93        let status = match artifact.status {
94            DerivedStatus::Complete => DerivedResultStatus::Succeeded,
95            DerivedStatus::Failed => DerivedResultStatus::FailedDeterministic,
96            DerivedStatus::Pending => DerivedResultStatus::Cancelled,
97        };
98        Self::new(key, handle, status)
99            .with_payload_opt(artifact.result)
100            .with_provenance(artifact.provenance)
101    }
102
103    /// Creates an immutable result record.
104    ///
105    /// - `key`: Exact derivation request identity.
106    /// - `handle`: Immutable artifact identity.
107    /// - `status`: Structured attempt outcome.
108    ///
109    /// A record with disposable retention and weak input references.
110    #[must_use]
111    /// # Returns
112    /// The operation result.
113    pub fn new(
114        key: DerivationKey,
115        handle: DerivedResultHandle,
116        status: DerivedResultStatus,
117    ) -> Self {
118        let profile = format!("{}/{}", key.profile.kind(), key.profile.version());
119        let inputs = key
120            .dependencies
121            .as_slice()
122            .iter()
123            .filter_map(|dependency| match dependency {
124                crate::Dependency::ExactRevision { entity, revision } => Some(EntityRevisionRef {
125                    entity_id: *entity,
126                    revision_id: *revision,
127                }),
128                _ => None,
129            })
130            .collect();
131        Self {
132            key,
133            handle,
134            status,
135            freshness: None,
136            failure: None,
137            retention: (
138                RetentionPolicy::Disposable,
139                RetentionStrength::WeakDependency,
140            ),
141            payload: None,
142            provenance: Provenance {
143                source: ProvenanceSource::System,
144                producer: None,
145                producer_version: None,
146                inputs,
147                profile: Some(profile),
148                note: None,
149            },
150        }
151    }

1//! Defines graph-independent schema validation contracts and reports.
2
3use crate::{Schema, matches_shape};
4use scriberig_core::{EntityId, KnowledgeValue, ObjectType};
5use std::collections::BTreeMap;
6
7/// Supplies the semantic data needed by schema validation.
8pub trait ValidationSubject {
9    /// Returns the subject entity type.
10    ///
11    /// # Returns
12    /// The declared entity type.
13    fn entity_type(&self) -> &ObjectType;
14    /// Returns semantic properties.
15    ///
16    /// # Returns
17    /// The subject's property map.
18    fn properties(&self) -> &BTreeMap<String, KnowledgeValue>;
19}
20
21/// Identifies where a validation issue occurred.
22#[derive(Clone, Debug, Eq, PartialEq)]
23pub enum ValidationLocation {
24    /// Whole entity.
25    Entity(EntityId),
26    /// Property key.
27    Property(String),
28    /// Relation type.
29    Relation(String),
30    /// Collection type.
31    Collection(String),
32}
33
34/// Severity of a schema validation issue.
35#[derive(Clone, Copy, Debug, Eq, PartialEq)]
36pub enum ValidationSeverity {
37    /// Recoverable concern.
38    Warning,
39    /// Invalid semantic data.
40    Error,
41}
42
43/// One structured validation issue.
44#[derive(Clone, Debug, Eq, PartialEq)]
45pub struct ValidationIssue {
46    /// Issue severity.
47    pub severity: ValidationSeverity,
48    /// Stable machine-readable code.
49    pub code: String,
50    /// Human-readable explanation.
51    pub message: String,
52    /// Semantic location.
53    pub location: Option<ValidationLocation>,
54}
55
56/// Collection of deterministic schema validation issues.
57#[derive(Clone, Debug, Default, Eq, PartialEq)]
58pub struct ValidationReport {
59    /// Reported issues.
60    pub issues: Vec<ValidationIssue>,
61}
62
63impl ValidationReport {
64    /// Returns whether any error issue exists.
65    ///
66    /// # Returns
67    /// `true` when at least one error issue is present.
68    #[must_use]
69    pub fn has_errors(&self) -> bool {
70        self.issues
71            .iter()
72            .any(|issue| issue.severity == ValidationSeverity::Error)
73    }
74}
75
76/// Validates required properties and declared value shapes.
77#[must_use]
78pub fn validate<S: ValidationSubject>(schema: &Schema, subject: &S) -> ValidationReport {
79    let mut report = ValidationReport::default();
80    let Some(entity) = schema
81        .entity_types
82        .iter()
83        .find(|item| item.entity_type == *subject.entity_type())
84    else {
85        report.issues.push(ValidationIssue {
86            severity: ValidationSeverity::Error,
87            code: "unknown_entity_type".to_owned(),
88            message: "entity type is not declared".to_owned(),
89            location: None,
90        });
91        return report;
92    };
93    for property in &entity.properties {
94        match subject.properties().get(&property.key) {
95            None if property.required => report.issues.push(ValidationIssue {
96                severity: ValidationSeverity::Error,
97                code: "missing_property".to_owned(),
98                message: "required property is missing".to_owned(),
99                location: Some(ValidationLocation::Property(property.key.clone())),
100            }),
101            Some(value) if !matches_shape(value, &property.value_shape) => {
102                report.issues.push(ValidationIssue {
103                    severity: ValidationSeverity::Error,
104                    code: "property_shape".to_owned(),
105                    message: "property value has the wrong shape".to_owned(),
106                    location: Some(ValidationLocation::Property(property.key.clone())),
107                });
108            }
109            _ => {}
110        }
111    }
112    report
113}

115#[cfg(test)]
116mod tests {
117    use super::{ValidationSubject, validate};
118    use crate::{Cardinality, EntityTypeSchema, PropertySchema, Schema, ValueShape};
119    use scriberig_core::{KnowledgeClass, KnowledgeValue, ObjectType};
120    use std::collections::BTreeMap;
121
122    /// Supplies a small subject fixture to validation tests.
123    struct Subject {
124        entity_type: ObjectType,
125        properties: BTreeMap<String, KnowledgeValue>,
126    }
127
128    impl ValidationSubject for Subject {
129        /// Returns the fixture entity type.
130        ///
131        /// # Returns
132        /// The fixture entity type.
133        fn entity_type(&self) -> &ObjectType {
134            &self.entity_type
135        }
136        /// Returns the fixture properties.
137        ///
138        /// # Returns
139        /// The fixture property map.
140        fn properties(&self) -> &BTreeMap<String, KnowledgeValue> {
141            &self.properties
142        }
143    }
144
145    /// Checks required properties and shape mismatches produce structured issues.
146    #[test]
147    fn validation_reports_structured_property_issues() {
148        let entity_type = ObjectType::new("test.note").expect("valid type");
149        let schema = Schema {
150            id: "test".to_owned(),
151            version: 1,
152            name: "Test".to_owned(),
153            entity_types: vec![EntityTypeSchema {
154                entity_type: entity_type.clone(),
155                name: "Note".to_owned(),
156                properties: vec![PropertySchema {
157                    key: "title".to_owned(),
158                    name: "Title".to_owned(),
159                    description: None,
160                    value_shape: ValueShape::String,
161                    required: true,
162                    knowledge_class: KnowledgeClass::Canonical,
163                }],
164                relations: Vec::new(),
165                collections: Vec::new(),
166            }],
167            relations: Vec::new(),
168            collections: Vec::new(),
169        };
170        let subject = Subject {
171            entity_type,
172            properties: BTreeMap::from([(String::from("title"), KnowledgeValue::Bool(true))]),
173        };
174        let report = validate(&schema, &subject);
175        assert!(report.has_errors());
176        assert_eq!(report.issues[0].code, "property_shape");
177        let _ = Cardinality::new(0, None).expect("valid cardinality");
178    }

6/// Identifies an installed domain independently of its Rust crate name.
7#[derive(Clone, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
8pub struct DomainId(String);
9
10impl DomainId {
11    /// Validates and constructs a domain identifier.
12    ///
13    /// # Parameters
14    /// - `value`: Stable nonempty domain identifier.
15    ///
16    /// # Returns
17    /// The domain identifier.
18    ///
19    /// # Errors
20    /// Returns an error when the identifier is empty or whitespace-only.
21    pub fn new(value: impl Into<String>) -> Result<Self, &'static str> {
22        let value = value.into();
23        if value.trim().is_empty() {
24            return Err("domain identifiers must not be empty");
25        }
26        Ok(Self(value))
27    }
28
29    /// Borrows the stable domain identifier.
30    ///
31    /// # Parameters
32    /// - `self`: Domain identifier to inspect.
33    ///
34    /// # Returns
35    /// The identifier text.
36    #[must_use]
37    pub fn as_str(&self) -> &str {
38        &self.0
39    }
40}
41
42/// Describes one installed semantic domain and its advertised functionality.
43#[derive(Clone, Debug, Eq, PartialEq)]
44pub struct DomainDescriptor {
45    /// Stable domain identifier.
46    pub id: DomainId,
47    /// Human-readable domain name.
48    pub name: String,
49    /// Version of the domain schema contract.
50    pub schema_version: u32,
51    /// Entity types supplied by this domain.
52    pub entity_types: Vec<ObjectType>,
53    /// Relation identifiers supplied by this domain.
54    pub relation_types: Vec<String>,
55    /// Capabilities supplied by this domain.
56    pub capabilities: Vec<CapabilityDescriptor>,
57}

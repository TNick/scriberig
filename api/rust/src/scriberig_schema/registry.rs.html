1//! Provides deterministic discovery of installed domains and capabilities.
2
3use std::collections::BTreeMap;
4
5use crate::{CapabilityDescriptor, DomainDescriptor, DomainId};
6use scriberig_core::ObjectType;
7
8/// Reports a failed domain registration.
9#[derive(Clone, Debug, Eq, PartialEq)]
10pub enum RegistryError {
11    /// A domain with the same stable identifier is already registered.
12    DuplicateDomain {
13        /// Identifier already present in the registry.
14        id: DomainId,
15    },
16    /// A capability is advertised for an entity type the domain does not provide.
17    UnknownEntityType {
18        /// Domain containing the invalid capability.
19        domain: DomainId,
20        /// Entity type referenced by the invalid capability.
21        entity_type: ObjectType,
22    },
23}
24
25/// Stores installed domain descriptors without depending on concrete adapters.
26#[derive(Clone, Debug, Default, Eq, PartialEq)]
27pub struct DomainRegistry {
28    domains: BTreeMap<DomainId, DomainDescriptor>,
29}
30
31impl DomainRegistry {
32    /// Creates an empty domain registry.
33    #[must_use]
34    pub const fn new() -> Self {
35        Self {
36            domains: BTreeMap::new(),
37        }
38    }
39
40    /// Registers one installed domain descriptor.
41    ///
42    /// # Parameters
43    /// - `self`: Registry to update.
44    /// - `descriptor`: Domain metadata to register.
45    ///
46    /// # Returns
47    /// Unit after successful registration.
48    ///
49    /// # Errors
50    /// Returns a duplicate or unknown-entity error when registration is invalid.
51    pub fn register(&mut self, descriptor: DomainDescriptor) -> Result<(), RegistryError> {
52        if self.domains.contains_key(&descriptor.id) {
53            return Err(RegistryError::DuplicateDomain { id: descriptor.id });
54        }
55        for capability in &descriptor.capabilities {
56            if !descriptor.entity_types.contains(&capability.entity_type) {
57                return Err(RegistryError::UnknownEntityType {
58                    domain: descriptor.id.clone(),
59                    entity_type: capability.entity_type.clone(),
60                });
61            }
62        }
63        self.domains.insert(descriptor.id.clone(), descriptor);
64        Ok(())
65    }
66
67    /// Finds a domain by stable identifier.
68    ///
69    /// # Parameters
70    /// - `self`: Registry to query.
71    /// - `id`: Domain identifier to find.
72    ///
73    /// # Returns
74    /// The registered descriptor, when present.
75    #[must_use]
76    pub fn domain(&self, id: &DomainId) -> Option<&DomainDescriptor> {
77        self.domains.get(id)
78    }
79
80    /// Lists installed domains in deterministic identifier order.
81    ///
82    /// # Parameters
83    /// - `self`: Registry to query.
84    ///
85    /// # Returns
86    /// Registered domain descriptors.
87    pub fn domains(&self) -> impl Iterator<Item = &DomainDescriptor> {
88        self.domains.values()
89    }
90
91    /// Collects capabilities available for one entity type.
92    ///
93    /// # Parameters
94    /// - `self`: Registry to query.
95    /// - `entity_type`: Entity type whose capabilities are requested.
96    ///
97    /// # Returns
98    /// Capabilities from installed and available providers in deterministic order.
99    #[must_use]
100    pub fn capabilities_for(&self, entity_type: &ObjectType) -> Vec<&CapabilityDescriptor> {
101        let mut capabilities = self
102            .domains
103            .values()
104            .flat_map(|domain| domain.capabilities.iter())
105            .filter(|capability| {
106                capability.entity_type == *entity_type
107                    && capability
108                        .provider
109                        .as_ref()
110                        .is_none_or(|provider| provider.available)
111            })
112            .collect::<Vec<_>>();
113        capabilities.sort_by(|left, right| left.id.cmp(&right.id));
114        capabilities
115    }
116}

118#[cfg(test)]
119mod tests {
120    use super::{DomainRegistry, RegistryError};
121    use crate::{
122        CapabilityCategory, CapabilityDescriptor, CapabilityId, DomainDescriptor, DomainId,
123        ProviderMetadata,
124    };
125    use scriberig_core::ObjectType;
126
127    /// Checks a registered domain can be found by stable identifier.
128    #[test]
129    fn domains_are_discoverable_by_id() {
130        let domain = descriptor("markdown", "markdown.section", true);
131        let id = domain.id.clone();
132        let mut registry = DomainRegistry::new();
133        registry
134            .register(domain)
135            .expect("registration should succeed");
136        assert_eq!(
137            registry.domain(&id).map(|item| item.name.as_str()),
138            Some("Markdown")
139        );
140    }
141
142    /// Checks optional providers do not advertise unavailable capabilities.
143    #[test]
144    fn unavailable_provider_is_not_advertised() {
145        let mut registry = DomainRegistry::new();
146        registry
147            .register(descriptor("rust", "rust.function", false))
148            .expect("registration should succeed");
149        let entity_type = ObjectType::new("rust.function").expect("valid type");
150        assert_eq!(registry.capabilities_for(&entity_type).len(), 0);
151    }
152
153    /// Checks a full-style registry exposes only its installed domains.
154    #[test]
155    fn full_style_registry_combines_installed_domains() {
156        let mut registry = DomainRegistry::new();
157        registry
158            .register(descriptor("markdown", "markdown.section", true))
159            .expect("Markdown registration should succeed");
160        registry
161            .register(descriptor("rust", "rust.function", true))
162            .expect("Rust registration should succeed");
163        let markdown = ObjectType::new("markdown.section").expect("valid type");
164        let rust = ObjectType::new("rust.function").expect("valid type");
165        assert_eq!(registry.capabilities_for(&markdown).len(), 1);
166        assert_eq!(registry.capabilities_for(&rust).len(), 1);
167    }
168
169    /// Checks duplicate domains are rejected deterministically.
170    #[test]
171    fn duplicate_domains_are_rejected() {
172        let mut registry = DomainRegistry::new();
173        registry
174            .register(descriptor("markdown", "markdown.section", true))
175            .expect("first registration should succeed");
176        let error = registry
177            .register(descriptor("markdown", "markdown.section", true))
178            .expect_err("duplicate should fail");
179        assert!(matches!(error, RegistryError::DuplicateDomain { .. }));
180    }
181
182    /// Checks capability identifiers reject unsafe transport characters.
183    #[test]
184    fn capability_ids_are_transport_safe() {
185        assert!(CapabilityId::new("projection.markdown").is_ok());
186        assert!(CapabilityId::new("projection/markdown").is_err());
187    }

189    /// Builds a test domain descriptor with one capability.
190    ///
191    /// # Parameters
192    /// - `domain`: Stable domain identifier.
193    /// - `entity_type`: Entity type advertised by the domain.
194    /// - `available`: Provider availability flag.
195    ///
196    /// # Returns
197    /// A deterministic domain descriptor.
198    fn descriptor(domain: &str, entity_type: &str, available: bool) -> DomainDescriptor {
199        DomainDescriptor {
200            id: DomainId::new(domain).expect("valid domain"),
201            name: "Markdown".to_owned(),
202            schema_version: 1.into(),
203            entity_types: vec![ObjectType::new(entity_type).expect("valid type")],
204            relation_types: Vec::new(),
205            capabilities: vec![CapabilityDescriptor {
206                id: CapabilityId::new("inspect.summary").expect("valid capability"),
207                entity_type: ObjectType::new(entity_type).expect("valid type"),
208                category: CapabilityCategory::Description,
209                description: "Inspect summary".to_owned(),
210                provider: Some(ProviderMetadata {
211                    id: domain.to_owned(),
212                    version: "1".to_owned(),
213                    profile: None,
214                    available,
215                }),
216            }],
217        }
218    }

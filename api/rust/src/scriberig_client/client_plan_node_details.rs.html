1//! Provides a focused Rust client method for node gate and evidence pages.
2
3use super::Client;
4use crate::error::ClientError;
5use scriberig_protocol::{
6    ProtocolTransport, Request, Response, SessionPlanNodeDetailsPage, SessionPlanNodeDetailsRequest,
7};
8
9impl<T: ProtocolTransport> Client<T> {
10    /// Reads one bounded page of a node's gates and evidence.
11    ///
12    /// # Parameters
13    /// - `request`: Session, plan, node, snapshot, and page range.
14    ///
15    /// # Returns
16    /// Gate definitions and associated evidence for the requested page.
17    ///
18    /// # Errors
19    /// Returns a structured server, transport, or response-mismatch error.
20    pub fn plan_node_details(
21        &self,
22        request: SessionPlanNodeDetailsRequest,
23    ) -> Result<SessionPlanNodeDetailsPage, ClientError> {
24        match self.send(Request::PlanNodeDetails(request))?.response {
25            Response::PlanNodeDetails(page) => Ok(page),
26            Response::Error(error) => Err(error.into()),
27            _ => Err(ClientError::UnexpectedResponse),
28        }
29    }
30}

1//! Provides semantic client methods for durable session detail.
2
3use super::Client;
4use crate::error::ClientError;
5use scriberig_protocol::{
6    AgentPresentationPreferencesReadRequest, AgentPresentationPreferencesResponse,
7    AgentPresentationPreferencesSetRequest, BranchReference, ModelRequestDetailRequest,
8    ModelRequestDetailResponse, ProtocolTransport, PublicAgentPresentationPreferences, Request,
9    Response, SessionReadRequest, SessionReadResponse, SessionReference, ToolResultDetailRequest,
10    ToolResultDetailResponse,
11};
12
13impl<T: ProtocolTransport> Client<T> {
14    /// Reads shared presentation preferences for the authenticated principal.
15    ///
16    /// # Parameters
17    /// - `self`: Client using the protocol transport.
18    ///
19    /// # Returns
20    /// Revisioned shared presentation defaults.
21    ///
22    /// # Errors
23    /// Returns a structured server error or response mismatch.
24    pub fn agent_presentation_preferences(
25        &self,
26    ) -> Result<AgentPresentationPreferencesResponse, ClientError> {
27        match self
28            .send(Request::AgentPresentationPreferencesRead(
29                AgentPresentationPreferencesReadRequest {},
30            ))?
31            .response
32        {
33            Response::AgentPresentationPreferences(response) => Ok(response),
34            Response::Error(error) => Err(ClientError::Protocol(error)),
35            _ => Err(ClientError::UnexpectedResponse),
36        }
37    }
38
39    /// Replaces shared preferences at the caller's observed revision.
40    ///
41    /// # Parameters
42    /// - `self`: Client using the protocol transport.
43    /// - `expected_revision`: Revision from the last preference read.
44    /// - `preferences`: Complete set of shared defaults.
45    ///
46    /// # Returns
47    /// Updated revisioned shared presentation defaults.
48    ///
49    /// # Errors
50    /// Returns a structured server error or response mismatch.
51    pub fn set_agent_presentation_preferences(
52        &self,
53        expected_revision: u64,
54        preferences: PublicAgentPresentationPreferences,
55    ) -> Result<AgentPresentationPreferencesResponse, ClientError> {
56        match self
57            .send(Request::AgentPresentationPreferencesSet(
58                AgentPresentationPreferencesSetRequest {
59                    expected_revision,
60                    preferences,
61                },
62            ))?
63            .response
64        {
65            Response::AgentPresentationPreferences(response) => Ok(response),
66            Response::Error(error) => Err(ClientError::Protocol(error)),
67            _ => Err(ClientError::UnexpectedResponse),
68        }
69    }
70
71    /// Reads one branch and its nested descendants on demand.
72    ///
73    /// # Parameters
74    /// - `self`: Client using the protocol transport.
75    /// - `session`: Stable session identity.
76    /// - `branch`: Stable branch identity whose activity is requested.
77    ///
78    /// # Returns
79    /// The selected branch subtree without unrelated branch histories.
80    ///
81    /// # Errors
82    /// Returns a structured server error or response mismatch.
83    pub fn read_session_branch(
84        &self,
85        session: impl Into<String>,
86        branch: impl Into<String>,
87    ) -> Result<SessionReadResponse, ClientError> {
88        match self
89            .send(Request::SessionRead(SessionReadRequest {
90                session: SessionReference { id: session.into() },
91                branch: Some(BranchReference { id: branch.into() }),
92            }))?
93            .response
94        {
95            Response::Session(response) => Ok(response),
96            Response::Error(error) => Err(ClientError::Protocol(error)),
97            _ => Err(ClientError::UnexpectedResponse),
98        }
99    }
100
101    /// Loads the captured safe model request for one invocation.
102    ///
103    /// # Parameters
104    /// - `self`: Client using the protocol transport.
105    /// - `session`: Stable session identity.
106    /// - `branch`: Stable branch identity.
107    /// - `invocation`: Model invocation activity identity.
108    ///
109    /// # Returns
110    /// Bounded provider-neutral request detail retained by the session.
111    ///
112    /// # Errors
113    /// Returns a structured server error or response mismatch.
114    pub fn model_request_detail(
115        &self,
116        session: impl Into<String>,
117        branch: impl Into<String>,
118        invocation: impl Into<String>,
119    ) -> Result<ModelRequestDetailResponse, ClientError> {
120        match self
121            .send(Request::ModelRequestDetail(ModelRequestDetailRequest {
122                session: SessionReference { id: session.into() },
123                branch: BranchReference { id: branch.into() },
124                invocation: invocation.into(),
125            }))?
126            .response
127        {
128            Response::ModelRequestDetail(response) => Ok(response),
129            Response::Error(error) => Err(ClientError::Protocol(error)),
130            _ => Err(ClientError::UnexpectedResponse),
131        }
132    }
133
134    /// Loads bounded safe detail for one durable tool result.
135    ///
136    /// # Parameters
137    /// - `self`: Client using the protocol transport.
138    /// - `session`: Stable session identity.
139    /// - `branch`: Stable branch identity.
140    /// - `result`: Tool result activity identity.
141    ///
142    /// # Returns
143    /// Whitelisted safe result fields and a truncation indicator.
144    ///
145    /// # Errors
146    /// Returns a structured server error or response mismatch.
147    pub fn tool_result_detail(
148        &self,
149        session: impl Into<String>,
150        branch: impl Into<String>,
151        result: impl Into<String>,
152    ) -> Result<ToolResultDetailResponse, ClientError> {
153        match self
154            .send(Request::ToolResultDetail(ToolResultDetailRequest {
155                session: SessionReference { id: session.into() },
156                branch: BranchReference { id: branch.into() },
157                result: result.into(),
158            }))?
159            .response
160        {
161            Response::ToolResultDetail(response) => Ok(response),
162            Response::Error(error) => Err(ClientError::Protocol(error)),
163            _ => Err(ClientError::UnexpectedResponse),
164        }
165    }

1//! Defines explicit and validated projection capabilities.
2
3use std::collections::BTreeSet;
4
5/// Identifies one operation or preservation guarantee of an adapter.
6#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
7#[non_exhaustive]
8pub enum ProjectionFeature {
9    /// Semantic state can be rendered.
10    Renderable,
11    /// An external representation can be parsed.
12    Parseable,
13    /// External edits may be imported.
14    Editable,
15    /// Parsing and rendering preserve semantic meaning.
16    RoundTripCapable,
17    /// Reconciliation can retain structure across edits.
18    StructurePreserving,
19    /// Rendering can preserve source style choices.
20    StylePreserving,
21}
22
23/// Declares supported projection operations and preservation guarantees.
24#[derive(Clone, Debug, Default, Eq, PartialEq)]
25pub struct ProjectionCapabilities {
26    features: BTreeSet<ProjectionFeature>,
27}
28
29/// Errors raised when capability declarations contradict prerequisites.
30#[derive(Clone, Copy, Debug, Eq, PartialEq)]
31pub enum CapabilityError {
32    /// Editing requires parsing external representations.
33    EditableRequiresParseable,
34    /// Round-trip capability requires both parsing and rendering.
35    RoundTripRequiresParseAndRender,
36}
37
38// Construct and inspect capability declarations.
39impl ProjectionCapabilities {
40    /// Creates a capability declaration after validating its implications.
41    ///
42    /// # Parameters
43    /// - `features`: Operations and guarantees supported by the adapter.
44    ///
45    /// # Returns
46    /// A valid capability declaration.
47    ///
48    /// # Errors
49    /// Returns an error when editing or round-trip guarantees lack prerequisites.
50    pub fn new(
51        features: impl IntoIterator<Item = ProjectionFeature>,
52    ) -> Result<Self, CapabilityError> {
53        let features: BTreeSet<_> = features.into_iter().collect();
54        if features.contains(&ProjectionFeature::Editable)
55            && !features.contains(&ProjectionFeature::Parseable)
56        {
57            return Err(CapabilityError::EditableRequiresParseable);
58        }
59        if features.contains(&ProjectionFeature::RoundTripCapable)
60            && !(features.contains(&ProjectionFeature::Renderable)
61                && features.contains(&ProjectionFeature::Parseable))
62        {
63            return Err(CapabilityError::RoundTripRequiresParseAndRender);
64        }
65        Ok(Self { features })
66    }
67
68    /// Reports whether a projection feature is supported.
69    ///
70    /// # Parameters
71    /// - `self`: Capability declaration to inspect.
72    /// - `feature`: Feature to look up.
73    ///
74    /// # Returns
75    /// `true` when the feature is present.
76    #[must_use]
77    pub fn supports(&self, feature: ProjectionFeature) -> bool {
78        self.features.contains(&feature)
79    }
80
81    /// Creates a render-only capability declaration.
82    ///
83    /// # Parameters
84    /// - `self`: No existing value is required.
85    ///
86    /// # Returns
87    /// Capabilities for read-only rendering.
88    #[must_use]
89    pub fn read_only() -> Self {
90        Self {
91            features: BTreeSet::from([ProjectionFeature::Renderable]),
92        }
93    }
94
95    /// Creates capabilities for an editable, renderable projection.
96    ///
97    /// # Parameters
98    /// - `self`: No existing value is required.
99    ///
100    /// # Returns
101    /// Capabilities with parsing and editing enabled.
102    #[must_use]
103    pub fn editable() -> Self {
104        Self {
105            features: BTreeSet::from([
106                ProjectionFeature::Renderable,
107                ProjectionFeature::Parseable,
108                ProjectionFeature::Editable,
109                ProjectionFeature::RoundTripCapable,
110                ProjectionFeature::StructurePreserving,
111                ProjectionFeature::StylePreserving,
112            ]),
113        }
114    }
115}

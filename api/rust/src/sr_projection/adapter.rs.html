11/// Groups exact state and specification inputs needed for reconciliation.
12#[derive(Clone, Debug, Eq, PartialEq)]
13pub struct ReconciliationContext {
14    /// Last synchronization base state.
15    pub base: crate::model::SemanticStateRef,
16    /// Current canonical state.
17    pub current: crate::model::SemanticStateRef,
18    /// Semantic scope being reconciled.
19    pub scope: crate::model::SemanticScopeRef,
20    /// Persistent projection specification.
21    pub specification: crate::model::ProjectionSpecRef,
22    /// Adapter-defined metadata.
23    pub metadata: std::collections::BTreeMap<String, Vec<u8>>,
24}
25
26/// Returns projection results with heap-allocated diagnostics.
27pub type ProjectionResult<Value> = Result<Value, Box<ProjectionDiagnostic>>;
28
29/// Connects one domain representation to generic projection operations.
30///
31/// # Public members
32/// - `capabilities`: Declares supported operations.
33/// - `render`: Produces an external representation.
34/// - `parse`: Parses an external representation into adapter-owned candidate data.
35/// - `reconcile`: Relates parsed data to exact canonical state.
36/// - `validate`: Checks adapter-specific candidate invariants.
37///
38/// # Private members
39/// - None.
40pub trait ProjectionAdapter {
41    /// Adapter-owned parsed semantic candidate.
42    type Candidate;
43    /// Adapter-owned proposed semantic migration.
44    type Migration;
45    /// Adapter-owned semantic conflict representation.
46    type Conflict;
47
48    /// Returns capabilities for the requested projection kind.
49    ///
50    /// # Parameters
51    /// - `self`: Adapter providing the contract.
52    /// - `kind`: Projection kind to inspect.
53    ///
54    /// # Returns
55    /// Capabilities declared for the kind.
56    fn capabilities(&self, kind: &ProjectionKind) -> ProjectionCapabilities;
57
58    /// Renders semantic state using the explicit request inputs.
59    ///
60    /// # Parameters
61    /// - `self`: Adapter performing rendering.
62    /// - `request`: Exact render inputs.
63    ///
64    /// # Returns
65    /// The rendered representation.
66    ///
67    /// # Errors
68    /// Returns a structured diagnostic when rendering fails.
69    fn render(&self, request: &RenderRequest) -> ProjectionResult<RenderedProjection>;
70
71    /// Parses an external representation without assigning canonical identity.
72    ///
73    /// # Parameters
74    /// - `self`: Adapter performing parsing.
75    /// - `kind`: External representation format.
76    /// - `bytes`: Exact external representation bytes.
77    ///
78    /// # Returns
79    /// An adapter-owned semantic candidate.
80    ///
81    /// # Errors
82    /// Returns a structured diagnostic when parsing fails.
83    fn parse(&self, kind: &ProjectionKind, bytes: &[u8]) -> ProjectionResult<Self::Candidate> {
84        let _ = (kind, bytes);
85        Err(Box::new(ProjectionDiagnostic::unsupported(
86            crate::diagnostic::DiagnosticStage::Parse,
87        )))
88    }
89
90    /// Reconciles a candidate against the exact state from which it was edited.
91    ///
92    /// # Parameters
93    /// - `self`: Adapter performing reconciliation.
94    /// - `candidate`: Parsed external candidate.
95    /// - `context`: Exact canonical and projection synchronization context.
96    ///
97    /// # Returns
98    /// A typed outcome distinguishing no-op, proposal, ambiguity, conflict,
99    /// and invalid external state.
100    fn reconcile(
101        &self,
102        candidate: Self::Candidate,
103        context: &ReconciliationContext,
104    ) -> ReconciliationOutcome<Self::Migration, Self::Conflict, Box<ProjectionDiagnostic>> {
105        let _ = (candidate, context);
106        ReconciliationOutcome::InvalidExternalState(Box::new(ProjectionDiagnostic::unsupported(
107            crate::diagnostic::DiagnosticStage::Reconcile,
108        )))
109    }
110
111    /// Validates an adapter-owned candidate before proposing canonical change.
112    ///
113    /// # Parameters
114    /// - `self`: Adapter performing validation.
115    /// - `candidate`: Parsed external candidate.
116    ///
117    /// # Returns
118    /// Success when the candidate satisfies adapter invariants.
119    ///
120    /// # Errors
121    /// Returns a structured diagnostic when validation fails.
122    fn validate(&self, candidate: &Self::Candidate) -> ProjectionResult<()> {
123        let _ = candidate;
124        Err(Box::new(ProjectionDiagnostic::unsupported(
125            crate::diagnostic::DiagnosticStage::Validate,
126        )))
127    }
128}

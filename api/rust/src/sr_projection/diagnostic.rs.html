1//! Defines structured projection lifecycle diagnostics.
2
3use crate::model::{ProjectionRef, SemanticScopeRef};
4
5/// Identifies the projection lifecycle stage associated with a diagnostic.
6#[derive(Clone, Copy, Debug, Eq, Hash, PartialEq)]
7pub enum DiagnosticStage {
8    /// Semantic state could not be rendered.
9    Render,
10    /// External representation observation failed.
11    Observe,
12    /// External representation could not be parsed.
13    Parse,
14    /// Parsed state could not be reconciled.
15    Reconcile,
16    /// Candidate semantic state failed validation.
17    Validate,
18    /// Proposed semantic change could not be committed.
19    Commit,
20    /// Updated canonical state could not be rendered again.
21    Rerender,
22}
23
24/// Carries machine-readable adapter-specific diagnostic detail.
25#[derive(Clone, Debug, Eq, Hash, PartialEq)]
26pub struct DiagnosticDetail {
27    /// Stable adapter or diagnostic namespace.
28    pub code: String,
29    /// Structured opaque detail bytes interpreted by the owning adapter.
30    pub data: Vec<u8>,
31}
32
33/// Describes a failure or concern at one projection lifecycle stage.
34#[derive(Clone, Debug, Eq, PartialEq)]
35pub struct ProjectionDiagnostic {
36    /// Lifecycle stage where the diagnostic occurred.
37    pub stage: DiagnosticStage,
38    /// Projection instance when it is known.
39    pub projection: Option<ProjectionRef>,
40    /// Semantic scope involved when it is known.
41    pub scope: Option<SemanticScopeRef>,
42    /// Structured domain-specific diagnostic payload.
43    pub detail: DiagnosticDetail,
44    /// Optional concise human-readable explanation.
45    pub message: Option<String>,
46    /// Whether retrying the same operation may succeed.
47    pub retryable: bool,
48}

1//! Defines generic and domain-extensible semantic operations.
2
3use scriberig_core::{EntityId, KnowledgeValue, ObjectType, RevisionId};
4
5/// Identifies a semantic collection position without file offsets.
6#[derive(Clone, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
7pub struct CollectionPosition {
8    /// Ordered collection index.
9    pub index: u64,
10}
11
12/// Universal semantic change operations.
13#[derive(Clone, Debug, PartialEq)]
14pub enum Operation {
15    /// Creates an entity with an initial revision.
16    CreateEntity {
17        /// Created entity identity.
18        entity: EntityId,
19        /// Initial entity revision.
20        revision: RevisionId,
21    },
22    /// Replaces an entity revision.
23    ReplaceRevision {
24        /// Changed entity identity.
25        entity: EntityId,
26        /// Replacement revision.
27        revision: RevisionId,
28    },
29    /// Removes an entity from canonical state.
30    RemoveEntity {
31        /// Removed entity identity.
32        entity: EntityId,
33    },
34    /// Adds a semantic relation.
35    AddRelation {
36        /// Source entity.
37        from: EntityId,
38        /// Stable relation name.
39        relation: String,
40        /// Target entity.
41        to: EntityId,
42    },
43    /// Removes a semantic relation.
44    RemoveRelation {
45        /// Source entity.
46        from: EntityId,
47        /// Stable relation name.
48        relation: String,
49        /// Target entity.
50        to: EntityId,
51    },
52    /// Inserts an ordered collection member.
53    InsertCollectionMember {
54        /// Collection entity.
55        collection: EntityId,
56        /// Inserted member.
57        member: EntityId,
58        /// Semantic insertion position.
59        position: CollectionPosition,
60    },
61    /// Removes an ordered collection member.
62    RemoveCollectionMember {
63        /// Collection entity.
64        collection: EntityId,
65        /// Removed member.
66        member: EntityId,
67    },
68    /// Moves an ordered collection member.
69    MoveCollectionMember {
70        /// Collection entity.
71        collection: EntityId,
72        /// Moved member.
73        member: EntityId,
74        /// Semantic destination position.
75        position: CollectionPosition,
76    },
77    /// Replaces collection structure while preserving semantic references.
78    ReplaceCollectionStructure {
79        /// Collection entity.
80        collection: EntityId,
81        /// New ordered members.
82        members: Vec<EntityId>,
83    },
84    /// Carries domain-specific semantic intent opaquely.
85    Domain {
86        /// Domain type namespace.
87        domain: ObjectType,
88        /// Domain operation identifier.
89        operation: String,
90        /// Opaque semantic payload.
91        payload: KnowledgeValue,
92    },
93}

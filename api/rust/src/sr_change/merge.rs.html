1//! Implements backend-independent three-way semantic merge.

3use crate::{ConflictKind, EntityDelta, MergeConflict};

8/// Exact snapshots supplied to a three-way merge.
9#[derive(Clone, Copy, Debug, Eq, PartialEq)]
10pub struct MergeInput {
11    /// Explicit common base snapshot identity.
12    pub base: SnapshotId,
13    /// Left branch snapshot identity.
14    pub left: SnapshotId,
15    /// Right branch snapshot identity.
16    pub right: SnapshotId,
17}
18
19/// Exact state supplied to an optional domain merge strategy.
20#[derive(Clone, Copy, Debug, Eq, PartialEq)]
21pub struct EntityMergeContext {
22    /// Entity being merged.
23    pub entity: EntityId,
24    /// Base revision, if present.
25    pub base: Option<RevisionId>,
26    /// Left revision, if present.
27    pub left: Option<RevisionId>,
28    /// Right revision, if present.
29    pub right: Option<RevisionId>,
30}
31
32/// Result returned by a domain strategy for one conflict.
33#[derive(Clone, Debug, Eq, PartialEq)]
34pub enum DomainMergeOutcome {
35    /// Strategy proved one revision safe.
36    Resolved(Option<RevisionId>),
37    /// Strategy deliberately retained a structured conflict.
38    Conflict(String),
39    /// Strategy does not apply to this entity.
40    NotApplicable,
41}
42
43/// Optional domain extension for safe same-entity merge.
44pub trait DomainMergeStrategy {
45    /// Resolves one exact conflicted entity without mutating snapshots.
46    /// # Parameters
47    /// - `context`: Exact base, left, and right revisions.
48    /// # Returns
49    /// A resolution, explicit conflict detail, or not-applicable result.
50    fn resolve(&self, context: EntityMergeContext) -> DomainMergeOutcome;
51}
52
53/// Successful three-way merge result without automatic promotion.
54#[derive(Clone, Debug, Eq, PartialEq)]
55pub struct MergeResult {
56    /// Explicit merge base.
57    pub base: SnapshotId,
58    /// Left input snapshot.
59    pub left: SnapshotId,
60    /// Right input snapshot.
61    pub right: SnapshotId,
62    /// Automatically or strategy-resolved entity deltas.
63    pub resolved_deltas: Vec<EntityDelta>,
64    /// Unresolved semantic conflicts.
65    pub conflicts: Vec<MergeConflict>,
66}
67
68/// Reports invalid merge input.
69#[derive(Clone, Debug, Eq, PartialEq)]
70pub enum MergeError {
71    /// Supplied snapshots do not match the identities in the request.
72    InputIdentityMismatch,
73}
74
75/// Merges three immutable snapshots using universal entity rules.
76/// # Parameters
77/// - `input`: IDs describing the merge request.
78/// - `base`: Explicit base snapshot.
79/// - `left`: Left branch snapshot.
80/// - `right`: Right branch snapshot.
81/// - `strategy`: Optional domain strategy.
82/// # Returns
83/// Resolved deltas and minimal unresolved conflicts.
84/// # Errors
85/// Returns an identity error when inputs do not match the request IDs.
86pub fn merge_snapshots(
87    input: MergeInput,
88    base: &Snapshot,
89    left: &Snapshot,
90    right: &Snapshot,
91    strategy: Option<&dyn DomainMergeStrategy>,
92) -> Result<MergeResult, MergeError> {
93    if input.base != base.id() || input.left != left.id() || input.right != right.id() {
94        return Err(MergeError::InputIdentityMismatch);
95    }
96    let base_map: std::collections::BTreeMap<_, _> = base.iter().collect();
97    let left_map: std::collections::BTreeMap<_, _> = left.iter().collect();
98    let right_map: std::collections::BTreeMap<_, _> = right.iter().collect();
99    let entities: BTreeSet<_> = base_map
100        .keys()
101        .chain(left_map.keys())
102        .chain(right_map.keys())
103        .copied()
104        .collect();
105    let mut result = MergeResult {
106        base: input.base,
107        left: input.left,
108        right: input.right,
109        resolved_deltas: Vec::new(),
110        conflicts: Vec::new(),
111    };
112    for entity in entities {
113        let b = base_map.get(&entity).copied();
114        let l = left_map.get(&entity).copied();
115        let r = right_map.get(&entity).copied();
116        let left_changed = l != b;
117        let right_changed = r != b;
118        if !left_changed && !right_changed {
119            continue;
120        }
121        if left_changed && !right_changed {
122            add_delta(&mut result.resolved_deltas, entity, b, l);
123            continue;
124        }
125        if right_changed && !left_changed {
126            add_delta(&mut result.resolved_deltas, entity, b, r);
127            continue;
128        }
129        if l == r {
130            add_delta(&mut result.resolved_deltas, entity, b, l);
131            continue;
132        }
133        let kind = if b.is_none() {
134            ConflictKind::CompetingCreate
135        } else if l.is_none() || r.is_none() {
136            ConflictKind::DeleteModify
137        } else {
138            ConflictKind::BothModified
139        };
140        let context = EntityMergeContext {
141            entity,
142            base: b,
143            left: l,
144            right: r,
145        };
146        if let Some(strategy) = strategy {
147            match strategy.resolve(context) {
148                DomainMergeOutcome::Resolved(revision) => {
149                    add_delta(&mut result.resolved_deltas, entity, b, revision);
150                    continue;
151                }
152                DomainMergeOutcome::Conflict(detail) => {
153                    result.conflicts.push(MergeConflict {
154                        entity,
155                        base: b,
156                        left: l,
157                        right: r,
158                        kind: ConflictKind::DomainConflict,
159                        path: None,
160                        detail: Some(detail),
161                    });
162                    continue;
163                }
164                DomainMergeOutcome::NotApplicable => {}
165            }
166        }
167        result.conflicts.push(MergeConflict {
168            entity,
169            base: b,
170            left: l,
171            right: r,
172            kind,
173            path: None,
174            detail: None,
175        });
176    }
177    Ok(result)
178}

180/// Adds a delta when the transition is semantically non-empty.
181///
182/// # Parameters
183/// - `deltas`: Output delta collection.
184/// - `entity`: Changed entity.
185/// - `before`: Base revision.
186/// - `after`: Result revision.
187fn add_delta(
188    deltas: &mut Vec<EntityDelta>,
189    entity: EntityId,
190    before: Option<RevisionId>,
191    after: Option<RevisionId>,
192) {
193    if let Ok(delta) = EntityDelta::new(entity, before, after) {
194        deltas.push(delta);
195    }

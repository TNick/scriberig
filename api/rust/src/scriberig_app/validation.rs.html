1//! Defines provenance-bearing validation evidence and conservative reuse.
2
3use scriberig_core::{EntityRevisionRef, Provenance, SnapshotId};
4
5/// Outcome of a validator, kept distinct from authorization or promotion.
6#[derive(Clone, Copy, Debug, Eq, PartialEq)]
7pub enum ValidationStatus {
8    /// Validation completed successfully.
9    Passed,
10    /// Validation completed and found diagnostics.
11    Failed,
12    /// Validator could not complete.
13    Error,
14    /// Validator intentionally did not run.
15    Skipped,
16    /// Outcome is not known.
17    Unknown,
18}
19
20/// Structured severity for one validation diagnostic.
21#[derive(Clone, Copy, Debug, Eq, PartialEq)]
22pub enum DiagnosticSeverity {
23    /// Informational diagnostic.
24    Info,
25    /// Warning diagnostic.
26    Warning,
27    /// Error diagnostic.
28    Error,
29}
30
31/// One structured validator diagnostic.
32#[derive(Clone, Debug, Eq, PartialEq)]
33pub struct ValidationDiagnostic {
34    /// Diagnostic severity.
35    pub severity: DiagnosticSeverity,
36    /// Stable category code.
37    pub category: String,
38    /// Human-readable message.
39    pub message: String,
40    /// Optional exact subject reference.
41    pub subject: Option<EntityRevisionRef>,
42    /// Optional source coordinates represented as an opaque label.
43    pub source: Option<String>,
44}
45
46/// Immutable validation evidence bound to exact candidate inputs.
47#[derive(Clone, Debug, Eq, PartialEq)]
48pub struct ValidationEvidence {
49    /// Validator identity.
50    pub validator: String,
51    /// Validator version/profile/configuration identity.
52    pub configuration: String,
53    /// Candidate snapshot, when validation was snapshot-scoped.
54    pub candidate: Option<SnapshotId>,
55    /// Exact immutable dependencies inspected by the validator.
56    pub dependencies: Option<Vec<EntityRevisionRef>>,
57    /// Structured outcome.
58    pub status: ValidationStatus,
59    /// Structured diagnostics.
60    pub diagnostics: Vec<ValidationDiagnostic>,
61    /// Produced artifact references.
62    pub artifacts: Vec<EntityRevisionRef>,
63    /// Origin and exact inputs for the evidence itself.
64    pub provenance: Provenance,
65}
66
67/// Returns whether evidence can be reused for a later candidate.
68///
69/// Evidence is reusable only when validator configuration, candidate snapshot,
70/// and complete dependency identities all match exactly.
71///
72/// # Parameters
73/// - `evidence`: Prior validation evidence.
74/// - `candidate`: Candidate snapshot being considered.
75/// - `dependencies`: Exact dependencies observed for the candidate.
76///
77/// # Returns
78/// `true` only when reuse is proven safe.
79#[must_use]
80pub fn evidence_applicable(
81    evidence: &ValidationEvidence,
82    candidate: Option<SnapshotId>,
83    dependencies: Option<&[EntityRevisionRef]>,
84) -> bool {
85    evidence.candidate == candidate
86        && evidence.dependencies.as_deref() == dependencies
87        && evidence.dependencies.is_some()
88}

90/// Supplies a candidate to an application-level validator.
91pub trait Validator {
92    /// Validator-specific error type.
93    type Error;
94    /// Validates one explicit candidate without promoting it.
95    ///
96    /// # Parameters
97    /// - `candidate`: Explicit snapshot and dependency context.
98    ///
99    /// # Returns
100    /// Structured validation evidence.
101    ///
102    /// # Errors
103    /// Returns the validator-specific execution error.
104    fn validate(&self, candidate: ValidationCandidate) -> Result<ValidationEvidence, Self::Error>;
105}
106
107/// Explicit candidate context supplied to a validator.
108#[derive(Clone, Debug, Eq, PartialEq)]
109pub struct ValidationCandidate {
110    /// Candidate snapshot identity.
111    pub snapshot: Option<SnapshotId>,
112    /// Exact dependencies available to the validator.
113    pub dependencies: Option<Vec<EntityRevisionRef>>,
114}

116#[cfg(test)]
117mod tests {
118    use super::{ValidationEvidence, ValidationStatus, evidence_applicable};
119    use scriberig_core::{
120        EntityId, EntityRevisionRef, Provenance, ProvenanceSource, RevisionId, SnapshotId,
121    };
122
123    /// Confirms evidence reuse is conservative and exact.
124    #[test]
125    fn evidence_requires_exact_dependencies() {
126        let dependency = EntityRevisionRef {
127            entity_id: EntityId::from_bytes([1; 16]),
128            revision_id: RevisionId::new([2; 32]),
129        };
130        let evidence = ValidationEvidence {
131            validator: "check".to_owned(),
132            configuration: "v1".to_owned(),
133            candidate: Some(SnapshotId::new([3; 32])),
134            dependencies: Some(vec![dependency]),
135            status: ValidationStatus::Passed,
136            diagnostics: Vec::new(),
137            artifacts: Vec::new(),
138            provenance: Provenance {
139                source: ProvenanceSource::System,
140                producer: None,
141                producer_version: None,
142                inputs: vec![dependency],
143                profile: None,
144                note: None,
145            },
146        };
147        assert!(evidence_applicable(
148            &evidence,
149            evidence.candidate,
150            Some(&[dependency])
151        ));
152        assert!(!evidence_applicable(&evidence, evidence.candidate, None));
153        assert!(!evidence_applicable(
154            &evidence,
155            Some(SnapshotId::new([4; 32])),
156            Some(&[dependency])
157        ));
158    }
159}

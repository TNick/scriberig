1//! Defines the backend contract for shared application operations.
2
3use super::{
4    ApplicationContext, ApplicationError, ApplicationRequest, ApplicationResult, EntityId,
5    MigrationDetail,
6};
7
8/// Provides semantic execution for the universal application request model.
9pub trait ApplicationBackend {
10    /// Reads canonical detail for one migration proposal.
11    ///
12    /// - `_context`: Authorized application context.
13    /// - `_id`: Stable migration handle.
14    ///
15    /// Snapshot-bound migration detail.
16    ///
17    /// Returns unsupported-operation when proposal reads are not configured.
18    /// # Returns
19    /// The operation result.
20    /// # Errors
21    /// Returns an error when the operation fails.
22    fn migration_detail(
23        &self,
24        _context: &ApplicationContext,
25        _id: &str,
26    ) -> Result<MigrationDetail, ApplicationError> {
27        Err(ApplicationError::UnsupportedOperation {
28            operation: "migration detail".to_owned(),
29        })
30    }
31
32    /// Executes one transport-independent application request.
33    ///
34    /// - `self`: Backend owning semantic subsystem providers.
35    /// - `context`: Caller, capability, and request context.
36    /// - `request`: Universal application request to execute.
37    ///
38    /// Structured application result owned by the canonical subsystem.
39    ///
40    /// Returns a structured application failure without transport details.
41    /// # Returns
42    /// The operation result.
43    /// # Errors
44    /// Returns an error when the operation fails.
45    fn execute(
46        &self,
47        context: &ApplicationContext,
48        request: &ApplicationRequest,
49    ) -> Result<ApplicationResult, ApplicationError>;
50
51    /// Resolves the current accepted snapshot for the request.
52    ///
53    /// - `self`: Backend owning snapshot/root state.
54    /// - `_context`: Caller context.
55    ///
56    /// The exact current snapshot.
57    ///
58    /// Returns an unsupported-operation error when no current root exists.
59    /// # Returns
60    /// The operation result.
61    /// # Errors
62    /// Returns an error when the operation fails.
63    fn current_snapshot(
64        &self,
65        _context: &ApplicationContext,
66    ) -> Result<scriberig_core::SnapshotId, ApplicationError> {
67        Err(ApplicationError::UnsupportedOperation {
68            operation: "current snapshot resolution".to_owned(),
69        })
70    }
71
72    /// Reads a session through the owning application subsystem.
73    ///
74    /// - `_context`: Application context for the request.
75    /// - `_request`: Session request to resolve.
76    ///
77    /// The requested session.
78    ///
79    /// Returns an unsupported-operation error by default.
80    /// # Returns
81    /// The operation result.
82    /// # Errors
83    /// Returns an error when the operation fails.
84    fn read_session(
85        &self,
86        _context: &ApplicationContext,
87        _request: &crate::SessionRequest,
88    ) -> Result<scriberig_session::Session, ApplicationError> {
89        Err(ApplicationError::UnsupportedOperation {
90            operation: "session read".to_owned(),
91        })
92    }
93
94    /// Atomically persists a session execution selection through its owner.
95    ///
96    /// - `_context`: Authorized application context.
97    /// - `_session`: Session snapshot with the updated selection.
98    /// - `_expected_revision`: Selection revision observed by the caller; the
99    ///   owner must compare it atomically with the persisted revision.
100    ///
101    /// Unit after the update is persisted.
102    ///
103    /// Returns conflict for a stale revision, or unsupported-operation when
104    /// the backend cannot atomically mutate session state.
105    /// # Returns
106    /// The operation result.
107    /// # Errors
108    /// Returns an error when the operation fails.
109    fn update_session_execution_selection(
110        &self,
111        _context: &ApplicationContext,
112        _session: &scriberig_session::Session,
113        _expected_revision: u64,
114    ) -> Result<(), ApplicationError> {
115        Err(ApplicationError::UnsupportedOperation {
116            operation: "session execution selection update".to_owned(),
117        })
118    }
119
120    /// Starts an agent command through the owning application subsystem.
121    ///
122    /// - `_context`: Application context for the request.
123    /// - `_request`: Agent command request to submit.
124    ///
125    /// The accepted operation identity.
126    ///
127    /// Returns an unsupported-operation error by default.
128    /// # Returns
129    /// The operation result.
130    /// # Errors
131    /// Returns an error when the operation fails.
132    fn submit_agent_command(
133        &self,
134        _context: &ApplicationContext,
135        _request: &crate::AgentCommandRequest,
136    ) -> Result<scriberig_core::OperationId, ApplicationError> {
137        Err(ApplicationError::UnsupportedOperation {
138            operation: "agent command submission".to_owned(),
139        })
140    }
141
142    /// Persists one complete changed session snapshot through its owning backend.
143    ///
144    /// - `_context`: Authorized application request context.
145    /// - `_session`: Complete provider-neutral session snapshot.
146    ///
147    /// Returns unsupported-operation when the backend has no session mutation.
148    /// # Returns
149    /// The operation result.
150    /// # Errors
151    /// Returns an error when the operation fails.
152    fn save_session(
153        &self,
154        _context: &ApplicationContext,
155        _session: &scriberig_session::Session,
156    ) -> Result<(), ApplicationError> {
157        Err(ApplicationError::UnsupportedOperation {
158            operation: "session mutation".to_owned(),
159        })
160    }
161
162    /// Atomically replaces one plan at its observed durable revision.
163    /// Newly settled worker results are committed on their exact branches with
164    /// the semantic plan revision; unrelated branch records remain untouched.
165    ///
166    /// - `_context`: Authorized application context.
167    /// - `_session`: Complete snapshot containing the proposed plan.
168    /// - `_plan`: Stable plan identity.
169    /// - `_expected_revision`: Plan revision observed by the caller.
170    ///
171    /// Returns unsupported-operation when atomic plan updates are unavailable.
172    /// # Returns
173    /// The operation result.
174    /// # Errors
175    /// Returns an error when the operation fails.
176    fn compare_and_swap_plan(
177        &self,
178        _context: &ApplicationContext,
179        _session: &scriberig_session::Session,
180        _plan: EntityId,
181        _expected_revision: u64,
182    ) -> Result<(), ApplicationError> {
183        Err(ApplicationError::UnsupportedOperation {
184            operation: "atomic plan update".to_owned(),
185        })
186    }
187}

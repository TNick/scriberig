27/// Holds bounded safe detail for one durable tool result.
28///
29/// # Public members
30/// - result: Stable tool-result identity.
31/// - detail: Allowlisted result fields safe for client presentation.
32/// - truncated: Whether the detail budget omitted safe values.
33///
34/// # Private members
35/// None.
36#[derive(Clone, Debug, Eq, PartialEq)]
37pub struct ToolResultDetail {
38    /// Stable tool-result identity.
39    pub result: EntityId,
40    /// Whitelisted provider-neutral result fields.
41    pub detail: serde_json::Value,
42    /// Whether safe fields were truncated to satisfy the response bound.
43    pub truncated: bool,
44}

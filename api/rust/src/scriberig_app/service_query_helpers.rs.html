3use scriberig_query::{DetailLevel as QueryDetailLevel, Inspection, TargetRef};

5use crate::{ApplicationError, DetailLevel, InspectRequest, InspectResult};

7/// Converts a query inspection into the application result shape.
8///
9/// # Parameters
10/// - `super`: Application helper module.
11/// - `inspection`: Query-layer inspection result.
12/// - `_detail`: Detail requested by the caller.
13///
14/// # Returns
15/// Snapshot-aware application inspection metadata.
16///
17/// # Errors
18/// Returns a revision mismatch when the query target is inconsistent.
19pub(super) fn inspect_result(
20    inspection: Inspection,
21    _detail: DetailLevel,
22) -> Result<InspectResult, ApplicationError> {
23    let entity = match inspection.subject {
24        TargetRef::Snapshot(target) => target.entity,
25        TargetRef::Revision(target) => target.entity,
26    };
27    let revision = match inspection.subject {
28        TargetRef::Revision(target) => {
29            if inspection
30                .revision
31                .is_some_and(|value| value != target.revision)
32            {
33                return Err(ApplicationError::RevisionMismatch {
34                    entity,
35                    revision: target.revision,
36                });
37            }
38            Some(target.revision)
39        }
40        TargetRef::Snapshot(_) => inspection.revision,
41    };
42    Ok(InspectResult {
43        entity,
44        revision,
45        snapshot: inspection.snapshot,
46        object_type: inspection.entity_type,
47        label: inspection.label,
48        description: inspection.description,
49        capabilities: inspection
50            .operations
51            .into_iter()
52            .map(|capability| capability.id)
53            .chain(inspection.views.into_iter().map(|capability| capability.id))
54            .collect(),
55        trust: None,
56    })
57}

59/// Converts an application inspection request into an exact query target.
60///
61/// # Parameters
62/// - `super`: Application helper module.
63/// - `request`: Application inspection request.
64/// - `current`: Resolved current snapshot for a current request.
65///
66/// # Returns
67/// Query-layer inspection request with an exact target.
68///
69/// # Errors
70/// Returns an internal error when current snapshot resolution is missing.
71pub(super) fn query_request(
72    request: &InspectRequest,
73    current: Option<scriberig_core::SnapshotId>,
74) -> Result<scriberig_query::InspectRequest, ApplicationError> {
75    let target = match request.snapshot {
76        crate::SnapshotResolution::At(snapshot) => {
77            TargetRef::Snapshot(scriberig_query::EntityInSnapshot {
78                entity: request.entity,
79                snapshot,
80            })
81        }
82        crate::SnapshotResolution::Revision(revision) => {
83            TargetRef::Revision(scriberig_query::EntityAtRevision {
84                entity: request.entity,
85                revision,
86            })
87        }
88        crate::SnapshotResolution::Current => {
89            TargetRef::Snapshot(scriberig_query::EntityInSnapshot {
90                entity: request.entity,
91                snapshot: current.ok_or_else(|| ApplicationError::Internal {
92                    message: "current snapshot was not resolved".to_owned(),
93                })?,
94            })
95        }
96    };
97    Ok(scriberig_query::InspectRequest {
98        target,
99        detail: match request.detail {
100            DetailLevel::Compact => QueryDetailLevel::Compact,
101            DetailLevel::Detailed => QueryDetailLevel::Structure,
102        },
103        view: None,
104    })
105}

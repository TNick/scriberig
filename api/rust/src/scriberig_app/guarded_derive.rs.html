1//! Enforces application capability checks before exposing derived artifacts.
2
3use crate::{ApplicationContext, CapabilityAction, CapabilityScope, OperationKind};
4use scriberig_core::RevisionId;
5use scriberig_derive::{Dependency, DerivedRepository, DerivedResultHandle, DerivedResultRecord};
6use std::collections::BTreeSet;
7
8/// Wraps a shared derived repository with dependency-aware application checks.
9pub struct GuardedDerivedRepository<R> {
10    /// Immutable cache repository shared by all principals.
11    inner: R,
12}
13
14impl<R> GuardedDerivedRepository<R> {
15    /// Creates a guarded view over a reusable derived repository.
16    ///
17    /// - `inner`: Shared policy-free derived repository.
18    ///
19    /// A repository adapter that checks dependencies on reads.
20    #[must_use]
21    pub const fn new(inner: R) -> Self {
22        Self { inner }
23    }
24
25    /// Borrows the shared repository without applying application policy.
26    ///
27    /// The underlying immutable cache repository.
28    #[must_use]
29    pub const fn inner(&self) -> &R {
30        &self.inner
31    }
32}
33
34impl<R: DerivedRepository> GuardedDerivedRepository<R> {
35    /// Returns a cached artifact only when every dependency is readable.
36    ///
37    /// - `context`: Principal capabilities for the cache request.
38    /// - `handle`: Exact immutable artifact handle.
39    ///
40    /// The cached artifact when retained and authorized; otherwise `None`.
41    ///
42    /// This lookup is infallible; inaccessible artifacts are represented by
43    /// `None`.
44    /// # Returns
45    /// The operation result.
46    /// # Errors
47    /// Returns an error when the operation fails.
48    pub fn get(
49        &self,
50        context: &ApplicationContext,
51        handle: DerivedResultHandle,
52    ) -> Option<&DerivedResultRecord> {
53        let mut visiting = BTreeSet::new();
54        self.authorized_record(context, handle, &mut visiting)
55    }
56
57    /// Checks one artifact and recursively checks nested derived dependencies.
58    fn authorized_record<'a>(
59        &'a self,
60        context: &ApplicationContext,
61        handle: DerivedResultHandle,
62        visiting: &mut BTreeSet<RevisionId>,
63    ) -> Option<&'a DerivedResultRecord> {
64        if !visiting.insert(handle.artifact_id) {
65            return None;
66        }
67        let record = self.inner.get(handle)?;
68        let authorized = record
69            .dependencies()
70            .as_slice()
71            .iter()
72            .all(|dependency| self.dependency_visible(context, dependency, visiting));
73        visiting.remove(&handle.artifact_id);
74        authorized.then_some(record)
75    }
76
77    /// Checks one declared dependency without inspecting its payload.
78    ///
79    /// - `context`: Principal capabilities for the cache request.
80    /// - `dependency`: Declared dependency to authorize.
81    /// - `visiting`: Handles currently being checked for cycle detection.
82    ///
83    /// `true` when the dependency and all nested dependencies are readable.
84    /// # Returns
85    /// The operation result.
86    fn dependency_visible(
87        &self,
88        context: &ApplicationContext,
89        dependency: &Dependency,
90        visiting: &mut BTreeSet<RevisionId>,
91    ) -> bool {
92        match dependency {
93            Dependency::ExactRevision { entity, .. } => {
94                can_read(context, &CapabilityScope::Entity(*entity), None)
95            }
96            Dependency::Snapshot(snapshot) => {
97                can_read(context, &CapabilityScope::Project, Some(*snapshot))
98            }
99            Dependency::Scope(_) | Dependency::Configuration(_) => {
100                can_read(context, &CapabilityScope::Project, None)
101            }
102            Dependency::DerivedResult(artifact_id) => self
103                .authorized_record(
104                    context,
105                    DerivedResultHandle {
106                        artifact_id: *artifact_id,
107                    },
108                    visiting,
109                )
110                .is_some(),
111        }
112    }
113}
114
115/// Checks query or inspect authority for one dependency scope.
116///
117/// - `context`: Principal capabilities for the cache request.
118/// - `scope`: Dependency scope to authorize.
119/// - `snapshot`: Exact snapshot constraint, when available.
120///
121/// `true` when query or inspect authority covers the dependency.
122/// # Returns
123/// The operation result.
124fn can_read(
125    context: &ApplicationContext,
126    scope: &CapabilityScope,
127    snapshot: Option<scriberig_core::SnapshotId>,
128) -> bool {
129    context.capabilities.permits(
130        &CapabilityAction::Query,
131        scope,
132        snapshot,
133        Some(OperationKind::Query),
134    ) || context.capabilities.permits(
135        &CapabilityAction::Inspect,
136        scope,
137        snapshot,
138        Some(OperationKind::Inspect),
139    )
140}

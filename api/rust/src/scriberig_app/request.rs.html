1//! Defines the transport-independent universal application request model.
2
3use crate::DomainOperationRequest;
4use scriberig_core::{RootName, SnapshotId};
5use scriberig_query::{FindRequest, FollowRequest, InspectRequest, ListRequest, QuerySpec};
6
7/// Resolves an application request to an exact snapshot before execution.
8#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
9pub enum SnapshotSelector {
10    /// Use this exact immutable snapshot.
11    Exact(SnapshotId),
12    /// Resolve this named root through application state.
13    NamedRoot(RootName),
14}
15
16/// Identifies a semantic diff request.
17#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
18pub struct DiffRequest {
19    /// Earlier exact snapshot.
20    pub from: SnapshotId,
21    /// Later exact snapshot.
22    pub to: SnapshotId,
23}
24
25/// Identifies exact candidates for an explicit three-way semantic merge.
26#[derive(Clone, Copy, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
27pub struct MergeRequest {
28    /// Exact common candidate base.
29    pub base: SnapshotId,
30    /// Exact candidate produced by the first dependency.
31    pub left: SnapshotId,
32    /// Exact candidate produced by the second dependency.
33    pub right: SnapshotId,
34}
35
36/// Identifies a history request for a semantic target.
37#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
38pub struct HistoryRequest {
39    /// Snapshot or named root from which history is resolved.
40    pub snapshot: SnapshotSelector,
41}
42
43/// Identifies a rendering request for a semantic target.
44#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
45pub struct RenderRequest {
46    /// Exact snapshot context for rendering.
47    pub snapshot: SnapshotSelector,
48    /// Registered projection/view identifier.
49    pub view: String,
50}
51
52/// Identifies a branch operation without implementing branch mutation.
53#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
54pub struct BranchRequest {
55    /// Root from which the branch is created.
56    pub base: SnapshotSelector,
57    /// Requested branch name.
58    pub name: RootName,
59}
60
61/// Identifies an application change-application request.
62#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
63pub struct ApplyRequest {
64    /// Exact base snapshot required by the change.
65    pub base_snapshot: SnapshotId,
66    /// Optional branch receiving the candidate context.
67    pub branch: Option<RootName>,
68    /// Stable migration/change handle.
69    pub change: String,
70}
71
72/// Identifies a commit request for an already validated change.
73#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
74pub struct CommitRequest {
75    /// Exact base snapshot being committed.
76    pub base_snapshot: SnapshotId,
77    /// Named branch whose root is promoted, or `main` when omitted.
78    pub branch: Option<RootName>,
79    /// Stable migration/change handle.
80    pub change: String,
81}
82
83/// Selects one universal application operation.
84#[derive(Clone, Debug, Eq, PartialEq)]
85pub enum ApplicationRequest {
86    /// Inspect an exact semantic target.
87    Inspect(InspectRequest),
88    /// List a snapshot-bound collection.
89    List(ListRequest),
90    /// Follow a snapshot-bound relation.
91    Follow(FollowRequest),
92    /// Discover semantic entities in an exact scope.
93    Find(FindRequest),
94    /// Evaluate a structural query in an exact scope.
95    Query(QuerySpec),
96    /// Compute a semantic diff.
97    Diff(DiffRequest),
98    /// Merge exact candidate snapshots without promoting a root.
99    Merge(MergeRequest),
100    /// Read semantic history.
101    History(HistoryRequest),
102    /// Render through a registered view.
103    Render(RenderRequest),
104    /// Create or update a branch context.
105    Branch(BranchRequest),
106    /// Apply a proposed change.
107    Apply(ApplyRequest),
108    /// Commit a validated change.
109    Commit(CommitRequest),
110    /// Execute a registered exact-target semantic domain operation.
111    DomainOperation(DomainOperationRequest),
112}

14/// Persists ordered model requests and tool invocation/result pairs.
15///
16/// - `service`: Application identity allocator.
17/// - `branch`: Session branch that owns the execution evidence.
18/// - `tool_evidence`: Ordered tool inputs and bounded outputs from the runtime.
19/// - `model_requests`: Ordered provider-neutral request projections.
20fn persist_execution_evidence<B>(
21    service: &ApplicationService<B>,
22    branch: &mut Branch,
23    tool_evidence: Vec<crate::ToolExecutionEvidence>,
24    model_requests: Vec<ModelRequestProjection>,
25) {
26    let mut ordered = tool_evidence
27        .into_iter()
28        .map(|item| (item.ordinal, ExecutionEvidence::Tool(item)))
29        .chain(
30            model_requests
31                .into_iter()
32                .map(|item| (item.ordinal, ExecutionEvidence::Model(item))),
33        )
34        .collect::<Vec<_>>();
35    ordered.sort_by_key(|(ordinal, _)| *ordinal);
36    let mut parent_model_invocation = None;
37    for (_, evidence) in ordered {
38        match evidence {
39            ExecutionEvidence::Tool(evidence) => {
40                let invocation = service.next_entity(0xB7);
41                let result = service.next_entity(0xB8);
42                branch.tool_invocations.push(ToolInvocation {
43                    id: invocation,
44                    name: evidence.name,
45                    input: evidence.input,
46                    parent_model_invocation,
47                    presentation_input: evidence.presentation_input,
48                    presentation_label: evidence.presentation_label,
49                    category: evidence.category,
50                });
51                if let Some(child_branch) = evidence.child_branch {
52                    if let Some(activity) = branch
53                        .activity
54                        .iter_mut()
55                        .find(|activity| **activity == ExecutionActivity::ChildBranch(child_branch))
56                    {
57                        *activity = ExecutionActivity::ChildBranchSpawned {
58                            branch: child_branch,
59                            parent_tool_invocation: invocation,
60                        };
61                    }
62                }
63                branch.tool_results.push(ToolResult {
64                    id: result,
65                    invocation,
66                    output: evidence.output,
67                    failed: evidence.failed,
68                    presentation_preview: None,
69                });
70                branch
71                    .activity
72                    .push(ExecutionActivity::ToolInvocation(invocation));
73                branch.activity.push(ExecutionActivity::ToolResult(result));
74            }
75            ExecutionEvidence::Model(request) => {
76                parent_model_invocation = Some(persist_model_request(service, branch, request));
77            }
78        }
79    }
80}

82/// Persists one request under its stable model-invocation activity identity.
83///
84/// - `service`: Application identity allocator.
85/// - `branch`: Branch that owns the invocation.
86/// - `request`: Bounded provider-neutral request projection.
87fn persist_model_request<B>(
88    service: &ApplicationService<B>,
89    branch: &mut Branch,
90    request: ModelRequestProjection,
91) -> scriberig_core::EntityId {
92    let invocation = service.next_entity(0xB6);
93    branch.model_invocations.push(ModelInvocation {
94        id: invocation,
95        branch: branch.id,
96        ordinal: request.ordinal,
97        agent_profile: None,
98        context_fingerprint: None,
99        provider_profile: None,
100        model: Some(request.model.clone()),
101        model_profile: None,
102        provider_model_id: None,
103        thinking_level: request.thinking_level.clone(),
104        request_fingerprint: None,
105        cache: CacheDisposition::NotUsed,
106        output: None,
107        request: Some(request),
108    });
109    branch
110        .activity
111        .push(ExecutionActivity::ModelInvocation(invocation));
112    invocation
113}

115/// Distinguishes provider requests from tool calls while preserving order.
116enum ExecutionEvidence {
117    /// Safe normalized model request.
118    Model(ModelRequestProjection),
119    /// Executed tool and bounded result evidence.
120    Tool(crate::ToolExecutionEvidence),

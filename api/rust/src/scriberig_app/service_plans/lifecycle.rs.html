95    /// Persists final completion after every node result is available.
96    ///
97    /// - `self`: Application service owning durable plan state.
98    /// - `context`: Authorized caller context.
99    /// - `request`: Session, plan, and observed revision.
100    ///
101    /// The completed plan with its coordinator claim released.
102    ///
103    /// Returns a revision conflict, incomplete-work error, or persistence failure.
104    /// # Returns
105    /// The operation result.
106    /// # Errors
107    /// Returns an error when the operation fails.
108    pub fn complete_plan(
109        &self,
110        context: &ApplicationContext,
111        request: &PlanLifecycleRequest,
112    ) -> Result<Plan, ApplicationError> {
113        let mut session = self.read_session(
114            context,
115            &crate::SessionRequest {
116                session: request.session,
117            },
118        )?;
119        let plan = session
120            .plans
121            .iter_mut()
122            .find(|plan| plan.id == request.plan)
123            .ok_or_else(|| ApplicationError::NotFound {
124                subject: request.plan.to_string(),
125            })?;
126        if plan.revision != request.expected_revision {
127            return Err(revision_conflict(request.plan));
128        }
129        let operation = plan
130            .coordinator_claim
131            .as_ref()
132            .and_then(|claim| claim.operation.clone());
133        let results = session
134            .branches
135            .iter()
136            .filter_map(|branch| branch.result.clone())
137            .collect::<Vec<_>>();
138        if !plan
139            .complete_if_satisfied(&results)
140            .map_err(|error| invalid_plan(format!("plan completion failed: {error:?}")))?
141        {
142            return Err(invalid_plan("plan still has unresolved node results"));
143        }
144        let completed = plan.clone();
145        self.compare_and_swap_plan(context, &session, request.plan, request.expected_revision)?;
146        self.operations.publish_plan_activity(
147            request.session,
148            request.plan,
149            None,
150            scriberig_session::SessionActivityKind::CoordinatorChanged,
151            scriberig_session::SessionActivityState::Completed,
152            "Plan coordinator completed",
153        );
154        if let Some(operation) = operation {
155            let _ = self
156                .operations
157                .transition(&operation, crate::OperationStatus::Completed);
158        }
159        Ok(completed)
160    }

162    /// Applies one optimistic plan lifecycle update and persists the session.
163    ///
164    /// - `self`: Application service.
165    /// - `context`: Authorized caller context.
166    /// - `request`: Session, plan, and observed revision.
167    /// - `status`: Requested lifecycle state.
168    ///
169    /// The authoritative updated plan.
170    ///
171    /// Returns conflict, invalid transition, lookup, or persistence failures.
172    /// # Returns
173    /// The operation result.
174    /// # Errors
175    /// Returns an error when the operation fails.
176    #[allow(clippy::too_many_lines)]
177    fn mutate_plan(
178        &self,
179        context: &ApplicationContext,
180        request: &PlanLifecycleRequest,
181        status: PlanStatus,
182    ) -> Result<Plan, ApplicationError> {
183        let mut session = self.read_session(
184            context,
185            &crate::SessionRequest {
186                session: request.session,
187            },
188        )?;
189        let plan = session
190            .plans
191            .iter_mut()
192            .find(|plan| plan.id == request.plan)
193            .ok_or_else(|| ApplicationError::NotFound {
194                subject: request.plan.to_string(),
195            })?;
196        if plan.revision != request.expected_revision {
197            return Err(revision_conflict(request.plan));
198        }
199        let previous_coordinator_operation = plan
200            .coordinator_claim
201            .as_ref()
202            .and_then(|claim| claim.operation.clone());
203        if status == PlanStatus::Running
204            && plan.status == PlanStatus::Running
205            && plan.coordinator_claim.is_some()
206        {
207            return Ok(plan.clone());
208        }
209        let coordinator_operation = if status == PlanStatus::Running {
210            let sequence = self
211                .next_agent_handle
212                .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
213            let operation = scriberig_core::OperationId::new(format!(
214                "plan-coordinator-{sequence}"
215            ))
216            .map_err(|error| ApplicationError::Internal {
217                message: error.to_string(),
218            })?;
219            Some(operation)
220        } else {
221            None
222        };
223        if status == PlanStatus::Running && plan.coordinator_claim.is_none() {
224            let owner = context
225                .correlation_id
226                .as_deref()
227                .or_else(|| {
228                    context
229                        .principal
230                        .as_ref()
231                        .map(|principal| principal.id.as_str())
232                })
233                .unwrap_or("application-runtime");
234            plan.start_with_coordinator_claim(
235                self.next_entity(0xC3),
236                owner,
237                coordinator_operation.clone(),
238            )
239            .map_err(|error| invalid_plan(format!("coordinator start failed: {error:?}")))?;
240            if let Some(operation) = &coordinator_operation {
241                self.operations
242                    .start(operation.clone(), crate::OperationKind::Plan)?;
243                if let Err(error) = self
244                    .operations
245                    .transition(operation, crate::OperationStatus::Running)
246                {
247                    let _ = self.operations.cancel(operation);
248                    return Err(error);
249                }
250            }
251        } else {
252            plan.transition(status)
253                .map_err(|error| invalid_plan(format!("invalid plan transition: {error:?}")))?;
254        }
255        let updated = plan.clone();
256        if let Err(error) =
257            self.compare_and_swap_plan(context, &session, request.plan, request.expected_revision)
258        {
259            if let Some(operation) = &coordinator_operation {
260                let _ = self.operations.cancel(operation);
261            }
262            return Err(error);
263        }
264        if status == PlanStatus::Paused || status == PlanStatus::Cancelled {
265            if let Some(operation) = previous_coordinator_operation {
266                let _ = self.operations.cancel(&operation);
267            }
268        }
269        self.operations.publish_plan_activity(
270            request.session,
271            request.plan,
272            None,
273            scriberig_session::SessionActivityKind::CoordinatorChanged,
274            match status {
275                PlanStatus::Failed => scriberig_session::SessionActivityState::Failed,
276                PlanStatus::Cancelled => scriberig_session::SessionActivityState::Cancelled,
277                _ => scriberig_session::SessionActivityState::Completed,
278            },
279            match status {
280                PlanStatus::Running => "Plan coordinator started",
281                PlanStatus::Paused => "Plan coordinator paused",
282                PlanStatus::Cancelled => "Plan coordinator cancelled",
283                PlanStatus::Failed => "Plan coordinator failed",
284                PlanStatus::Draft | PlanStatus::Planned | PlanStatus::Completed => {
285                    "Plan coordinator changed"
286                }
287            },
288        );
289        Ok(updated)
290    }

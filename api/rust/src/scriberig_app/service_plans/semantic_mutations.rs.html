1//! Applies shared summary, evidence, gate, and waiver mutations.
2
3use crate::{
4    ApplicationBackend, ApplicationContext, ApplicationError, ApplicationService,
5    PlanSemanticMutation, PlanSemanticMutationRequest, SessionBackend,
6};
7use scriberig_core::EntityId;
8use scriberig_session::{Plan, PlanEvidenceAssociation, PlanGate, PlanGateWaiver, PlanNodeSummary};
9
10use super::validation::{invalid_plan, revision_conflict};
11
12impl<B: ApplicationBackend> ApplicationService<B> {
13    /// Applies one typed semantic mutation at an exact plan revision.
14    ///
15    /// Actor provenance is derived from the trusted application context.
16    ///
17    /// # Parameters
18    /// - `self`: Application service owning plan persistence.
19    /// - `context`: Authorized caller context.
20    /// - `request`: Plan revision and typed summary/evidence/gate mutation.
21    ///
22    /// The authoritative updated plan.
23    ///
24    /// Returns validation, conflict, lookup, or persistence failures.
25    /// # Returns
26    /// The updated plan.
27    /// # Errors
28    /// Returns an error when the operation fails.
29    pub fn mutate_plan_semantics(
30        &self,
31        context: &ApplicationContext,
32        request: &PlanSemanticMutationRequest,
33    ) -> Result<Plan, ApplicationError> {
34        let mut session = self.read_session(
35            context,
36            &crate::SessionRequest {
37                session: request.session,
38            },
39        )?;
40        let plan = session
41            .plans
42            .iter_mut()
43            .find(|plan| plan.id == request.plan)
44            .ok_or_else(|| ApplicationError::NotFound {
45                subject: request.plan.to_string(),
46            })?;
47        if plan.revision != request.expected_revision {
48            return Err(revision_conflict(request.plan));
49        }
50        let actor = context
51            .principal
52            .as_ref()
53            .and_then(|principal| principal.id.parse::<EntityId>().ok());
54        match request.mutation.clone() {
55            PlanSemanticMutation::UpdateSummary {
56                node,
57                text,
58                sources,
59            } => plan.update_summary(PlanNodeSummary {
60                id: self.next_entity(0xC8),
61                node,
62                text,
63                sources,
64            }),
65            PlanSemanticMutation::AttachEvidence {
66                owner,
67                role,
68                evidence,
69                note,
70            } => plan.attach_evidence(PlanEvidenceAssociation {
71                id: self.next_entity(0xC9),
72                owner,
73                role,
74                evidence,
75                note,
76                actor,
77            }),
78            PlanSemanticMutation::AddGate {
79                owner,
80                phase,
81                requirement,
82            } => plan.add_gate(PlanGate {
83                id: self.next_entity(0xCA),
84                owner,
85                phase,
86                requirement,
87                waiver: None,
88            }),
89            PlanSemanticMutation::WaiveGate { gate, reason } => plan.waive_gate(
90                gate,
91                PlanGateWaiver {
92                    reason,
93                    actor,
94                    operation: None,
95                },
96            ),
97        }
98        .map_err(|error| invalid_plan(format!("invalid semantic plan mutation: {error:?}")))?;
99        let updated = plan.clone();
100        self.compare_and_swap_plan(context, &session, request.plan, request.expected_revision)?;
101        self.operations.publish_plan_activity(
102            request.session,
103            request.plan,
104            None,
105            scriberig_session::SessionActivityKind::PlanRevised,
106            scriberig_session::SessionActivityState::Completed,
107            "Plan semantics updated",
108        );
109        Ok(updated)
110    }
111}

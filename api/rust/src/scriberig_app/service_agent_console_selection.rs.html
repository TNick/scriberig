3use super::{
4    ApplicationBackend, ApplicationContext, ApplicationError, ApplicationService, EntityId,
5    ModelDescriptor, ModelProfile, SelectionError, SelectionStoreError, Session, empty_session,
6    ensure_revision, invalid_selection, selection_error,

9impl<B: ApplicationBackend> ApplicationService<B> {
10    /// Resolves an enabled profile and its model descriptor.
11    ///
12    /// # Parameters
13    /// - `self`: Application service owning the model library.
14    /// - `profile_id`: Stable model-profile identity.
15    ///
16    /// # Returns
17    /// The profile and its validated model descriptor.
18    ///
19    /// # Errors
20    /// Returns not-found or invalid-request for unavailable configuration.
21    pub(super) fn resolve_profile(
22        &self,
23        profile_id: &str,
24    ) -> Result<(&ModelProfile, &ModelDescriptor), ApplicationError> {
25        let profile =
26            self.model_library
27                .profile(profile_id)
28                .ok_or_else(|| ApplicationError::NotFound {
29                    subject: format!("model profile {profile_id}"),
30                })?;
31        if !profile.enabled {
32            return Err(invalid_selection("model profile is disabled"));
33        }
34        let model =
35            self.model_library
36                .model(&profile.model)
37                .ok_or_else(|| ApplicationError::NotFound {
38                    subject: format!("model {}", profile.model.as_str()),
39                })?;
40        profile
41            .validate_for_model(model)
42            .map_err(invalid_selection)?;
43        Ok((profile, model))
44    }
45
46    /// Loads session state and identifies the subsystem that owns persistence.
47    ///
48    /// # Parameters
49    /// - `self`: Application service providing session storage.
50    /// - `context`: Authorized request context for backend reads.
51    /// - `session`: Stable session identity.
52    ///
53    /// # Returns
54    /// The session and whether an application backend owns its persistence.
55    ///
56    /// # Errors
57    /// Returns an application error for failed or unauthorized reads.
58    pub(super) fn selection_session(
59        &self,
60        context: &ApplicationContext,
61        session: EntityId,
62    ) -> Result<(Session, bool), ApplicationError> {
63        if let Some(value) = self.load_session_state(session)? {
64            return Ok((value, false));
65        }
66        match self
67            .backend()
68            .read_session(context, &crate::SessionRequest { session })
69        {
70            Ok(value) => Ok((value, true)),
71            Err(error) if matches!(error.category(), "unsupported_operation" | "not_found") => {
72                Ok((empty_session(session), false))
73            }
74            Err(error) => Err(error),
75        }
76    }
77
78    /// Persists a selection update through its owning session boundary.
79    ///
80    /// # Parameters
81    /// - `self`: Application service providing session storage.
82    /// - `context`: Authorized request context for backend writes.
83    /// - `session`: Complete session with the updated execution selection.
84    /// - `backend_owned`: Whether the backend supplied the current session.
85    /// - `expected_revision`: Selection revision observed by the caller.
86    ///
87    /// # Returns
88    /// Unit after selection persistence succeeds.
89    ///
90    /// # Errors
91    /// Returns persistence errors or unsupported-operation for read-only state.
92    pub(super) fn persist_selection(
93        &self,
94        context: &ApplicationContext,
95        session: Session,
96        backend_owned: bool,
97        expected_revision: u64,
98    ) -> Result<(), ApplicationError> {
99        if backend_owned {
100            return self.backend().update_session_execution_selection(
101                context,
102                &session,
103                expected_revision,
104            );
105        }
106        if let Some(store) = &self.session_store {
107            return store
108                .compare_and_swap_execution_selection(&session, expected_revision)
109                .map_err(|error| match error {
110                    SelectionStoreError::RevisionConflict { expected, actual } => selection_error(
111                        SelectionError::RevisionConflict { expected, actual },
112                        session.id,
113                    ),
114                    SelectionStoreError::Persistence(message) => {
115                        ApplicationError::Internal { message }
116                    }
117                });
118        }
119        let mut sessions = self
120            .sessions
121            .lock()
122            .map_err(|_| ApplicationError::Internal {
123                message: "application session lock failed".to_owned(),
124            })?;
125        let actual_revision = sessions
126            .get(&session.id)
127            .map_or(0, |current| current.execution.revision);
128        ensure_revision(expected_revision, actual_revision, session.id)?;
129        if let Some(current) = sessions.get_mut(&session.id) {
130            current.execution = session.execution;
131        } else {
132            sessions.insert(session.id, session);
133        }
134        Ok(())
135    }
136}

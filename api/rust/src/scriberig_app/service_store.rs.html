11/// Provides exact inspection over canonical revision, snapshot, and root stores.
12pub struct StoreInspectionBackend<S> {
13    store: S,
14    current_root: scriberig_core::RootName,
15}
16
17impl<S> StoreInspectionBackend<S> {
18    /// Creates an inspection backend using the conventional `main` root.
19    ///
20    /// - `store`: Revision, snapshot, and root backend.
21    ///
22    /// A storage-backed inspection provider.
23    ///
24    /// # Panics
25    /// Panics only if the fixed root name is invalid.
26    #[must_use]
27    /// # Returns
28    /// The operation result.
29    pub fn new(store: S) -> Self {
30        Self {
31            store,
32            current_root: scriberig_core::RootName::new("main").expect("main is a valid root name"),
33        }
34    }
35
36    /// Returns the underlying storage backend.
37    ///
38    /// - `self`: Storage-backed provider.
39    ///
40    /// The backend used for reads.
41    #[must_use]
42    pub const fn store(&self) -> &S {
43        &self.store
44    }
45}
46
47impl<S> ApplicationBackend for StoreInspectionBackend<S>
48where
49    S: RevisionStore + SnapshotStore + RootStore,
50{
51    /// Executes an exact snapshot or revision inspection.
52    ///
53    /// - `self`: Storage-backed inspection provider.
54    /// - `_context`: Trusted application context.
55    /// - `request`: Universal application request.
56    ///
57    /// Compact inspection metadata from canonical storage.
58    ///
59    /// Returns a structured not-found or backend failure.
60    #[allow(clippy::too_many_lines)]
61    /// # Returns
62    /// The operation result.
63    /// # Errors
64    /// Returns an error when the operation fails.
65    fn execute(
66        &self,
67        _context: &ApplicationContext,
68        request: &ApplicationRequest,
69    ) -> Result<ApplicationResult, ApplicationError> {
70        let ApplicationRequest::Inspect(request) = request else {
71            let ApplicationRequest::List(request) = request else {
72                if let ApplicationRequest::Find(request) = request {
73                    return find_snapshot_entities(&self.store, request);
74                }
75                return Err(ApplicationError::UnsupportedOperation {
76                    operation: "storage backend read operation".to_owned(),
77                });
78            };
79            if request.collection.collection_type != "snapshot.entities" {
80                return Err(ApplicationError::UnsupportedCapability {
81                    operation: "list".to_owned(),
82                    capability: request.collection.collection_type.clone(),
83                });
84            }
85            let entities = self
86                .store
87                .iter(request.subject.snapshot)
88                .map_err(|error| storage_snapshot_error(&error))?;
89            let start = request
90                .page
91                .continuation
92                .as_ref()
93                .map_or(Some(0), |token| {
94                    token.offset_for("list", request.subject.snapshot)
95                })
96                .ok_or_else(|| ApplicationError::InvalidRequest {
97                    operation: "list".to_owned(),
98                    message: "invalid snapshot entity continuation".to_owned(),
99                })?;
100            let end = start.saturating_add(request.page.limit).min(entities.len());
101            if start > end {
102                return Err(ApplicationError::InvalidRequest {
103                    operation: "list".to_owned(),
104                    message: "snapshot entity continuation is out of range".to_owned(),
105                });
106            }
107            let items = entities
108                .iter()
109                .skip(start)
110                .take(end.saturating_sub(start))
111                .copied()
112                .map(|(entity, revision)| {
113                    let entity_type = self
114                        .store
115                        .get(revision)
116                        .ok()
117                        .flatten()
118                        .map(|value| value.object_type().clone());
119                    NavigationEntry {
120                        target: TargetRef::Snapshot(EntityInSnapshot {
121                            entity,
122                            snapshot: request.subject.snapshot,
123                        }),
124                        entity_type,
125                        label: None,
126                        description: None,
127                        metadata: Some("snapshot entity".to_owned()),
128                    }
129                })
130                .collect();
131            return Ok(ApplicationResult::List(scriberig_query::ListResult {
132                subject: request.subject,
133                snapshot: request.subject.snapshot,
134                collection: CollectionRef {
135                    owner: request.collection.owner,
136                    collection_type: request.collection.collection_type.clone(),
137                },
138                page: Page {
139                    items,
140                    next: (end < entities.len()).then(|| {
141                        scriberig_query::ContinuationToken::for_page(
142                            "list",
143                            request.subject.snapshot,
144                            end,
145                        )
146                    }),
147                },
148            }));
149        };
150        let (subject, revision_id, snapshot) = match request.target {
151            TargetRef::Snapshot(target) => {
152                let revision = self
153                    .store
154                    .resolve(target.snapshot, target.entity)
155                    .map_err(|error| storage_snapshot_error(&error))?
156                    .ok_or_else(|| ApplicationError::NotFound {
157                        subject: target.entity.to_string(),
158                    })?;
159                (request.target, revision, Some(target.snapshot))
160            }
161            TargetRef::Revision(target) => (request.target, target.revision, None),
162        };
163        let revision = self
164            .store
165            .get(revision_id)
166            .map_err(|error| storage_revision_error(&error))?
167            .ok_or_else(|| ApplicationError::NotFound {
168                subject: revision_id.to_string(),
169            })?;
170        Ok(ApplicationResult::Inspect(Inspection {
171            subject,
172            snapshot,
173            revision: Some(revision.id()),
174            entity_type: Some(revision.object_type().clone()),
175            label: None,
176            description: None,
177            properties: Vec::new(),
178            counts: Vec::new(),
179            relations: Vec::new(),
180            collections: Vec::new(),
181            views: Vec::new(),
182            operations: Vec::new(),
183            provenance: Vec::new(),
184            information_cost: None,
185            detail: request.detail,
186        }))
187    }
188
189    /// Resolves the conventional current `main` root.
190    ///
191    /// - `self`: Storage-backed inspection provider.
192    /// - `_context`: Trusted application context.
193    ///
194    /// The exact snapshot promoted on `main`.
195    ///
196    /// Returns not-found when `main` has not been promoted.
197    /// # Returns
198    /// The operation result.
199    /// # Errors
200    /// Returns an error when the operation fails.
201    fn current_snapshot(
202        &self,
203        _context: &ApplicationContext,
204    ) -> Result<scriberig_core::SnapshotId, ApplicationError> {
205        self.store
206            .read_root(&self.current_root)
207            .ok_or_else(|| ApplicationError::NotFound {
208                subject: self.current_root.to_string(),
209            })
210    }
211}

213/// Finds entities by exact text over stable identity and type metadata.
214///
215/// - `store`: Revision and snapshot storage backend.
216/// - `request`: Snapshot-scoped exact-text discovery request.
217///
218/// A bounded public find result.
219///
220/// Returns a structured application error for unsupported scopes or pages.
221fn find_snapshot_entities<S>(
222    store: &S,
223    request: &scriberig_query::FindRequest,
224) -> Result<ApplicationResult, ApplicationError>
225where
226    S: RevisionStore + SnapshotStore,
227{
228    let scriberig_query::QueryScope::Snapshot(snapshot) = request.scope else {
229        return Err(ApplicationError::InvalidRequest {
230            operation: "find".to_owned(),
231            message: "storage find requires a snapshot scope".to_owned(),
232        });
233    };
234    if request.page.continuation.is_some() {
235        return Err(ApplicationError::InvalidRequest {
236            operation: "find".to_owned(),
237            message: "storage find has no next page".to_owned(),
238        });
239    }
240    let query = request.query.to_lowercase();
241    let hits = store
242        .iter(snapshot)
243        .map_err(|error| storage_snapshot_error(&error))?
244        .into_iter()
245        .filter_map(|(entity, revision)| {
246            let value = store.get(revision).ok().flatten()?;
247            let entity_text = entity.to_string();
248            let type_text = value.object_type().to_string();
249            (entity_text.to_lowercase().contains(&query)
250                || type_text.to_lowercase().contains(&query))
251            .then_some(scriberig_query::FindHit {
252                target: TargetRef::Snapshot(EntityInSnapshot { entity, snapshot }),
253                entity_type: Some(value.object_type().clone()),
254                label: None,
255                preview: Some(type_text),
256                score: None,
257                anchor: None,
258                provenance: None,
259            })
260        })
261        .take(request.page.limit)
262        .collect();
263    Ok(ApplicationResult::Find(scriberig_query::FindResult {
264        scope: request.scope.clone(),
265        strategy: request.strategy.clone(),
266        page: Page {
267            items: hits,
268            next: None,
269        },
270        information_cost: None,
271    }))
272}

274/// Maps a revision backend failure to a safe application error.
275///
276/// - `error`: Revision backend failure.
277///
278/// Safe application error preserving no backend internals.
279/// # Returns
280/// The operation result.
281fn storage_revision_error(error: &scriberig_store::StoreError) -> ApplicationError {
282    ApplicationError::Internal {
283        message: format!("revision backend failure: {error:?}"),
284    }

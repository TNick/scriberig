97    /// Creates a one-action capability for a scope test.
98    ///
99    /// # Parameters
100    /// - `scope`: Scope granted by the test capability.
101    /// - `action`: Action granted by the test capability.
102    ///
103    /// # Returns
104    /// A capability containing the requested single action.
105    fn capability(scope: CapabilityScope, action: CapabilityAction) -> Capability {
106        Capability {
107            scope,
108            actions: [action].into_iter().collect(),
109            constraints: CapabilityConstraints::unrestricted(),
110        }
111    }

341            actions: BTreeSet::default(),
342            constraints: CapabilityConstraints::unrestricted(),

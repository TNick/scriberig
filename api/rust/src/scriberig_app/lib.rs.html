35mod tests {

44    /// Verifies terminal operation states remain distinguishable from active states.
45    #[test]
46    fn operation_status_has_explicit_terminal_states() {
47        assert!(!OperationStatus::Queued.is_terminal());
48        assert!(!OperationStatus::Running.is_terminal());
49        assert!(OperationStatus::Completed.is_terminal());
50        assert!(OperationStatus::Failed.is_terminal());
51        assert!(OperationStatus::Cancelled.is_terminal());
52    }
53
54    /// Verifies application errors expose stable categories without text parsing.
55    #[test]
56    fn application_error_categories_are_structured() {
57        let error = ApplicationError::RevisionMismatch {
58            entity: EntityId::new([1; 16]),
59            revision: RevisionId::new([2; 32]),
60        };
61        assert_eq!(error.category(), "revision_mismatch");
62        assert_ne!(error.to_string(), "");
63    }
64
65    /// Verifies events retain stable handles rather than owning canonical state.
66    #[test]
67    fn events_retain_snapshot_and_revision_handles() {
68        let event = ApplicationEvent::EntityChanged {
69            entity: EntityId::new([1; 16]),
70            revision: RevisionId::new([2; 32]),
71        };
72        assert!(matches!(event, ApplicationEvent::EntityChanged { .. }));
73        let snapshot = SnapshotId::new([3; 32]);
74        assert_eq!(snapshot.as_bytes(), &[3; 32]);
75    }
76
77    /// Verifies inspection request and result types preserve snapshot identity.
78    #[test]
79    fn inspection_preserves_requested_snapshot() {
80        let entity = EntityId::new([4; 16]);
81        let snapshot = SnapshotId::new([5; 32]);
82        let request = InspectRequest {
83            entity,
84            snapshot: SnapshotResolution::At(snapshot),
85            detail: DetailLevel::Compact,
86        };
87        let result = InspectResult {
88            entity,
89            revision: Some(RevisionId::new([6; 32])),
90            snapshot: Some(snapshot),
91            object_type: None,
92            label: None,
93            description: None,
94            capabilities: vec!["inspect.revision".to_owned()],
95            trust: None,
96        };
97        assert_eq!(request.snapshot, SnapshotResolution::At(snapshot));
98        assert_eq!(result.snapshot, Some(snapshot));
99    }

120    /// Verifies snapshot selectors preserve exact and named-root semantics.
121    #[test]
122    fn snapshot_selectors_remain_distinct() {
123        let snapshot = SnapshotId::new([7; 32]);
124        let root = RootName::new("main").expect("valid root");

131    /// Verifies capability descriptors and domain requests retain semantic handles.
132    #[test]
133    fn domain_operation_request_is_structured() {
134        let entity = EntityId::new([8; 16]);
135        let selection = Selection {
136            snapshot: SnapshotId::new([9; 32]),
137            entity,
138            revision: None,
139            relation_path: Vec::new(),
140            projection_anchor: None,
141        };
142        let capability = CapabilityDescriptor {
143            id: CapabilityId::new("command.rename").expect("valid capability"),
144            kind: CapabilityKind::DomainOperation,
145            label: "Rename".to_owned(),
146            description: "Rename an entity".to_owned(),
147            entity_types: vec![ObjectType::new("rust.function").expect("valid type")],
148            available: true,
149            derived: false,
150            estimated_items: None,
151        };
152        let request = DomainOperationRequest {
153            capability: capability.id.clone(),
154            subject: selection,
155            context: BranchContext {
156                base_snapshot: SnapshotId::new([9; 32]),
157                branch: None,
158            },
159            arguments: Vec::new(),
160        };
161        assert_eq!(request.capability, capability.id);
162        assert_eq!(request.subject.entity, entity);
163    }

16/// Wraps reusable read providers with application capability enforcement.
17pub struct GuardedQueryService<S> {
18    /// Policy-free provider used only after request-level authorization.
19    inner: S,
20}
21
22impl<S> GuardedQueryService<S> {
23    /// Creates a guarded application read service.
24    ///
25    /// # Parameters
26    /// - `inner`: Reusable provider that supplies semantic read operations.
27    ///
28    /// # Returns
29    /// A provider whose exposed results are checked against request grants.
30    #[must_use]
31    pub const fn new(inner: S) -> Self {
32        Self { inner }
33    }
34
35    /// Borrows the wrapped reusable provider for intentional low-level use.
36    ///
37    /// # Returns
38    /// The policy-free provider.
39    #[must_use]
40    pub const fn inner(&self) -> &S {
41        &self.inner
42    }
43
44    /// Returns a structured denial without including hidden content.
45    ///
46    /// # Parameters
47    /// - `action`: Action that the caller requested.
48    /// - `scope`: Scope that the caller requested.
49    ///
50    /// # Returns
51    /// A safe application permission error.
52    fn denied(action: &CapabilityAction, scope: &CapabilityScope) -> ApplicationError {
53        ApplicationError::PermissionDenied {
54            action: format!("{action:?}"),
55            scope: format!("{scope:?}"),
56        }
57    }
58
59    /// Checks one operation-level capability.
60    ///
61    /// # Parameters
62    /// - `context`: Caller grants to evaluate.
63    /// - `action`: Action required by the operation.
64    /// - `scope`: Semantic scope exposed by the operation.
65    /// - `snapshot`: Exact snapshot constraint, when available.
66    /// - `operation`: Universal operation kind.
67    ///
68    /// # Returns
69    /// `Ok(())` when one grant covers the request.
70    ///
71    /// # Errors
72    /// Returns a structured permission denial when no grant covers it.
73    fn authorize(
74        context: &ApplicationContext,
75        action: &CapabilityAction,
76        scope: &CapabilityScope,
77        snapshot: Option<scriberig_core::SnapshotId>,
78        operation: OperationKind,
79    ) -> Result<(), ApplicationError> {
80        if context
81            .capabilities
82            .permits(action, scope, snapshot, Some(operation))
83        {
84            Ok(())
85        } else {
86            Err(Self::denied(action, scope))
87        }
88    }
89}

91impl<S: QueryService> QueryService for GuardedQueryService<S> {
92    /// Authorizes and delegates one application inspection.
93    ///
94    /// # Parameters
95    /// - `self`: Guarded provider.
96    /// - `context`: Caller grants and exact request context.
97    /// - `request`: Entity inspection request.
98    ///
99    /// # Returns
100    /// The provider's inspection result after authorization.
101    ///
102    /// # Errors
103    /// Returns a structured permission denial or provider error.
104    fn inspect(
105        &self,
106        context: &ApplicationContext,
107        request: &InspectRequest,
108    ) -> Result<InspectResult, ApplicationError> {
109        let scope = CapabilityScope::Entity(request.entity);
110        Self::authorize(
111            context,
112            &CapabilityAction::Inspect,
113            &scope,
114            match request.snapshot {
115                crate::SnapshotResolution::At(snapshot) => Some(snapshot),
116                _ => context.snapshot,
117            },
118            crate::OperationKind::Inspect,
119        )?;
120        let result = self.inner.inspect(context, request)?;
121        Ok(result)
122    }
123}

125impl<S: InspectService> GuardedQueryService<S> {
126    /// Authorizes and delegates a reusable exact-target inspection.
127    ///
128    /// # Parameters
129    /// - `self`: Guarded provider.
130    /// - `context`: Caller grants and exact request context.
131    /// - `request`: Query-crate inspection request.
132    ///
133    /// # Returns
134    /// The inspection result when its entity is authorized.
135    ///
136    /// # Errors
137    /// Returns a structured permission denial or provider error.
138    pub fn inspect_target(
139        &self,
140        context: &ApplicationContext,
141        request: &scriberig_query::InspectRequest,
142    ) -> Result<Inspection, ApplicationError> {
143        let entity = target_entity(request.target);
144        let scope = CapabilityScope::Entity(entity);
145        Self::authorize(
146            context,
147            &CapabilityAction::Inspect,
148            &scope,
149            target_snapshot(request.target),
150            crate::OperationKind::Inspect,
151        )?;
152        if let Some(view) = &request.view {
153            let projection = CapabilityScope::Projection {
154                entity,
155                projection: view.0.clone(),
156            };
157            Self::authorize(
158                context,
159                &CapabilityAction::Render,
160                &projection,
161                target_snapshot(request.target),
162                OperationKind::Render,
163            )?;
164        }
165        self.inner
166            .inspect(request)
167            .map_err(query_error_to_application)
168    }
169}

171impl<S: TraverseService> GuardedQueryService<S> {
172    /// Authorizes collection access and filters restricted entries.
173    ///
174    /// # Parameters
175    /// - `self`: Guarded provider.
176    /// - `context`: Caller grants and exact request context.
177    /// - `request`: Collection listing request.
178    ///
179    /// # Returns
180    /// A page containing only authorized navigation entries.
181    ///
182    /// # Errors
183    /// Returns a structured denial or provider error.
184    pub fn list(
185        &self,
186        context: &ApplicationContext,
187        request: &ListRequest,
188    ) -> Result<ListResult, ApplicationError> {
189        let scope = CapabilityScope::Entity(request.subject.entity);
190        Self::authorize(
191            context,
192            &CapabilityAction::Query,
193            &scope,
194            Some(request.subject.snapshot),
195            crate::OperationKind::List,
196        )?;
197        let mut result = self
198            .inner
199            .list(request)
200            .map_err(query_error_to_application)?;
201        filter_navigation(context, &mut result.page.items);
202        Ok(result)
203    }
204
205    /// Authorizes relation access and filters restricted entries.
206    ///
207    /// # Parameters
208    /// - `self`: Guarded provider.
209    /// - `context`: Caller grants and exact request context.
210    /// - `request`: Relation traversal request.
211    ///
212    /// # Returns
213    /// A page containing only authorized navigation entries.
214    ///
215    /// # Errors
216    /// Returns a structured denial or provider error.
217    pub fn follow(
218        &self,
219        context: &ApplicationContext,
220        request: &FollowRequest,
221    ) -> Result<FollowResult, ApplicationError> {
222        let scope = CapabilityScope::Relation {
223            entity: request.subject.entity,
224            relation: request.relation.relation_type.clone(),
225        };
226        Self::authorize(
227            context,
228            &CapabilityAction::Query,
229            &scope,
230            Some(request.subject.snapshot),
231            crate::OperationKind::Follow,
232        )?;
233        let mut result = self
234            .inner
235            .follow(request)
236            .map_err(query_error_to_application)?;
237        filter_navigation(context, &mut result.page.items);
238        Ok(result)
239    }
240}

308/// Authorizes every entity named by a query scope before evaluation.
309///
310/// # Parameters
311/// - `context`: Caller grants to evaluate.
312/// - `scope`: Exact query scope to authorize.
313/// - `action`: Action required by the query.
314/// - `operation`: Universal operation kind.
315///
316/// # Returns
317/// `Ok(())` when the complete scope is authorized.
318///
319/// # Errors
320/// Returns a denial or schema error before the provider is evaluated.
321fn authorize_query_scope(
322    context: &ApplicationContext,
323    scope: &scriberig_query::QueryScope,
324    action: &CapabilityAction,
325    operation: OperationKind,
326) -> Result<(), ApplicationError> {
327    match scope {
328        scriberig_query::QueryScope::Snapshot(snapshot) => GuardedQueryService::<()>::authorize(
329            context,
330            action,
331            &CapabilityScope::Project,
332            Some(*snapshot),
333            operation,
334        ),
335        scriberig_query::QueryScope::Entity { snapshot, entity } => {
336            GuardedQueryService::<()>::authorize(
337                context,
338                action,
339                &CapabilityScope::Entity(*entity),
340                Some(*snapshot),
341                operation,
342            )
343        }
344        scriberig_query::QueryScope::ExplicitEntities { snapshot, entities } => {
345            if entities.is_empty() {
346                return Err(ApplicationError::SchemaViolation {
347                    message: "query scope must contain an entity".to_owned(),
348                });
349            }
350            for entity in entities {
351                GuardedQueryService::<()>::authorize(
352                    context,
353                    action,
354                    &CapabilityScope::Entity(*entity),
355                    Some(*snapshot),
356                    operation,
357                )?;
358            }
359            Ok(())
360        }
361    }

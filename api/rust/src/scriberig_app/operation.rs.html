5/// Classifies the semantic work performed by an operation.
6#[derive(Clone, Copy, Debug, Eq, PartialEq)]
7pub enum OperationKind {
8    /// Inspects one semantic target.
9    Inspect,
10    /// Lists a typed collection.
11    List,
12    /// Follows a typed relation.
13    Follow,
14    /// Discovers semantic entities.
15    Find,
16    /// Evaluates a structural query.
17    Query,
18    /// Computes a semantic diff.
19    Diff,
20    /// Reads change history.
21    History,
22    /// Renders a semantic target through a registered view.
23    Render,
24    /// Creates or updates a branch context.
25    Branch,
26    /// Applies a proposed change.
27    Apply,
28    /// Commits a validated change.
29    Commit,
30}
31
32/// Describes the lifecycle state of an operation.
33#[derive(Clone, Copy, Debug, Eq, PartialEq)]
34pub enum OperationStatus {
35    /// Operation is waiting to start.
36    Queued,
37    /// Operation is currently running.
38    Running,
39    /// Operation completed successfully.
40    Completed,
41    /// Operation failed with a structured application error.
42    Failed,
43    /// Operation was cancelled before completion.
44    Cancelled,
45}
46
47impl OperationStatus {
48    /// Reports whether the operation will not produce further progress.
49    ///
50    /// # Parameters
51    /// - `self`: Status to classify.
52    ///
53    /// # Returns
54    /// `true` for completed, failed, and cancelled states.
55    #[must_use]
56    pub const fn is_terminal(self) -> bool {
57        matches!(self, Self::Completed | Self::Failed | Self::Cancelled)
58    }
59}

61/// Carries concise progress information without embedding mutable result data.
62#[derive(Clone, Debug, Eq, PartialEq)]
63pub struct OperationProgress {
64    /// Current operation status.
65    pub status: OperationStatus,
66    /// Optional completed work count.
67    pub completed: Option<u64>,
68    /// Optional total work count.
69    pub total: Option<u64>,
70    /// Optional human-readable phase name.
71    pub phase: Option<String>,
72}

74/// Retains the inspectable state of one tracked application operation.
75#[derive(Clone, Debug, Eq, PartialEq)]
76pub struct OperationRecord {
77    /// Stable operation identity.
78    pub reference: OperationId,
79    /// Universal operation family.
80    pub kind: OperationKind,
81    /// Exact base snapshot, when the operation has one.
82    pub base_snapshot: Option<scriberig_core::SnapshotId>,
83    /// Optional named branch context.
84    pub branch: Option<scriberig_core::RootName>,
85    /// Current lifecycle state.
86    pub status: OperationStatus,
87    /// Optional coarse progress information.
88    pub progress: Option<OperationProgress>,
89    /// Stable handles produced by a completed operation.
90    pub result_handles: Vec<String>,
91    /// Structured failure when the operation failed.
92    pub error: Option<crate::ApplicationError>,
93}
94
95impl OperationRecord {
96    /// Creates a queued operation record.
97    ///
98    /// # Parameters
99    /// - `reference`: Stable operation identity.
100    /// - `kind`: Universal operation family.
101    ///
102    /// # Returns
103    /// A queued operation record without mutable result data.
104    #[must_use]
105    pub fn queued(reference: OperationId, kind: OperationKind) -> Self {
106        Self {
107            reference,
108            kind,
109            base_snapshot: None,
110            branch: None,
111            status: OperationStatus::Queued,
112            progress: None,
113            result_handles: Vec::new(),
114            error: None,
115        }
116    }
117
118    /// Transitions the operation to a new lifecycle state.
119    ///
120    /// # Parameters
121    /// - `self`: Operation record to update.
122    /// - `status`: Next lifecycle status.
123    ///
124    /// # Returns
125    /// The updated operation record.
126    ///
127    /// # Errors
128    /// Returns an application error when a terminal operation is changed.
129    pub fn transition(mut self, status: OperationStatus) -> Result<Self, crate::ApplicationError> {
130        if self.status.is_terminal() {
131            return Err(crate::ApplicationError::OperationFailed {
132                operation: self.reference.to_string(),
133                message: "terminal operation cannot transition".to_owned(),
134            });
135        }
136        self.status = status;
137        if status.is_terminal() {
138            self.progress = Some(OperationProgress {
139                status,
140                completed: self.progress.as_ref().and_then(|value| value.completed),
141                total: self.progress.as_ref().and_then(|value| value.total),
142                phase: self.progress.as_ref().and_then(|value| value.phase.clone()),
143            });
144        }
145        Ok(self)
146    }
147}

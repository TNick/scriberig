1//! Defines the structured universal application result model.
2
3use scriberig_core::{RevisionId, RootName, SnapshotId};
4use scriberig_query::{FindResult, FollowResult, Inspection, ListResult, QueryResult};
5
6/// Describes one immutable entity transition in a migration proposal.
7#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
8pub struct MigrationDelta {
9    /// Entity whose canonical revision changes.
10    pub entity: scriberig_core::EntityId,
11    /// Revision in the base snapshot, when present.
12    pub before: Option<RevisionId>,
13    /// Revision in the candidate snapshot, when present.
14    pub after: Option<RevisionId>,
15}
16
17/// Exposes the canonical, snapshot-bound detail of one migration proposal.
18#[derive(Clone, Debug, Eq, PartialEq, serde::Deserialize, serde::Serialize)]
19pub struct MigrationDetail {
20    /// Stable migration handle.
21    pub id: String,
22    /// Exact snapshot from which the proposal starts.
23    pub base: SnapshotId,
24    /// Exact candidate snapshot produced by the proposal.
25    pub result: SnapshotId,
26    /// Deterministically ordered entity transitions.
27    pub deltas: Vec<MigrationDelta>,
28}
29
30/// Summarizes a semantic operation result without flattening it to text.
31#[derive(Clone, Debug, Eq, PartialEq)]
32pub struct OperationSummary {
33    /// Stable operation or result handle.
34    pub handle: String,
35    /// Exact snapshot involved when known.
36    pub snapshot: Option<SnapshotId>,
37    /// Exact revision involved when known.
38    pub revision: Option<RevisionId>,
39    /// Named branch context used by the operation, when applicable.
40    pub branch: Option<RootName>,
41    /// Structured semantic diff when this is a diff operation.
42    pub diff: Option<sr_change::SemanticDiff>,
43    /// Structured migration history when this is a history operation.
44    pub history: Option<Vec<sr_change::MigrationReference>>,
45    /// Structured rendered projection when this is a render operation.
46    pub rendered: Option<sr_projection::RenderedProjection>,
47    /// Validation outcome when the operation performed validation.
48    pub validation: Option<crate::ValidationStatus>,
49    /// Registered policy that produced the current validation outcome.
50    pub validation_profile: Option<String>,
51    /// Safe validation diagnostics associated with the operation.
52    pub validation_diagnostics: Vec<String>,
53    /// Safe conflict summaries associated with the operation.
54    pub conflicts: Vec<String>,
55}
56
57/// Identifies a tracked operation that has started.
58#[derive(Clone, Debug, Eq, PartialEq)]
59pub struct OperationStarted {
60    /// Stable tracked operation identifier.
61    pub operation: crate::OperationId,
62    /// Universal operation family.
63    pub kind: crate::OperationKind,
64}
65
66/// Returns one structured universal application result.
67#[derive(Clone, Debug, PartialEq)]
68pub enum ApplicationResult {
69    /// Result of entity inspection.
70    Inspect(Inspection),
71    /// Result of collection listing.
72    List(ListResult),
73    /// Result of relation traversal.
74    Follow(FollowResult),
75    /// Result of semantic discovery.
76    Find(FindResult),
77    /// Result of structural query evaluation.
78    Query(QueryResult),
79    /// Result of semantic diff computation.
80    Diff(OperationSummary),
81    /// Result of history lookup.
82    History(OperationSummary),
83    /// Result of projection rendering.
84    Render(OperationSummary),
85    /// Result of a branch operation.
86    Branch(OperationSummary),
87    /// Result of applying a change.
88    Apply(OperationSummary),
89    /// Result of committing a change.
90    Commit(OperationSummary),
91    /// Reference to a long-running operation.
92    OperationStarted(OperationStarted),
93}

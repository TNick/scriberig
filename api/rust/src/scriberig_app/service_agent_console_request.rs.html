1//! Implements shared safe model-request detail reads.
2
3use crate::{
4    ApplicationBackend, ApplicationContext, ApplicationError, ApplicationService, SessionBackend,
5    SessionRequest,
6};
7use scriberig_core::EntityId;
8
9/// Loads the session and resolves one captured request by stable identities.
10///
11/// - `service`: Application service that owns session reads.
12/// - `context`: Authorized request context.
13/// - `session`: Session containing the invocation.
14/// - `branch`: Branch containing the invocation.
15/// - `invocation`: Stable model-invocation identity.
16///
17/// The captured safe request projection.
18///
19/// Returns authorization or lookup errors when detail is unavailable.
20pub(super) fn model_request_detail<B: ApplicationBackend>(
21    service: &ApplicationService<B>,
22    context: &ApplicationContext,
23    session: EntityId,
24    branch: EntityId,
25    invocation: EntityId,
26) -> Result<scriberig_session::ModelRequestProjection, ApplicationError> {
27    let session = SessionBackend::read_session(service, context, &SessionRequest { session })?;
28    let branch = session
29        .branches
30        .iter()
31        .find(|candidate| candidate.id == branch)
32        .ok_or_else(|| ApplicationError::NotFound {
33            subject: branch.to_string(),
34        })?;
35    let model_invocation = branch
36        .model_invocations
37        .iter()
38        .find(|candidate| candidate.id == invocation)
39        .ok_or_else(|| ApplicationError::NotFound {
40            subject: invocation.to_string(),
41        })?;
42    model_invocation
43        .request
44        .clone()
45        .ok_or_else(|| ApplicationError::NotFound {
46            subject: invocation.to_string(),
47        })
48}

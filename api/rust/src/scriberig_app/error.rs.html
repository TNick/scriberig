1//! Defines structured application-operation failures.
2
3use scriberig_core::{EntityId, RevisionId, SnapshotId};
4use std::fmt::{self, Display, Formatter};
5
6/// Reports a semantic application failure without transport-specific details.
7#[derive(Clone, Debug, Eq, PartialEq)]
8pub enum ApplicationError {
9    /// Requested semantic data does not exist.
10    NotFound {
11        /// Stable subject that could not be found.
12        subject: String,
13    },
14    /// Input violates a domain or schema rule.
15    SchemaViolation {
16        /// Safe schema diagnostic.
17        message: String,
18    },
19    /// The operation was based on an obsolete snapshot.
20    StaleBaseSnapshot {
21        /// Snapshot supplied by the caller.
22        requested: SnapshotId,
23        /// Snapshot currently accepted by the application.
24        current: SnapshotId,
25    },
26    /// The requested semantic merge or reconciliation is ambiguous.
27    SemanticConflict {
28        /// Stable subject involved in the conflict.
29        subject: String,
30    },
31    /// Validation produced a failure for the requested candidate.
32    ValidationFailure {
33        /// Safe validation diagnostic.
34        message: String,
35    },
36    /// The caller lacks authority for the requested operation.
37    PermissionDenied {
38        /// Requested action.
39        action: String,
40        /// Requested authority scope.
41        scope: String,
42    },
43    /// The operation is not supported by this application configuration.
44    UnsupportedOperation {
45        /// Stable unsupported operation name.
46        operation: String,
47    },
48    /// A projection is not synchronized with the requested semantic state.
49    ProjectionOutOfSync {
50        /// Stable projection subject.
51        subject: String,
52    },
53    /// A long-running operation was cancelled.
54    OperationCancelled {
55        /// Stable operation name.
56        operation: String,
57    },
58    /// An implementation failure with a safe, backend-neutral message.
59    Internal {
60        /// Safe implementation diagnostic.
61        message: String,
62    },
63    /// The requested revision does not belong to the requested entity.
64    RevisionMismatch {
65        /// Entity identity supplied by the caller.
66        entity: EntityId,
67        /// Revision identity supplied by the caller.
68        revision: RevisionId,
69    },
70}
71
72impl ApplicationError {
73    /// Returns the stable category name used by adapters.
74    ///
75    /// # Parameters
76    /// - `self`: Error to classify.
77    ///
78    /// # Returns
79    /// A stable category label.
80    #[must_use]
81    pub const fn category(&self) -> &'static str {
82        match self {
83            Self::NotFound { .. } => "not_found",
84            Self::SchemaViolation { .. } => "schema_violation",
85            Self::StaleBaseSnapshot { .. } => "stale_base_snapshot",
86            Self::SemanticConflict { .. } => "semantic_conflict",
87            Self::ValidationFailure { .. } => "validation_failure",
88            Self::PermissionDenied { .. } => "permission_denied",
89            Self::UnsupportedOperation { .. } => "unsupported_operation",
90            Self::ProjectionOutOfSync { .. } => "projection_out_of_sync",
91            Self::OperationCancelled { .. } => "operation_cancelled",
92            Self::Internal { .. } => "internal",
93            Self::RevisionMismatch { .. } => "revision_mismatch",
94        }
95    }
96}

98impl Display for ApplicationError {
99    /// Formats an application error for a human-facing adapter.
100    ///
101    /// # Parameters
102    /// - `self`: Error to format.
103    /// - `formatter`: Destination formatter.
104    ///
105    /// # Returns
106    /// A formatting result.
107    ///
108    /// # Errors
109    /// Returns a formatting error when the destination rejects the write.
110    fn fmt(&self, formatter: &mut Formatter<'_>) -> fmt::Result {
111        match self {
112            Self::NotFound { subject } => write!(formatter, "subject not found: {subject}"),
113            Self::SchemaViolation { message } => {
114                write!(formatter, "schema violation: {message}")
115            }
116            Self::StaleBaseSnapshot { requested, current } => {
117                write!(
118                    formatter,
119                    "stale snapshot {requested}; current is {current}"
120                )
121            }
122            Self::SemanticConflict { subject } => {
123                write!(formatter, "semantic conflict: {subject}")
124            }
125            Self::ValidationFailure { message } => {
126                write!(formatter, "validation failed: {message}")
127            }
128            Self::PermissionDenied { action, scope } => {
129                write!(formatter, "permission denied for {action} in {scope}")
130            }
131            Self::UnsupportedOperation { operation } => {
132                write!(formatter, "unsupported operation: {operation}")
133            }
134            Self::ProjectionOutOfSync { subject } => {
135                write!(formatter, "projection out of sync: {subject}")
136            }
137            Self::OperationCancelled { operation } => {
138                write!(formatter, "operation cancelled: {operation}")
139            }
140            Self::Internal { message } => {
141                write!(formatter, "internal application error: {message}")
142            }
143            Self::RevisionMismatch { entity, revision } => {
144                write!(
145                    formatter,
146                    "revision {revision} does not belong to entity {entity}"
147                )
148            }
149        }
150    }
151}

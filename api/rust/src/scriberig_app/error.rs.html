1//! Defines structured application-operation failures.

3use scriberig_core::{EntityId, RevisionId, SnapshotId};
4use std::fmt::{self, Display, Formatter};
5
6/// Reports a semantic application failure without transport-specific details.
7#[derive(Clone, Debug, Eq, PartialEq)]
8pub enum ApplicationError {
9    /// Request parameters are inconsistent or incomplete.
10    InvalidRequest {
11        /// Universal operation being requested.
12        operation: String,
13        /// Safe request diagnostic.
14        message: String,
15    },
16    /// Requested semantic data does not exist.
17    NotFound {
18        /// Stable subject that could not be found.
19        subject: String,
20    },
21    /// Input violates a domain or schema rule.
22    SchemaViolation {
23        /// Safe schema diagnostic.
24        message: String,
25    },
26    /// The operation was based on an obsolete snapshot.
27    StaleBaseSnapshot {
28        /// Snapshot supplied by the caller.
29        requested: SnapshotId,
30        /// Snapshot currently accepted by the application.
31        current: SnapshotId,
32    },
33    /// The requested semantic merge or reconciliation is ambiguous.
34    SemanticConflict {
35        /// Stable subject involved in the conflict.
36        subject: String,
37    },
38    /// Validation produced a failure for the requested candidate.
39    ValidationFailure {
40        /// Safe validation diagnostic.
41        message: String,
42    },
43    /// The caller lacks authority for the requested operation.
44    PermissionDenied {
45        /// Requested action.
46        action: String,
47        /// Requested authority scope.
48        scope: String,
49    },
50    /// The operation is not supported by this application configuration.
51    UnsupportedOperation {
52        /// Stable unsupported operation name.
53        operation: String,
54    },
55    /// A projection is not synchronized with the requested semantic state.
56    ProjectionOutOfSync {
57        /// Stable projection subject.
58        subject: String,
59    },
60    /// A long-running operation was cancelled.
61    OperationCancelled {
62        /// Stable operation name.
63        operation: String,
64    },
65    /// An implementation failure with a safe, backend-neutral message.
66    Internal {
67        /// Safe implementation diagnostic.
68        message: String,
69    },
70    /// The requested revision does not belong to the requested entity.
71    RevisionMismatch {
72        /// Entity identity supplied by the caller.
73        entity: EntityId,
74        /// Revision identity supplied by the caller.
75        revision: RevisionId,
76    },
77    /// A requested capability is not installed or available.
78    UnsupportedCapability {
79        /// Universal operation requesting the capability.
80        operation: String,
81        /// Stable capability identifier.
82        capability: String,
83    },
84    /// The requested state transition has competing semantic candidates.
85    Conflict {
86        /// Stable subject involved in the conflict.
87        subject: String,
88        /// Structured conflict detail.
89        details: Box<ConflictDetails>,
90    },
91    /// Multiple semantic candidates remain possible.
92    Ambiguous {
93        /// Scope in which candidates were found.
94        scope: String,
95        /// Stable candidate handles.
96        candidates: Vec<String>,
97        /// Safe reason for the ambiguity.
98        reason: String,
99    },
100    /// The requested exact snapshot is unavailable.
101    SnapshotUnavailable {
102        /// Snapshot required by the operation.
103        snapshot: SnapshotId,
104    },
105    /// A tracked operation failed with structured context.
106    OperationFailed {
107        /// Stable operation identity or kind.
108        operation: String,
109        /// Safe failure diagnostic.
110        message: String,
111    },
112}
113
114/// Preserves semantic handles associated with a conflict.
115#[derive(Clone, Debug, Default, Eq, PartialEq)]
116pub struct ConflictDetails {
117    /// Base revision used by the attempted operation.
118    pub base_revision: Option<RevisionId>,
119    /// Revision expected by the operation.
120    pub expected_revision: Option<RevisionId>,
121    /// Revision currently present.
122    pub current_revision: Option<RevisionId>,
123    /// Competing left revision or migration handle.
124    pub left: Option<String>,
125    /// Competing right revision or migration handle.
126    pub right: Option<String>,
127    /// Relation or collection operation involved in the conflict.
128    pub operation: Option<String>,
129}

131impl ApplicationError {
132    /// Returns the stable category name used by adapters.
133    ///
134    /// # Parameters
135    /// - `self`: Error to classify.
136    ///
137    /// # Returns
138    /// A stable category label.
139    #[must_use]
140    pub const fn category(&self) -> &'static str {
141        match self {
142            Self::InvalidRequest { .. } => "invalid_request",
143            Self::NotFound { .. } => "not_found",
144            Self::SchemaViolation { .. } => "schema_violation",
145            Self::StaleBaseSnapshot { .. } => "stale_base_snapshot",
146            Self::SemanticConflict { .. } => "semantic_conflict",
147            Self::ValidationFailure { .. } => "validation_failure",
148            Self::PermissionDenied { .. } => "permission_denied",
149            Self::UnsupportedOperation { .. } => "unsupported_operation",
150            Self::ProjectionOutOfSync { .. } => "projection_out_of_sync",
151            Self::OperationCancelled { .. } => "operation_cancelled",
152            Self::Internal { .. } => "internal",
153            Self::RevisionMismatch { .. } => "revision_mismatch",
154            Self::UnsupportedCapability { .. } => "unsupported_capability",
155            Self::Conflict { .. } => "conflict",
156            Self::Ambiguous { .. } => "ambiguous",
157            Self::SnapshotUnavailable { .. } => "snapshot_unavailable",
158            Self::OperationFailed { .. } => "operation_failed",
159        }
160    }
161}

163impl Display for ApplicationError {
164    /// Formats an application error for a human-facing adapter.
165    ///
166    /// # Parameters
167    /// - `self`: Error to format.
168    /// - `formatter`: Destination formatter.
169    ///
170    /// # Returns
171    /// A formatting result.
172    ///
173    /// # Errors
174    /// Returns a formatting error when the destination rejects the write.
175    fn fmt(&self, formatter: &mut Formatter<'_>) -> fmt::Result {
176        match self {
177            Self::InvalidRequest { operation, message } => {
178                write!(formatter, "invalid {operation} request: {message}")
179            }
180            Self::NotFound { subject } => write!(formatter, "subject not found: {subject}"),
181            Self::SchemaViolation { message } => {
182                write!(formatter, "schema violation: {message}")
183            }
184            Self::StaleBaseSnapshot { requested, current } => {
185                write!(
186                    formatter,
187                    "stale snapshot {requested}; current is {current}"
188                )
189            }
190            Self::SemanticConflict { subject } => {
191                write!(formatter, "semantic conflict: {subject}")
192            }
193            Self::ValidationFailure { message } => {
194                write!(formatter, "validation failed: {message}")
195            }
196            Self::PermissionDenied { action, scope } => {
197                write!(formatter, "permission denied for {action} in {scope}")
198            }
199            Self::UnsupportedOperation { operation } => {
200                write!(formatter, "unsupported operation: {operation}")
201            }
202            Self::ProjectionOutOfSync { subject } => {
203                write!(formatter, "projection out of sync: {subject}")
204            }
205            Self::OperationCancelled { operation } => {
206                write!(formatter, "operation cancelled: {operation}")
207            }
208            Self::Internal { message } => {
209                write!(formatter, "internal application error: {message}")
210            }
211            Self::RevisionMismatch { entity, revision } => {
212                write!(
213                    formatter,
214                    "revision {revision} does not belong to entity {entity}"
215                )
216            }
217            Self::UnsupportedCapability {
218                operation,
219                capability,
220            } => write!(
221                formatter,
222                "unsupported capability {capability} for {operation}"
223            ),
224            Self::Conflict { subject, .. } => write!(formatter, "conflict for subject {subject}"),
225            Self::Ambiguous { scope, .. } => write!(formatter, "ambiguous result in {scope}"),
226            Self::SnapshotUnavailable { snapshot } => {
227                write!(formatter, "snapshot unavailable: {snapshot}")
228            }
229            Self::OperationFailed { operation, message } => {
230                write!(formatter, "operation {operation} failed: {message}")
231            }
232        }
233    }
234}

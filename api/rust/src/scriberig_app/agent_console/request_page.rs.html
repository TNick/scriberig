5/// One bounded fragment from a safe provider-neutral request field.
6#[derive(Clone, Debug, Eq, PartialEq)]
7pub struct ModelRequestDetailFragment {
8    /// Fragment category such as `message`, `tool_call`, or `tool`.
9    pub kind: String,
10    /// Zero-based item index within the fragment category.
11    pub index: u64,
12    /// Parent message index for tool-call fragments.
13    pub parent_index: Option<u64>,
14    /// Field name represented by this fragment.
15    pub field: String,
16    /// Unicode-scalar offset within the field.
17    pub field_offset: u64,
18    /// Message role when the field belongs to a message.
19    pub role: Option<String>,
20    /// Safe text at this position in the field.
21    pub text: String,
22}

24/// One bounded page of safe request detail for an invocation.
25#[derive(Clone, Debug, Eq, PartialEq)]
26pub struct ModelRequestDetailPage {
27    /// Safe content fragments included in this page.
28    pub fragments: Vec<ModelRequestDetailFragment>,
29    /// First Unicode-scalar offset represented by this page.
30    pub offset: u64,
31    /// Total Unicode-scalar length of the ordered safe detail stream.
32    pub total_chars: u64,
33    /// Offset to request next, absent at the end of the stream.
34    pub next_offset: Option<u64>,
35    /// Whether all safe text and tool content was captured.
36    pub content_complete: bool,
37}

39/// One ordered field in the bounded request detail stream.
40///
41/// # Public members
42/// None.
43///
44/// # Private members
45/// - `kind`: Category used to identify the source item.
46/// - `index`: Position within its category.
47/// - `parent_index`: Parent message position for tool calls.
48/// - `field`: Stable field label.
49/// - `role`: Message role when the field belongs to a message.
50/// - `text`: Safe text carried by this field.
51#[derive(Clone, Debug)]
52struct DetailField {
53    /// Category used to identify the source item.
54    kind: &'static str,
55    /// Position within its category.
56    index: u64,
57    /// Parent message position for tool calls.
58    parent_index: Option<u64>,
59    /// Stable field label.
60    field: &'static str,
61    /// Message role when the field belongs to a message.
62    role: Option<String>,
63    /// Safe text carried by this field.
64    text: String,
65}

67/// Projects one bounded page from a stored provider-neutral request.
68///
69/// The character budget applies across message text, tool calls, and tool
70/// definitions. It can split a field so a single large value cannot create an
71/// unbounded transport response.
72///
73/// # Parameters
74/// - `request`: Durable safe request projection.
75/// - `offset`: Requested Unicode-scalar offset in the ordered detail stream.
76/// - `max_chars`: Requested page size, clamped to 1 through 32,768.
77///
78/// # Returns
79/// One page of safe request text and its continuation offset.
80#[must_use]
81pub fn model_request_detail_page(
82    request: &ModelRequestProjection,
83    offset: u64,
84    max_chars: u32,
85) -> ModelRequestDetailPage {
86    let fields = detail_fields(request);
87    let total_chars = fields.iter().fold(0_u64, |total, field| {
88        total.saturating_add(field.text.chars().count() as u64)
89    });
90    let start = offset.min(total_chars);
91    let end = start
92        .saturating_add(u64::from(max_chars.clamp(1, 32_768)))
93        .min(total_chars);
94    let mut field_start = 0_u64;
95    let mut fragments = Vec::new();
96
97    for field in fields {
98        let field_end = field_start.saturating_add(field.text.chars().count() as u64);
99        let fragment_start = start.max(field_start);
100        let fragment_end = end.min(field_end);
101        if fragment_start < fragment_end {
102            if let (Ok(local_start), Ok(local_end)) = (
103                usize::try_from(fragment_start - field_start),
104                usize::try_from(fragment_end - field_start),
105            ) {
106                let chunk = field
107                    .text
108                    .chars()
109                    .skip(local_start)
110                    .take(local_end - local_start)
111                    .collect();
112                fragments.push(ModelRequestDetailFragment {
113                    kind: field.kind.to_owned(),
114                    index: field.index,
115                    parent_index: field.parent_index,
116                    field: field.field.to_owned(),
117                    field_offset: fragment_start - field_start,
118                    role: field.role,
119                    text: chunk,
120                });
121            }
122        }
123        field_start = field_end;
124    }
125
126    ModelRequestDetailPage {
127        fragments,
128        offset: start,
129        total_chars,
130        next_offset: (end < total_chars).then_some(end),
131        content_complete: request.content_complete,
132    }
133}

135/// Lists safe request fields in their stable presentation order.
136///
137/// # Parameters
138/// - `request`: Durable safe request projection.
139///
140/// # Returns
141/// Ordered fields with their message, tool-call, or tool identity.
142fn detail_fields(request: &ModelRequestProjection) -> Vec<DetailField> {
143    let mut fields = Vec::new();
144    append_message_fields(&mut fields, request);
145    append_tool_fields(&mut fields, request);
146    fields
147}

149/// Adds provider-visible message fields in request order.
150///
151/// # Parameters
152/// - `fields`: Ordered safe request fields being collected.
153/// - `request`: Durable safe request projection.
154fn append_message_fields(fields: &mut Vec<DetailField>, request: &ModelRequestProjection) {
155    let mut tool_call_index = 0_u64;
156    if let Some(messages) = request.full_messages.as_ref() {
157        for (index, message) in messages.iter().enumerate() {
158            let message_index = index as u64;
159            let role = Some(message.role.clone());
160            if let Some(tool_name) = &message.tool_name {
161                fields.push(DetailField {
162                    kind: "message",
163                    index: message_index,
164                    parent_index: None,
165                    field: "tool_name",
166                    role: role.clone(),
167                    text: tool_name.clone(),
168                });
169            }
170            if let Some(tool_call_id) = &message.tool_call_id {
171                fields.push(DetailField {
172                    kind: "message",
173                    index: message_index,
174                    parent_index: None,
175                    field: "tool_call_id",
176                    role: role.clone(),
177                    text: tool_call_id.clone(),
178                });
179            }
180            if message.parts.is_empty() {
181                fields.push(DetailField {
182                    kind: "message",
183                    index: message_index,
184                    parent_index: None,
185                    field: "content",
186                    role: role.clone(),
187                    text: message.content.clone(),
188                });
189                for call in &message.tool_calls {
190                    append_tool_call_fields(
191                        fields,
192                        &mut tool_call_index,
193                        message_index,
194                        &message.role,
195                        call,
196                    );
197                }
198            } else {
199                for (part_index, part) in message.parts.iter().enumerate() {
200                    match part {
201                        scriberig_session::ModelRequestDetailPart::Text(text) => {
202                            fields.push(DetailField {
203                                kind: "message_part",
204                                index: part_index as u64,
205                                parent_index: Some(message_index),
206                                field: "content",
207                                role: role.clone(),
208                                text: text.clone(),
209                            });
210                        }
211                        scriberig_session::ModelRequestDetailPart::ToolCall(call) => {
212                            append_tool_call_fields(
213                                fields,
214                                &mut tool_call_index,
215                                message_index,
216                                &message.role,
217                                call,
218                            );
219                        }
220                    }
221                }
222            }
223        }
224    } else {
225        for (index, message) in request.messages.iter().enumerate() {
226            fields.push(DetailField {
227                kind: "message",
228                index: index as u64,
229                parent_index: None,
230                field: "content",
231                role: Some(message.role.clone()),
232                text: message.content.clone(),
233            });
234        }
235    }
236}

238/// Adds safe tool schemas after all request messages.
239///
240/// # Parameters
241/// - `fields`: Ordered safe request fields being collected.
242/// - `request`: Durable safe request projection.
243fn append_tool_fields(fields: &mut Vec<DetailField>, request: &ModelRequestProjection) {
244    if let Some(tools) = request.full_tools.as_ref() {
245        for (index, tool) in tools.iter().enumerate() {
246            let index = index as u64;
247            push_tool_field(fields, index, "name", tool.name.clone());
248            push_tool_field(fields, index, "description", tool.description.clone());
249            push_tool_field(fields, index, "parameters", tool.parameters.clone());
250            if let Some(sampling) = &tool.constrained_sampling {
251                push_tool_field(fields, index, "constrained_sampling", sampling.to_string());
252            }
253        }
254    } else {
255        for (index, tool) in request.tools.iter().enumerate() {
256            let index = index as u64;
257            push_tool_field(fields, index, "name", tool.name.clone());
258            push_tool_field(fields, index, "description", tool.description.clone());
259            if let Some(parameters) = &tool.parameters {
260                push_tool_field(fields, index, "parameters", parameters.clone());
261            }
262        }
263    }
264}

266/// Adds one safe field for a registered tool schema.
267///
268/// # Parameters
269/// - `fields`: Ordered safe request fields being collected.
270/// - `index`: Tool index in the request.
271/// - `field`: Stable tool-schema field name.
272/// - `text`: Safe tool-schema value.
273fn push_tool_field(fields: &mut Vec<DetailField>, index: u64, field: &'static str, text: String) {

275        kind: "tool",
276        index,
277        parent_index: None,
278        field,
279        role: None,
280        text,
281    });
282}

284/// Adds a normalized tool call's safe identity and arguments.
285///
286/// # Parameters
287/// - `fields`: Ordered safe request fields being collected.
288/// - `tool_call_index`: Mutable global tool-call index.
289/// - `message_index`: Parent message index.
290/// - `role`: Parent provider-neutral message role.
291/// - `call`: Safe normalized tool-call data.
292fn append_tool_call_fields(
293    fields: &mut Vec<DetailField>,
294    tool_call_index: &mut u64,
295    message_index: u64,
296    role: &str,
297    call: &ModelRequestToolCall,
298) {
299    push_tool_call_field(
300        fields,
301        *tool_call_index,
302        message_index,
303        role,
304        "name",
305        call.name.clone(),
306    );
307    push_tool_call_field(
308        fields,
309        *tool_call_index,
310        message_index,
311        role,
312        "arguments",
313        call.arguments.clone(),
314    );
315    if let Some(id) = &call.id {
316        push_tool_call_field(
317            fields,
318            *tool_call_index,
319            message_index,
320            role,
321            "id",
322            id.clone(),
323        );
324    }
325    if let Some(namespace) = &call.namespace {
326        push_tool_call_field(
327            fields,
328            *tool_call_index,
329            message_index,
330            role,
331            "namespace",
332            namespace.clone(),
333        );
334    }
335    *tool_call_index = (*tool_call_index).saturating_add(1);
336}

8#[derive(serde::Deserialize)]
9/// Configuration envelope for curated model entries.
10struct CatalogInput {
11    /// Models available through one configured provider connection.
12    models: Vec<ModelInput>,
13}
14
15#[derive(serde::Deserialize)]
16/// One provider model and its model-specific thinking capabilities.
17struct ModelInput {
18    /// Stable Scriberig model identity.
19    id: String,
20    /// Safe user-facing model name.
21    display_name: String,
22    /// Provider-native model identifier.
23    provider_model_id: String,
24    /// Thinking levels supported by this model.
25    #[serde(default)]
26    thinking_levels: Vec<ThinkingLevelInput>,
27    /// Optional model default thinking level.
28    default_thinking_level: Option<String>,
29}
30
31#[derive(serde::Deserialize)]
32/// One advertised model-specific thinking level.
33struct ThinkingLevelInput {
34    /// Stable level identity interpreted by the adapter.
35    id: String,
36    /// Safe user-facing label.
37    label: String,
38    /// Optional safe description.
39    description: Option<String>,
40    /// Optional presentation order.
41    order: Option<u32>,
42}

44/// Builds a validated model library from trusted non-secret catalog JSON.
45///
46/// Provider-specific `provider_parameters` fields in the input are ignored by
47/// the application projection and must be consumed only by the runtime adapter.
48/// When the JSON is absent, a single fallback model is exposed without thinking
49/// levels. Every configured model gets one enabled profile on the local
50/// connection.
51///
52/// # Parameters
53/// - `json`: Optional catalog object with a `models` array.
54/// - `fallback_model`: Runtime model exposed when no catalog is configured.
55///
56/// # Returns
57/// Safe, validated model descriptors and selectable profiles.
58///
59/// # Errors
60/// Returns an error for malformed or internally inconsistent catalog data.
61pub fn model_library_from_json(
62    json: Option<&str>,
63    fallback_model: &str,
64) -> Result<ModelLibrary, &'static str> {
65    let input = match json {
66        Some(value) => serde_json::from_str::<CatalogInput>(value)
67            .map_err(|_| "model catalog JSON is invalid")?,
68        None => CatalogInput {
69            models: vec![ModelInput {
70                id: fallback_model.to_owned(),
71                display_name: fallback_model.to_owned(),
72                provider_model_id: fallback_model.to_owned(),
73                thinking_levels: Vec::new(),
74                default_thinking_level: None,
75            }],
76        },
77    };
78    let mut models = Vec::new();
79    let mut profiles = Vec::new();
80    for model_input in input.models {
81        let model_id = ModelId::new(model_input.id)?;
82        let levels = model_input
83            .thinking_levels
84            .into_iter()
85            .enumerate()
86            .map(|(index, level)| {
87                Ok(ThinkingLevelDescriptor {
88                    id: ThinkingLevelId::new(level.id)?,
89                    label: level.label,
90                    description: level.description,
91                    order: level
92                        .order
93                        .unwrap_or(u32::try_from(index + 1).unwrap_or(u32::MAX)),
94                })
95            })
96            .collect::<Result<Vec<_>, &'static str>>()?;
97        let default_level = model_input
98            .default_thinking_level
99            .map(ThinkingLevelId::new)
100            .transpose()?;
101        let profile_id = format!("{}-profile", model_id.as_str());
102        models.push(ModelDescriptor {
103            id: model_id.clone(),
104            display_name: model_input.display_name.clone(),
105            provider_profile: "local".to_owned(),
106            provider_model_id: model_input.provider_model_id,
107            capabilities: ModelCapabilities {
108                streaming: true,
109                tool_calls: true,
110                structured_output: false,
111            },
112            thinking_levels: levels,
113            default_thinking_level: default_level.clone(),
114        });
115        profiles.push(ModelProfile {
116            id: profile_id,
117            display_name: model_input.display_name,
118            model: model_id,
119            connection: "local".to_owned(),
120            credential: None,
121            default_thinking_level: default_level,
122            enabled: true,
123        });
124    }
125    ModelLibrary::new(models, profiles)
126}

128#[cfg(test)]
129mod tests {
130    use super::model_library_from_json;
131
132    /// Excludes private provider wire data from returned public descriptors.
133    #[test]
134    fn configured_catalog_projects_safe_model_specific_levels() {
135        let library = model_library_from_json(
136            Some(
137                r#"{"models":[{"id":"one","display_name":"One","provider_model_id":"wire-one","thinking_levels":[{"id":"deep","label":"Deep","provider_parameters":{"secret":"fake-secret-marker"}}]},{"id":"two","display_name":"Two","provider_model_id":"wire-two","thinking_levels":[{"id":"deep","label":"Deep"}]}]}"#,
138            ),
139            "fallback",
140        )
141        .expect("valid catalog");
142        let serialized = serde_json::to_string(&library).expect("serialize library");
143
144        assert_eq!(library.models.len(), 2);
145        assert_eq!(library.profiles.len(), 2);
146        assert_eq!(library.models[0].thinking_levels[0].id.as_str(), "deep");
147        assert_eq!(library.models[1].thinking_levels[0].id.as_str(), "deep");
148        assert!(!serialized.contains("provider_parameters"));
149        assert!(!serialized.contains("fake-secret-marker"));
150    }
151
152    /// Exposes the configured fallback when catalog JSON is absent.
153    #[test]
154    fn absent_catalog_exposes_fallback_model_without_levels() {
155        let library = model_library_from_json(None, "runtime-default").expect("fallback");
156
157        assert_eq!(library.models[0].provider_model_id, "runtime-default");
158        assert_eq!(library.models[0].thinking_levels, Vec::new());
159        assert_eq!(library.profiles[0].model.as_str(), "runtime-default");
160    }

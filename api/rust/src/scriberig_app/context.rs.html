7/// Identifies the principal on whose behalf an operation runs.
8#[derive(Clone, Debug, Eq, PartialEq)]
9pub struct Principal {
10    /// Stable caller identity used for provenance and audit.
11    pub id: String,
12}
13
14impl Principal {
15    /// Creates a principal from a stable identity string.
16    ///
17    /// # Parameters
18    /// - `id`: Stable principal identity.
19    ///
20    /// # Returns
21    /// The principal descriptor.
22    #[must_use]
23    pub fn new(id: impl Into<String>) -> Self {
24        Self { id: id.into() }
25    }
26}

28/// Carries the capabilities granted to one application request.
29#[derive(Clone, Debug, Default, Eq, PartialEq)]
30pub struct CapabilityContext {
31    /// Stable capabilities granted by the trusted runtime boundary.
32    pub granted: Vec<CapabilityId>,
33    /// Scoped capability grants supplied by the trusted runtime boundary.
34    pub grants: Vec<Capability>,
35}
36
37impl CapabilityContext {
38    /// Creates a capability context from granted identifiers.
39    ///
40    /// # Parameters
41    /// - `granted`: Capabilities granted to the caller.
42    ///
43    /// # Returns
44    /// The request capability context.
45    #[must_use]
46    pub fn new(granted: impl IntoIterator<Item = CapabilityId>) -> Self {
47        Self {
48            granted: granted.into_iter().collect(),
49            grants: Vec::new(),
50        }
51    }
52
53    /// Creates a capability context from complete scoped grants.
54    ///
55    /// # Parameters
56    /// - `grants`: Explicit capabilities to evaluate for this request.
57    ///
58    /// # Returns
59    /// A context containing the supplied grants.
60    #[must_use]
61    pub fn with_grants(grants: impl IntoIterator<Item = Capability>) -> Self {
62        Self {
63            granted: Vec::new(),
64            grants: grants.into_iter().collect(),
65        }
66    }
67
68    /// Checks whether a named capability is granted.
69    ///
70    /// # Parameters
71    /// - `self`: Capability context to inspect.
72    /// - `capability`: Capability to find.
73    ///
74    /// # Returns
75    /// `true` when the capability is present.
76    #[must_use]
77    pub fn allows(&self, capability: &CapabilityId) -> bool {
78        self.granted.iter().any(|candidate| candidate == capability)
79    }
80
81    /// Reports whether any scoped grant covers one application read.
82    ///
83    /// # Parameters
84    /// - `self`: Capability context to inspect.
85    /// - `action`: Action required by the read operation.
86    /// - `scope`: Semantic scope exposed by the operation.
87    /// - `snapshot`: Exact snapshot bound to the operation.
88    /// - `operation`: Universal operation kind, when known.
89    ///
90    /// # Returns
91    /// `true` when a complete scoped grant covers the request.
92    #[must_use]
93    pub fn permits(
94        &self,
95        action: &CapabilityAction,
96        scope: &CapabilityScope,
97        snapshot: Option<SnapshotId>,
98        operation: Option<OperationKind>,
99    ) -> bool {
100        self.grants
101            .iter()
102            .any(|grant| grant.covers(action, scope, snapshot, operation))
103    }
104}

106/// Carries immutable request context into shared application operations.
107#[derive(Clone, Debug, Default, Eq, PartialEq)]
108pub struct ApplicationContext {
109    /// Optional exact snapshot under which the request executes.
110    pub snapshot: Option<SnapshotId>,
111    /// Optional caller identity for provenance and audit.
112    pub principal: Option<Principal>,
113    /// Capabilities supplied by the trusted runtime boundary.
114    pub capabilities: CapabilityContext,
115    /// Optional correlation identifier for tracing one request.
116    pub correlation_id: Option<String>,
117}
118
119impl ApplicationContext {
120    /// Creates an empty context for an embedded operation.
121    ///
122    /// # Returns
123    /// A context with no snapshot, principal, or capabilities.
124    #[must_use]
125    pub fn empty() -> Self {
126        Self::default()
127    }
128
129    /// Binds the context to an exact snapshot.
130    ///
131    /// # Parameters
132    /// - `self`: Context to update.
133    /// - `snapshot`: Exact snapshot identity.
134    ///
135    /// # Returns
136    /// The updated context.
137    #[must_use]
138    pub const fn with_snapshot(mut self, snapshot: SnapshotId) -> Self {
139        self.snapshot = Some(snapshot);
140        self
141    }
142}

9/// Decodes a protocol command into an application command.
10/// - `super`: Parent-module imports used by this helper.
11/// - `kind`: Command kind to decode.
12/// - `payload`: Serialized command payload.
13///
14/// The decoded application command.
15///
16/// Returns an error when the payload is invalid.
17/// # Returns
18/// The operation result.
19/// # Errors
20/// Returns an error when the operation fails.
21pub(super) fn decode_command(
22    kind: &str,
23    payload: &serde_json::Value,
24) -> Result<(OperationKind, ApplicationRequest), ApplicationError> {
25    /// Implements `invalid`.
26    /// - `kind`: Argument named `kind`.
27    /// - `error`: Argument named `error`.
28    ///
29    /// # Returns
30    /// The operation result.
31    fn invalid(kind: &str, error: impl std::fmt::Display) -> ApplicationError {
32        ApplicationError::InvalidRequest {
33            operation: kind.to_owned(),
34            message: error.to_string(),
35        }
36    }
37
38    match kind {
39        "apply" => serde_json::from_value::<ApplyRequest>(payload.clone())
40            .map(|request| (OperationKind::Apply, ApplicationRequest::Apply(request)))
41            .map_err(|error| invalid(kind, error)),
42        "branch" => serde_json::from_value::<BranchRequest>(payload.clone())
43            .map(|request| (OperationKind::Branch, ApplicationRequest::Branch(request)))
44            .map_err(|error| invalid(kind, error)),
45        "commit" => serde_json::from_value::<CommitRequest>(payload.clone())
46            .map(|request| (OperationKind::Commit, ApplicationRequest::Commit(request)))
47            .map_err(|error| invalid(kind, error)),
48        "diff" => serde_json::from_value::<DiffRequest>(payload.clone())
49            .map(|request| (OperationKind::Diff, ApplicationRequest::Diff(request)))
50            .map_err(|error| invalid(kind, error)),
51        "domain_operation" => decode_domain_operation(payload)
52            .map(|request| {
53                (
54                    OperationKind::DomainOperation,
55                    ApplicationRequest::DomainOperation(request),
56                )
57            })
58            .map_err(|error| invalid(kind, error)),
59        "render" => serde_json::from_value::<RenderRequest>(payload.clone())
60            .map(|request| (OperationKind::Render, ApplicationRequest::Render(request)))
61            .map_err(|error| invalid(kind, error)),
62        _ => Err(ApplicationError::UnsupportedOperation {
63            operation: format!("command {kind}"),
64        }),
65    }
66}

68/// Converts a protocol domain operation payload to the application contract.
69///
70/// - `payload`: JSON-encoded typed domain operation command.
71///
72/// Exact application request with parsed semantic identifiers.
73///
74/// Returns a decoding error when an identifier or argument is invalid.
75/// # Returns
76/// The operation result.
77/// # Errors
78/// Returns an error when the operation fails.
79fn decode_domain_operation(payload: &serde_json::Value) -> Result<DomainOperationRequest, String> {
80    let command: scriberig_protocol::DomainOperationCommand =
81        serde_json::from_value(payload.clone()).map_err(|error| error.to_string())?;
82    let entity = command
83        .entity
84        .id
85        .parse::<scriberig_core::EntityId>()
86        .map_err(|error| error.to_string())?;
87    let snapshot = command
88        .snapshot
89        .id
90        .parse::<scriberig_core::SnapshotId>()
91        .map_err(|error| error.to_string())?;
92    let revision = command
93        .revision
94        .map(|value| value.parse::<scriberig_core::RevisionId>())
95        .transpose()
96        .map_err(|error| error.to_string())?;
97    let branch = command
98        .branch
99        .map(|value| value.parse::<scriberig_core::RootName>())
100        .transpose()
101        .map_err(|error| error.to_string())?;
102    let arguments = command
103        .arguments
104        .into_iter()
105        .map(|(key, value)| {
106            let value = if let Some(text) = value.as_str() {
107                scriberig_query::QueryValue::Text(text.to_owned())
108            } else if let Some(integer) = value.as_i64() {
109                scriberig_query::QueryValue::Integer(integer)
110            } else if let Some(boolean) = value.as_bool() {
111                scriberig_query::QueryValue::Boolean(boolean)
112            } else {
113                return Err(format!(
114                    "invalid argument {key}: expected a string, integer, or boolean"
115                ));
116            };
117            Ok((key, value))
118        })
119        .collect::<Result<Vec<_>, String>>()?;
120    Ok(DomainOperationRequest {
121        capability: scriberig_app::CapabilityId::new(command.capability).map_err(str::to_owned)?,
122        subject: scriberig_query::Selection {
123            entity,
124            snapshot,
125            revision,
126            relation_path: Vec::new(),
127            projection_anchor: None,
128        },
129        context: scriberig_app::BranchContext {
130            base_snapshot: snapshot,
131            branch,
132        },
133        arguments,
134    })
135}

137#[cfg(test)]
138mod tests {
139    use super::{ApplicationRequest, OperationKind, decode_command};
140
141    /// Decodes exact typed domain operation payloads at the protocol boundary.
142    #[test]
143    fn domain_operation_command_preserves_exact_scope_and_arguments() {
144        let entity = scriberig_core::EntityId::new();
145        let snapshot = scriberig_core::SnapshotId::new([1; 32]);
146        let payload = serde_json::json!({
147            "capability": "code.add_function",
148            "entity": {"id": entity.to_string()},
149            "snapshot": {"id": snapshot.to_string()},
150            "revision": null,
151            "branch": "root_working",
152            "arguments": {"name": "greet", "exported": true, "arity": 1}
153        });
154        let (kind, request) =
155            decode_command("domain_operation", &payload).expect("decode semantic domain command");
156        assert_eq!(kind, OperationKind::DomainOperation);
157        let ApplicationRequest::DomainOperation(request) = request else {
158            panic!("decoded incompatible application request");
159        };
160        assert_eq!(request.capability.as_str(), "code.add_function");
161        assert_eq!(request.subject.entity, entity);
162        assert_eq!(request.context.base_snapshot, snapshot);
163        assert_eq!(request.context.branch.unwrap().as_str(), "working");
164        assert_eq!(request.arguments.len(), 3);
165    }
166
167    /// Rejects non-scalar domain arguments before application dispatch.
168    #[test]
169    fn domain_operation_rejects_structured_arguments() {
170        let entity = scriberig_core::EntityId::new();
171        let snapshot = scriberig_core::SnapshotId::new([1; 32]);
172        let payload = serde_json::json!({
173            "capability": "code.add_function",
174            "entity": {"id": entity.to_string()},
175            "snapshot": {"id": snapshot.to_string()},
176            "arguments": {"body": {"nested": true}}
177        });
178        assert!(decode_command("domain_operation", &payload).is_err());
179    }

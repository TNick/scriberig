3use super::{
4    ApplicationError, ApplicationRequest, ApplyRequest, BranchRequest, CommitRequest, DiffRequest,
5    OperationKind, RenderRequest,
6};

8/// Decodes a protocol command into an application command.
9/// - `super`: Parent-module imports used by this helper.
10/// - `kind`: Command kind to decode.
11/// - `payload`: Serialized command payload.
12///
13/// # Returns
14/// The decoded application command.
15///
16/// # Errors
17/// Returns an error when the payload is invalid.
18pub(super) fn decode_command(
19    kind: &str,
20    payload: &serde_json::Value,
21) -> Result<(OperationKind, ApplicationRequest), ApplicationError> {
22    /// Implements `invalid`.
23    /// - `kind`: Argument named `kind`.
24    /// - `error`: Argument named `error`.
25    ///
26    /// # Returns
27    /// The value produced by the declaration.
28    fn invalid(kind: &str, error: impl std::fmt::Display) -> ApplicationError {
29        ApplicationError::InvalidRequest {
30            operation: kind.to_owned(),
31            message: error.to_string(),
32        }
33    }
34
35    match kind {
36        "apply" => serde_json::from_value::<ApplyRequest>(payload.clone())
37            .map(|request| (OperationKind::Apply, ApplicationRequest::Apply(request)))
38            .map_err(|error| invalid(kind, error)),
39        "branch" => serde_json::from_value::<BranchRequest>(payload.clone())
40            .map(|request| (OperationKind::Branch, ApplicationRequest::Branch(request)))
41            .map_err(|error| invalid(kind, error)),
42        "commit" => serde_json::from_value::<CommitRequest>(payload.clone())
43            .map(|request| (OperationKind::Commit, ApplicationRequest::Commit(request)))
44            .map_err(|error| invalid(kind, error)),
45        "diff" => serde_json::from_value::<DiffRequest>(payload.clone())
46            .map(|request| (OperationKind::Diff, ApplicationRequest::Diff(request)))
47            .map_err(|error| invalid(kind, error)),
48        "render" => serde_json::from_value::<RenderRequest>(payload.clone())
49            .map(|request| (OperationKind::Render, ApplicationRequest::Render(request)))
50            .map_err(|error| invalid(kind, error)),
51        _ => Err(ApplicationError::UnsupportedOperation {
52            operation: format!("command {kind}"),
53        }),
54    }
55}

13/// Converts a durable session into a compact provider-neutral response.
14///
15/// # Parameters
16/// - `super`: Server protocol module namespace used by the response type.
17/// - `session`: Durable session value returned by the application boundary.
18///
19/// # Returns
20/// Session tree containing ordered activity and compact result handles.
21pub(super) fn session_response(
22    session: scriberig_session::Session,
23) -> scriberig_protocol::SessionReadResponse {
24    let session_id = session.id;
25    let execution = session.execution;
26    let mut branches = session
27        .branches
28        .into_iter()
29        .map(|branch| {
30            let activity = activity::branch_execution_activity(&session_id.to_string(), &branch);
31            scriberig_protocol::SessionBranch {
32                id: branch.id.to_string(),
33                parent: branch.parent.map(|value| value.to_string()),
34                children: branch.children.iter().map(ToString::to_string).collect(),
35                title: branch.title,
36                status: match branch.status {
37                    scriberig_session::BranchStatus::Active => {
38                        scriberig_protocol::SessionBranchStatus::Active
39                    }
40                    scriberig_session::BranchStatus::Completed => {
41                        scriberig_protocol::SessionBranchStatus::Completed
42                    }
43                    scriberig_session::BranchStatus::Abandoned => {
44                        scriberig_protocol::SessionBranchStatus::Abandoned
45                    }
46                },
47                turns: branch
48                    .turns
49                    .into_iter()
50                    .map(|turn| scriberig_protocol::SessionTurn {
51                        id: turn.id.to_string(),
52                        ordinal: turn.ordinal,
53                        has_human_input: turn.human.is_some(),
54                        has_agent_output: turn.agent.is_some(),
55                        agent_output: turn.agent.map(|value| value.text),
56                        tool_invocation: turn.tool_invocation.map(|value| value.to_string()),
57                        tool_result: turn.tool_result.map(|value| value.to_string()),
58                    })
59                    .collect(),
60                activity,
61                result: branch
62                    .result
63                    .map(|result| scriberig_protocol::SessionBranchResult {
64                        id: result.id.to_string(),
65                        references: result
66                            .references
67                            .into_iter()
68                            .map(|value| value.to_string())
69                            .collect(),
70                        result_handles: result.result_handles,
71                    }),
72            }
73        })
74        .collect::<Vec<_>>();
75    branches.sort_by(|left, right| left.id.cmp(&right.id));
76    scriberig_protocol::SessionReadResponse {
77        session: scriberig_protocol::SessionReference {
78            id: session_id.to_string(),
79        },
80        branches,
81        execution: scriberig_protocol::SessionExecutionSelection {
82            revision: execution.revision,
83            model_profile: execution.model_profile,
84            model: execution.model,
85            thinking_level: execution.thinking_level,
86            history: execution
87                .history
88                .into_iter()
89                .map(|change| scriberig_protocol::SessionExecutionChange {
90                    revision: change.revision,
91                    kind: change.kind,
92                    summary: change.summary,
93                    model: change.model,
94                    thinking_level: change.thinking_level,
95                })
96                .collect(),
97        },
98    }
99}

101/// Extracts stable result handles without exposing canonical result payloads.
102/// - `super`: Parent-module imports used by this helper.
103/// - `result`: Argument named `result`.
104///
105/// # Returns
106/// The value produced by the declaration.
107pub(super) fn result_handles(result: &scriberig_app::ApplicationResult) -> Vec<String> {
108    let summary = match result {
109        scriberig_app::ApplicationResult::Diff(value)
110        | scriberig_app::ApplicationResult::History(value)
111        | scriberig_app::ApplicationResult::Render(value)
112        | scriberig_app::ApplicationResult::Branch(value)
113        | scriberig_app::ApplicationResult::Apply(value)
114        | scriberig_app::ApplicationResult::Commit(value) => Some(value),
115        _ => None,
116    };
117    summary.map_or_else(Vec::new, |value| vec![value.handle.clone()])
118}

120/// Tests whether a protocol event matches a case-insensitive kind filter.
121/// - `super`: Parent-module imports used by this helper.
122///
123/// # Parameters
124/// - `event`: Event whose kind is being tested.
125/// - `event_kinds`: Requested event kind names; empty selects every event.
126///
127/// # Returns
128/// `true` when the event is selected by the subscription.
129pub(super) fn event_matches(event: &Event, event_kinds: &[String]) -> bool {
130    event_kinds.is_empty()
131        || event_kinds.iter().any(|kind| {
132            kind.eq_ignore_ascii_case(&format!("{:?}", event.kind))
133                || kind.eq_ignore_ascii_case(&format!("{:?}", event.kind).replace('_', ""))
134        })
135}

137/// Preserves safe semantic handles in protocol error details.
138///
139/// # Parameters
140/// - `error`: Application error being serialized.
141///
142/// # Returns
143/// JSON-safe structured details without backend internals.
144pub fn error_details(error: &ApplicationError) -> serde_json::Value {
145    match error {
146        ApplicationError::NotFound { subject } => serde_json::json!({"subject": subject}),
147        ApplicationError::InvalidRequest { operation, message }
148        | ApplicationError::OperationFailed { operation, message } => {
149            serde_json::json!({"operation": operation, "message": message})
150        }
151        ApplicationError::UnsupportedCapability {
152            operation,
153            capability,
154        } => serde_json::json!({"operation": operation, "capability": capability}),
155        ApplicationError::Conflict { subject, details } => serde_json::json!({
156            "subject": subject,
157            "base_revision": details.base_revision.map(|value| value.to_string()),
158            "expected_revision": details.expected_revision.map(|value| value.to_string()),
159            "current_revision": details.current_revision.map(|value| value.to_string()),
160            "left": details.left.as_ref(),
161            "right": details.right.as_ref(),
162            "operation": details.operation,
163        }),
164        ApplicationError::Ambiguous {
165            scope,
166            candidates,
167            reason,
168        } => serde_json::json!({"scope": scope, "candidates": candidates, "reason": reason}),
169        ApplicationError::SnapshotUnavailable { snapshot } => {
170            serde_json::json!({"snapshot": snapshot.to_string()})
171        }
172        _ => serde_json::json!({}),
173    }
174}

176/// Converts an application operation record to its public lifecycle form.
177/// - `super`: Parent-module imports used by this helper.
178/// - `record`: Argument named `record`.
179///
180/// # Returns
181/// The value produced by the declaration.
182pub(super) fn operation_from_record(record: scriberig_app::OperationRecord) -> Operation {
183    Operation {
184        reference: scriberig_protocol::OperationReference {
185            id: record.reference.to_string(),
186        },
187        kind: match record.kind {
188            scriberig_app::OperationKind::Inspect => {
189                scriberig_protocol::operation::OperationKind::Inspect
190            }
191            scriberig_app::OperationKind::Query => {
192                scriberig_protocol::operation::OperationKind::Query
193            }
194            scriberig_app::OperationKind::Apply => {
195                scriberig_protocol::operation::OperationKind::Apply
196            }
197            scriberig_app::OperationKind::List => {
198                scriberig_protocol::operation::OperationKind::List
199            }
200            scriberig_app::OperationKind::Follow => {
201                scriberig_protocol::operation::OperationKind::Follow
202            }
203            scriberig_app::OperationKind::Find => {
204                scriberig_protocol::operation::OperationKind::Find
205            }
206            scriberig_app::OperationKind::Diff => {
207                scriberig_protocol::operation::OperationKind::Diff
208            }
209            scriberig_app::OperationKind::History => {
210                scriberig_protocol::operation::OperationKind::History
211            }
212            scriberig_app::OperationKind::Render => {
213                scriberig_protocol::operation::OperationKind::Render
214            }
215            scriberig_app::OperationKind::Branch => {
216                scriberig_protocol::operation::OperationKind::Branch
217            }
218            scriberig_app::OperationKind::Agent => {
219                scriberig_protocol::operation::OperationKind::Agent
220            }
221            scriberig_app::OperationKind::Commit => {
222                scriberig_protocol::operation::OperationKind::Commit
223            }
224        },
225        status: match record.status {
226            scriberig_app::OperationStatus::Queued => {
227                scriberig_protocol::operation::OperationStatus::Queued
228            }
229            scriberig_app::OperationStatus::Running => {
230                scriberig_protocol::operation::OperationStatus::Running
231            }
232            scriberig_app::OperationStatus::Completed => {
233                scriberig_protocol::operation::OperationStatus::Completed
234            }
235            scriberig_app::OperationStatus::Failed => {
236                scriberig_protocol::operation::OperationStatus::Failed
237            }
238            scriberig_app::OperationStatus::Cancelled => {
239                scriberig_protocol::operation::OperationStatus::Cancelled
240            }
241        },
242        progress: record.progress.map(|progress| {
243            let percent = progress
244                .total
245                .filter(|total| *total > 0)
246                .map_or(0, |total| {
247                    progress
248                        .completed
249                        .unwrap_or_default()
250                        .saturating_mul(100)
251                        .checked_div(total)
252                        .unwrap_or(0)
253                        .min(100) as u8
254                });
255            scriberig_protocol::OperationProgress {
256                percent,
257                message: progress.phase,
258            }
259        }),
260        result_snapshot: None,
261        result_handles: record.result_handles,
262        agent_result: record.agent_result.map(agent_result),
263        error: record.error.map(|error| error.to_string()),
264    }
265}

267/// Converts an application agent result into a provider-neutral protocol value.
268///
269/// # Parameters
270/// - `result`: Durable application result.
271///
272/// # Returns
273/// Safe protocol result without provider credentials or DTOs.
274///
275/// # Errors
276/// This conversion is infallible because all identifiers have already been
277/// validated by the application layer.
278fn agent_result(result: scriberig_app::AgentResult) -> scriberig_protocol::AgentResult {
279    scriberig_protocol::AgentResult {
280        session: scriberig_protocol::SessionReference {
281            id: result.session.to_string(),
282        },
283        root_branch: scriberig_protocol::BranchReference {
284            id: result.root_branch.to_string(),
285        },
286        final_output: result.final_output,
287        branch_result: result.branch_result.to_string(),
288        produced_changes: result
289            .produced_changes
290            .into_iter()
291            .map(|value| scriberig_protocol::EntityReference {
292                id: value.to_string(),
293            })
294            .collect(),
295        validation_results: result.validation_results,
296        unresolved_questions: result.unresolved_questions,
297        candidate_changes: result
298            .candidate_changes
299            .into_iter()
300            .map(|value| scriberig_protocol::CandidateChangeReference {
301                migration: value.migration,
302                entity: value
303                    .entity
304                    .map(|id| scriberig_protocol::EntityReference { id: id.to_string() }),
305                candidate_snapshot: value
306                    .candidate_snapshot
307                    .map(|id| scriberig_protocol::SnapshotReference { id: id.to_string() }),
308                validation: value.validation,
309            })
310            .collect(),
311    }

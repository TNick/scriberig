13/// Converts a durable session into a compact provider-neutral response.
14///
15/// # Parameters
16/// - `super`: Server protocol module namespace used by the response type.
17/// - `session`: Durable session value returned by the application boundary.
18///
19/// # Returns
20/// Session tree containing ordered activity and compact result handles.
21pub(super) fn session_response(
22    session: scriberig_session::Session,
23) -> scriberig_protocol::SessionReadResponse {
24    let session_id = session.id;
25    let execution = session.execution;
26    let mut branches = session
27        .branches
28        .into_iter()
29        .map(|branch| {
30            let activity = activity::branch_execution_activity(&session_id.to_string(), &branch);
31            scriberig_protocol::SessionBranch {
32                id: branch.id.to_string(),
33                parent: branch.parent.map(|value| value.to_string()),
34                children: branch.children.iter().map(ToString::to_string).collect(),
35                title: branch.title,
36                status: match branch.status {
37                    scriberig_session::BranchStatus::Active => {
38                        scriberig_protocol::SessionBranchStatus::Active
39                    }
40                    scriberig_session::BranchStatus::Completed => {
41                        scriberig_protocol::SessionBranchStatus::Completed
42                    }
43                    scriberig_session::BranchStatus::Abandoned => {
44                        scriberig_protocol::SessionBranchStatus::Abandoned
45                    }
46                },
47                turns: branch
48                    .turns
49                    .into_iter()
50                    .map(|turn| scriberig_protocol::SessionTurn {
51                        id: turn.id.to_string(),
52                        ordinal: turn.ordinal,
53                        has_human_input: turn.human.is_some(),
54                        has_agent_output: turn.agent.is_some(),
55                        agent_output: turn.agent.map(|value| value.text),
56                        tool_invocation: turn.tool_invocation.map(|value| value.to_string()),
57                        tool_result: turn.tool_result.map(|value| value.to_string()),
58                    })
59                    .collect(),
60                activity,
61                result: branch
62                    .result
63                    .map(|result| scriberig_protocol::SessionBranchResult {
64                        id: result.id.to_string(),
65                        references: result
66                            .references
67                            .into_iter()
68                            .map(|value| value.to_string())
69                            .collect(),
70                        result_handles: result.result_handles,
71                    }),
72            }
73        })
74        .collect::<Vec<_>>();
75    branches.sort_by(|left, right| left.id.cmp(&right.id));
76    scriberig_protocol::SessionReadResponse {
77        session: scriberig_protocol::SessionReference {
78            id: session_id.to_string(),
79        },
80        branches,
81        execution: scriberig_protocol::SessionExecutionSelection {
82            revision: execution.revision,
83            model_profile: execution.model_profile,
84            model: execution.model,
85            thinking_level: execution.thinking_level,
86        },
87    }
88}

90/// Extracts stable result handles without exposing canonical result payloads.
91/// - `super`: Parent-module imports used by this helper.
92/// - `result`: Argument named `result`.
93///
94/// # Returns
95/// The value produced by the declaration.
96pub(super) fn result_handles(result: &scriberig_app::ApplicationResult) -> Vec<String> {
97    let summary = match result {
98        scriberig_app::ApplicationResult::Diff(value)
99        | scriberig_app::ApplicationResult::History(value)
100        | scriberig_app::ApplicationResult::Render(value)
101        | scriberig_app::ApplicationResult::Branch(value)
102        | scriberig_app::ApplicationResult::Apply(value)
103        | scriberig_app::ApplicationResult::Commit(value) => Some(value),
104        _ => None,
105    };
106    summary.map_or_else(Vec::new, |value| vec![value.handle.clone()])
107}

109/// Tests whether a protocol event matches a case-insensitive kind filter.
110/// - `super`: Parent-module imports used by this helper.
111///
112/// # Parameters
113/// - `event`: Event whose kind is being tested.
114/// - `event_kinds`: Requested event kind names; empty selects every event.
115///
116/// # Returns
117/// `true` when the event is selected by the subscription.
118pub(super) fn event_matches(event: &Event, event_kinds: &[String]) -> bool {
119    event_kinds.is_empty()
120        || event_kinds.iter().any(|kind| {
121            kind.eq_ignore_ascii_case(&format!("{:?}", event.kind))
122                || kind.eq_ignore_ascii_case(&format!("{:?}", event.kind).replace('_', ""))
123        })
124}

126/// Preserves safe semantic handles in protocol error details.
127///
128/// # Parameters
129/// - `error`: Application error being serialized.
130///
131/// # Returns
132/// JSON-safe structured details without backend internals.
133pub fn error_details(error: &ApplicationError) -> serde_json::Value {
134    match error {
135        ApplicationError::NotFound { subject } => serde_json::json!({"subject": subject}),
136        ApplicationError::InvalidRequest { operation, message }
137        | ApplicationError::OperationFailed { operation, message } => {
138            serde_json::json!({"operation": operation, "message": message})
139        }
140        ApplicationError::UnsupportedCapability {
141            operation,
142            capability,
143        } => serde_json::json!({"operation": operation, "capability": capability}),
144        ApplicationError::Conflict { subject, details } => serde_json::json!({
145            "subject": subject,
146            "base_revision": details.base_revision.map(|value| value.to_string()),
147            "expected_revision": details.expected_revision.map(|value| value.to_string()),
148            "current_revision": details.current_revision.map(|value| value.to_string()),
149            "left": details.left.as_ref(),
150            "right": details.right.as_ref(),
151            "operation": details.operation,
152        }),
153        ApplicationError::Ambiguous {
154            scope,
155            candidates,
156            reason,
157        } => serde_json::json!({"scope": scope, "candidates": candidates, "reason": reason}),
158        ApplicationError::SnapshotUnavailable { snapshot } => {
159            serde_json::json!({"snapshot": snapshot.to_string()})
160        }
161        _ => serde_json::json!({}),
162    }
163}

165/// Converts an application operation record to its public lifecycle form.
166/// - `super`: Parent-module imports used by this helper.
167/// - `record`: Argument named `record`.
168///
169/// # Returns
170/// The value produced by the declaration.
171pub(super) fn operation_from_record(record: scriberig_app::OperationRecord) -> Operation {
172    Operation {
173        reference: scriberig_protocol::OperationReference {
174            id: record.reference.to_string(),
175        },
176        kind: match record.kind {
177            scriberig_app::OperationKind::Inspect => {
178                scriberig_protocol::operation::OperationKind::Inspect
179            }
180            scriberig_app::OperationKind::Query => {
181                scriberig_protocol::operation::OperationKind::Query
182            }
183            scriberig_app::OperationKind::Apply => {
184                scriberig_protocol::operation::OperationKind::Apply
185            }
186            scriberig_app::OperationKind::List => {
187                scriberig_protocol::operation::OperationKind::List
188            }
189            scriberig_app::OperationKind::Follow => {
190                scriberig_protocol::operation::OperationKind::Follow
191            }
192            scriberig_app::OperationKind::Find => {
193                scriberig_protocol::operation::OperationKind::Find
194            }
195            scriberig_app::OperationKind::Diff => {
196                scriberig_protocol::operation::OperationKind::Diff
197            }
198            scriberig_app::OperationKind::History => {
199                scriberig_protocol::operation::OperationKind::History
200            }
201            scriberig_app::OperationKind::Render => {
202                scriberig_protocol::operation::OperationKind::Render
203            }
204            scriberig_app::OperationKind::Branch => {
205                scriberig_protocol::operation::OperationKind::Branch
206            }
207            scriberig_app::OperationKind::Agent => {
208                scriberig_protocol::operation::OperationKind::Agent
209            }
210            scriberig_app::OperationKind::Commit => {
211                scriberig_protocol::operation::OperationKind::Commit
212            }
213        },
214        status: match record.status {
215            scriberig_app::OperationStatus::Queued => {
216                scriberig_protocol::operation::OperationStatus::Queued
217            }
218            scriberig_app::OperationStatus::Running => {
219                scriberig_protocol::operation::OperationStatus::Running
220            }
221            scriberig_app::OperationStatus::Completed => {
222                scriberig_protocol::operation::OperationStatus::Completed
223            }
224            scriberig_app::OperationStatus::Failed => {
225                scriberig_protocol::operation::OperationStatus::Failed
226            }
227            scriberig_app::OperationStatus::Cancelled => {
228                scriberig_protocol::operation::OperationStatus::Cancelled
229            }
230        },
231        progress: record.progress.map(|progress| {
232            let percent = progress
233                .total
234                .filter(|total| *total > 0)
235                .map_or(0, |total| {
236                    progress
237                        .completed
238                        .unwrap_or_default()
239                        .saturating_mul(100)
240                        .checked_div(total)
241                        .unwrap_or(0)
242                        .min(100) as u8
243                });
244            scriberig_protocol::OperationProgress {
245                percent,
246                message: progress.phase,
247            }
248        }),
249        result_snapshot: None,
250        result_handles: record.result_handles,
251        agent_result: record.agent_result.map(agent_result),
252        error: record.error.map(|error| error.to_string()),
253    }
254}

256/// Converts an application agent result into a provider-neutral protocol value.
257///
258/// # Parameters
259/// - `result`: Durable application result.
260///
261/// # Returns
262/// Safe protocol result without provider credentials or DTOs.
263///
264/// # Errors
265/// This conversion is infallible because all identifiers have already been
266/// validated by the application layer.
267fn agent_result(result: scriberig_app::AgentResult) -> scriberig_protocol::AgentResult {
268    scriberig_protocol::AgentResult {
269        session: scriberig_protocol::SessionReference {
270            id: result.session.to_string(),
271        },
272        root_branch: scriberig_protocol::BranchReference {
273            id: result.root_branch.to_string(),
274        },
275        final_output: result.final_output,
276        branch_result: result.branch_result.to_string(),
277        produced_changes: result
278            .produced_changes
279            .into_iter()
280            .map(|value| scriberig_protocol::EntityReference {
281                id: value.to_string(),
282            })
283            .collect(),
284        validation_results: result.validation_results,
285        unresolved_questions: result.unresolved_questions,
286        candidate_changes: result
287            .candidate_changes
288            .into_iter()
289            .map(|value| scriberig_protocol::CandidateChangeReference {
290                migration: value.migration,
291                entity: value
292                    .entity
293                    .map(|id| scriberig_protocol::EntityReference { id: id.to_string() }),
294                candidate_snapshot: value
295                    .candidate_snapshot
296                    .map(|id| scriberig_protocol::SnapshotReference { id: id.to_string() }),
297                validation: value.validation,
298            })
299            .collect(),
300    }

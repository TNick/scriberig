19/// Creates the deterministic upstream Faux provider and its selected model.
20///
21/// - `entity`: Selected semantic entity.
22/// - `snapshot`: Selected exact snapshot.
23/// - `accepted`: Deterministic accepted source text.
24///
25/// # Returns
26/// A deterministic upstream Faux provider and its selected model.
27pub(super) fn faux_provider(
28    entity: EntityId,
29    snapshot: SnapshotId,
30    accepted: &str,
31) -> (Arc<dyn rpi_ai::Provider>, rpi_ai::Model) {
32    let provider = FauxProvider::new(
33        FauxScript::new()
34            .with_tool_call(
35                "inspect",
36                json!({"entity": entity.to_string(), "snapshot": snapshot.to_string(), "detail": "detailed"}),
37            )
38            .with_tool_call("project_propose", json!({"text": accepted}))
39            .with_text("Candidate ready for review."),
40    );
41    let model = provider.default_model().clone();
42    (provider, model)
43}

45/// Creates branch-scoped semantic tools for the project workflow.
46///
47/// - `tool`: Owned project service adapter.
48/// - `branch`: Canonical ephemeral branch identity for this invocation.
49/// - `context`: Trusted `ScribeRig` request context.
50///
51/// # Returns
52/// Snapshot-bound inspect and candidate tools.
53///
54/// # Errors
55/// Returns an error if the branch catalog rejects a semantic tool.
56pub(super) fn project_tools(
57    tool: Arc<ProjectAgentTool>,
58    branch: EntityId,
59    context: ApplicationContext,
60) -> Result<Vec<Arc<SemanticAgentTool>>, String> {
61    let inspect = ProviderTool {
62        name: "inspect".to_owned(),
63        description: "Inspect selected semantic entity.".to_owned(),
64        parameters: json!({"type": "object", "required": ["entity", "snapshot"], "properties": {
65            "entity": {"type": "string"}, "snapshot": {"type": "string"}, "detail": {"type": "string"}
66        }})
67        .into(),
68        constrained_sampling: None,
69    };
70    let propose = ProviderTool {
71        name: "project_propose".to_owned(),
72        description: "Create a reviewed source candidate.".to_owned(),
73        parameters: json!({"type": "object", "required": ["text"], "properties": {
74            "text": {"type": "string"}
75        }})
76        .into(),
77        constrained_sampling: None,
78    };
79    let catalog = ToolCatalog::new(vec![
80        tool_descriptor(inspect.clone(), false, ToolCategory::SemanticRead),
81        tool_descriptor(propose.clone(), true, ToolCategory::SemanticMutation),
82    ]);
83    let inspect_tool = SemanticAgentTool::new(
84        &inspect,
85        "Inspect selected source",
86        branch,
87        catalog.clone(),
88        CapabilityContext::default(),
89        {
90            let tool = Arc::clone(&tool);
91            let context = context.clone();
92            move |_id, arguments, _signal, _update| {
93                let tool = Arc::clone(&tool);
94                let context = context.clone();
95                async move {
96                    let entity = parse_id::<EntityId>(&arguments, "entity")?;
97                    let snapshot = parse_id::<SnapshotId>(&arguments, "snapshot")?;
98                    let result = tool
99                        .invoke(
100                            &context,
101                            ToolInput {
102                                entity,
103                                snapshot: Some(snapshot),
104                                detail: scriberig_agent_core::ToolDetail::Detailed,
105                            },
106                        )
107                        .map_err(|_| AgentError::tool("semantic inspect failed"))?;
108                    Ok(AgentToolResult::text(format!(
109                        "{}: {} {}",
110                        result.entity,
111                        result.summary.unwrap_or_default(),
112                        result.handles.join(", ")
113                    )))
114                }
115            }
116        },
117    )?;
118    let propose_tool = SemanticAgentTool::new(
119        &propose,
120        "Create project candidate",
121        branch,
122        catalog,
123        CapabilityContext::default(),
124        move |_id, arguments, _signal, _update| {
125            let tool = Arc::clone(&tool);
126            let context = context.clone();
127            async move {
128                let result = tool
129                    .invoke_named(&context, "project_propose", &arguments)
130                    .map_err(|_| AgentError::tool("candidate proposal failed"))?;
131                let id = result
132                    .handles
133                    .first()
134                    .ok_or_else(|| AgentError::tool("candidate identity missing"))?;
135                Ok(AgentToolResult::text(format!("candidate created: {id}")))
136            }
137        },
138    )?;
139    Ok(vec![Arc::new(inspect_tool), Arc::new(propose_tool)])
140}

142/// Creates a descriptor retaining `ScribeRig` semantic tool policy.
143///
144/// - `definition`: Provider-facing schema.
145/// - `state_changing`: Whether the operation creates a candidate.
146/// - `category`: Semantic operation category.
147///
148/// # Returns
149/// A bounded tool descriptor for the trusted catalog.
150fn tool_descriptor(
151    definition: ProviderTool,
152    state_changing: bool,
153    category: ToolCategory,
154) -> ToolDescriptor {
155    ToolDescriptor {
156        definition,
157        category,
158        result_shape: "bounded semantic result".to_owned(),
159        invocation_label: None,
160        state_changing,
161        execution_mode: ToolExecutionMode::Atomic,
162        required_capability: None,
163    }
164}

166/// Parses one stable identifier from provider-supplied tool arguments.
167///
168/// - `arguments`: Provider-supplied JSON object.
169/// - `name`: Required string property.
170///
171/// # Returns
172/// The exact parsed semantic identifier.
173///
174/// # Errors
175/// Returns a generic tool error when parsing fails.
176fn parse_id<T: std::str::FromStr>(arguments: &Value, name: &str) -> Result<T, AgentError> {
177    arguments
178        .get(name)
179        .and_then(Value::as_str)
180        .and_then(|value| value.parse().ok())
181        .ok_or_else(|| AgentError::tool("invalid semantic tool identifier"))
182}

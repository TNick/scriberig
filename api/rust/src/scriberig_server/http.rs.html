23/// Handles one local HTTP connection.
24///
25/// # Parameters
26/// - `stream`: Connected client socket.
27/// - `state`: Application and fixture state.
28///
29/// # Returns
30/// I/O result after one response is written.
31///
32/// # Errors
33/// Returns an I/O error when the request or response cannot be read or written.
34pub fn handle_connection<S: ApplicationService>(
35    mut stream: TcpStream,
36    state: &HttpState<S>,
37) -> std::io::Result<()> {
38    let mut request = vec![0_u8; 64 * 1024];
39    let bytes_read = stream.read(&mut request)?;
40    let text = String::from_utf8_lossy(&request[..bytes_read]);
41    let Some((head, body)) = text.split_once("\r\n\r\n") else {
42        return write_json(
43            &mut stream,
44            400,
45            &json!({"code": "invalid_request", "message": "malformed HTTP request"}),
46        );
47    };
48    let mut lines = head.lines();
49    let Some(request_line) = lines.next() else {
50        return write_json(
51            &mut stream,
52            400,
53            &json!({"code": "invalid_request", "message": "missing request line"}),
54        );
55    };
56    let mut request_parts = request_line.split_whitespace();
57    let method = request_parts.next().unwrap_or_default();
58    let path = request_parts.next().unwrap_or_default();
59    let parsed = match (method, path) {
60        ("GET", "/api/health") => Ok(json!({"status": "ready", "database": "ready"})),
61        ("POST", "/api/protocol") => protocol(state, body),
62        ("POST", "/api/protocol/capabilities") => Ok(capabilities(state)),
63        ("POST", "/api/protocol/inspect") => inspect(state, body),
64        ("POST", "/api/protocol/list") => list(state, body),
65        _ => Err((
66            404,
67            json!({"code": "not_found", "message": "route not found"}),
68        )),
69    };
70    match parsed {
71        Ok(value) => write_json(&mut stream, 200, &value),
72        Err((status, value)) => write_json(&mut stream, status, &value),
73    }
74}

76/// Dispatches an authoritative protocol envelope through shared application semantics.
77fn protocol<S: ApplicationService>(
78    state: &HttpState<S>,
79    body: &str,
80) -> Result<Value, (u16, Value)> {
81    let request: scriberig_protocol::RequestEnvelope =
82        serde_json::from_str(body).map_err(|error| {
83            (
84                400,
85                json!({"code": "invalid_request", "message": error.to_string()}),
86            )
87        })?;
88    let response = Dispatcher::new(&state.service).dispatch(request);
89    serde_json::to_value(response).map_err(|error| {
90        (
91            500,
92            json!({"code": "internal", "message": error.to_string()}),
93        )
94    })
95}

97/// Returns the capabilities required for development bootstrap.
98fn capabilities<S>(state: &HttpState<S>) -> Value {
99    json!({
100        "protocolMajor": 1,
101        "features": ["inspect", "list"],
102        "capabilities": [
103            {"id": "inspect.entity", "description": "Inspect one entity", "available": true},
104            {"id": "list.children", "description": "List root children", "available": true}
105        ],
106        "root": {
107            "entityId": state.fixture.root_entity.to_string(),
108            "snapshotId": state.fixture.snapshot.to_string()
109        }
110    })
111}

113/// Maps one browser inspection request through the application boundary.
114fn inspect<S: ApplicationService>(state: &HttpState<S>, body: &str) -> Result<Value, (u16, Value)> {
115    let value: Value = serde_json::from_str(body).map_err(|error| {
116        (
117            400,
118            json!({"code": "invalid_request", "message": error.to_string()}),
119        )
120    })?;
121    let entity_text = value
122        .get("entityId")
123        .and_then(Value::as_str)
124        .unwrap_or_default();
125    let entity = EntityId::from_str(entity_text).map_err(|_| {
126        (
127            400,
128            json!({"code": "invalid_request", "message": "invalid entity id"}),
129        )
130    })?;
131    let snapshot = snapshot_resolution(&value, state.fixture.snapshot)?;
132    let result = state
133        .service
134        .inspect(
135            &ApplicationContext::empty(),
136            &InspectRequest {
137                entity,
138                snapshot,
139                detail: scriberig_app::DetailLevel::Compact,
140            },
141        )
142        .map_err(|error| {
143            (
144                404,
145                json!({"code": error.category(), "message": error.to_string()}),
146            )
147        })?;
148    Ok(object_description(
149        &result.entity.to_string(),
150        result.snapshot,
151        result.revision,
152        result.object_type.map(|value| value.to_string()),
153        result.label.as_deref().or(result.description.as_deref()),
154        None,
155    ))
156}

158/// Returns one bounded page of development root children.
159fn list<S: ApplicationService>(state: &HttpState<S>, body: &str) -> Result<Value, (u16, Value)> {
160    let value: Value = serde_json::from_str(body).map_err(|error| {
161        (
162            400,
163            json!({"code": "invalid_request", "message": error.to_string()}),
164        )
165    })?;
166    let entity = value
167        .get("entityId")
168        .and_then(Value::as_str)
169        .unwrap_or_default();
170    if entity != state.fixture.root_entity.to_string() {
171        return Ok(json!({"items": [], "nextCursor": null}));
172    }
173    let items = state
174        .fixture
175        .child_entities
176        .iter()
177        .map(|child| {
178            object_description(
179                child.to_string().as_str(),
180                Some(state.fixture.snapshot),
181                None,
182                Some("scriberig.demo.entity".to_owned()),
183                Some("Development entity"),
184                None,
185            )["summary"]
186                .clone()
187        })
188        .collect::<Vec<_>>();
189    Ok(json!({"items": items, "nextCursor": null}))
190}

192/// Resolves the browser snapshot selector without silently changing exact IDs.
193///
194/// # Parameters
195/// - `value`: Browser request containing a snapshot selector.
196/// - `current`: Current development snapshot used for the `current` selector.
197///
198/// # Returns
199/// The application snapshot resolution.
200///
201/// # Errors
202/// Returns a transport status and structured error body for invalid selectors.
203fn snapshot_resolution(
204    value: &Value,
205    current: SnapshotId,
206) -> Result<SnapshotResolution, (u16, Value)> {
207    match value
208        .get("snapshot")
209        .and_then(|item| item.get("kind"))
210        .and_then(Value::as_str)
211    {
212        Some("exact") => {
213            let id = value["snapshot"]["snapshotId"].as_str().unwrap_or_default();
214            SnapshotId::from_str(id)
215                .map(SnapshotResolution::At)
216                .map_err(|_| {
217                    (
218                        400,
219                        json!({"code": "invalid_request", "message": "invalid snapshot id"}),
220                    )
221                })
222        }
223        Some("current") | None => Ok(SnapshotResolution::At(current)),
224        Some(_) => Err((
225            400,
226            json!({"code": "invalid_request", "message": "unsupported snapshot selector"}),
227        )),
228    }
229}

231/// Builds the public semantic object description shape consumed by the web client.
232///
233/// # Parameters
234/// - `entity`: Entity identifier to expose.
235/// - `snapshot`: Snapshot identity, when available.
236/// - `revision`: Revision identity, when available.
237/// - `object_type`: Semantic type, when available.
238/// - `label`: Compact display label, when available.
239/// - `child_count`: Known child count, when available.
240///
241/// # Returns
242/// JSON-compatible semantic summary data.
243fn object_description(
244    entity: &str,
245    snapshot: Option<SnapshotId>,
246    revision: Option<scriberig_core::RevisionId>,
247    object_type: Option<String>,
248    label: Option<&str>,
249    child_count: Option<usize>,
250) -> Value {
251    json!({
252        "summary": {
253            "ref": {"entityId": entity, "snapshotId": snapshot.map(|value| value.to_string()), "revisionId": revision.map(|value| value.to_string())},
254            "typeId": object_type.unwrap_or_else(|| "scriberig.unknown".to_owned()),
255            "label": label,
256            "childCount": child_count
257        }
258    })
259}

261/// Writes a JSON response with a minimal HTTP/1.1 envelope.
262///
263/// # Parameters
264/// - `stream`: Connected client socket.
265/// - `status`: HTTP status code.
266/// - `body`: JSON response body.
267///
268/// # Returns
269/// Unit after the response is written.
270///
271/// # Errors
272/// Returns an I/O error when the response cannot be written.
273fn write_json(stream: &mut TcpStream, status: u16, body: &Value) -> std::io::Result<()> {
274    let body = serde_json::to_vec(&body).unwrap_or_else(|_| b"{\"code\":\"internal\"}".to_vec());
275    let reason = match status {
276        200 => "OK",
277        400 => "Bad Request",
278        404 => "Not Found",
279        _ => "Error",
280    };
281    write!(
282        stream,
283        "HTTP/1.1 {status} {reason}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n",
284        body.len()
285    )?;
286    stream.write_all(&body)
287}

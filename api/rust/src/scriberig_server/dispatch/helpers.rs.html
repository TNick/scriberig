15/// Restricts a session response to one branch and its descendants.
16///
17/// - `response`: Durable session payload before filtering.
18/// - `branch`: Branch identifier to retain in the projection.
19///
20/// The original session response reduced to the selected branch subtree.
21/// # Returns
22/// The operation result.
23pub fn select_branch_subtree(response: SessionReadResponse, branch: &str) -> SessionReadResponse {
24    let mut selected = std::collections::BTreeSet::from([branch.to_owned()]);
25    let mut changed = true;
26    while changed {
27        changed = false;
28        for candidate in &response.branches {
29            if candidate
30                .parent
31                .as_ref()
32                .is_some_and(|parent| selected.contains(parent))
33                && selected.insert(candidate.id.clone())
34            {
35                changed = true;
36            }
37        }
38    }
39
40    let branches = response
41        .branches
42        .into_iter()
43        .filter(|candidate| selected.contains(&candidate.id))
44        .collect();
45
46    SessionReadResponse {
47        branches,
48        ..response
49    }
50}

52/// Converts an application inspection result to protocol references.
53///
54/// - `result`: Application inspection result.
55///
56/// Protocol inspection response with stable references.
57/// # Returns
58/// The operation result.
59pub fn map_inspect(result: scriberig_app::InspectResult) -> InspectEntityResponse {
60    InspectEntityResponse {
61        entity: EntityReference {
62            id: result.entity.to_string(),
63        },
64        revision: RevisionReference {
65            entity: EntityReference {
66                id: result.entity.to_string(),
67            },
68            id: result
69                .revision
70                .map_or_else(String::new, |value| value.to_string()),
71        },
72        snapshot: SnapshotReference {
73            id: result
74                .snapshot
75                .map_or_else(String::new, |value| value.to_string()),
76        },
77        object_type: result.object_type.map(|value| value.to_string()),
78        summary: result.description.or(result.label),
79    }
80}

82/// Converts an application error without exposing restricted payloads.
83///
84/// - `error`: Application error to classify.
85/// - `correlation_id`: Request correlation identity.
86///
87/// Structured protocol error.
88/// # Returns
89/// The operation result.
90pub fn map_error(error: &ApplicationError, correlation_id: &str) -> ProtocolError {
91    let code = match error.category() {
92        "not_found" => ProtocolErrorCode::NotFound,
93        "invalid_request" => ProtocolErrorCode::InvalidRequest,
94        "permission_denied" => ProtocolErrorCode::PermissionDenied,
95        "schema_violation" => ProtocolErrorCode::SchemaViolation,
96        "stale_base_snapshot" => ProtocolErrorCode::StaleBaseSnapshot,
97        "semantic_conflict" => ProtocolErrorCode::SemanticConflict,
98        "validation_failure" => ProtocolErrorCode::ValidationFailure,
99        "unsupported_operation" => ProtocolErrorCode::UnsupportedOperation,
100        "unsupported_capability" => ProtocolErrorCode::UnsupportedCapability,
101        "conflict" => ProtocolErrorCode::Conflict,
102        "ambiguous" => ProtocolErrorCode::Ambiguous,
103        "snapshot_unavailable" => ProtocolErrorCode::SnapshotUnavailable,
104        "operation_failed" => ProtocolErrorCode::OperationFailed,
105        "projection_out_of_sync" => ProtocolErrorCode::ProjectionOutOfSync,
106        "operation_cancelled" => ProtocolErrorCode::OperationCancelled,
107        "cursor_too_old" => ProtocolErrorCode::CursorTooOld,
108        _ => ProtocolErrorCode::Internal,
109    };
110    ProtocolError {
111        code,
112        message: error.to_string(),
113        details: Some(error_details(error)),
114        correlation_id: Some(correlation_id.to_owned()),
115    }
116}

118/// Builds the trusted server context for an exposed command family.
119///
120/// - `kind`: Stable command or capability identifier.
121///
122/// A context granting only the requested public command capability.
123/// # Returns
124/// The operation result.
125pub(super) fn command_context(kind: &str) -> ApplicationContext {
126    let mut capabilities = vec![CapabilityId::new(kind).expect("protocol command capability")];
127    if matches!(kind, "agent_start" | "agent_continue" | "agent_plan") {
128        capabilities.push(CapabilityId::new("session_branch_tree").expect("session capability"));
129    }
130    ApplicationContext {
131        capabilities: CapabilityContext::new(capabilities),
132        ..ApplicationContext::empty()
133    }
134}

136/// Grants the trusted session-selection action for one exact session.
137///
138/// - `session`: Stable session identity from the protocol request.
139///
140/// Application context authorizing session model configuration.
141/// # Returns
142/// The operation result.
143pub(super) fn session_selection_context(session: &str) -> ApplicationContext {
144    ApplicationContext {
145        capabilities: CapabilityContext {
146            granted: vec![
147                CapabilityId::new("session_branch_tree").expect("session read capability"),
148            ],
149            grants: vec![Capability {
150                scope: CapabilityScope::Session(session.to_owned()),
151                actions: [CapabilityAction::ModifyBranch].into_iter().collect(),
152                constraints: CapabilityConstraints {
153                    operation: Some(OperationKind::Agent),
154                    ..CapabilityConstraints::unrestricted()
155                },
156            }],
157        },
158        ..ApplicationContext::empty()
159    }

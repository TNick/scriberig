3use super::server_commands::decode_command;
4use super::server_helpers::{operation_from_record, result_handles};
5use crate::server::ApplicationService as _;
6use scriberig_app::{
7    ApplicationBackend, ApplicationContext, ApplicationError, ApplicationService, OperationId,
8};
9use scriberig_protocol::Operation;

11/// Decodes and executes one command through the shared application boundary.
12///
13/// - `service`: Shared application service.
14/// - `context`: Trusted request context.
15/// - `kind`: Stable command family.
16/// - `payload`: Typed command payload in JSON form.
17///
18/// The accepted or completed shared operation.
19///
20/// Returns a structured application failure.
21pub(super) fn execute<B: ApplicationBackend>(
22    service: &ApplicationService<B>,
23    context: &ApplicationContext,
24    kind: &str,
25    payload: &serde_json::Value,
26) -> Result<Operation, ApplicationError> {
27    if matches!(kind, "agent_start" | "agent_continue") {
28        return submit_agent_command(service, context, kind, payload);
29    }
30    let (operation_kind, request) = decode_command(kind, payload)?;
31    let operation_id = OperationId::new(format!(
32        "command-{kind}-{}",
33        std::time::SystemTime::now()
34            .duration_since(std::time::UNIX_EPOCH)
35            .map_err(|error| ApplicationError::Internal {
36                message: error.to_string(),
37            })?
38            .as_nanos()
39    ))
40    .map_err(|error| ApplicationError::Internal {
41        message: error.to_string(),
42    })?;
43    service
44        .operations()
45        .start(operation_id.clone(), operation_kind)?;
46    service
47        .operations()
48        .transition(&operation_id, scriberig_app::OperationStatus::Running)?;
49    let result = service.execute(context, &request);
50    match result {
51        Ok(result) => {
52            let handles = result_handles(&result);
53            Ok(operation_from_record(
54                service.operations().complete(&operation_id, handles)?,
55            ))
56        }
57        Err(error) => {
58            let _ = service.operations().fail(&operation_id, error.clone());
59            Err(error)
60        }
61    }
62}

64/// Parses and submits a protocol agent command through the shared session API.
65///
66/// - `service`: Shared application service.
67/// - `context`: Trusted request context.
68/// - `kind`: Agent command family.
69/// - `payload`: Agent command payload in JSON form.
70///
71/// The accepted agent operation.
72///
73/// Returns a structured validation, authorization, or operation error.
74fn submit_agent_command<B: ApplicationBackend>(
75    service: &ApplicationService<B>,
76    context: &ApplicationContext,
77    kind: &str,
78    payload: &serde_json::Value,
79) -> Result<Operation, ApplicationError> {
80    let session = payload
81        .get("session")
82        .and_then(serde_json::Value::as_str)
83        .ok_or_else(|| ApplicationError::InvalidRequest {
84            operation: kind.to_owned(),
85            message: "session is required".to_owned(),
86        })?
87        .parse::<scriberig_core::EntityId>()
88        .map_err(|error| ApplicationError::InvalidRequest {
89            operation: kind.to_owned(),
90            message: format!("invalid session: {error}"),
91        })?;
92    let branch = payload
93        .get("branch")
94        .and_then(serde_json::Value::as_str)
95        .map(str::parse::<scriberig_core::EntityId>)
96        .transpose()
97        .map_err(|error| ApplicationError::InvalidRequest {
98            operation: kind.to_owned(),
99            message: format!("invalid branch: {error}"),
100        })?;
101    let prompt = payload
102        .get("prompt")
103        .and_then(serde_json::Value::as_str)
104        .filter(|value| !value.trim().is_empty())
105        .ok_or_else(|| ApplicationError::InvalidRequest {
106            operation: kind.to_owned(),
107            message: "prompt is required".to_owned(),
108        })?;
109    let selection = match payload
110        .get("selection")
111        .filter(|value| !value.is_null())
112        .cloned()
113    {
114        None => None,
115        Some(value) => {
116            let selection = serde_json::from_value::<scriberig_protocol::AgentSelection>(value)
117                .map_err(|error| ApplicationError::InvalidRequest {
118                    operation: kind.to_owned(),
119                    message: format!("invalid semantic selection: {error}"),
120                })?;
121            Some(scriberig_app::AgentSelection {
122                project: selection
123                    .project_id
124                    .as_deref()
125                    .map(str::parse)
126                    .transpose()
127                    .map_err(|error| ApplicationError::InvalidRequest {
128                        operation: kind.to_owned(),
129                        message: format!("invalid selected project: {error}"),
130                    })?,
131                root: selection
132                    .root
133                    .as_ref()
134                    .map(|root| root.id.parse())
135                    .transpose()
136                    .map_err(|error| ApplicationError::InvalidRequest {
137                        operation: kind.to_owned(),
138                        message: format!("invalid selected root: {error}"),
139                    })?
140                    .unwrap_or(selection.entity.id.parse().map_err(|error| {
141                        ApplicationError::InvalidRequest {
142                            operation: kind.to_owned(),
143                            message: format!("invalid selected entity: {error}"),
144                        }
145                    })?),
146                entity: selection.entity.id.parse().map_err(|error| {
147                    ApplicationError::InvalidRequest {
148                        operation: kind.to_owned(),
149                        message: format!("invalid selected entity: {error}"),
150                    }
151                })?,
152                snapshot: selection.snapshot.id.parse().map_err(|error| {
153                    ApplicationError::InvalidRequest {
154                        operation: kind.to_owned(),
155                        message: format!("invalid selected snapshot: {error}"),
156                    }
157                })?,
158            })
159        }
160    };
161    let operation = scriberig_app::SessionBackend::submit_agent_command(
162        service,
163        context,
164        &scriberig_app::AgentCommandRequest {
165            session,
166            branch,
167            anchor: None,
168            prompt: prompt.to_owned(),
169            selection,
170        },
171    )?;
172    service.operation_status(&operation.to_string())
173}

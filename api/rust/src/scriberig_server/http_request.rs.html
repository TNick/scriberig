1//! Provides bounded HTTP header parsing shared by the local server adapter.
2
3use std::sync::atomic::{AtomicU64, Ordering};
4
5static REQUEST_SEQUENCE: AtomicU64 = AtomicU64::new(1);
6
7/// Returns a request header value without accepting malformed names.
8pub(super) fn header_value<'a>(head: &'a str, wanted: &str) -> Option<&'a str> {
9    head.lines().skip(1).find_map(|line| {
10        let (name, value) = line.split_once(':')?;
11        name.eq_ignore_ascii_case(wanted).then_some(value.trim())
12    })
13}
14
15/// Allows only the configured exact origin, never a wildcard.
16///
17/// - `super`: Internal server-module visibility.
18/// - `head`: HTTP request headers.
19/// - `configured`: Optional configured exact origin.
20///
21/// The allowed origin when the request matches configuration.
22/// # Returns
23/// The operation result.
24pub(super) fn allowed_origin(head: &str, configured: Option<&str>) -> Option<String> {
25    let origin = header_value(head, "origin")?;
26    configured
27        .filter(|value| !value.is_empty() && *value != "*")
28        .filter(|value| *value == origin)
29        .map(ToOwned::to_owned)
30}

32/// Returns a bounded request correlation identifier from headers or local state.
33///
34/// - `super`: Internal server-module visibility.
35/// - `head`: HTTP request headers.
36///
37/// A bounded request correlation identifier.
38/// # Returns
39/// The operation result.
40pub(super) fn request_id(head: &str) -> String {
41    let supplied = head.lines().skip(1).find_map(|line| {
42        let (name, value) = line.split_once(':')?;
43        (name.eq_ignore_ascii_case("x-request-id")).then_some(value.trim())
44    });
45    let value = supplied
46        .filter(|value| !value.is_empty() && value.len() <= 128)
47        .map(ToOwned::to_owned);
48    value.unwrap_or_else(|| format!("http-{}", REQUEST_SEQUENCE.fetch_add(1, Ordering::Relaxed)))
49}

14/// Projects ordered execution references without exposing raw tool payloads.
15///
16/// # Parameters
17/// - `session`: Stable identity of the containing session.
18/// - `branch`: Durable branch containing execution records.
19///
20/// # Returns
21/// Safe activity items in the branch's recorded causal order.
22pub(super) fn branch_execution_activity(
23    session: &str,
24    branch: &Branch,
25) -> Vec<SessionActivityItem> {
26    branch
27        .activity
28        .iter()
29        .enumerate()
30        .filter_map(|(sequence, activity)| project_item(session, branch, sequence, activity))
31        .collect()
32}

34/// Contains one safe projection of durable branch activity.
35///
36/// Fields retain only display-safe summaries and stable causal references.
37struct ActivityProjection {
38    /// Durable activity identity.
39    id: EntityId,
40    /// Optional parent activity identity.
41    parent: Option<String>,
42    /// Provider-neutral activity kind.
43    kind: Kind,
44    /// Current lifecycle state.
45    state: State,
46    /// Bounded display summary.
47    summary: String,
48    /// Selected model identifier, when this is a model request.
49    model: Option<String>,
50    /// Selected thinking level, when this is a model request.
51    thinking_level: Option<String>,
52    /// Public tool name, when this is a tool invocation.
53    tool_name: Option<String>,
54}
55
56/// Projects one ordered activity into its public protocol representation.
57///
58/// # Parameters
59/// - `session`: Stable session identity.
60/// - `branch`: Durable branch containing the activity.
61/// - `sequence`: Position in the durable causal activity list.
62/// - `activity`: Durable activity reference to project.
63///
64/// # Returns
65/// Safe protocol activity with stable parent and durable references.
66fn project_item(
67    session: &str,
68    branch: &Branch,
69    sequence: usize,
70    activity: &Activity,
71) -> Option<SessionActivityItem> {
72    let projection = activity_projection(activity, branch)?;
73    let (tool_arguments, tool_result_preview) = activity_tool_details(activity, branch);
74    let id = projected_activity_id(projection.kind, projection.parent.is_none(), projection.id);
75    let parent = match (projection.kind, projection.parent) {
76        (Kind::AssistantOutput, None) => Some(format!("{}:human", projection.id)),
77        (_, parent) => parent,
78    };
79    Some(SessionActivityItem {
80        id,
81        session: session.to_owned(),
82        sequence: u64::try_from(sequence).unwrap_or(u64::MAX),
83        branch: branch.id.to_string(),
84        parent,
85        kind: projection.kind,
86        state: projection.state,
87        summary: bounded_summary(&projection.summary),
88        durable_ref: Some(projection.id.to_string()),
89        model: projection.model,
90        thinking_level: projection.thinking_level,
91        tool_name: projection.tool_name,
92        tool_arguments,
93        tool_result_preview,
94    })
95}

97/// Converts one durable activity reference to display-safe activity fields.
98///
99/// # Parameters
100/// - `activity`: Durable activity reference.
101/// - `branch`: Branch containing referenced records.
102///
103/// # Returns
104/// Safe activity fields when the referenced durable record exists.
105fn activity_projection(activity: &Activity, branch: &Branch) -> Option<ActivityProjection> {
106    match activity {
107        Activity::ModelInvocation(id) => project_model_invocation(*id, branch),
108        Activity::HumanInput(id) => project_human_input(*id, branch),
109        Activity::ModelOutput(id) | Activity::AgentOutput(id) => {
110            project_assistant_output(*id, branch)
111        }
112        Activity::ToolInvocation(id) => project_tool_invocation(*id, branch),
113        Activity::ToolResult(id) => project_tool_result(*id, branch),
114        Activity::ChildBranch(id) => Some(child_branch_projection(*id, None)),
115        Activity::ChildBranchSpawned {
116            branch,
117            parent_tool_invocation,
118        } => Some(child_branch_projection(
119            *branch,
120            Some(*parent_tool_invocation),
121        )),
122        Activity::BranchResult(id) => project_branch_result(*id, branch),
123    }
124}

126/// Projects one model invocation and its selected model metadata.
127///
128/// # Parameters
129/// - `id`: Durable model-invocation identity.
130/// - `branch`: Branch containing model invocation records.
131///
132/// # Returns
133/// Safe model invocation fields when the record exists.
134fn project_model_invocation(id: EntityId, branch: &Branch) -> Option<ActivityProjection> {
135    let invocation = branch.model_invocations.iter().find(|item| item.id == id)?;
136    let model = invocation.model.clone();
137    let summary = model.as_ref().map_or_else(
138        || "Model request".to_owned(),
139        |value| format!("Model request: {value}"),
140    );
141    Some(ActivityProjection {
142        id,
143        parent: None,
144        kind: Kind::ModelInvocation,
145        state: model_invocation_state(branch.status, invocation.output.is_some()),
146        summary,
147        model,
148        thinking_level: invocation.thinking_level.clone(),
149        tool_name: None,
150    })
151}

153/// Resolves model invocation lifecycle state from branch and output records.
154///
155/// # Parameters
156/// - `branch`: Current lifecycle state of the branch.
157/// - `has_output`: Whether an output record already exists.
158///
159/// # Returns
160/// Public model invocation lifecycle state.
161fn model_invocation_state(branch: BranchStatus, has_output: bool) -> State {
162    match branch {
163        BranchStatus::Completed => State::Completed,
164        BranchStatus::Active if has_output => State::Completed,
165        BranchStatus::Active => State::Running,
166        BranchStatus::Abandoned => State::Failed,
167    }
168}

170/// Projects one human-input marker without exposing prompt content.
171///
172/// # Parameters
173/// - `id`: Durable turn identity.
174/// - `branch`: Branch containing the input turn.
175///
176/// # Returns
177/// Safe human-input marker when the turn exists.
178fn project_human_input(id: EntityId, branch: &Branch) -> Option<ActivityProjection> {
179    branch
180        .turns
181        .iter()
182        .any(|turn| turn.id == id)
183        .then(|| ActivityProjection {
184            id,
185            parent: None,
186            kind: Kind::HumanInput,
187            state: State::Completed,
188            summary: "Human input".to_owned(),
189            model: None,
190            thinking_level: None,
191            tool_name: None,
192        })
193}

195/// Projects a model or agent output summary without the raw input record.
196///
197/// # Parameters
198/// - `id`: Durable output or turn identity.
199/// - `branch`: Branch containing output records.
200///
201/// # Returns
202/// Safe assistant-output fields when the record exists.
203fn project_assistant_output(id: EntityId, branch: &Branch) -> Option<ActivityProjection> {
204    let (parent, summary) =
205        if let Some(output) = branch.model_outputs.iter().find(|item| item.id == id) {
206            (
207                Some(output.invocation.to_string()),
208                output.text.as_ref().map_or_else(
209                    || "Assistant output".to_owned(),
210                    |value| bounded_summary(value),
211                ),
212            )
213        } else {
214            let turn = branch.turns.iter().find(|turn| turn.id == id)?;
215            (None, bounded_summary(&turn.agent.as_ref()?.text))
216        };
217    Some(ActivityProjection {
218        id,
219        parent,
220        kind: Kind::AssistantOutput,
221        state: State::Completed,
222        summary,
223        model: None,
224        thinking_level: None,
225        tool_name: None,
226    })
227}

229/// Projects a tool invocation using its safe presentation label.
230///
231/// # Parameters
232/// - `id`: Durable tool-invocation identity.
233/// - `branch`: Branch containing invocation and result records.
234///
235/// # Returns
236/// Safe tool invocation fields when the record exists.
237fn project_tool_invocation(id: EntityId, branch: &Branch) -> Option<ActivityProjection> {
238    let invocation = branch.tool_invocations.iter().find(|item| item.id == id)?;
239    let has_result = branch
240        .tool_results
241        .iter()
242        .any(|result| result.invocation == invocation.id);
243    Some(ActivityProjection {
244        id,
245        parent: invocation
246            .parent_model_invocation
247            .map(|parent| parent.to_string()),
248        kind: Kind::ToolInvocation,
249        state: if has_result {
250            State::Completed
251        } else {
252            State::Running
253        },
254        summary: invocation
255            .presentation_label
256            .clone()
257            .unwrap_or_else(|| format!("Tool: {}", invocation.name)),
258        model: None,
259        thinking_level: None,
260        tool_name: Some(invocation.name.clone()),
261    })
262}

264/// Projects a tool result without serializing raw output bytes.
265///
266/// # Parameters
267/// - `id`: Durable tool-result identity.
268/// - `branch`: Branch containing the result record.
269///
270/// # Returns
271/// Safe bounded tool-result fields when the record exists.
272fn project_tool_result(id: EntityId, branch: &Branch) -> Option<ActivityProjection> {
273    let result = branch.tool_results.iter().find(|item| item.id == id)?;
274    Some(ActivityProjection {
275        id,
276        parent: Some(result.invocation.to_string()),
277        kind: Kind::ToolResult,
278        state: if result.failed {
279            State::Failed
280        } else {
281            State::Completed
282        },
283        summary: tool_result_summary(result, result.failed),
284        model: None,
285        thinking_level: None,
286        tool_name: None,
287    })
288}

290/// Projects a child-branch reference and optional spawning tool parent.
291///
292/// # Parameters
293/// - `id`: Child branch identity.
294/// - `parent`: Optional spawning tool invocation identity.
295///
296/// # Returns
297/// Safe child-branch fields.
298fn child_branch_projection(id: EntityId, parent: Option<EntityId>) -> ActivityProjection {
299    ActivityProjection {
300        id,
301        parent: parent.map(|parent| parent.to_string()),
302        kind: Kind::BranchStarted,
303        state: State::Running,
304        summary: "Child branch".to_owned(),
305        model: None,
306        thinking_level: None,
307        tool_name: None,
308    }
309}

311/// Projects a branch result and its terminal status.
312///
313/// # Parameters
314/// - `id`: Durable branch-result identity.
315/// - `branch`: Branch containing the result.
316///
317/// # Returns
318/// Safe branch-result fields when the result exists.
319fn project_branch_result(id: EntityId, branch: &Branch) -> Option<ActivityProjection> {
320    let result = branch.result.as_ref().filter(|result| result.id == id)?;
321    Some(ActivityProjection {
322        id,
323        parent: None,
324        kind: Kind::BranchCompleted,
325        state: match result.status {
326            BranchStatus::Completed => State::Completed,
327            BranchStatus::Active => State::Running,
328            BranchStatus::Abandoned => State::Failed,
329        },
330        summary: result
331            .summary
332            .as_deref()
333            .map_or_else(|| "Branch result".to_owned(), bounded_summary),
334        model: None,
335        thinking_level: None,
336        tool_name: None,
337    })
338}

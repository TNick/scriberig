1//! Provides deterministic semantic data for the local development server.
2
3use scriberig_core::{EncodingVersion, EntityId, ObjectType, RootName, SnapshotId};
4use scriberig_store::{Revision, RootStore, Snapshot, SnapshotChange, SnapshotStore};
5use scriberig_store_sqlite::SqliteStore;
6
7/// Identifies the entities exposed by the local development fixture.
8#[derive(Clone, Debug)]
9pub struct DevelopmentFixture {
10    /// Stable root entity shown when the browser starts.
11    pub root_entity: EntityId,
12    /// Stable children available through the root collection.
13    pub child_entities: Vec<EntityId>,
14    /// Snapshot containing the fixture entities.
15    pub snapshot: SnapshotId,
16}
17
18/// Creates or reuses the deterministic browser development fixture.
19///
20/// # Parameters
21/// - `store`: SQLite store receiving the fixture when it is absent.
22///
23/// # Returns
24/// Fixture identities used by the HTTP bootstrap response.
25///
26/// # Errors
27/// Returns a storage error when the fixture cannot be created or inspected.
28pub fn ensure_fixture(store: &SqliteStore) -> Result<DevelopmentFixture, String> {
29    let root = RootName::new("main").map_err(|error| error.to_string())?;
30    if let Some(snapshot) = store.read_root(&root) {
31        let entities = store.iter(snapshot).map_err(|error| format!("{error:?}"))?;
32        if let Some((root_entity, _)) = entities.first() {
33            return Ok(DevelopmentFixture {
34                root_entity: *root_entity,
35                child_entities: entities.iter().skip(1).map(|(entity, _)| *entity).collect(),
36                snapshot,
37            });
38        }
39    }
40
41    let entities = [
42        EntityId::from_bytes([1; 16]),
43        EntityId::from_bytes([2; 16]),
44        EntityId::from_bytes([3; 16]),
45    ];
46    let revisions = entities
47        .iter()
48        .enumerate()
49        .map(|(index, _entity)| {
50            let object_type =
51                ObjectType::new("scriberig.demo.entity").map_err(|error| error.to_string())?;
52            let payload = format!("development-{}", index + 1).into_bytes();
53            Revision::new(EncodingVersion::V1, object_type, payload)
54                .map_err(|error| format!("{error:?}"))
55        })
56        .collect::<Result<Vec<_>, String>>()?;
57    let base = Snapshot::empty().id();
58    let changes = revisions
59        .iter()
60        .zip(entities)
61        .map(|(revision, entity)| SnapshotChange::Set(entity, revision.id()))
62        .collect::<Vec<_>>();
63    let snapshot = store
64        .apply_revisions_and_promote(root, None, base, &revisions, &changes)
65        .map_err(|error| format!("{error:?}"))?;
66    Ok(DevelopmentFixture {
67        root_entity: entities[0],
68        child_entities: entities[1..].to_vec(),
69        snapshot,
70    })
71}

73#[cfg(test)]
74mod tests {
75    use super::ensure_fixture;
76    use scriberig_core::RootName;
77    use scriberig_store::RootStore;
78    use scriberig_store_sqlite::SqliteStore;
79
80    /// Verifies an empty database receives a usable semantic starting point.
81    #[test]
82    fn empty_store_gets_development_fixture() {
83        let store = SqliteStore::open_in_memory().expect("store");
84
85        let fixture = ensure_fixture(&store).expect("fixture");
86
87        assert_eq!(fixture.child_entities.len(), 2);
88        let root = RootName::new("main").expect("root");
89        assert_eq!(store.read_root(&root), Some(fixture.snapshot));
90    }
91
92    /// Verifies bootstrapping is idempotent for an already initialized store.
93    #[test]
94    fn existing_fixture_is_reused() {
95        let store = SqliteStore::open_in_memory().expect("store");
96
97        let first = ensure_fixture(&store).expect("first fixture");
98        let second = ensure_fixture(&store).expect("second fixture");
99
100        assert_eq!(first.root_entity, second.root_entity);
101        assert_eq!(first.child_entities, second.child_entities);
102        assert_eq!(first.snapshot, second.snapshot);
103    }
104}

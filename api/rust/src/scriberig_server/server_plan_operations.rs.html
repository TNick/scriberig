11/// Applies one typed plan request using application-owned validation.
12///
13/// - `service`: Shared application service.
14/// - `context`: Trusted capability context.
15/// - `request`: Typed plan operation request.
16///
17/// Safe projection of the authoritative plan state.
18///
19/// Returns authorization, validation, revision conflict, or persistence errors.
20#[expect(
21    clippy::too_many_lines,
22    reason = "This dispatcher keeps the protocol request variants visibly mapped to application operations."
23)]
24pub(super) fn apply<B: ApplicationBackend>(
25    service: &ApplicationService<B>,
26    context: &ApplicationContext,
27    request: &scriberig_protocol::Request,
28) -> Result<scriberig_protocol::SessionPlan, ApplicationError> {
29    let plan = match request {
30        scriberig_protocol::Request::PlanCreate(request) => service.create_plan(
31            context,
32            &scriberig_app::PlanCreateRequest {
33                session: parse_entity(&request.session.id)?,
34                anchor: anchor(&request.anchor)?,
35                goal: request.goal.clone(),
36                nodes: request.nodes.iter().map(node_draft).collect(),
37                dependencies: request.dependencies.clone(),
38            },
39        )?,
40        scriberig_protocol::Request::PlanLifecycle(request) => {
41            let app_request = scriberig_app::PlanLifecycleRequest {
42                session: parse_entity(&request.session.id)?,
43                plan: parse_entity(&request.plan)?,
44                expected_revision: request.expected_revision,
45            };
46            request_action(service, context, &app_request, request.action)?
47        }
48        scriberig_protocol::Request::PlanAssignment(request) => service.set_plan_node_assignment(
49            context,
50            &scriberig_app::PlanAssignmentRequest {
51                session: parse_entity(&request.session.id)?,
52                plan: parse_entity(&request.plan)?,
53                node: parse_entity(&request.node)?,
54                expected_revision: request.expected_revision,
55                assignment: request.assignment.clone().map(assignment),
56            },
57        )?,
58        scriberig_protocol::Request::PlanReplan(request) => service.replan(
59            context,
60            &scriberig_app::PlanReplanRequest {
61                session: parse_entity(&request.session.id)?,
62                plan: parse_entity(&request.plan)?,
63                expected_revision: request.expected_revision,
64                additions: request.additions.iter().map(node_draft).collect(),
65                dependencies: request
66                    .dependencies
67                    .iter()
68                    .map(|(prerequisite, dependent)| {
69                        Ok((node_reference(prerequisite)?, node_reference(dependent)?))
70                    })
71                    .collect::<Result<Vec<_>, ApplicationError>>()?,
72                removed_dependencies: request
73                    .removed_dependencies
74                    .iter()
75                    .map(|(prerequisite, dependent)| {
76                        Ok((parse_entity(prerequisite)?, parse_entity(dependent)?))
77                    })
78                    .collect::<Result<Vec<_>, ApplicationError>>()?,
79                superseded: request
80                    .superseded
81                    .iter()
82                    .map(|(node, key)| Ok((parse_entity(node)?, key.clone())))
83                    .collect::<Result<Vec<_>, ApplicationError>>()?,
84                assignment_changes: request
85                    .assignment_changes
86                    .iter()
87                    .map(|(node, value)| Ok((parse_entity(node)?, value.clone().map(assignment))))
88                    .collect::<Result<Vec<_>, ApplicationError>>()?,
89                preserved_terminal_nodes: request
90                    .preserved_terminal_nodes
91                    .iter()
92                    .map(|node| parse_entity(node))
93                    .collect::<Result<Vec<_>, _>>()?,
94                reason: request.reason.clone(),
95            },
96        )?,
97        scriberig_protocol::Request::PlanRetry(request) => service.retry_plan_node(
98            context,
99            &scriberig_app::PlanRetryRequest {
100                session: parse_entity(&request.session.id)?,
101                plan: parse_entity(&request.plan)?,
102                node: parse_entity(&request.node)?,
103                expected_revision: request.expected_revision,
104            },
105        )?,
106        scriberig_protocol::Request::PlanNextStep(request) => service.add_plan_next_step(
107            context,
108            &scriberig_app::PlanNextStepRequest {
109                session: parse_entity(&request.session.id)?,
110                plan: parse_entity(&request.plan)?,
111                expected_revision: request.expected_revision,
112                title: request.title.clone(),
113                purpose: request.purpose.clone(),
114                expected_result: request.expected_result.as_ref().map(|expected| {
115                    scriberig_session::ExpectedResult {
116                        kind: expected.kind.clone(),
117                        requires_handle: expected.requires_handle,
118                    }
119                }),
120                semantic_scope: request.semantic_scope.clone(),
121                assignment: request.assignment.clone().map(assignment),
122            },
123        )?,
124        scriberig_protocol::Request::PlanDecompose(request) => service.decompose_plan_node(
125            context,
126            &scriberig_app::PlanDecomposeRequest {
127                session: parse_entity(&request.session.id)?,
128                plan: parse_entity(&request.plan)?,
129                parent: parse_entity(&request.parent)?,
130                expected_revision: request.expected_revision,
131                children: request
132                    .children
133                    .iter()
134                    .map(|child| scriberig_app::PlanChildDraft {
135                        title: child.title.clone(),
136                        purpose: child.purpose.clone(),
137                        expected_result: child.expected_result.as_ref().map(|expected| {
138                            scriberig_session::ExpectedResult {
139                                kind: expected.kind.clone(),
140                                requires_handle: expected.requires_handle,
141                            }
142                        }),
143                        semantic_scope: child.semantic_scope.clone(),
144                        assignment: child.assignment.clone().map(assignment),
145                    })
146                    .collect(),
147                required: request.required,
148            },
149        )?,
150        scriberig_protocol::Request::PlanSemanticMutation(request) => service
151            .mutate_plan_semantics(
152                context,
153                &scriberig_app::PlanSemanticMutationRequest {
154                    session: parse_entity(&request.session.id)?,
155                    plan: parse_entity(&request.plan)?,
156                    expected_revision: request.expected_revision,
157                    mutation: semantic_mutation(request.mutation.clone())?,
158                },
159            )?,
160        _ => {
161            return Err(ApplicationError::InvalidRequest {
162                operation: "plan".to_owned(),
163                message: "request is not a plan operation".to_owned(),
164            });
165        }
166    };
167    let session_id = match request {
168        scriberig_protocol::Request::PlanCreate(request) => parse_entity(&request.session.id)?,
169        scriberig_protocol::Request::PlanLifecycle(request) => parse_entity(&request.session.id)?,
170        scriberig_protocol::Request::PlanAssignment(request) => parse_entity(&request.session.id)?,
171        scriberig_protocol::Request::PlanReplan(request) => parse_entity(&request.session.id)?,
172        scriberig_protocol::Request::PlanRetry(request) => parse_entity(&request.session.id)?,
173        scriberig_protocol::Request::PlanNextStep(request) => parse_entity(&request.session.id)?,
174        scriberig_protocol::Request::PlanDecompose(request) => parse_entity(&request.session.id)?,
175        scriberig_protocol::Request::PlanSemanticMutation(request) => {
176            parse_entity(&request.session.id)?
177        }
178        _ => {
179            return Err(ApplicationError::InvalidRequest {
180                operation: "plan".to_owned(),
181                message: "request is not a plan operation".to_owned(),
182            });
183        }
184    };
185    let session = service.read_session(
186        context,
187        &scriberig_app::SessionRequest {
188            session: session_id,
189        },
190    )?;
191    let results = session
192        .branches
193        .iter()
194        .filter_map(|branch| branch.result.clone())
195        .collect::<Vec<_>>();
196    super::server_helpers::session_plan::project(plan, Some(&results))
197}

199/// Applies the requested plan lifecycle operation.
200///
201/// - `service`: Shared application service.
202/// - `context`: Trusted caller context.
203/// - `request`: Stable plan and expected revision.
204/// - `action`: Typed lifecycle operation.
205///
206/// The updated durable plan.
207///
208/// Returns conflict, invalid lifecycle, or persistence failures.
209fn request_action<B: ApplicationBackend>(
210    service: &ApplicationService<B>,
211    context: &ApplicationContext,
212    request: &scriberig_app::PlanLifecycleRequest,
213    action: scriberig_protocol::SessionPlanAction,
214) -> Result<scriberig_session::Plan, ApplicationError> {
215    match action {
216        scriberig_protocol::SessionPlanAction::Start => service.start_plan(context, request),
217        scriberig_protocol::SessionPlanAction::Pause => service.pause_plan(context, request),
218        scriberig_protocol::SessionPlanAction::Resume => service.resume_plan(context, request),
219        scriberig_protocol::SessionPlanAction::Stop => service.stop_plan(context, request),
220        scriberig_protocol::SessionPlanAction::Complete => service.complete_plan(context, request),
221    }
222}

224/// Converts one protocol node draft to application input.
225///
226/// - `value`: Provider-neutral structured proposal node.
227///
228/// Untrusted draft validated by the application service.
229/// # Returns
230/// The operation result.
231fn node_draft(value: &scriberig_protocol::SessionPlanNodeDraft) -> scriberig_app::PlanNodeDraft {
232    scriberig_app::PlanNodeDraft {
233        key: value.key.clone(),
234        parent_index: value.parent_index,
235        title: value.title.clone(),
236        purpose: value.purpose.clone(),
237        expected_result: value.expected_result.as_ref().map(|expected| {
238            scriberig_session::ExpectedResult {
239                kind: expected.kind.clone(),
240                requires_handle: expected.requires_handle,
241            }
242        }),
243        semantic_scope: value.semantic_scope.clone(),
244        assignment: value.assignment.clone().map(assignment),
245    }
246}

265/// Converts one protocol node reference to application identity.
266///
267/// - `value`: Existing node or request-local proposed key.
268///
269/// Shared application node reference.
270///
271/// Returns invalid-request when an existing identity is malformed.
272/// # Returns
273/// The operation result.
274/// # Errors
275/// Returns an error when the operation fails.
276fn node_reference(
277    value: &scriberig_protocol::SessionPlanNodeReference,
278) -> Result<scriberig_app::PlanNodeReference, ApplicationError> {
279    Ok(match value {
280        scriberig_protocol::SessionPlanNodeReference::Existing { id } => {
281            scriberig_app::PlanNodeReference::Existing(parse_entity(id)?)
282        }
283        scriberig_protocol::SessionPlanNodeReference::Proposed { key } => {
284            scriberig_app::PlanNodeReference::Proposed(key.clone())
285        }
286    })
287}

289/// Converts a stable protocol identity to a shared entity id.
290///
291/// - `value`: Serialized stable identity.
292///
293/// Typed entity identity.
294///
295/// Returns invalid-request when the identity is malformed.
296/// # Returns
297/// The operation result.
298/// # Errors
299/// Returns an error when the operation fails.
300fn parse_entity(value: &str) -> Result<scriberig_core::EntityId, ApplicationError> {
301    value
302        .parse()
303        .map_err(|error| ApplicationError::InvalidRequest {
304            operation: "plan".to_owned(),
305            message: format!("invalid plan identity: {error}"),
306        })
307}

309/// Converts a protocol continuation anchor to shared session identity.
310///
311/// - `value`: Exact provider-neutral continuation anchor.
312///
313/// Session-domain continuation anchor.
314///
315/// Returns invalid-request when an anchor identity is malformed.
316/// # Returns
317/// The operation result.
318/// # Errors
319/// Returns an error when the operation fails.
320fn anchor(
321    value: &scriberig_protocol::SessionContinuationAnchor,
322) -> Result<scriberig_session::ContinuationAnchor, ApplicationError> {
323    Ok(match value {
324        scriberig_protocol::SessionContinuationAnchor::BranchHead { branch } => {
325            scriberig_session::ContinuationAnchor::BranchHead {
326                branch: parse_entity(branch)?,
327            }
328        }
329        scriberig_protocol::SessionContinuationAnchor::AssistantOutput { branch, turn } => {
330            scriberig_session::ContinuationAnchor::AssistantOutput {
331                branch: parse_entity(branch)?,
332                turn: parse_entity(turn)?,
333            }
334        }
335        scriberig_protocol::SessionContinuationAnchor::HumanInput { branch, turn } => {
336            scriberig_session::ContinuationAnchor::HumanInput {
337                branch: parse_entity(branch)?,
338                turn: parse_entity(turn)?,
339            }
340        }
341        scriberig_protocol::SessionContinuationAnchor::BranchResult { branch, result } => {
342            scriberig_session::ContinuationAnchor::BranchResult {
343                branch: parse_entity(branch)?,
344                result: parse_entity(result)?,
345            }
346        }
347    })

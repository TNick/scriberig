7/// Builds the public semantic object description shape consumed by the web client.
8///
9/// - `entity`: Entity identifier to expose.
10/// - `snapshot`: Snapshot identity, when available.
11/// - `revision`: Revision identity, when available.
12/// - `object_type`: Semantic type, when available.
13/// - `label`: Compact display label, when available.
14/// - `child_count`: Known child count, when available.
15/// - `super`: Parent-module visibility marker.
16///
17/// JSON-compatible semantic summary data.
18/// # Returns
19/// The operation result.
20pub(super) fn object_description(
21    entity: &str,
22    snapshot: Option<SnapshotId>,
23    revision: Option<scriberig_core::RevisionId>,
24    object_type: Option<String>,
25    label: Option<&str>,
26    child_count: Option<usize>,
27) -> Value {
28    json!({
29        "summary": {
30            "ref": {"entityId": entity, "snapshotId": snapshot.map(|value| value.to_string()), "revisionId": revision.map(|value| value.to_string())},
31            "typeId": object_type.unwrap_or_else(|| "scriberig.unknown".to_owned()),
32            "label": label,
33            "childCount": child_count
34        }
35    })
36}

38/// Writes a JSON response with a minimal HTTP/1.1 envelope.
39///
40/// - `stream`: Connected client socket.
41/// - `status`: HTTP status code.
42/// - `body`: JSON response body.
43/// - `origin`: Optional allowed CORS origin.
44/// - `super`: Parent-module visibility marker.
45///
46/// Unit after the response is written.
47///
48/// Returns an I/O error when the response cannot be written.
49/// # Returns
50/// The operation result.
51/// # Errors
52/// Returns an error when the operation fails.
53pub(super) fn write_json(
54    stream: &mut TcpStream,
55    status: u16,
56    body: &Value,
57    origin: Option<&str>,
58) -> std::io::Result<()> {
59    let body = serde_json::to_vec(&body).unwrap_or_else(|_| b"{\"code\":\"internal\"}".to_vec());
60    let reason = match status {
61        200 => "OK",
62        400 => "Bad Request",
63        404 => "Not Found",
64        _ => "Error",
65    };
66    let cors = origin.map_or_else(String::new, |value| {
67        format!("Access-Control-Allow-Origin: {value}\r\nVary: Origin\r\n")
68    });
69    write!(
70        stream,
71        "HTTP/1.1 {status} {reason}\r\nContent-Type: application/json\r\n{cors}Content-Length: {}\r\nConnection: close\r\n\r\n",
72        body.len()
73    )?;
74    stream.write_all(&body)
75}

77/// Resolves the browser snapshot selector without silently changing exact IDs.
78///
79/// - `value`: Browser request containing a snapshot selector.
80/// - `current`: Current development snapshot used for the `current` selector.
81///
82/// The application snapshot resolution.
83///
84/// Returns a transport status and structured error body for invalid selectors.
85/// # Returns
86/// The operation result.
87/// # Errors
88/// Returns an error when the selector is invalid.
89pub(super) fn snapshot_resolution(
90    value: &Value,
91    current: SnapshotId,
92) -> Result<SnapshotResolution, (u16, Value)> {
93    match value
94        .get("snapshot")
95        .and_then(|item| item.get("kind"))
96        .and_then(Value::as_str)
97    {
98        Some("exact") => {
99            let id = value["snapshot"]["snapshotId"].as_str().unwrap_or_default();
100            SnapshotId::from_str(id)
101                .map(SnapshotResolution::At)
102                .map_err(|_| {
103                    (
104                        400,
105                        json!({"code": "invalid_request", "message": "invalid snapshot id"}),
106                    )
107                })
108        }
109        Some("current") | None => Ok(SnapshotResolution::At(current)),
110        Some(_) => Err((
111            400,
112            json!({"code": "invalid_request", "message": "unsupported snapshot selector"}),
113        )),
114    }

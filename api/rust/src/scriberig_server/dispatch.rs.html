21/// Server-side protocol dispatcher for the initial vertical slice.
22pub struct Dispatcher<S> {
23    service: S,
24}

26impl<S> Dispatcher<S> {
27    /// Creates a dispatcher around an application service.
28    ///
29    /// # Parameters
30    /// - `service`: Application implementation receiving semantic requests.
31    ///
32    /// # Returns
33    /// A protocol dispatcher.
34    #[must_use]
35    pub fn new(service: S) -> Self {
36        Self { service }
37    }
38}

40impl<S: ApplicationService> Dispatcher<S> {
41    /// Dispatches one correlated protocol envelope.
42    ///
43    /// # Parameters
44    /// - `request`: Correlated protocol request.
45    ///
46    /// # Returns
47    /// Correlated typed response or structured error.
48    #[must_use]
49    pub fn dispatch(&self, request: RequestEnvelope) -> ResponseEnvelope {
50        let correlation_id = request.correlation_id.clone();
51        let response = if request.version.major == ProtocolVersion::default().major {
52            self.dispatch_request(request.request, &correlation_id)
53        } else {
54            Response::Error(ProtocolError {
55                code: ProtocolErrorCode::UnsupportedVersion,
56                message: "unsupported protocol major version".to_owned(),
57                details: Some(serde_json::json!({"major": request.version.major})),
58                correlation_id: Some(correlation_id.clone()),
59            })
60        };
61        ResponseEnvelope {
62            version: ProtocolVersion::default(),
63            correlation_id,
64            response,
65        }
66    }
67
68    /// Dispatches a version-validated typed request.
69    ///
70    /// # Parameters
71    /// - `request`: Version-validated typed request.
72    /// - `correlation_id`: Request correlation identity.
73    ///
74    /// # Returns
75    /// Typed protocol response.
76    #[allow(clippy::too_many_lines)]
77    fn dispatch_request(&self, request: Request, correlation_id: &str) -> Response {
78        match request {
79            Request::Inspect(request) => self.inspect(request, correlation_id),
80            Request::Capabilities(_) => self.capabilities(),
81            Request::OperationStatus(request) => self
82                .service
83                .operation_status(&request.operation.id)
84                .map_or_else(
85                    |error| Response::Error(helpers::map_error(&error, correlation_id)),
86                    |operation| Response::OperationStatus(OperationStatusResponse { operation }),
87                ),
88            Request::CancelOperation(request) => self
89                .service
90                .cancel_operation(&request.operation.id)
91                .map_or_else(
92                    |error| Response::Error(helpers::map_error(&error, correlation_id)),
93                    |operation| Response::OperationCancelled(OperationStatusResponse { operation }),
94                ),
95            Request::Subscribe(request) => self
96                .service
97                .events_after(&request.event_kinds, request.after.unwrap_or_default())
98                .map_or_else(
99                    |error| Response::Error(helpers::map_error(&error, correlation_id)),
100                    Response::Events,
101                ),
102            Request::Query(request) => {
103                let Ok(specification) = serde_json::from_value::<QuerySpec>(request.specification)
104                else {
105                    return Response::Error(ProtocolError {
106                        code: ProtocolErrorCode::SchemaViolation,
107                        message: "invalid query specification".to_owned(),
108                        details: None,
109                        correlation_id: Some(correlation_id.to_owned()),
110                    });
111                };
112                self.service
113                    .query(&ApplicationContext::empty(), &specification)
114                    .map_or_else(
115                        |error| Response::Error(helpers::map_error(&error, correlation_id)),
116                        |result| {
117                            Response::Query(scriberig_protocol::QueryResponse {
118                                result: serde_json::to_value(result).unwrap_or_default(),
119                            })
120                        },
121                    )
122            }
123            Request::List(request) => {
124                let Ok(specification) =
125                    serde_json::from_value::<scriberig_query::ListRequest>(request.specification)
126                else {
127                    return Response::Error(ProtocolError {
128                        code: ProtocolErrorCode::SchemaViolation,
129                        message: "invalid list specification".to_owned(),
130                        details: None,
131                        correlation_id: Some(correlation_id.to_owned()),
132                    });
133                };
134                self.service
135                    .list(&ApplicationContext::empty(), &specification)
136                    .map_or_else(
137                        |error| Response::Error(helpers::map_error(&error, correlation_id)),
138                        |result| {
139                            Response::List(scriberig_protocol::ListResponse {
140                                result: serde_json::to_value(result).unwrap_or_default(),
141                            })
142                        },
143                    )
144            }
145            Request::Find(request) => {
146                let Ok(specification) =
147                    serde_json::from_value::<scriberig_query::FindRequest>(request.specification)
148                else {
149                    return Response::Error(ProtocolError {
150                        code: ProtocolErrorCode::SchemaViolation,
151                        message: "invalid find specification".to_owned(),
152                        details: None,
153                        correlation_id: Some(correlation_id.to_owned()),
154                    });
155                };
156                self.service
157                    .find(&ApplicationContext::empty(), &specification)
158                    .map_or_else(
159                        |error| Response::Error(helpers::map_error(&error, correlation_id)),
160                        |result| {
161                            Response::Find(scriberig_protocol::FindResponse {
162                                result: serde_json::to_value(result).unwrap_or_default(),
163                            })
164                        },
165                    )
166            }
167            Request::Command(request) => self
168                .service
169                .command(
170                    &command_context(&request.kind),
171                    &request.kind,
172                    &request.payload,
173                )
174                .map_or_else(
175                    |error| Response::Error(helpers::map_error(&error, correlation_id)),
176                    |operation| Response::CommandAccepted(OperationStatusResponse { operation }),
177                ),
178            Request::AgentStart(request) => self.agent_command(
179                "agent_start",
180                &serde_json::json!({
181                    "session": request.session.id,
182                    "prompt": request.prompt,
183                    "selection": request.selection,
184                }),
185                None,
186                correlation_id,
187            ),
188            Request::AgentContinue(request) => self.agent_command(
189                "agent_continue",
190                &serde_json::json!({
191                    "session": request.session.id,
192                    "branch": request.branch.id,
193                    "prompt": request.prompt,
194                    "selection": request.selection,
195                }),
196                Some((&request.session, &request.branch)),
197                correlation_id,
198            ),
199            Request::SessionRead(request) => self.session_read(&request, correlation_id),
200            Request::Project(_) => Response::Error(ProtocolError {
201                code: ProtocolErrorCode::UnsupportedOperation,
202                message: "project operations require the HTTP project catalog".to_owned(),
203                details: None,
204                correlation_id: Some(correlation_id.to_owned()),
205            }),
206        }
207    }

209    /// Builds the capability response from the injected application service.
210    ///
211    /// # Returns
212    /// The public runtime capability response.
213    fn capabilities(&self) -> Response {
214        let capabilities = self
215            .service
216            .capabilities()
217            .into_iter()
218            .map(|capability| CapabilityInfo {
219                id: capability.id.to_string(),
220                description: capability.description,
221                available: capability.available,
222                domain: capability.scope,
223            })
224            .collect();
225        Response::Capabilities(CapabilitiesResponse {
226            protocol: ProtocolVersion::default(),
227            capabilities,
228            features: vec![RuntimeFeature {
229                id: "inspect".to_owned(),
230                enabled: true,
231            }],
232        })
233    }

235    /// Reads one durable session through the application boundary.
236    ///
237    /// # Parameters
238    /// - `request`: Typed session read request.
239    /// - `correlation_id`: Request correlation identity.
240    ///
241    /// # Returns
242    /// Typed session response or protocol error.
243    fn session_read(
244        &self,
245        request: &scriberig_protocol::SessionReadRequest,
246        correlation_id: &str,
247    ) -> Response {
248        self.service
249            .read_session(&command_context("session_branch_tree"), &request.session.id)
250            .map_or_else(
251                |error| Response::Error(helpers::map_error(&error, correlation_id)),
252                |response| {
253                    Response::Session(request.branch.as_ref().map_or(response.clone(), |branch| {
254                        helpers::select_branch_subtree(response, &branch.id)
255                    }))
256                },
257            )
258    }

260    /// Dispatches one typed agent request through the shared command boundary.
261    ///
262    /// # Parameters
263    /// - `kind`: Stable agent operation identifier.
264    /// - `payload`: Provider-neutral request payload.
265    /// - `references`: Session and optional branch references for the response.
266    /// - `correlation_id`: Request correlation identity.
267    ///
268    /// # Returns
269    /// Typed accepted agent response or protocol error.
270    fn agent_command(
271        &self,
272        kind: &str,
273        payload: &serde_json::Value,
274        references: Option<(
275            &scriberig_protocol::SessionReference,
276            &scriberig_protocol::BranchReference,
277        )>,
278        correlation_id: &str,
279    ) -> Response {
280        if !self
281            .service
282            .capabilities()
283            .iter()
284            .any(|capability| capability.id.as_str() == kind && capability.available)
285        {
286            return Response::Error(helpers::map_error(
287                &scriberig_app::ApplicationError::UnsupportedOperation {
288                    operation: kind.to_owned(),
289                },
290                correlation_id,
291            ));
292        }
293        let session = references.as_ref().map_or_else(
294            || {
295                payload
296                    .get("session")
297                    .and_then(serde_json::Value::as_str)
298                    .unwrap_or_default()
299                    .to_owned()
300            },
301            |(session, _)| session.id.clone(),
302        );
303        self.service
304            .command(&command_context(kind), kind, payload)
305            .map_or_else(
306                |error| Response::Error(helpers::map_error(&error, correlation_id)),
307                |operation| {
308                    let branch = references.map(|(_, branch)| branch.clone()).or_else(|| {
309                        self.service
310                            .read_session(&command_context("session_branch_tree"), &session)
311                            .ok()
312                            .and_then(|value| value.branches.last().cloned())
313                            .map(|branch| scriberig_protocol::BranchReference {
314                                id: branch.id.clone(),
315                            })
316                    });
317                    Response::AgentAccepted(scriberig_protocol::AgentAcceptedResponse {
318                        operation: OperationStatusResponse { operation },
319                        session: scriberig_protocol::SessionReference { id: session },
320                        branch,
321                    })
322                },
323            )
324    }

326    /// Maps one protocol inspection request to the shared application request.
327    ///
328    /// # Parameters
329    /// - `request`: Protocol inspection request.
330    /// - `correlation_id`: Request correlation identity.
331    ///
332    /// # Returns
333    /// Typed protocol inspection response.
334    fn inspect(
335        &self,
336        request: scriberig_protocol::InspectEntityRequest,
337        correlation_id: &str,
338    ) -> Response {
339        let Ok(entity) = EntityId::from_str(&request.entity.id) else {
340            return Response::Error(ProtocolError {
341                code: ProtocolErrorCode::SchemaViolation,
342                message: "invalid entity identifier".to_owned(),
343                details: None,
344                correlation_id: Some(correlation_id.to_owned()),
345            });
346        };
347        let snapshot = match request.snapshot {
348            scriberig_protocol::SnapshotSelector::Current
349            | scriberig_protocol::SnapshotSelector::NamedRoot(_) => SnapshotResolution::Current,
350            scriberig_protocol::SnapshotSelector::Exact(reference) => {
351                match SnapshotId::from_str(&reference.id) {
352                    Ok(value) => SnapshotResolution::At(value),
353                    Err(_) => {
354                        return Response::Error(ProtocolError {
355                            code: ProtocolErrorCode::SchemaViolation,
356                            message: "invalid snapshot identifier".to_owned(),
357                            details: None,
358                            correlation_id: Some(correlation_id.to_owned()),
359                        });
360                    }
361                }
362            }
363        };
364        let result = self.service.inspect(
365            &ApplicationContext::empty(),
366            &InspectRequest {
367                entity,
368                snapshot,
369                detail: match request.detail {
370                    scriberig_protocol::request::DetailLevel::Compact => DetailLevel::Compact,
371                    scriberig_protocol::request::DetailLevel::Deep => DetailLevel::Detailed,
372                },
373            },
374        );
375        result.map_or_else(
376            |error| Response::Error(helpers::map_error(&error, correlation_id)),
377            |result| Response::Inspect(helpers::map_inspect(result)),
378        )
379    }

382/// Builds the trusted server context for an exposed command family.
383///
384/// # Parameters
385/// - `kind`: Stable command or capability identifier.
386///
387/// # Returns
388/// A context granting only the requested public command capability.
389fn command_context(kind: &str) -> ApplicationContext {
390    let mut capabilities = vec![CapabilityId::new(kind).expect("protocol command capability")];
391    if matches!(kind, "agent_start" | "agent_continue") {
392        capabilities.push(CapabilityId::new("session_branch_tree").expect("session capability"));
393    }
394    ApplicationContext {
395        capabilities: CapabilityContext::new(capabilities),
396        ..ApplicationContext::empty()
397    }

9/// Dispatches one operation-segment request through the public protocol.
10///
11/// - state: Shared application service and fixture state.
12/// - operation: Stable operation identifier from the URL.
13/// - body: JSON request payload.
14/// - `correlation_id`: Request correlation identity.
15///
16/// A serialized typed protocol response envelope.
17///
18/// Returns a transport status and structured body for malformed or unknown
19/// operations.
20pub(super) fn dispatch<S: ApplicationService + 'static>(
21    state: &HttpState<S>,
22    operation: &str,
23    body: &str,
24    correlation_id: String,
25) -> Result<Value, (u16, Value)> {
26    let request = decode_request(operation, body)?;
27    if let scriberig_protocol::Request::Project(project) = request {
28        return serde_json::to_value(super::project_routes::protocol(
29            state,
30            &project,
31            correlation_id,
32        ))
33        .map_err(|error| {
34            (
35                500,
36                json!({"code": "internal", "message": error.to_string()}),
37            )
38        });
39    }
40    let lifecycle_handler = state.plan_lifecycle_handler.as_ref().map(|handler| {
41        let handler = std::sync::Arc::clone(handler);
42        std::sync::Arc::new(
43            move |service: &&S,
44                  context: &scriberig_app::ApplicationContext,
45                  request: &scriberig_app::PlanLifecycleRequest,
46                  action: scriberig_app::PlanLifecycleAction| {
47                handler(*service, context, request, action)
48            },
49        ) as std::sync::Arc<crate::http::PlanLifecycleHandler<&S>>
50    });
51    let response = Dispatcher::for_principal(&state.service, state.principal.clone())
52        .dispatch_with_plan_lifecycle(
53            scriberig_protocol::RequestEnvelope {
54                version: scriberig_protocol::ProtocolVersion::default(),
55                correlation_id,
56                request,
57            },
58            lifecycle_handler.as_deref(),
59        );
60    serde_json::to_value(response).map_err(|error| {
61        (
62            500,
63            json!({"code": "internal", "message": error.to_string()}),
64        )
65    })
66}

68/// Decodes one stable operation identifier into a protocol request variant.
69///
70/// - `operation`: Stable operation identifier from the URL.
71/// - `body`: JSON request payload.
72///
73/// The corresponding typed protocol request.
74///
75/// Returns a transport status and structured body when decoding fails.
76/// # Returns
77/// The operation result.
78/// # Errors
79/// Returns an error when the operation fails.
80fn decode_request(
81    operation: &str,
82    body: &str,
83) -> Result<scriberig_protocol::Request, (u16, Value)> {
84    let invalid = |error: serde_json::Error| {
85        (
86            400,
87            json!({"code": "invalid_request", "message": error.to_string()}),
88        )
89    };
90    let request = match operation {
91        "inspect" => {
92            scriberig_protocol::Request::Inspect(serde_json::from_str(body).map_err(invalid)?)
93        }
94        "capabilities" => {
95            scriberig_protocol::Request::Capabilities(serde_json::from_str(body).map_err(invalid)?)
96        }
97        "operation" | "operation_status" => scriberig_protocol::Request::OperationStatus(
98            serde_json::from_str(body).map_err(invalid)?,
99        ),
100        "cancel" | "cancel_operation" => scriberig_protocol::Request::CancelOperation(
101            serde_json::from_str(body).map_err(invalid)?,
102        ),
103        "subscribe" => {
104            scriberig_protocol::Request::Subscribe(serde_json::from_str(body).map_err(invalid)?)
105        }
106        "query" => scriberig_protocol::Request::Query(serde_json::from_str(body).map_err(invalid)?),
107        "list" => scriberig_protocol::Request::List(serde_json::from_str(body).map_err(invalid)?),
108        "find" => scriberig_protocol::Request::Find(serde_json::from_str(body).map_err(invalid)?),
109        "command" => {
110            scriberig_protocol::Request::Command(serde_json::from_str(body).map_err(invalid)?)
111        }
112        "agent_start" => {
113            scriberig_protocol::Request::AgentStart(serde_json::from_str(body).map_err(invalid)?)
114        }
115        "agent_new_session" => scriberig_protocol::Request::AgentNewSession(
116            serde_json::from_str(body).map_err(invalid)?,
117        ),
118        "agent_continue" => {
119            scriberig_protocol::Request::AgentContinue(serde_json::from_str(body).map_err(invalid)?)
120        }
121        "agent_continue_at" => scriberig_protocol::Request::AgentContinueAt(
122            serde_json::from_str(body).map_err(invalid)?,
123        ),
124        "session" | "session_read" => {
125            scriberig_protocol::Request::SessionRead(serde_json::from_str(body).map_err(invalid)?)
126        }
127        "plan_create" => {
128            scriberig_protocol::Request::PlanCreate(serde_json::from_str(body).map_err(invalid)?)
129        }
130        "plan_lifecycle" => {
131            scriberig_protocol::Request::PlanLifecycle(serde_json::from_str(body).map_err(invalid)?)
132        }
133        "plan_assignment" => scriberig_protocol::Request::PlanAssignment(
134            serde_json::from_str(body).map_err(invalid)?,
135        ),
136        "plan_replan" => {
137            scriberig_protocol::Request::PlanReplan(serde_json::from_str(body).map_err(invalid)?)
138        }
139        "model_request_detail" => scriberig_protocol::Request::ModelRequestDetail(
140            serde_json::from_str(body).map_err(invalid)?,
141        ),
142        "tool_result_detail" => scriberig_protocol::Request::ToolResultDetail(
143            serde_json::from_str(body).map_err(invalid)?,
144        ),
145        "model_library" => {
146            scriberig_protocol::Request::ModelLibrary(serde_json::from_str(body).map_err(invalid)?)
147        }
148        "agent_presentation_preferences" => {
149            scriberig_protocol::Request::AgentPresentationPreferencesRead(
150                serde_json::from_str(body).map_err(invalid)?,
151            )
152        }
153        "set_agent_presentation_preferences" => {
154            scriberig_protocol::Request::AgentPresentationPreferencesSet(
155                serde_json::from_str(body).map_err(invalid)?,
156            )
157        }
158        "session_model_selection" => scriberig_protocol::Request::SessionModelSelection(
159            serde_json::from_str(body).map_err(invalid)?,
160        ),
161        "session_thinking_selection" => scriberig_protocol::Request::SessionThinkingSelection(
162            serde_json::from_str(body).map_err(invalid)?,
163        ),
164        "project" => {
165            scriberig_protocol::Request::Project(serde_json::from_str(body).map_err(invalid)?)
166        }
167        _ => {
168            return Err((
169                404,
170                json!({
171                    "code": "unsupported_operation",
172                    "message": format!("unknown protocol operation: {operation}"),
173                }),
174            ));
175        }
176    };
177    Ok(request)
178}

180#[cfg(test)]
181mod tests {
182    use super::decode_request;
183
184    /// Verifies HTTP operations decode model-library and selection requests.
185    #[test]
186    fn model_library_and_selection_operations_decode() {
187        assert!(matches!(
188            decode_request("model_library", "{}"),
189            Ok(scriberig_protocol::Request::ModelLibrary(_))
190        ));
191        assert!(matches!(
192            decode_request(
193                "model_request_detail",
194                r#"{"session":{"id":"session-1"},"branch":{"id":"branch-1"},"invocation":"invocation-1"}"#
195            ),
196            Ok(scriberig_protocol::Request::ModelRequestDetail(_))
197        ));
198        assert!(matches!(
199            decode_request(
200                "tool_result_detail",
201                r#"{"session":{"id":"session-1"},"branch":{"id":"branch-1"},"result":"result-1"}"#
202            ),
203            Ok(scriberig_protocol::Request::ToolResultDetail(_))
204        ));
205        assert!(matches!(
206            decode_request(
207                "session_model_selection",
208                r#"{"session":{"id":"session-1"},"expected_revision":2,"model_profile":"profile-1"}"#
209            ),
210            Ok(scriberig_protocol::Request::SessionModelSelection(_))
211        ));
212        assert!(matches!(
213            decode_request(
214                "session_thinking_selection",
215                r#"{"session":{"id":"session-1"},"expected_revision":3,"thinking_level":"deep-dive"}"#
216            ),
217            Ok(scriberig_protocol::Request::SessionThinkingSelection(_))
218        ));
219    }

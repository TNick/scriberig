28/// Request for the deterministic project-agent workflow.
29#[derive(Debug, Deserialize)]
30struct AgentInput {
31    /// Project receiving the candidate.
32    project: String,
33    /// Exact selected entity.
34    entity: String,
35    /// Exact selected snapshot.
36    snapshot: String,
37    /// User instruction sent to the provider.
38    prompt: String,
39}
40
41/// Records the semantic scope observed by the provider-facing tool.
42#[derive(Clone, Copy)]
43struct ObservedScope {
44    /// Entity received by the inspect tool.
45    entity: EntityId,
46    /// Snapshot received by the inspect tool.
47    snapshot: SnapshotId,
48}
49
50/// Project-backed provider tool used by the deterministic golden provider.
51struct ProjectAgentTool {
52    /// Shared project store used by `ScribeRig` project operations.
53    project_store: Arc<SqliteStore>,
54    /// Project selected by the user.
55    project: scriberig_core::ProjectId,
56    /// Scope expected from the provider.
57    expected: ObservedScope,
58    /// Scope actually observed by the tool.
59    observed: Arc<Mutex<Option<ObservedScope>>>,
60    /// Result sink receiving the candidate identity created by proposal tools.
61    candidate_id: Arc<Mutex<Option<String>>>,
62}
63
64impl Tool for ProjectAgentTool {
65    /// Returns the stable provider tool identifier.
66    ///
67    /// The stable project tool name.
68    /// # Returns
69    /// The operation result.
70    fn name(&self) -> &'static str {
71        "scriberig.project"
72    }
73
74    /// Inspects the selected entity at the requested exact snapshot.
75    ///
76    /// - `_context`: Agent execution context.
77    /// - `input`: Structured semantic inspection input.
78    ///
79    /// The selected entity's bounded semantic result.
80    ///
81    /// Returns a tool error when scope or project state is invalid.
82    /// # Returns
83    /// The operation result.
84    /// # Errors
85    /// Returns an error when the operation fails.
86    fn invoke(
87        &self,
88        _context: &ApplicationContext,
89        input: ToolInput,
90    ) -> Result<ToolResult, ToolError> {
91        let snapshot = input.snapshot.ok_or_else(|| invalid_scope("snapshot"))?;
92        if input.entity != self.expected.entity || snapshot != self.expected.snapshot {
93            return Err(ToolError::Application {
94                category: "semantic_scope_mismatch".to_owned(),
95                message: "agent inspected outside its selected entity and snapshot".to_owned(),
96            });
97        }
98        self.observed
99            .lock()
100            .map_err(|_| tool_error("scope lock poisoned"))?
101            .replace(ObservedScope {
102                entity: input.entity,
103                snapshot,
104            });
105        let entity = self
106            .project_store
107            .project_entities(self.project)
108            .map_err(|error| tool_error(&error.to_string()))?
109            .into_iter()
110            .find(|value| value.entity == input.entity)
111            .ok_or_else(|| tool_error("selected entity unavailable"))?;
112        Ok(ToolResult {
113            entity: input.entity,
114            revision: None,
115            snapshot: Some(snapshot),
116            object_type: Some(entity.kind),
117            summary: entity.name,
118            handles: vec![entity.relative_path],
119        })
120    }
121
122    /// Creates the normal reviewed candidate through the project operation.
123    ///
124    /// - `_context`: Agent execution context.
125    /// - `name`: Provider tool name.
126    /// - `arguments`: Structured proposal arguments.
127    ///
128    /// The normal migration-candidate tool result.
129    ///
130    /// Returns a tool error when proposal arguments or project validation fail.
131    /// # Returns
132    /// The operation result.
133    /// # Errors
134    /// Returns an error when the operation fails.
135    fn invoke_named(
136        &self,
137        _context: &ApplicationContext,
138        name: &str,
139        arguments: &Value,
140    ) -> Result<ToolResult, ToolError> {
141        if name != "project_propose" {
142            return Err(tool_error("unsupported project agent tool"));
143        }
144        if let Some(effects) = arguments.get("effects") {
145            let result = aggregate::propose(
146                &self.project_store,
147                self.project,
148                self.expected.entity,
149                self.expected.snapshot,
150                effects,
151            )?;
152            let candidate_id = result
153                .handles
154                .first()
155                .cloned()
156                .ok_or_else(|| tool_error("candidate identity missing"))?;
157            self.candidate_id
158                .lock()
159                .map_err(|_| tool_error("candidate result lock poisoned"))?
160                .replace(candidate_id);
161            return Ok(result);
162        }
163        let text = arguments
164            .get("text")
165            .and_then(Value::as_str)
166            .ok_or_else(|| invalid_scope("text"))?;
167        let source = self
168            .project_store
169            .project_files(self.project)
170            .map_err(|error| tool_error(&error.to_string()))?
171            .into_iter()
172            .find(|file| file.relative_path == "src/lib.rs")
173            .ok_or_else(|| tool_error("source unavailable"))?;
174        let edited = String::from_utf8_lossy(&source.bytes).replace("left + right", text);
175        let candidate = ProjectService::new(Arc::clone(&self.project_store))
176            .propose_candidate(self.project, self.expected.entity, edited)
177            .map_err(|error| tool_error(&error.to_string()))?;
178        self.candidate_id
179            .lock()
180            .map_err(|_| tool_error("candidate result lock poisoned"))?
181            .replace(candidate.id.clone());
182        Ok(ToolResult {
183            entity: self.expected.entity,
184            revision: None,
185            snapshot: Some(self.expected.snapshot),
186            object_type: Some("migration_candidate".to_owned()),
187            summary: Some("candidate created".to_owned()),
188            handles: vec![candidate.id],
189        })
190    }
191}

193/// Runs the deterministic provider through inspect and candidate tools.
194pub(super) fn agent<S: ApplicationService>(
195    state: &HttpState<S>,
196    body: &str,
197) -> Result<Value, (u16, Value)> {
198    let input: AgentInput =
199        serde_json::from_str(body).map_err(|error| (400, json!({"message": error.to_string()})))?;
200    let project = input
201        .project
202        .parse::<scriberig_core::ProjectId>()
203        .map_err(|_| (400, json!({"message": "invalid project"})))?;
204    let entity = input
205        .entity
206        .parse::<EntityId>()
207        .map_err(|_| (400, json!({"message": "invalid entity"})))?;
208    let snapshot = input
209        .snapshot
210        .parse::<SnapshotId>()
211        .map_err(|_| (400, json!({"message": "invalid snapshot"})))?;
212    let source = source_for(state, project)?;
213    let accepted = source.replace("left + right", "left.saturating_add(right)");
214    let observed = Arc::new(Mutex::new(None));
215    let candidate_id: Arc<Mutex<Option<String>>> = Arc::new(Mutex::new(None));
216    let branch = EntityId::new();
217    let context = ApplicationContext::empty();
218    let tool = Arc::new(ProjectAgentTool {
219        project_store: Arc::clone(
220            state
221                .project_store
222                .as_ref()
223                .ok_or_else(|| (503, json!({"message": "project store unavailable"})))?,
224        ),
225        project,
226        expected: ObservedScope { entity, snapshot },
227        observed: Arc::clone(&observed),
228        candidate_id: Arc::clone(&candidate_id),
229    });
230    let context_inputs = BranchExecutionContext::new(
231        project.to_string(),
232        branch.to_string(),
233        "project candidate proposal",
234        input.prompt.clone(),
235    )
236    .with_project_scope(ProjectSemanticScope {
237        project: project.to_string(),
238        snapshot: snapshot.to_string(),
239        root: project.to_string(),
240        selection: Some(entity.to_string()),
241    });
242    let compiled_context = ContextCompiler::new(ContextPolicy::default()).compile(&context_inputs);
243    let (provider, model) = faux_provider(entity, snapshot, &accepted);
244    let tools = project_tools(Arc::clone(&tool), branch, context.clone())
245        .map_err(|error| (500, json!({"message": error})))?;
246    let invocation = UpstreamBranchInvocation {
247        branch,
248        provider,
249        model,
250        request_secret: None,
251        bearer: false,
252        applied_settings: None,
253        provider_profile: Some("scriberig-deterministic-project-agent".to_owned()),
254        thinking_level: rpi_ai::ThinkingLevel::Off,
255        context: compiled_context,
256        tools,
257    };
258    let result = run_upstream_branch_invocation_blocking(invocation, input.prompt)
259        .map_err(|failure| (500, json!({"message": failure.error.to_string()})))?;
260    let candidate = candidate_id
261        .lock()
262        .map_err(|_| (500, json!({"message": "candidate result lock poisoned"})))?
263        .clone()
264        .ok_or_else(|| (500, json!({"message": "agent did not create a candidate"})))?;
265    let store = Arc::clone(
266        state
267            .project_store
268            .as_ref()
269            .ok_or_else(|| (503, json!({"message": "project store unavailable"})))?,
270    );
271    let scope = observed
272        .lock()
273        .map_err(|_| (500, json!({"message": "scope lock poisoned"})))?
274        .ok_or_else(|| {
275            (
276                500,
277                json!({"message": "agent did not inspect selected scope"}),
278            )
279        })?;
280    candidate_response(store, &candidate, scope, &result.output)
281}

283/// Builds the reviewed-candidate response after validating persisted state.
284///
285/// - `store`: Project store used by the project service.
286/// - `candidate`: Exact candidate identity created by the branch tool.
287/// - `scope`: Entity and snapshot observed by the inspect operation.
288/// - `output`: Bounded final response from the upstream agent.
289///
290/// The response payload for the existing project-agent route.
291///
292/// # Returns
293/// JSON containing the reviewed candidate and exact base/scope references.
294///
295/// # Errors
296/// Returns an HTTP error when the candidate is missing or invalid.
297fn candidate_response(
298    store: Arc<SqliteStore>,
299    candidate: &str,
300    scope: ObservedScope,
301    output: &str,
302) -> Result<Value, (u16, Value)> {
303    let detail = ProjectService::new(store)
304        .candidate(candidate)
305        .map_err(|error| (500, json!({"message": error.message})))?
306        .ok_or_else(|| (500, json!({"message": "candidate disappeared"})))?;
307    let (validation, apply_enabled) = match &detail.validation {
308        scriberig_project::CandidateValidation::Pending => ("pending", false),
309        scriberig_project::CandidateValidation::Passed => ("passed", true),
310        scriberig_project::CandidateValidation::Failed { diagnostics } => {
311            return Err((500, json!({"message": diagnostics.join("\n")})));
312        }
313    };
314    Ok(json!({
315        "candidate": candidate,
316        "base": detail.base_snapshot.to_string(),
317        "path": detail.path,
318        "old": String::from_utf8_lossy(&detail.old),
319        "new": String::from_utf8_lossy(&detail.new),
320        "validation": validation,
321        "applyEnabled": apply_enabled,
322        "scope": {"entity": scope.entity.to_string(), "snapshot": scope.snapshot.to_string()},
323        "output": output,
324    }))
325}

327/// Reads the currently captured Rust source for a project.
328///
329/// - `state`: HTTP state owning project storage.
330/// - `project`: Project whose source is requested.
331///
332/// The captured Rust source text.
333///
334/// Returns an HTTP error tuple when storage or source lookup fails.
335fn source_for<S: ApplicationService>(
336    state: &HttpState<S>,
337    project: scriberig_core::ProjectId,
338) -> Result<String, (u16, Value)> {
339    let store = state
340        .project_store
341        .as_ref()
342        .ok_or_else(|| (503, json!({"message": "project store unavailable"})))?;
343    store
344        .project_files(project)
345        .map_err(|error| (500, json!({"message": error.to_string()})))?
346        .into_iter()
347        .find(|file| file.relative_path == "src/lib.rs")
348        .map(|file| String::from_utf8_lossy(&file.bytes).into_owned())
349        .ok_or_else(|| (404, json!({"message": "source unavailable"})))
350}

352/// Creates a provider-facing invalid-input error.
353///
354/// - `value`: Missing input name.
355///
356/// A normalized tool error.
357/// # Returns
358/// The operation result.
359fn invalid_scope(value: &str) -> ToolError {
360    tool_error(&format!("missing {value}"))
361}
362
363/// Creates a provider-facing application error.
364///
365/// - `message`: Safe application diagnostic.
366///
367/// A normalized tool error.
368/// # Returns
369/// The operation result.
370fn tool_error(message: &str) -> ToolError {
371    ToolError::Application {
372        category: "project_agent".to_owned(),
373        message: message.to_owned(),
374    }

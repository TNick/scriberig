3/// Builds the public identifier for a projected activity item.
4///
5/// - `kind`: Public category of the activity.
6/// - `has_no_parent`: Whether the source activity has no causal parent.
7/// - `id`: Durable source entity identity.
8///
9/// Stable identifier with a role suffix for human and standalone output rows.
10/// # Returns
11/// The operation result.
12pub(super) fn projected_activity_id(
13    kind: scriberig_protocol::SessionActivityKind,
14    has_no_parent: bool,
15    id: scriberig_core::EntityId,
16) -> String {
17    match (kind, has_no_parent) {
18        (scriberig_protocol::SessionActivityKind::HumanInput, _) => format!("{id}:human"),
19        (scriberig_protocol::SessionActivityKind::AssistantOutput, true) => {
20            format!("{id}:assistant")
21        }
22        _ => id.to_string(),
23    }
24}

26/// Projects bounded safe detail for one tool activity.
27///
28/// - `activity`: Durable activity reference being projected.
29/// - `branch`: Branch containing the corresponding tool record.
30///
31/// Safe invocation arguments or result preview, with the other value absent.
32/// # Returns
33/// The operation result.
34pub(super) fn activity_tool_details(
35    activity: &scriberig_session::ExecutionActivity,
36    branch: &scriberig_session::Branch,
37) -> (Option<serde_json::Value>, Option<serde_json::Value>) {
38    match activity {
39        scriberig_session::ExecutionActivity::ToolInvocation(id) => (
40            branch
41                .tool_invocations
42                .iter()
43                .find(|invocation| invocation.id == *id)
44                .and_then(|invocation| invocation.presentation_input.as_deref())
45                .and_then(bounded_tool_arguments),
46            None,
47        ),
48        scriberig_session::ExecutionActivity::ToolResult(id) => (
49            None,
50            branch
51                .tool_results
52                .iter()
53                .find(|result| result.id == *id)
54                .and_then(|result| bounded_tool_result_preview(&result.output)),
55        ),
56        _ => (None, None),
57    }
58}

60/// Bounds display text by Unicode scalar values.
61///
62/// - `value`: Externally meaningful text.
63///
64/// At most 512 characters with an ellipsis when truncated.
65/// # Returns
66/// The operation result.
67pub(super) fn bounded_summary(value: &str) -> String {
68    let mut chars = value.chars();
69    let mut summary = chars.by_ref().take(512).collect::<String>();
70    if chars.next().is_some() {
71        summary.push('…');
72    }
73    summary
74}

76/// Decodes and bounds persisted schema-redacted tool argument detail.
77///
78/// - `bytes`: Safe structured argument bytes stored with the invocation.
79///
80/// The parsed argument value when its JSON form is at most 4096 characters.
81/// # Returns
82/// The operation result.
83pub(super) fn bounded_tool_arguments(bytes: &[u8]) -> Option<serde_json::Value> {
84    let value = serde_json::from_slice::<serde_json::Value>(bytes).ok()?;
85    (value.to_string().chars().count() <= 4096).then_some(value)
86}

88/// Projects only safe result metadata for bounded full-mode display.
89///
90/// - `bytes`: Structured result bytes retained by the trusted tool boundary.
91///
92/// A whitelisted object bounded to 4096 serialized characters.
93/// # Returns
94/// The operation result.
95pub(super) fn bounded_tool_result_preview(bytes: &[u8]) -> Option<serde_json::Value> {
96    let value = serde_json::from_slice::<serde_json::Value>(bytes).ok()?;
97    let object = value.as_object()?;
98    let mut preview = serde_json::Map::new();
99    for key in ["entity", "revision", "snapshot", "object_type", "summary"] {
100        if let Some(serde_json::Value::String(text)) = object.get(key) {
101            let bounded = text.chars().take(512).collect::<String>();
102            preview.insert(key.to_owned(), serde_json::Value::String(bounded));
103        }
104    }
105    if let Some(serde_json::Value::Array(handles)) = object.get("handles") {
106        let handles = handles
107            .iter()
108            .filter_map(serde_json::Value::as_str)
109            .take(20)
110            .map(|handle| handle.chars().take(128).collect::<String>())
111            .collect::<Vec<_>>();
112        preview.insert("handles".to_owned(), serde_json::json!(handles));
113    }
114    let preview = serde_json::Value::Object(preview);
115    (preview.to_string().chars().count() <= 4096).then_some(preview)
116}

118/// Reads the tool boundary's safe summary, falling back to a generic status.
119///
120/// - `result`: Durable result bytes returned by the trusted tool boundary.
121/// - `failed`: Whether the tool execution failed.
122///
123/// A bounded result summary without exposing arbitrary output fields.
124/// # Returns
125/// The operation result.
126pub(super) fn tool_result_summary(result: &scriberig_session::ToolResult, failed: bool) -> String {
127    let fallback = if failed {
128        "Tool failed"
129    } else {
130        "Tool completed"
131    };
132    serde_json::from_slice::<serde_json::Value>(&result.output)
133        .ok()
134        .and_then(|value| {
135            value
136                .get("summary")
137                .and_then(serde_json::Value::as_str)
138                .map(str::to_owned)
139        })
140        .filter(|summary| !summary.trim().is_empty())
141        .map_or_else(|| fallback.to_owned(), |summary| bounded_summary(&summary))
142}

1//! Advertises protocol capabilities supported by the application façade.
2
3use super::RuntimeCapability;
4
5/// Builds the static capability list supported by this server façade.
6///
7/// # Parameters
8/// - `super`: Parent module containing the protocol application adapter.
9///
10/// # Returns
11/// Available inspect, explorer, and agent session capabilities.
12pub(super) fn capabilities() -> Vec<RuntimeCapability> {
13    [
14        ("inspect_entity", "Inspect one entity"),
15        ("explorer_list", "List entities in an exact snapshot"),
16        ("agent_start", "Start provider-neutral agent work"),
17        ("agent_continue", "Continue work on a session branch"),
18    ]
19    .into_iter()
20    .map(|(id, description)| {
21        RuntimeCapability::new(
22            scriberig_app::CapabilityId::new(id).expect("static capability"),
23            description,
24            true,
25        )
26    })
27    .collect()
28}

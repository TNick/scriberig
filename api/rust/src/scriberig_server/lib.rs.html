1//! Adapts remote protocol requests and events onto shared Scriberig application operations.
2
3use std::error::Error;
4use std::io::{Read, Write};
5use std::net::{TcpListener, TcpStream};
6use std::path::Path;
7
8/// Serve the development health endpoint over HTTP.
9///
10/// # Parameters
11/// - `host`: Interface address to bind.
12/// - `port`: TCP port to bind.
13/// - `database_path`: SQLite file to initialize before accepting requests.
14///
15/// # Returns
16/// Unit after the listener stops accepting connections.
17///
18/// # Errors
19/// Returns an error when the database, listener, or a client connection fails.
20pub fn serve(
21    host: &str,
22    port: u16,
23    database_path: &Path,
24) -> Result<(), Box<dyn Error + Send + Sync>> {
25    scriberig_store_sqlite::initialize_development_database(database_path)?;
26    let listener = TcpListener::bind((host, port))?;
27    println!("Scriberig API listening on http://{host}:{port}");
28    println!("SQLite database: {}", database_path.display());
29
30    for connection in listener.incoming() {
31        handle_connection(connection?)?;
32    }
33    Ok(())
34}
35
36/// Handle one HTTP request from the development web proxy.
37///
38/// # Parameters
39/// - `stream`: Connected client socket.
40///
41/// # Returns
42/// Unit after the HTTP response is written.
43///
44/// # Errors
45/// Returns an error when the request or response cannot be read or written.
46fn handle_connection(mut stream: TcpStream) -> std::io::Result<()> {
47    let mut request = [0_u8; 2048];
48    let bytes_read = stream.read(&mut request)?;
49    let request_line = std::str::from_utf8(&request[..bytes_read])
50        .ok()
51        .and_then(|text| text.lines().next())
52        .unwrap_or_default();
53    let (status, body) = if request_line.starts_with("GET /api/health ") {
54        ("200 OK", r#"{"status":"ready","database":"ready"}"#)
55    } else {
56        ("404 Not Found", r#"{"error":"not found"}"#)
57    };
58
59    write!(
60        stream,
61        "HTTP/1.1 {status}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
62        body.len()
63    )
64}

22/// Serve the development health endpoint over HTTP.
23///
24/// # Parameters
25/// - `host`: Interface address to bind.
26/// - `port`: TCP port to bind.
27/// - `database_path`: SQLite file to initialize before accepting requests.
28///
29/// # Returns
30/// Unit after the listener stops accepting connections.
31///
32/// # Errors
33/// Returns an error when the database or listener cannot be initialized.
34pub fn serve(
35    host: &str,
36    port: u16,
37    database_path: &Path,
38) -> Result<(), Box<dyn Error + Send + Sync>> {
39    serve_with_origin(host, port, database_path, None, &[])
40}

42/// Serves the API with an optional exact browser origin policy.
43///
44/// # Parameters
45/// - `host`: Interface address to bind.
46/// - `port`: TCP port to bind.
47/// - `database_path`: SQLite file to initialize before accepting requests.
48/// - `allowed_origin`: Exact browser origin allowed for CORS responses.
49/// - `project_roots`: Filesystem roots allowed for project onboarding.
50///
51/// # Returns
52/// Unit after the listener stops accepting connections.
53///
54/// # Errors
55/// Returns an error for invalid origin configuration, database failure, or
56/// listener initialization failure.
57pub fn serve_with_origin(
58    host: &str,
59    port: u16,
60    database_path: &Path,
61    allowed_origin: Option<&str>,
62    project_roots: &[std::path::PathBuf],
63) -> Result<(), Box<dyn Error + Send + Sync>> {
64    if allowed_origin.is_some_and(|origin| origin.is_empty() || origin == "*") {
65        return Err("allowed origin must be a non-empty exact origin".into());
66    }
67    let store = Arc::new(scriberig_store_sqlite::SqliteStore::open(database_path)?);
68    let fixture = ensure_fixture(&store).map_err(std::io::Error::other)?;
69    let read = scriberig_app::StoreInspectionBackend::new(Arc::clone(&store));
70    let operations = scriberig_app::StoreOperationProvider::new(Arc::clone(&store));
71    let backend = scriberig_app::ApplicationBackendComposition::new(read, operations);
72    let session_store = Arc::new(scriberig_session_sqlite::SqliteSessionStore::open(
73        database_path,
74    )?);
75    let service = scriberig_app::ApplicationService::new(backend).with_session_store(session_store);
76    let state = http::HttpState {
77        service,
78        fixture,
79        allowed_origin: allowed_origin.map(ToOwned::to_owned),
80        project_store: Some(Arc::clone(&store)),
81        allowed_project_roots: canonical_project_roots(project_roots)?,
82    };
83    let listener = TcpListener::bind((host, port))?;
84    println!("Scriberig API listening on http://{host}:{port}");
85    println!("SQLite database: {}", database_path.display());
86
87    let state = std::sync::Arc::new(state);
88    for connection in listener.incoming() {
89        match connection {
90            Ok(connection) => {
91                let state = std::sync::Arc::clone(&state);
92                std::thread::spawn(move || {
93                    if let Err(error) = http::handle_connection(connection, &state) {
94                        eprintln!("Scriberig API client connection failed: {error}");
95                    }
96                });
97            }
98            Err(error) => {
99                eprintln!("Scriberig API accept failed: {error}");
100            }
101        }
102    }
103    Ok(())
104}

106/// Resolves configured project roots, defaulting to the current directory.
107///
108/// # Parameters
109/// - `roots`: Operator-provided project roots.
110///
111/// # Returns
112/// Canonical roots used by the HTTP path policy.
113///
114/// # Errors
115/// Returns an error when a configured root cannot be resolved.
116fn canonical_project_roots(
117    roots: &[std::path::PathBuf],
118) -> Result<Vec<std::path::PathBuf>, Box<dyn Error + Send + Sync>> {
119    let roots = if roots.is_empty() {
120        vec![std::env::current_dir()?]
121    } else {
122        roots.to_vec()
123    };
124    roots
125        .into_iter()
126        .map(|root| std::fs::canonicalize(root).map_err(Into::into))
127        .collect()
128}

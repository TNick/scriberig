12/// Converts one typed protocol semantic mutation.
13///
14/// - `value`: Protocol mutation with string-encoded identities.
15///
16/// Shared application mutation.
17/// # Returns
18/// The converted mutation.
19/// # Errors
20/// Returns an invalid-request error for malformed identities.
21pub(super) fn semantic_mutation(
22    value: scriberig_protocol::SessionPlanSemanticMutation,
23) -> Result<PlanSemanticMutation, ApplicationError> {
24    Ok(match value {
25        scriberig_protocol::SessionPlanSemanticMutation::UpdateSummary {
26            node,
27            text,
28            sources,
29        } => PlanSemanticMutation::UpdateSummary {
30            node: parse_entity(&node)?,
31            text,
32            sources: sources
33                .into_iter()
34                .map(semantic_reference)
35                .collect::<Result<Vec<_>, _>>()?,
36        },
37        scriberig_protocol::SessionPlanSemanticMutation::AttachEvidence {
38            owner,
39            role,
40            evidence,
41            note,
42        } => PlanSemanticMutation::AttachEvidence {
43            owner: match owner {
44                scriberig_protocol::SessionPlanEvidenceOwner::Node(node) => {
45                    PlanEvidenceOwner::Node(parse_entity(&node)?)
46                }
47                scriberig_protocol::SessionPlanEvidenceOwner::Gate(gate) => {
48                    PlanEvidenceOwner::Gate(parse_entity(&gate)?)
49                }
50            },
51            role: match role {
52                scriberig_protocol::SessionPlanEvidenceRole::Supports => PlanEvidenceRole::Supports,
53                scriberig_protocol::SessionPlanEvidenceRole::Contradicts => {
54                    PlanEvidenceRole::Contradicts
55                }
56                scriberig_protocol::SessionPlanEvidenceRole::Informational => {
57                    PlanEvidenceRole::Informational
58                }
59            },
60            evidence: PlanEvidenceReference {
61                kind: evidence_kind(evidence.kind),
62                reference: semantic_reference(evidence.reference)?,
63                attributes: evidence.attributes,
64            },
65            note,
66        },
67        scriberig_protocol::SessionPlanSemanticMutation::AddGate {
68            owner,
69            phase,
70            requirement,
71        } => PlanSemanticMutation::AddGate {
72            owner: parse_entity(&owner)?,
73            phase: match phase {
74                scriberig_protocol::SessionPlanGatePhase::Entry => PlanGatePhase::Entry,
75                scriberig_protocol::SessionPlanGatePhase::Completion => PlanGatePhase::Completion,
76            },
77            requirement: gate_requirement(requirement)?,
78        },
79        scriberig_protocol::SessionPlanSemanticMutation::WaiveGate { gate, reason } => {
80            PlanSemanticMutation::WaiveGate {
81                gate: parse_entity(&gate)?,
82                reason,
83            }
84        }
85    })
86}

88/// Converts one exact protocol source reference.
89///
90/// - `value`: Entity, revision, and snapshot identities from the request.
91///
92/// Shared application semantic reference.
93/// # Returns
94/// The converted reference.
95/// # Errors
96/// Returns an invalid-request error for malformed identities.
97fn semantic_reference(
98    value: scriberig_protocol::SessionSemanticReference,
99) -> Result<PlanSemanticReference, ApplicationError> {
100    Ok(PlanSemanticReference {
101        entity: parse_entity(&value.entity)?,
102        revision: value
103            .revision
104            .map(|revision| RevisionId::from_str(&revision))
105            .transpose()
106            .map_err(invalid_identity)?,
107        snapshot: value
108            .snapshot
109            .map(|snapshot| SnapshotId::from_str(&snapshot))
110            .transpose()
111            .map_err(invalid_identity)?,
112    })
113}

115/// Converts one protocol evidence kind.
116///
117/// - `value`: Typed protocol evidence category.
118///
119/// Shared session evidence category.
120/// # Returns
121/// The converted evidence kind.
122fn evidence_kind(value: scriberig_protocol::SessionPlanEvidenceKind) -> PlanEvidenceKind {
123    match value {
124        scriberig_protocol::SessionPlanEvidenceKind::BranchResult => PlanEvidenceKind::BranchResult,
125        scriberig_protocol::SessionPlanEvidenceKind::ValidationResult => {
126            PlanEvidenceKind::ValidationResult
127        }
128        scriberig_protocol::SessionPlanEvidenceKind::Decision => PlanEvidenceKind::Decision,
129        scriberig_protocol::SessionPlanEvidenceKind::HumanApproval => {
130            PlanEvidenceKind::HumanApproval
131        }
132        scriberig_protocol::SessionPlanEvidenceKind::Candidate => PlanEvidenceKind::Candidate,
133        scriberig_protocol::SessionPlanEvidenceKind::Artifact => PlanEvidenceKind::Artifact,
134        scriberig_protocol::SessionPlanEvidenceKind::Failure => PlanEvidenceKind::Failure,
135        scriberig_protocol::SessionPlanEvidenceKind::Other(kind) => PlanEvidenceKind::Other(kind),
136    }
137}

139/// Converts one protocol gate requirement.
140///
141/// - `value`: Typed, non-executable protocol gate requirement.
142///
143/// Shared session gate requirement.
144/// # Returns
145/// The converted requirement.
146/// # Errors
147/// Returns an invalid-request error for malformed identities.
148fn gate_requirement(
149    value: scriberig_protocol::SessionPlanGateRequirement,
150) -> Result<PlanGateRequirement, ApplicationError> {
151    Ok(match value {
152        scriberig_protocol::SessionPlanGateRequirement::EvidencePresent { kind } => {
153            PlanGateRequirement::EvidencePresent {
154                kind: evidence_kind(kind),
155            }
156        }
157        scriberig_protocol::SessionPlanGateRequirement::ValidationPassed { profile } => {
158            PlanGateRequirement::ValidationPassed { profile }
159        }
160        scriberig_protocol::SessionPlanGateRequirement::DecisionResolved { decision } => {
161            PlanGateRequirement::DecisionResolved {
162                decision: parse_entity(&decision)?,
163            }
164        }
165        scriberig_protocol::SessionPlanGateRequirement::HumanApproval { policy, role } => {
166            PlanGateRequirement::HumanApproval { policy, role }
167        }
168        scriberig_protocol::SessionPlanGateRequirement::SemanticCondition {
169            reference,
170            predicate,
171        } => PlanGateRequirement::SemanticCondition {
172            reference: semantic_reference(reference)?,
173            predicate: match predicate {
174                scriberig_protocol::SessionPlanSemanticPredicate::EntityExists => {
175                    PlanSemanticPredicate::EntityExists
176                }
177                scriberig_protocol::SessionPlanSemanticPredicate::ExactRevision => {
178                    PlanSemanticPredicate::ExactRevision
179                }
180            },
181        },
182        scriberig_protocol::SessionPlanGateRequirement::RequiredChildCompleted { child } => {
183            PlanGateRequirement::RequiredChildCompleted {
184                child: parse_entity(&child)?,
185            }
186        }
187    })
188}

190/// Parses one stable semantic entity identity from protocol text.
191///
192/// - `value`: Protocol string form of an entity identity.
193///
194/// Parsed entity identity.
195/// # Returns
196/// The parsed identity.
197/// # Errors
198/// Returns invalid-request when the string is malformed.
199fn parse_entity(value: &str) -> Result<EntityId, ApplicationError> {
200    EntityId::from_str(value).map_err(invalid_identity)
201}
202
203/// Creates a structured invalid-identity error.
204///
205/// - `error`: Parsing failure from a typed identity.
206///
207/// An application invalid-request error.
208/// # Returns
209/// The structured invalid-identity error.
210fn invalid_identity(error: impl std::fmt::Display) -> ApplicationError {
211    ApplicationError::InvalidRequest {
212        operation: "plan semantic mutation".to_owned(),
213        message: error.to_string(),
214    }

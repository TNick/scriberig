1//! Defines canonical and derived code-domain classifications.
2/// Classifies whether code knowledge is canonical or derived.
3#[derive(Clone, Copy, Debug, Eq, PartialEq)]
4pub enum InformationClass {
5    /// Source-authored semantic state.
6    Canonical,
7    /// Computed or generated state.
8    Derived,
9}
10/// Returns the class for a field or artifact kind.
11///
12/// - `name`: Stable field name.
13///
14/// Classification when recognized.
15#[must_use]
16/// # Returns
17/// The operation result.
18pub fn classify(name: &str) -> Option<InformationClass> {
19    if matches!(
20        name,
21        "declaration" | "parameter" | "statement" | "expression" | "comment" | "code_group"
22    ) {
23        Some(InformationClass::Canonical)
24    } else if matches!(
25        name,
26        "inferred_type" | "call_graph" | "control_flow" | "description" | "diagnostic"
27    ) {
28        Some(InformationClass::Derived)
29    } else {
30        None
31    }
32}

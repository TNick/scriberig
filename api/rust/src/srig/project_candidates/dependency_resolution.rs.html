1//! Reports bounded dependency acquisition evidence for project candidates.
2
3use scriberig_agent_core::{ToolError, ToolResult};
4use scriberig_project::CandidateValidation;
5
6use super::{candidate_id, validation_name};
7use crate::agent_tool::LocalAgentTool;
8
9impl<B: scriberig_app::ApplicationBackend> LocalAgentTool<'_, B> {
10    /// Acquires dependencies fixed by a candidate lockfile in isolation.
11    ///
12    /// - `arguments`: Stable candidate identity.
13    /// - `cancelled`: Shared branch cancellation flag.
14    ///
15    /// Durable dependency acquisition evidence bound to the candidate.
16    ///
17    /// Returns invalid input, stale-base, cancellation, or workspace failure.
18    /// Acquisition failures are returned as durable failed evidence.
19    /// # Returns
20    /// The operation result.
21    /// # Errors
22    /// Returns an error when the operation fails.
23    pub(crate) fn project_resolve_dependencies(
24        &self,
25        arguments: &serde_json::Value,
26        cancelled: &std::sync::atomic::AtomicBool,
27    ) -> Result<ToolResult, ToolError> {
28        let id = candidate_id(arguments)?;
29        let candidate = self
30            .project_service()?
31            .resolve_candidate_dependencies_cancellable(id, cancelled)
32            .map_err(|error| Self::project_error(&error))?;
33        let (summary, resolution) = match &candidate.validation {
34            CandidateValidation::Pending => (
35                format!(
36                    "locked dependency acquisition remains pending for candidate {}",
37                    candidate.id
38                ),
39                "pending",
40            ),
41            CandidateValidation::Passed => (
42                format!(
43                    "locked dependency acquisition passed for candidate {}",
44                    candidate.id
45                ),
46                "passed",
47            ),
48            CandidateValidation::Failed { diagnostics } => (
49                format!(
50                    "locked dependency acquisition failed for candidate {}: {}",
51                    candidate.id,
52                    diagnostics
53                        .iter()
54                        .take(8)
55                        .cloned()
56                        .collect::<Vec<_>>()
57                        .join("; ")
58                ),
59                "failed",
60            ),
61        };
62        let mut result = Self::result(
63            summary,
64            vec![
65                format!("candidate:{}", candidate.id),
66                format!("snapshot:{}", candidate.base_snapshot),
67                format!("dependency_resolution:{resolution}"),
68                format!("validation:{}", validation_name(&candidate.validation)),
69                "validation_profile:cargo-dependency-resolution".to_owned(),
70            ],
71        );
72        result.snapshot = Some(candidate.base_snapshot);
73        result.object_type = Some("candidate_dependency_resolution".to_owned());
74        Ok(result)
75    }
76}

41/// Runs provider-backed agent commands submitted through the native TUI.
42pub(super) struct NativeTuiAgentRunner {
43    /// Configured adapters keyed by the model library's safe connection IDs.
44    providers: std::collections::BTreeMap<String, NativeProvider>,
45    /// Model identifier selected by the local runtime configuration.
46    model: String,
47    /// Safe models and selectable profiles exposed to the application service.
48    model_library: scriberig_app::ModelLibrary,
49    /// Upstream catalog retained as the model selection authority.
50    upstream_models: Vec<rpi_ai::Model>,
51    /// Shared project store used by application-backed agent tools.
52    project_store: Arc<SqliteStore>,
53    /// In-process guard against starting two coordinator loops for one plan.
54    active_coordinators: Mutex<HashSet<EntityId>>,
55}
56
57impl NativeTuiAgentRunner {
58    /// Builds the TUI agent runtime from trusted process configuration.
59    ///
60    /// - `project_store`: Shared durable project and semantic store.
61    /// - `super`: Parent module containing the native runtime composition.
62    ///
63    /// A runner configured from shared application settings and optional TUI overrides.
64    ///
65    /// Returns a structured provider configuration failure.
66    /// # Returns
67    /// The operation result.
68    /// # Errors
69    /// Returns an error when the operation fails.
70    pub(super) fn from_environment(project_store: Arc<SqliteStore>) -> Result<Self, ProviderError> {
71        let provider_kind = std::env::var("SCRIBE_AGENT_PROVIDER")
72            .unwrap_or_else(|_| "openai_compatible".to_owned());
73        let shared_config = if provider_kind == "openai_compatible"
74            && std::env::var_os("SCRIBE_AGENT_BASE_URL").is_none()
75        {
76            crate::agent_config::resolve_provider_config()
77                .map_err(|error| ProviderError::Configuration(error.message))?
78                .map(|(config, _)| config)
79        } else {
80            None
81        };
82        let configured_model = shared_config
83            .as_ref()
84            .map_or("default", |config| config.model.as_str());
85        let model =
86            std::env::var("SCRIBE_AGENT_MODEL").unwrap_or_else(|_| configured_model.to_owned());
87        let catalog = provider_config::catalog::from_environment(&model)?;
88        let mut providers = std::collections::BTreeMap::new();
89        if let Some(provider) =
90            provider_config::from_environment(&provider_kind, &model, shared_config)?
91        {
92            providers.insert("local".to_owned(), provider);
93        }
94        for connection in &catalog.connections {
95            let provider = provider_config::from_connection(connection, &model)?;
96            providers.insert(connection.id.clone(), provider);
97        }
98        Ok(Self {
99            providers,
100            model,
101            model_library: catalog.library,
102            upstream_models: catalog.models,
103            project_store,
104            active_coordinators: Mutex::new(HashSet::new()),
105        })
106    }
107
108    /// Returns the safe model library installed in the application service.
109    ///
110    /// Configured model descriptors and selectable profiles.
111    /// # Returns
112    /// The operation result.
113    pub(super) fn model_library(&self) -> scriberig_app::ModelLibrary {
114        self.model_library.clone()
115    }
116
117    /// Returns the upstream catalog used for model resolution.
118    ///
119    /// # Returns
120    /// A copy of the upstream model catalog.
121    pub(super) fn upstream_models(&self) -> Vec<rpi_ai::Model> {
122        self.upstream_models.clone()
123    }
124}

126/// Resolves the latest durable project scope produced in a session.
127///
128/// - `service`: Shared application boundary owning durable session state.
129/// - `context`: Trusted authorization context for session reads.
130/// - `session`: Session receiving the next user instruction.
131///
132/// Exact project, root, selected entity, and snapshot when the session has an
133/// onboarding result.
134///
135/// Returns session read or malformed durable-handle failures.
136fn previous_workspace_selection<B: ApplicationBackend>(
137    service: &ApplicationService<B>,
138    context: &ApplicationContext,
139    session: EntityId,
140) -> Result<Option<AgentSelection>, ApplicationError> {
141    let saved = match SessionBackend::read_session(service, context, &SessionRequest { session }) {
142        Ok(saved) => saved,
143        Err(ApplicationError::NotFound { .. }) => return Ok(None),
144        Err(error) => return Err(error),
145    };
146    let Some(handles) = saved.branches.iter().rev().find_map(|branch| {
147        let handles = &branch.result.as_ref()?.result_handles;
148        ["project", "snapshot", "root"]
149            .iter()
150            .all(|kind| handle_value(handles, kind).is_some())
151            .then_some(handles)
152    }) else {
153        return Ok(None);
154    };
155    let project = handle_value(handles, "project").expect("complete workspace handle set");
156    let snapshot = handle_value(handles, "snapshot").expect("complete workspace handle set");
157    let root = handle_value(handles, "root").expect("complete workspace handle set");
158    let entity = handle_value(handles, "entity").unwrap_or(root);
159    Ok(Some(AgentSelection {
160        project: Some(parse_workspace_handle("project", project)?),
161        root: parse_workspace_handle("root", root)?,
162        entity: parse_workspace_handle("entity", entity)?,
163        snapshot: parse_workspace_handle("snapshot", snapshot)?,
164    }))
165}

167/// Parses one typed identity retained by a prior agent branch.
168///
169/// - `name`: Human-readable identity kind for diagnostics.
170/// - `value`: Durable string identity.
171///
172/// The typed identity.
173///
174/// Returns an invalid-request error when the durable identity is malformed.
175fn parse_workspace_handle<T: std::str::FromStr>(
176    name: &str,
177    value: &str,
178) -> Result<T, ApplicationError>
179where
180    T::Err: std::fmt::Display,
181{
182    value
183        .parse()
184        .map_err(|error| ApplicationError::InvalidRequest {
185            operation: "agent_continue".to_owned(),
186            message: format!("invalid durable {name} reference: {error}"),
187        })
188}

190/// Finds one value in a durable branch result handle set.
191///
192/// - `handles`: Stable branch result handles.
193/// - `kind`: Handle prefix to find.
194///
195/// The value after `kind:` when present.
196fn handle_value<'a>(handles: &'a [String], kind: &str) -> Option<&'a str> {
197    handles
198        .iter()
199        .find_map(|handle| handle.strip_prefix(&format!("{kind}:")))
200}

202/// Converts an exact workspace scope into durable result references.
203///
204/// - `selection`: Prior project context to preserve on the continuation result.
205///
206/// Stable project, snapshot, and root handles, or an empty set without scope.
207/// # Returns
208/// The operation result.
209fn selection_handles(selection: Option<&AgentSelection>) -> Vec<String> {
210    selection.map_or_else(Vec::new, |selection| {
211        let mut handles = vec![
212            format!("snapshot:{}", selection.snapshot),
213            format!("root:{}", selection.root),
214            format!("entity:{}", selection.entity),
215        ];
216        if let Some(project) = selection.project {
217            handles.push(format!("project:{project}"));
218        }
219        handles
220    })
221}

223/// Executes one model loop with the shared semantic tool and child-branch adapters.
224///
225/// - `super`: Parent executable composition module.
226/// - `provider`: Cloneable provider adapter for parent and child branches.
227/// - `model`: Configured model identifier.
228/// - `service`: Shared application operation boundary.
229/// - `project_store`: Shared project catalog and artifact store.
230/// - `context`: Trusted request capabilities and scope.
231/// - `session`: Durable session receiving branch activity.
232/// - `branch`: Exact root branch executing this turn.
233/// - `prompt`: User-authored instruction.
234///
235/// Provider-neutral runtime output and tool observations.
236///
237/// Returns a bounded runtime, provider, or tool failure.
238#[allow(clippy::too_many_arguments)]
239pub(super) fn execute_agent<P, B>(
240    provider: P,
241    model: &str,
242    thinking_level: Option<&str>,
243    service: &ApplicationService<B>,
244    project_store: &Arc<SqliteStore>,
245    context: &ApplicationContext,
246    session: EntityId,
247    branch: EntityId,
248    operation: scriberig_app::OperationId,
249    prompt: &str,
250    selection: Option<&AgentSelection>,
251    cancelled: &std::sync::atomic::AtomicBool,
252) -> Result<scriberig_agent_runtime::AgentResult, AgentError>
253where
254    P: ProviderAdapter + Clone,
255    B: ApplicationBackend,
256{
257    let cache = ModelCache::new();
258    let semantic_scope =
259        selection.map(
260            |selection| scriberig_agent_runtime::context::ProjectSemanticScope {
261                project: selection
262                    .project
263                    .map_or_else(|| "unknown".to_owned(), |project| project.to_string()),
264                snapshot: selection.snapshot.to_string(),
265                root: selection.root.to_string(),
266                selection: Some(selection.entity.to_string()),
267            },
268        );
269    let scope = ChildExecutionScope {
270        session,
271        parent_branch: branch,
272        depth: 0,
273        active_branches: Arc::new(std::sync::atomic::AtomicUsize::new(0)),
274        semantic_scope: semantic_scope.clone(),
275    };
276    let mut children = LocalChildExecutor {
277        service,
278        operation: Some(operation.clone()),
279        provider: provider.clone(),
280        model: model.to_owned(),
281        thinking_level: thinking_level.map(str::to_owned),
282        cache: cache.clone(),
283        project_store: Arc::clone(project_store),
284    };
285    let mut model_provider = ProviderModel::new(provider, model)
286        .with_thinking_level(thinking_level.map(str::to_owned))
287        .with_catalog(inspect_catalog())
288        .with_capabilities(context.capabilities.clone())
289        .with_cache(cache, CachePolicy::ReuseExact)
290        .with_cache_namespace(branch.to_string());
291    let runtime_context = selection.map_or_else(
292        || context.clone(),
293        |selection| context.clone().with_snapshot(selection.snapshot),
294    );
295    let compiled = context::compile_branch_context(
296        service,
297        &runtime_context,
298        session,
299        branch,
300        prompt,
301        semantic_scope,
302    )?;
303    model_provider = model_provider.with_compiled_context(compiled);
304    let mut runtime = AgentRuntime::new(
305        model_provider,
306        LocalAgentTool::new(service).with_project_store(Arc::clone(project_store)),
307        32,
308    )
309    .with_observed_operation(service.operations().clone(), operation)
310    .with_branch_id(branch.to_string());
311    runtime.run_with_child_executor_at(&scope, context, prompt, cancelled, &mut children)
312}

217/// Executes recursively delegated branches through the public application API.
218pub(super) struct LocalChildExecutor<'a, B, P> {
219    /// Application service owning session and operation state.
220    pub(super) service: &'a ApplicationService<B>,
221    /// Root operation that owns transient child activity.
222    pub(super) operation: Option<scriberig_app::OperationId>,
223    /// Cloneable provider adapter used for each child request.
224    pub(super) provider: P,
225    /// Model identifier shared by child provider requests.
226    pub(super) model: String,
227    /// Thinking capability snapshotted for this execution tree.
228    pub(super) thinking_level: Option<String>,
229    /// Shared exact-response cache scoped by branch namespace.
230    pub(super) cache: ModelCache,
231    /// Shared project store used by nested recovery tools.
232    pub(super) project_store: std::sync::Arc<SqliteStore>,
233}
234
235impl<B, P> ChildExecutor for LocalChildExecutor<'_, B, P>
236where
237    B: ApplicationBackend,
238    P: ProviderAdapter + Clone,
239{
240    /// Creates, executes, and completes one durable child branch.
241    ///
242    /// # Parameters
243    /// - `context`: Authorized application context inherited by the child.
244    /// - `scope`: Durable session and parent branch identity.
245    /// - `request`: Validated child purpose and prompt.
246    /// - `cancelled`: Shared cancellation flag for the execution tree.
247    ///
248    /// # Returns
249    /// Compact child summary and durable result handle.
250    ///
251    /// # Errors
252    /// Returns a safe failure when child creation or execution fails.
253    fn execute_child(
254        &mut self,
255        context: &ApplicationContext,
256        scope: Option<ChildExecutionScope>,
257        request: ChildRequest,
258        cancelled: &std::sync::atomic::AtomicBool,
259    ) -> Result<ChildResult, String> {
260        let scope = scope.ok_or_else(|| "durable child scope is missing".to_owned())?;
261        let prompt = request.prompt.clone();
262        let handle = self
263            .service
264            .start_local_child_agent(
265                scope.session,
266                scope.parent_branch,
267                "delegated agent branch",
268                format!(
269                    "{}; expected result: {}",
270                    request.purpose, request.expected_result
271                ),
272                prompt.clone(),
273            )
274            .map_err(|error| error.to_string())?;
275        let child_scope = ChildExecutionScope {
276            session: handle.session,
277            parent_branch: handle.child_branch,
278            depth: scope.depth + 1,
279            active_branches: std::sync::Arc::clone(&scope.active_branches),
280            semantic_scope: scope.semantic_scope.clone(),
281        };
282        let mut nested = Self {
283            service: self.service,
284            operation: self.operation.clone(),
285            provider: self.provider.clone(),
286            model: self.model.clone(),
287            thinking_level: self.thinking_level.clone(),
288            cache: self.cache.clone(),
289            project_store: std::sync::Arc::clone(&self.project_store),
290        };
291        let child_capabilities =
292            delegated_capabilities(&context.capabilities, &request.capabilities);
293        let child_context = ApplicationContext {
294            capabilities: child_capabilities.clone(),
295            ..context.clone()
296        };
297        let model = ProviderModel::new(self.provider.clone(), self.model.clone())
298            .with_thinking_level(self.thinking_level.clone())
299            .with_catalog(inspect_catalog())
300            .with_capabilities(child_capabilities)
301            .with_cache(self.cache.clone(), CachePolicy::ReuseExact)
302            .with_cache_namespace(child_scope.parent_branch.to_string());
303        let model = with_child_context(model, &child_scope, &request);
304        let result = scriberig_agent_runtime::AgentRuntime::new(
305            model,
306            LocalAgentTool::new(self.service)
307                .with_project_store(std::sync::Arc::clone(&self.project_store)),
308            32,
309        )
310        .with_branch_id(handle.child_branch.to_string());
311        let result = if let Some(operation) = &self.operation {
312            result.with_observed_operation(self.service.operations().clone(), operation.clone())
313        } else {
314            result
315        }
316        .run_with_child_executor_at(
317            &child_scope,
318            &child_context,
319            prompt,
320            cancelled,
321            &mut nested,
322        );
323        let result = match result {
324            Ok(result) => result,
325            Err(error) => {
326                let message = error.to_string();
327                let _ = self.service.fail_local_child_agent(&handle, &message);
328                return Err(message);
329            }
330        };
331        let result_handles = compact_result_handles(&result);
332        let completed = self
333            .service
334            .complete_local_child_agent_with_evidence(
335                &handle,
336                result.output,
337                result_handles.clone(),
338                result.tool_evidence,
339                result.model_requests,
340            )
341            .map_err(|error| error.to_string())?;
342        Ok(ChildResult {
343            summary: completed.final_output,
344            handles: std::iter::once(completed.branch_result.to_string())
345                .chain(result_handles)
346                .collect(),
347            branch: Some(handle.child_branch),
348        })
349    }
350}

352/// Adds a bounded project context to one delegated child model request.
353///
354/// # Parameters
355/// - `model`: Configured provider model for this child branch.
356/// - `scope`: Child session, branch, and optional semantic scope.
357/// - `request`: Validated child purpose and prompt.
358///
359/// # Returns
360/// The model with compiled context when a semantic scope is present.
361fn with_child_context<P>(
362    model: ProviderModel<P>,
363    scope: &ChildExecutionScope,
364    request: &ChildRequest,
365) -> ProviderModel<P> {
366    let Some(semantic_scope) = scope.semantic_scope.clone() else {
367        return model;
368    };
369    let compiled = scriberig_agent_runtime::context::ContextCompiler::new(
370        scriberig_agent_runtime::context::ContextPolicy::default(),
371    )
372    .compile(
373        &scriberig_agent_runtime::context::BranchExecutionContext::new(
374            scope.session.to_string(),
375            scope.parent_branch.to_string(),
376            &request.purpose,
377            &request.prompt,
378        )
379        .with_project_scope(semantic_scope),
380    );
381    model.with_compiled_context(compiled)
382}

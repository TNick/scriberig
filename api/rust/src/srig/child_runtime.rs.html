26/// Executes recursively delegated branches through the public application API.
27pub(super) struct LocalChildExecutor<'a, B, P> {
28    /// Application service owning session and operation state.
29    pub(super) service: &'a ApplicationService<B>,
30    /// Root operation that owns transient child activity.
31    pub(super) operation: Option<scriberig_app::OperationId>,
32    /// Cloneable provider adapter used for each child request.
33    pub(super) provider: P,
34    /// Model identifier shared by child provider requests.
35    pub(super) model: String,
36    /// Thinking capability snapshotted for this execution tree.
37    pub(super) thinking_level: Option<String>,
38    /// Shared exact-response cache scoped by branch namespace.
39    pub(super) cache: ModelCache,
40    /// Shared project store used by nested recovery tools.
41    pub(super) project_store: std::sync::Arc<SqliteStore>,
42}
43
44impl<B, P> ChildExecutor for LocalChildExecutor<'_, B, P>
45where
46    B: ApplicationBackend,
47    P: ProviderAdapter + Clone,
48{
49    /// Creates, executes, and completes one durable child branch.
50    ///
51    /// - `context`: Authorized application context inherited by the child.
52    /// - `scope`: Durable session and parent branch identity.
53    /// - `request`: Validated child purpose and prompt.
54    /// - `cancelled`: Shared cancellation flag for the execution tree.
55    ///
56    /// Compact child summary and durable result handle.
57    ///
58    /// Returns a safe failure when child creation or execution fails.
59    /// # Returns
60    /// The operation result.
61    /// # Errors
62    /// Returns an error when the operation fails.
63    fn execute_child(
64        &mut self,
65        context: &ApplicationContext,
66        scope: Option<ChildExecutionScope>,
67        request: ChildRequest,
68        cancelled: &std::sync::atomic::AtomicBool,
69    ) -> Result<ChildResult, String> {
70        let scope = scope.ok_or_else(|| "durable child scope is missing".to_owned())?;
71        let prompt = request.prompt.clone();
72        let handle = self
73            .service
74            .start_local_child_agent(
75                scope.session,
76                scope.parent_branch,
77                "delegated agent branch",
78                format!(
79                    "{}; expected result: {}",
80                    request.purpose, request.expected_result
81                ),
82                prompt.clone(),
83            )
84            .map_err(|error| error.to_string())?;
85        let child_scope = ChildExecutionScope {
86            session: handle.session,
87            parent_branch: handle.child_branch,
88            depth: scope.depth + 1,
89            active_branches: std::sync::Arc::clone(&scope.active_branches),
90            semantic_scope: scope.semantic_scope.clone(),
91        };
92        let mut nested = Self {
93            service: self.service,
94            operation: self.operation.clone(),
95            provider: self.provider.clone(),
96            model: self.model.clone(),
97            thinking_level: self.thinking_level.clone(),
98            cache: self.cache.clone(),
99            project_store: std::sync::Arc::clone(&self.project_store),
100        };
101        let child_capabilities =
102            delegated_capabilities(&context.capabilities, &request.capabilities);
103        let child_context = ApplicationContext {
104            capabilities: child_capabilities.clone(),
105            ..context.clone()
106        };
107        let model = ProviderModel::new(self.provider.clone(), self.model.clone())
108            .with_thinking_level(self.thinking_level.clone())
109            .with_catalog(inspect_catalog())
110            .with_capabilities(child_capabilities)
111            .with_cache(self.cache.clone(), CachePolicy::ReuseExact)
112            .with_cache_namespace(child_scope.parent_branch.to_string());
113        let model = with_child_context(model, &child_scope, &request);
114        let result = scriberig_agent_runtime::AgentRuntime::new(
115            model,
116            LocalAgentTool::new(self.service)
117                .with_project_store(std::sync::Arc::clone(&self.project_store)),
118            64,
119        )
120        .with_branch_id(handle.child_branch.to_string());
121        let result = if let Some(operation) = &self.operation {
122            result.with_observed_operation(self.service.operations().clone(), operation.clone())
123        } else {
124            result
125        }
126        .run_with_child_executor_at(
127            &child_scope,
128            &child_context,
129            prompt,
130            cancelled,
131            &mut nested,
132        );
133        let result = match result {
134            Ok(result) => result,
135            Err(error) => {
136                let message = error.to_string();
137                let _ = self.service.fail_local_child_agent(&handle, &message);
138                return Err(message);
139            }
140        };
141        let result_handles = compact_result_handles(&result);
142        let completed = self
143            .service
144            .complete_local_child_agent_with_evidence(
145                &handle,
146                result.output,
147                result_handles.clone(),
148                result.tool_evidence,
149                result.model_requests,
150            )
151            .map_err(|error| error.to_string())?;
152        Ok(ChildResult {
153            summary: completed.final_output,
154            handles: std::iter::once(completed.branch_result.to_string())
155                .chain(result_handles)
156                .collect(),
157            branch: Some(handle.child_branch),
158        })
159    }
160}

162/// Adds a bounded project context to one delegated child model request.
163///
164/// - `model`: Configured provider model for this child branch.
165/// - `scope`: Child session, branch, and optional semantic scope.
166/// - `request`: Validated child purpose and prompt.
167///
168/// The model with compiled context when a semantic scope is present.
169fn with_child_context<P>(
170    model: ProviderModel<P>,
171    scope: &ChildExecutionScope,
172    request: &ChildRequest,
173) -> ProviderModel<P> {
174    let Some(semantic_scope) = scope.semantic_scope.clone() else {
175        return model;
176    };
177    let compiled = scriberig_agent_runtime::context::ContextCompiler::new(
178        scriberig_agent_runtime::context::ContextPolicy::default(),
179    )
180    .compile(
181        &scriberig_agent_runtime::context::BranchExecutionContext::new(
182            scope.session.to_string(),
183            scope.parent_branch.to_string(),
184            &request.purpose,
185            &request.prompt,
186        )
187        .with_project_scope(semantic_scope),
188    );
189    model.with_compiled_context(compiled)
190}

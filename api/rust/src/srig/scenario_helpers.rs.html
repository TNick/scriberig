1//! Builds upstream message values for `ScribeRig` semantic scenarios.
2
3use rpi_ai::types::{Context, Message, UserContent};
4use rpi_ai::{AssistantMessage, Content, Model, StopReason};
5
6/// Returns the latest user-authored text in an upstream context.
7///
8/// # Parameters
9/// - `context`: Provider context supplied by `rpi-agent`.
10///
11/// # Returns
12/// The most recent user text, or an empty string when none is present.
13pub(crate) fn user_prompt(context: &Context) -> String {
14    context
15        .messages
16        .iter()
17        .rev()
18        .find_map(|message| match message {
19            Message::User(message) => match &message.content {
20                UserContent::Text(text) => Some(text.clone()),
21                UserContent::Blocks(blocks) => Some(text_content(blocks)),
22            },
23            _ => None,
24        })
25        .unwrap_or_default()
26}
27
28/// Returns text from prior tool results in provider message order.
29///
30/// # Parameters
31/// - `context`: Provider context supplied by `rpi-agent`.
32///
33/// # Returns
34/// Text blocks from tool results, joined within each message.
35pub(crate) fn tool_messages(context: &Context) -> Vec<String> {
36    context
37        .messages
38        .iter()
39        .filter_map(|message| match message {
40            Message::ToolResult(result) => Some(text_content(&result.content)),
41            _ => None,
42        })
43        .collect()
44}

46/// Creates a normal upstream final assistant message.
47///
48/// # Parameters
49/// - `model`: Exact upstream model bound to the scenario.
50/// - `text`: Provider-visible final text.
51///
52/// # Returns
53/// An upstream message with a stop finish reason.
54pub(crate) fn final_response(model: &Model, text: &str) -> AssistantMessage {
55    let mut message = AssistantMessage::empty(
56        model.api.clone(),
57        model.provider.clone(),
58        model.id.clone(),
59        0,
60    );
61    message.content.push(Content::text(text));
62    message.stop_reason = StopReason::Stop;
63    message
64}

66/// Creates an upstream assistant message requesting one semantic tool.
67///
68/// # Parameters
69/// - `model`: Exact upstream model bound to the scenario.
70/// - `name`: `ScribeRig` semantic tool name.
71/// - `arguments`: Structured tool arguments.
72///
73/// # Returns
74/// An upstream message with one tool call and a tool-use finish reason.
75pub(crate) fn tool_call(
76    model: &Model,
77    name: &str,
78    arguments: serde_json::Value,
79) -> AssistantMessage {
80    let mut message = AssistantMessage::empty(
81        model.api.clone(),
82        model.provider.clone(),
83        model.id.clone(),
84        0,
85    );
86    message
87        .content
88        .push(Content::ToolCall(rpi_ai::types::ToolCall {
89            kind: rpi_ai::types::ToolCallType,
90            id: format!("scenario-{name}"),
91            name: name.to_owned(),
92            arguments,
93            thought_signature: None,
94            namespace: None,
95        }));
96    message.stop_reason = StopReason::ToolUse;
97    message
98}

100/// Creates an upstream terminal error message for invalid scenario state.
101///
102/// # Parameters
103/// - `model`: Exact upstream model bound to the scenario.
104/// - `message`: Safe scenario failure explanation.
105///
106/// # Returns
107/// An upstream message marked with an error finish reason.
108pub(crate) fn error_response(model: &Model, message: String) -> AssistantMessage {
109    AssistantMessage::terminal(
110        model.api.clone(),
111        model.provider.clone(),
112        model.id.clone(),
113        StopReason::Error,
114        message,
115        0,
116    )
117}

119/// Joins text blocks while dropping non-text details from scenario inspection.
120///
121/// # Parameters
122/// - `blocks`: Provider content blocks in message order.
123///
124/// # Returns
125/// Text blocks separated by newlines.
126fn text_content(blocks: &[Content]) -> String {
127    blocks
128        .iter()
129        .filter_map(|block| match block {
130            Content::Text(text) => Some(text.text.as_str()),
131            _ => None,
132        })
133        .collect::<Vec<_>>()
134        .join("\n")
135}

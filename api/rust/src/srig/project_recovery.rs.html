1//! Implements agent-assisted project import recovery operations.
2
3use scriberig_agent_core::{ToolError, ToolResult};
4use scriberig_app::ApplicationBackend;
5use scriberig_project::{ProjectService, ScanOptions};
6
7use crate::agent_tool::LocalAgentTool;
8use crate::agent_tool_helpers::invalid_argument;

10impl<B: ApplicationBackend> LocalAgentTool<'_, B> {
11    /// Imports a scanned project and returns structured recovery diagnostics.
12    ///
13    /// # Parameters
14    /// - `arguments`: Provider arguments containing a project identity.
15    /// - `super`: Parent agent tool module visibility marker.
16    ///
17    /// # Returns
18    /// Import summary or diagnostic handles for source repair.
19    ///
20    /// # Errors
21    /// Returns invalid arguments or a structured import failure.
22    pub(super) fn project_import(
23        &self,
24        arguments: &serde_json::Value,
25    ) -> Result<ToolResult, ToolError> {
26        let project_id = Self::project_id(arguments)?;
27        let manifest_id = Self::manifest_id(arguments)?;
28        let project = self
29            .project_service()?
30            .import_manifest(project_id, &manifest_id);
31        match project {
32            Ok(report) => Ok(Self::result(
33                format!(
34                    "project import succeeded: {} file(s), {} entity(ies), snapshot {}",
35                    report.files, report.entities, report.snapshot
36                ),
37                vec![report.snapshot.to_string()],
38            )),
39            Err(error) => Err(Self::project_error(&error)),
40        }
41    }
42
43    /// Rescans a project after an authorized source repair.
44    ///
45    /// # Parameters
46    /// - `arguments`: Provider arguments containing project path and Git policy.
47    /// - `super`: Parent agent tool module visibility marker.
48    ///
49    /// # Returns
50    /// Scan summary and the stable project handle.
51    ///
52    /// # Errors
53    /// Returns invalid arguments or a filesystem and Git selection failure.
54    pub(super) fn project_rescan(
55        &self,
56        arguments: &serde_json::Value,
57    ) -> Result<ToolResult, ToolError> {
58        let service = self.project_service()?;
59        let path = arguments
60            .get("path")
61            .and_then(serde_json::Value::as_str)
62            .ok_or_else(|| invalid_argument("path"))?;
63        let name = arguments
64            .get("name")
65            .and_then(serde_json::Value::as_str)
66            .map(str::to_owned);
67        let git_tracked = arguments
68            .get("git_tracked")
69            .and_then(serde_json::Value::as_bool)
70            .unwrap_or(false);
71        let report = service
72            .scan_with_options(name, path, ScanOptions { git_tracked })
73            .map_err(|error| Self::project_error(&error))?;
74        Ok(Self::result(
75            format!(
76                "project rescan succeeded: {} file(s), project {}",
77                report.files.len(),
78                report.project.id
79            ),
80            vec![report.project.id.to_string()],
81        ))
82    }
83
84    /// Repairs one project-relative file before an explicit rescan and retry.
85    ///
86    /// # Parameters
87    /// - `arguments`: Provider arguments containing project, relative path, and content.
88    /// - `super`: Parent agent tool module visibility marker.
89    ///
90    /// # Returns
91    /// A stable repaired-file handle.
92    ///
93    /// # Errors
94    /// Returns invalid arguments, path traversal, missing project, or filesystem failures.
95    pub(super) fn project_repair_file(
96        &self,
97        arguments: &serde_json::Value,
98    ) -> Result<ToolResult, ToolError> {
99        let store = self
100            .project_store
101            .as_ref()
102            .ok_or_else(|| ToolError::Application {
103                category: "project_recovery_unavailable".to_owned(),
104                message: "project recovery is not configured for this runtime".to_owned(),
105            })?;
106        let project = Self::project_id(arguments)?;
107        let relative = arguments
108            .get("path")
109            .and_then(serde_json::Value::as_str)
110            .ok_or_else(|| invalid_argument("path"))?;
111        let content = arguments
112            .get("content")
113            .and_then(serde_json::Value::as_str)
114            .ok_or_else(|| invalid_argument("content"))?;
115        let relative_path = std::path::Path::new(relative);
116        if relative_path.is_absolute()
117            || relative_path
118                .components()
119                .any(|component| matches!(component, std::path::Component::ParentDir))
120        {
121            return Err(ToolError::Application {
122                category: "invalid_request".to_owned(),
123                message: "repair path must remain project-relative".to_owned(),
124            });
125        }
126        let record = store
127            .project(project)
128            .map_err(|error| ToolError::Application {
129                category: "storage_error".to_owned(),
130                message: error.to_string(),
131            })?
132            .ok_or_else(|| ToolError::Application {
133                category: "project_not_found".to_owned(),
134                message: project.to_string(),
135            })?;
136        let path = record.path.join(relative_path);
137        std::fs::write(&path, content).map_err(|error| ToolError::Application {
138            category: "filesystem_error".to_owned(),
139            message: error.to_string(),
140        })?;
141        Ok(Self::result(
142            format!("repaired project file {relative}"),
143            vec![relative.to_owned()],
144        ))
145    }

147    /// Resolves the project service required by recovery operations.
148    ///
149    /// # Parameters
150    /// - `super`: Parent agent tool module.
151    ///
152    /// # Returns
153    /// The configured project service.
154    ///
155    /// # Errors
156    /// Returns a stable unavailable error when this runtime has no project store.
157    pub(super) fn project_service(&self) -> Result<&ProjectService, ToolError> {
158        self.project.as_ref().ok_or_else(|| ToolError::Application {
159            category: "project_recovery_unavailable".to_owned(),
160            message: "project recovery is not configured for this runtime".to_owned(),
161        })
162    }

164    /// Parses a project identity from provider arguments.
165    ///
166    /// # Parameters
167    /// - `super`: Parent agent tool module.
168    /// - `arguments`: Provider argument object.
169    ///
170    /// # Returns
171    /// Parsed project identity.
172    ///
173    /// # Errors
174    /// Returns a structured invalid-request failure.
175    pub(super) fn project_id(
176        arguments: &serde_json::Value,
177    ) -> Result<scriberig_core::ProjectId, ToolError> {
178        let value = arguments
179            .get("project")
180            .and_then(serde_json::Value::as_str)
181            .ok_or_else(|| invalid_argument("project"))?;
182        value
183            .parse()
184            .map_err(|error| invalid_argument(&format!("project {value} ({error})")))
185    }
186
187    /// Parses the exact manifest digest argument.
188    ///
189    /// # Parameters
190    /// - `super`: Parent agent tool module.
191    /// - `arguments`: Provider argument object.
192    ///
193    /// # Returns
194    /// Lowercase hexadecimal manifest identity.
195    ///
196    /// # Errors
197    /// Returns invalid-request when the digest is missing or malformed.
198    pub(super) fn manifest_id(arguments: &serde_json::Value) -> Result<String, ToolError> {
199        let value = arguments
200            .get("manifest_id")
201            .and_then(serde_json::Value::as_str)
202            .ok_or_else(|| invalid_argument("manifest_id"))?;
203        if value.len() != 64 || !value.bytes().all(|byte| byte.is_ascii_hexdigit()) {
204            return Err(invalid_argument(
205                "manifest_id (expected 64 hexadecimal characters)",
206            ));
207        }
208        Ok(value.to_ascii_lowercase())
209    }

211    /// Converts a project failure into a bounded provider-facing result.
212    ///
213    /// # Parameters
214    /// - `super`: Parent agent tool module.
215    /// - `error`: Structured project failure.
216    ///
217    /// # Returns
218    /// Provider-neutral diagnostic failure.
219    pub(super) fn project_error(error: &scriberig_project::ProjectError) -> ToolError {
220        let diagnostics = error
221            .diagnostics
222            .iter()
223            .map(|diagnostic| {
224                format!(
225                    "{}:{}:{}:{}",
226                    diagnostic.path, diagnostic.adapter, diagnostic.code, diagnostic.message
227                )
228            })
229            .collect::<Vec<_>>()
230            .join(" | ");
231        ToolError::Application {
232            category: error.code.to_owned(),
233            message: format!("{} diagnostics={diagnostics}", error.message),
234        }
235    }

1//! Retains bounded references produced by one agent branch.
2
3/// Keeps recent stable handles from tool and child results.
4///
5/// # Parameters
6/// - `result`: Completed child runtime result.
7/// - `super`: Parent runtime module that owns result persistence.
8///
9/// # Returns
10/// Deduplicated handles suitable for a durable branch result.
11pub fn compact_result_handles(result: &scriberig_agent_runtime::AgentResult) -> Vec<String> {
12    let mut primary = Vec::new();
13    let mut details = Vec::new();
14    for message in result.messages.iter().rev() {
15        let recent = match message {
16            scriberig_agent_runtime::AgentMessage::Tool(tool) => &tool.handles,
17            scriberig_agent_runtime::AgentMessage::Child(child) => &child.handles,
18            scriberig_agent_runtime::AgentMessage::User(_) => continue,
19        };
20        for handle in recent {
21            if !primary.contains(handle) && !details.contains(handle) {
22                let destination = if is_primary_handle(handle) {
23                    &mut primary
24                } else {
25                    &mut details
26                };
27                if destination.len() < 32 {
28                    destination.push(handle.clone());
29                }
30            }
31        }
32        if primary.len() >= 32 {
33            break;
34        }
35    }
36    primary.extend(
37        details
38            .into_iter()
39            .take(32usize.saturating_sub(primary.len())),
40    );
41    primary
42}

44/// Prioritizes typed references that callers need to continue durable work.
45///
46/// # Parameters
47/// - `handle`: Stable result handle emitted by a tool or child branch.
48///
49/// # Returns
50/// `true` for core project, candidate, validation, or workspace identity handles.
51fn is_primary_handle(handle: &str) -> bool {
52    [
53        "candidate:",
54        "validation:",
55        "project:",
56        "snapshot:",
57        "manifest:",
58        "root:",
59        "entity:",
60        "path:",
61        "source_digest:",
62    ]
63    .iter()
64    .any(|prefix| handle.starts_with(prefix))
65}

67#[cfg(test)]
68mod tests {
69    use super::compact_result_handles;
70    use scriberig_agent_core::ToolResult;
71    use scriberig_agent_runtime::{AgentMessage, AgentResult};
72    use scriberig_core::EntityId;
73
74    /// Keeps validation and candidate references when detail handles fill the bounded result.
75    #[test]
76    fn primary_candidate_handles_survive_detail_compaction() {
77        let result = AgentResult {
78            output: String::new(),
79            tool_evidence: Vec::new(),
80            model_requests: Vec::new(),
81            messages: vec![AgentMessage::Tool(ToolResult {
82                entity: EntityId::from_bytes([0; 16]),
83                revision: None,
84                snapshot: None,
85                object_type: None,
86                summary: None,
87                handles: (0..20)
88                    .map(|index| format!("create:src/{index}.rs"))
89                    .chain(["candidate:c1".to_owned(), "validation:passed".to_owned()])
90                    .collect(),
91            })],
92        };
93
94        let handles = compact_result_handles(&result);
95        assert!(handles.contains(&"candidate:c1".to_owned()));
96        assert!(handles.contains(&"validation:passed".to_owned()));
97        assert!(handles.len() <= 32);
98    }

1//! Restricts delegated child grants to explicitly requested capabilities.
2
3use scriberig_app::{Capability, CapabilityAction, CapabilityContext};
4
5/// Restricts child grants to the explicitly delegated capability names.
6///
7/// - `parent`: Capabilities available to the parent branch.
8/// - `requested`: Capability identifiers delegated by the parent model.
9///
10/// A child capability context that is no broader than the parent request.
11/// # Returns
12/// The operation result.
13pub(super) fn delegated_capabilities(
14    parent: &CapabilityContext,
15    requested: &[String],
16) -> CapabilityContext {
17    let requested = requested
18        .iter()
19        .map(String::as_str)
20        .collect::<std::collections::BTreeSet<_>>();
21    let granted = parent
22        .granted
23        .iter()
24        .filter(|capability| requested.contains(capability.as_str()))
25        .cloned()
26        .collect();
27    let grants = parent
28        .grants
29        .iter()
30        .filter_map(|grant| {
31            let actions = grant
32                .actions
33                .iter()
34                .filter(|action| match action {
35                    CapabilityAction::Inspect => requested.contains("inspect.entity"),
36                    CapabilityAction::Query => requested.contains("query"),
37                    CapabilityAction::Custom(name) => requested.contains(name.as_str()),
38                    _ => false,
39                })
40                .cloned()
41                .collect::<std::collections::BTreeSet<_>>();
42            (!actions.is_empty()).then(|| Capability {
43                actions,
44                ..grant.clone()
45            })
46        })
47        .collect();
48    CapabilityContext { granted, grants }
49}

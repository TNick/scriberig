1//! Extracts stable semantic handles from upstream tool results.
2
3/// Parses ordered handles from a tool-result projection.
4///
5/// # Parameters
6/// - `message`: Text projected from an upstream tool result.
7///
8/// # Returns
9/// The parsed stable semantic handles.
10///
11/// # Errors
12/// Returns an error when the result omits or corrupts its handle list.
13pub(crate) fn parse_handles(message: &str) -> Result<Vec<String>, String> {
14    if let Ok(value) = serde_json::from_str::<serde_json::Value>(message)
15        && let Some(handles) = value.get("handles").and_then(serde_json::Value::as_array)
16    {
17        return Ok(handles
18            .iter()
19            .filter_map(serde_json::Value::as_str)
20            .map(str::to_owned)
21            .collect());
22    }
23    let values = message
24        .split_once("handles=[")
25        .and_then(|(_, value)| value.split_once(']'))
26        .map(|(values, _)| values)
27        .ok_or_else(|| "tool result omitted handles".to_owned())?;
28    serde_json::from_str(&format!("[{values}]"))
29        .map_err(|error| format!("invalid tool result handles: {error}"))
30}

1//! Composes CLI commands that consume the public remote protocol client.

3type Result<T> = std::result::Result<T, Box<dyn std::error::Error + Send + Sync>>;
4
5/// Creates a semantic client for a public protocol endpoint.
6///
7/// # Parameters
8/// - `endpoint`: HTTP endpoint selected by the CLI.
9///
10/// # Returns
11/// A protocol client over the endpoint.
12///
13/// # Errors
14/// Returns an endpoint parsing or transport setup error.
15fn client(endpoint: &str) -> Result<scriberig_client::Client<scriberig_client::HttpTransport>> {
16    Ok(scriberig_client::Client::new(
17        scriberig_client::HttpTransport::from_endpoint(endpoint)?,
18    ))
19}

21/// Prints capabilities exposed by a remote application.
22///
23/// # Parameters
24/// - `super`: Module namespace containing the command composition helpers.
25/// - `args`: Parsed remote capability options.
26///
27/// # Returns
28/// Unit after printing the capability response.
29///
30/// # Errors
31/// Returns a client or protocol failure.
32pub(super) fn capabilities(args: &scriberig_cli::RemoteArgs) -> Result<()> {
33    let response = client(&args.connect)?.capabilities()?;
34    if args.json {
35        println!("{}", scriberig_cli::output::render_json(&response));
36    } else {
37        for capability in response.capabilities {
38            println!(
39                "{}\t{}\t{}",
40                capability.id, capability.available, capability.description
41            );
42        }
43    }
44    Ok(())
45}

47/// Executes one remote operation lifecycle command.
48///
49/// # Parameters
50/// - `super`: Module namespace containing the command composition helpers.
51/// - `args`: Parsed operation command.
52///
53/// # Returns
54/// Unit after printing the protocol response.
55///
56/// # Errors
57/// Returns a client or protocol failure.
58pub(super) fn operation(args: scriberig_cli::OperationArgs) -> Result<()> {
59    match args.command {
60        scriberig_cli::OperationCommand::Status { operation, remote } => {
61            let response = client(&remote.connect)?.operation_status(operation)?;
62            println!("{}", scriberig_cli::output::render_json(&response));
63        }
64        scriberig_cli::OperationCommand::Cancel { operation, remote } => {
65            let response = client(&remote.connect)?.cancel_operation(operation)?;
66            println!("{}", scriberig_cli::output::render_json(&response));
67        }
68    }
69    Ok(())
70}

72/// Reads buffered lifecycle or streaming events.
73///
74/// # Parameters
75/// - `super`: Module namespace containing the command composition helpers.
76/// - `args`: Parsed event filter and cursor.
77///
78/// # Returns
79/// Unit after printing the event response.
80///
81/// # Errors
82/// Returns a client or protocol failure.
83pub(super) fn events(args: scriberig_cli::EventsArgs) -> Result<()> {
84    let events = if let Some(after) = args.after {
85        client(&args.remote.connect)?.subscribe_after(args.event_kinds, Some(after))?
86    } else {
87        client(&args.remote.connect)?.subscribe(args.event_kinds)?
88    };
89    println!("{}", scriberig_cli::output::render_json(&events));
90    Ok(())
91}

93/// Reads a durable session tree or branch activity view.
94///
95/// # Parameters
96/// - `super`: Module namespace containing the command composition helpers.
97/// - `args`: Parsed session read command.
98///
99/// # Returns
100/// Unit after printing the session response.
101///
102/// # Errors
103/// Returns a client or protocol failure.
104pub(super) fn session(args: scriberig_cli::SessionArgs) -> Result<()> {
105    match args.command {
106        scriberig_cli::SessionCommand::Tree { session, remote } => {
107            let response = client(&remote.connect)?.read_session(session)?;
108            println!("{}", scriberig_cli::output::render_json(&response));
109        }
110        scriberig_cli::SessionCommand::Activity {
111            session,
112            branch,
113            remote,
114        } => {
115            let response = client(&remote.connect)?.read_session_branch(session, branch)?;
116            println!("{}", scriberig_cli::output::render_json(&response));
117        }
118    }
119    Ok(())
120}

122/// Continues one remote agent branch.
123///
124/// # Parameters
125/// - `super`: Module namespace containing the command composition helpers.
126/// - `args`: Parsed session, branch, prompt, and endpoint.
127///
128/// # Returns
129/// Unit after printing the accepted operation handle.
130///
131/// # Errors
132/// Returns a client or protocol failure.
133pub(super) fn agent_continue(args: scriberig_cli::AgentContinueArgs) -> Result<()> {
134    let response = client(&args.remote.connect)?.continue_agent_with_handles(
135        args.session,
136        args.branch,
137        args.prompt,
138    )?;
139    if args.remote.json {
140        println!("{}", scriberig_cli::output::render_json(&response));
141    } else {
142        println!("{}", response.operation.operation.reference.id);
143    }
144    Ok(())
145}

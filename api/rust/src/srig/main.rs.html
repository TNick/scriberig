24/// Start the integrated `srig` executable and report runtime failures.
25///
26/// # Returns
27/// Unit after the selected command completes.
28///
29/// # Errors
30/// Returns an error when the API server cannot start or serve a request.
31#[allow(clippy::too_many_lines)]
32fn main() -> Result<(), Box<dyn std::error::Error + Send + Sync>> {
33    let command = scriberig_cli::parse();
34    match command {
35        scriberig_cli::Command::Serve(args) => {
36            scriberig_server::serve_with_origin(
37                &args.host,
38                args.port,
39                &args.db,
40                args.allow_origin.as_deref(),
41                &args.project_roots,
42            )?;
43        }
44        scriberig_cli::Command::Tui(args) => tui_command::run(&args)?,
45        scriberig_cli::Command::Run(args) | scriberig_cli::Command::Agent(args) => {
46            run_shared_agent(&args)?;
47        }
48        scriberig_cli::Command::Maintenance(args) => {
49            use scriberig_store::ProtectedRoots;
50            let store = SqliteStore::open(&args.db)?;
51            let report = store.integrity_check(&ProtectedRoots::default())?;
52            println!(
53                "maintenance: {} corruption issue(s), {} unreachable snapshot(s)",
54                report.corrupt.len(),
55                report.unreachable.snapshots.len()
56            );
57            if report.is_corrupt() {
58                return Err("maintenance integrity check found corruption".into());
59            }
60        }
61        scriberig_cli::Command::Inspect(args) => {
62            let store = SqliteStore::open(&args.db)?;
63            let service = ApplicationService::new(StoreInspectionBackend::new(store));
64            let service = GuardedQueryService::new(service);
65            let context = ApplicationContext {
66                capabilities: CapabilityContext::with_grants([Capability {
67                    scope: CapabilityScope::Project,
68                    actions: [CapabilityAction::Inspect]
69                        .into_iter()
70                        .collect::<BTreeSet<_>>(),
71                    constraints: CapabilityConstraints::unrestricted(),
72                }]),
73                principal: Some(scriberig_app::Principal::new("local-cli")),
74                ..ApplicationContext::empty()
75            };
76            let result =
77                scriberig_cli::dispatch(&service, &context, scriberig_cli::Command::Inspect(args));
78            match result {
79                Ok(Some(output)) => println!("{output}"),
80                Ok(None) => {}
81                Err(error) => {
82                    eprintln!("{}", error.message);
83                    return Err(format!("inspect failed with status {}", error.exit_code).into());
84                }
85            }
86        }
87        scriberig_cli::Command::Capabilities(args) => {
88            remote_commands::capabilities(&args)?;
89        }
90        scriberig_cli::Command::Operation(args) => {
91            remote_commands::operation(args)?;
92        }
93        scriberig_cli::Command::Events(args) => {
94            remote_commands::events(args)?;
95        }
96        scriberig_cli::Command::Session(args) => remote_commands::session(args)?,
97        scriberig_cli::Command::AgentContinue(args) => {
98            remote_commands::agent_continue(args)?;
99        }
100        scriberig_cli::Command::Project(args) => {
101            scriberig_cli::project::run(&args)?;
102        }
103    }
104    Ok(())
105}

107/// Executes a headless prompt through the shared application agent boundary.
108///
109/// # Parameters
110/// - `args`: Parsed prompt, database, and output settings.
111///
112/// # Returns
113/// Unit after the terminal operation result is rendered.
114///
115/// # Errors
116/// Returns an error when the local store, session operation, or output fails.
117fn run_shared_agent(
118    args: &scriberig_cli::RunArgs,
119) -> Result<(), Box<dyn std::error::Error + Send + Sync>> {
120    let store = Arc::new(SqliteStore::open(&args.db)?);
121    let backend = ApplicationBackendComposition::new(
122        StoreInspectionBackend::new(Arc::clone(&store)),
123        StoreOperationProvider::new(Arc::clone(&store)),
124    );
125    let session_store = Arc::new(scriberig_session_sqlite::SqliteSessionStore::open(
126        &args.db,
127    )?);
128    let service = ApplicationService::new(backend).with_session_store(session_store);
129    let session = fresh_session_id()?;
130    let context = ApplicationContext {
131        capabilities: scriberig_app::CapabilityContext::with_grants([Capability {
132            scope: CapabilityScope::Project,
133            actions: agent_actions(),
134            constraints: CapabilityConstraints::unrestricted(),
135        }]),
136        principal: Some(scriberig_app::Principal::new("local-agent")),
137        ..ApplicationContext::empty()
138    };
139    let cancellation = scriberig_agent_runtime::cancellation_flag();
140    let cancellation_handler = Arc::clone(&cancellation);
141    ctrlc::set_handler(move || {
142        cancellation_handler.store(true, std::sync::atomic::Ordering::Release);
143    })?;
144    let request = scriberig_app::AgentCommandRequest {
145        session,
146        branch: None,
147        prompt: args.prompt.clone(),
148    };
149    let handle = service.start_local_agent(&request)?;
150    let scope = ChildExecutionScope {
151        session,
152        parent_branch: handle.root_branch,
153        depth: 0,
154        active_branches: Arc::new(std::sync::atomic::AtomicUsize::new(0)),
155    };
156    let runtime_result =
157        match run_composed_model(args, &service, &store, &context, &scope, &cancellation) {
158            Ok(result) => result,
159            Err(error) => {
160                if matches!(error, scriberig_agent_runtime::AgentError::Cancelled) {
161                    let _ = service.operations().cancel(&handle.operation);
162                    eprintln!("agent operation cancelled: {}", handle.operation);
163                    std::process::exit(130);
164                }
165                let _ = service.operations().fail(
166                    &handle.operation,
167                    scriberig_app::ApplicationError::Internal {
168                        message: error.to_string(),
169                    },
170                );
171                return Err(Box::new(error));
172            }
173        };
174    service.complete_local_agent(&handle, runtime_result.output)?;
175    let operation = handle.operation;
176    let record = service.operations().get(&operation)?;
177    let result = record
178        .agent_result
179        .ok_or("agent operation completed without a result")?;
180    let format = args.format;
181    println!(
182        "{}",
183        scriberig_cli::output::render_agent_result(&result, &record.reference.to_string(), format,)
184    );
185    eprintln!(
186        "{}",
187        scriberig_cli::output::render_agent_diagnostics(&result, &record.reference.to_string())
188    );
189    Ok(())
190}

192/// Executes one local model loop using either the deterministic provider or a
193/// configured OpenAI-compatible provider.
194///
195/// # Parameters
196/// - `args`: Provider and prompt configuration.
197/// - `context`: Authorized application context for the runtime.
198///
199/// # Returns
200/// The completed provider-neutral runtime result.
201///
202/// # Errors
203/// Returns a provider, tool, budget, or cancellation failure.
204fn run_composed_model<B: scriberig_app::ApplicationBackend>(
205    args: &scriberig_cli::RunArgs,
206    service: &scriberig_app::ApplicationService<B>,
207    project_store: &Arc<SqliteStore>,
208    context: &ApplicationContext,
209    scope: &ChildExecutionScope,
210    cancellation: &std::sync::atomic::AtomicBool,
211) -> Result<scriberig_agent_runtime::AgentResult, scriberig_agent_runtime::AgentError> {
212    let output_callback =
213        (matches!(args.format, scriberig_cli::RunFormat::Human) && !args.no_stream).then(|| {
214            std::sync::Arc::new(|text: &str| eprintln!("agent output: {text}"))
215                as std::sync::Arc<dyn Fn(&str) + Send + Sync>
216        });
217    if let Some(base_url) = &args.base_url {
218        let credential_source = args
219            .api_key_env
220            .clone()
221            .map_or(CredentialSource::None, CredentialSource::Environment);
222        let config = ProviderConfig::new(
223            args.provider.clone(),
224            base_url,
225            args.model.clone(),
226            credential_source,
227        );
228        let provider =
229            scriberig_agent_runtime::openai_compatible::OpenAiCompatibleProvider::new(config)
230                .map_err(|error| {
231                    scriberig_agent_runtime::AgentError::Provider(error.to_string())
232                })?;
233        return execute_model(
234            provider,
235            args.model.clone(),
236            service,
237            project_store,
238            context,
239            scope,
240            &args.prompt,
241            cancellation,
242            output_callback,
243        );
244    }
245    let runtime = scriberig_agent_runtime::AgentRuntime::new(
246        scriberig_agent_runtime::EchoProvider,
247        LocalAgentTool::new(service).with_project_store(Arc::clone(project_store)),
248        8,
249    );
250    let mut runtime = if let Some(callback) = output_callback {
251        runtime.with_output_callback(callback)
252    } else {
253        runtime
254    };
255    runtime.run(context, &args.prompt, cancellation)
256}

258/// Runs one provider-neutral model through the shared runtime loop.
259///
260/// # Parameters
261/// - `provider`: Model decision provider.
262/// - `context`: Trusted application context.
263/// - `prompt`: Human instruction.
264/// - `cancellation`: Shared cancellation flag.
265///
266/// # Returns
267/// Completed runtime output and tool observations.
268///
269/// # Errors
270/// Returns the runtime's structured failure.
271#[allow(clippy::too_many_arguments)]
272fn execute_model<
273    P: scriberig_agent_runtime::provider::ProviderAdapter + Clone,
274    B: scriberig_app::ApplicationBackend,
275>(
276    provider: P,
277    model: String,
278    service: &scriberig_app::ApplicationService<B>,
279    project_store: &Arc<SqliteStore>,
280    context: &ApplicationContext,
281    scope: &ChildExecutionScope,
282    prompt: &str,
283    cancellation: &std::sync::atomic::AtomicBool,
284    output_callback: Option<scriberig_agent_runtime::OutputCallback>,
285) -> Result<scriberig_agent_runtime::AgentResult, scriberig_agent_runtime::AgentError> {
286    let catalog = inspect_catalog();
287    let cache = scriberig_agent_runtime::cache::ModelCache::new();
288    let mut children = LocalChildExecutor {
289        service,
290        provider: provider.clone(),
291        model: model.clone(),
292        cache: cache.clone(),
293        project_store: Arc::clone(project_store),
294    };
295    let runtime = scriberig_agent_runtime::AgentRuntime::new(
296        ProviderModel::new(provider, model)
297            .with_catalog(catalog)
298            .with_capabilities(context.capabilities.clone())
299            .with_cache(
300                cache,
301                scriberig_agent_runtime::cache::CachePolicy::ReuseExact,
302            )
303            .with_cache_namespace(scope.parent_branch.to_string()),
304        LocalAgentTool::new(service).with_project_store(Arc::clone(project_store)),
305        8,
306    );
307    let mut runtime = if let Some(callback) = output_callback {
308        runtime.with_output_callback(callback)
309    } else {
310        runtime
311    };
312    runtime.run_with_child_executor_at(scope, context, prompt, cancellation, &mut children)
313}

315/// Allocates a fresh session identity for one local command invocation.
316///
317/// # Returns
318/// A session identity derived from the current process time.
319///
320/// # Errors
321/// Returns an error when the system clock is before the Unix epoch.
322fn fresh_session_id() -> Result<scriberig_core::EntityId, std::time::SystemTimeError> {
323    let nanos = std::time::SystemTime::now()
324        .duration_since(std::time::UNIX_EPOCH)?
325        .as_nanos();
326    let mut bytes = [0_u8; 16];
327    bytes.copy_from_slice(&nanos.to_be_bytes());
328    Ok(scriberig_core::EntityId::from_bytes(bytes))
329}

331/// Builds the least-authority context for local read-only product surfaces.
332///
333/// # Parameters
334/// - `principal`: Stable identity assigned to the local surface.
335///
336/// # Returns
337/// An application context granting inspection only.
338fn local_read_context(principal: &str) -> ApplicationContext {
339    ApplicationContext {
340        capabilities: CapabilityContext::with_grants([Capability {
341            scope: CapabilityScope::Project,
342            actions: [
343                CapabilityAction::Inspect,
344                CapabilityAction::Custom("agent_start".to_owned()),
345                CapabilityAction::Custom("agent_continue".to_owned()),
346                CapabilityAction::Custom("session_branch_tree".to_owned()),
347            ]
348            .into_iter()
349            .collect::<BTreeSet<_>>(),
350            constraints: CapabilityConstraints::unrestricted(),
351        }]),
352        principal: Some(scriberig_app::Principal::new(principal)),
353        ..ApplicationContext::empty()
354    }
355}

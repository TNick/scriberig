10/// Builds one configured provider adapter for the TUI runtime.
11///
12/// - `provider_kind`: Trusted provider selection from process configuration.
13/// - `model`: Configured model name.
14/// - `catalog`: Safe model descriptors and provider-private thinking mappings.
15/// - `shared_config`: Provider resolved from the shared application config.
16/// - `super`: Parent native agent runtime module.
17///
18/// Configured provider, or `None` when remote configuration is incomplete.
19///
20/// Returns a structured error for an unsupported provider or scenario.
21/// # Returns
22/// The operation result.
23/// # Errors
24/// Returns an error when the operation fails.
25pub(super) fn from_environment(
26    provider_kind: &str,
27    model: &str,
28    catalog: &catalog::ProviderCatalog,
29    shared_config: Option<ProviderConfig>,
30) -> Result<Option<NativeProvider>, ProviderError> {
31    match provider_kind {
32        "deterministic" => deterministic_provider(),
33        "openai_compatible" => {
34            let config = configured_provider_config(
35                std::env::var("SCRIBE_AGENT_BASE_URL").ok(),
36                std::env::var("SCRIBE_AGENT_API_KEY_ENV").ok(),
37                model,
38                shared_config,
39            );
40            config
41                .map(|config| openai_compatible(config, catalog))
42                .transpose()
43        }
44        unknown => Err(ProviderError::Configuration(format!(
45            "unsupported SCRIBE_AGENT_PROVIDER: {unknown}"
46        ))),
47    }
48}

50/// Selects the legacy TUI override or shared application provider settings.
51///
52/// - `legacy_base_url`: Optional explicit TUI endpoint override.
53/// - `legacy_credential`: Optional environment variable name for that endpoint.
54/// - `model`: Effective model selected for the TUI request.
55/// - `shared_config`: Provider settings resolved by the shared configuration.
56///
57/// Provider settings for one TUI request, without resolving credential values.
58/// # Returns
59/// The selected safe provider configuration, when configured.
60fn configured_provider_config(
61    legacy_base_url: Option<String>,
62    legacy_credential: Option<String>,
63    model: &str,
64    shared_config: Option<ProviderConfig>,
65) -> Option<ProviderConfig> {
66    legacy_base_url
67        .map(|base_url| {
68            let credential =
69                legacy_credential.map_or(CredentialSource::None, CredentialSource::Environment);
70            ProviderConfig::new("local", base_url, model.to_owned(), credential)
71        })
72        .or_else(|| {
73            shared_config.map(|mut config| {
74                model.clone_into(&mut config.model);
75                config
76            })
77        })
78}

80/// Builds an OpenAI-compatible adapter with catalog-specific thinking mappings.
81///
82/// - `config`: Effective endpoint, model, and indirect credential reference.
83/// - `catalog`: Safe descriptors and provider-private thinking mappings.
84///
85/// The configured native provider adapter.
86///
87/// Returns an error when the provider configuration is invalid.
88/// # Returns
89/// The configured provider.
90/// # Errors
91/// Returns an error when the operation fails.
92fn openai_compatible(
93    config: ProviderConfig,
94    catalog: &catalog::ProviderCatalog,
95) -> Result<NativeProvider, ProviderError> {
96    let mut provider =
97        scriberig_agent_runtime::openai_compatible::OpenAiCompatibleProvider::new(config)?;
98    for (model, level, parameters) in &catalog.thinking_parameters {
99        provider = provider.with_model_thinking_parameters(
100            model.clone(),
101            level.clone(),
102            parameters.clone(),
103        );
104    }
105    Ok(NativeProvider::OpenAiCompatible(Box::new(provider)))
106}

108/// Builds one named OpenAI-compatible adapter from credential-free references.
109///
110/// - `connection`: Safe adapter ID and environment variable names.
111/// - `catalog`: Model descriptors and adapter-only request mappings.
112/// - `fallback_model`: Model name used for provider configuration defaults.
113///
114/// A configured provider adapter for that connection.
115///
116/// Returns a configuration failure when the endpoint is missing or unsupported.
117/// # Returns
118/// The configured provider adapter.
119/// # Errors
120/// Returns an error when the operation fails.
121pub(super) fn from_connection(
122    connection: &catalog::ProviderConnection,
123    catalog: &catalog::ProviderCatalog,
124    fallback_model: &str,
125) -> Result<NativeProvider, ProviderError> {
126    if connection.provider != "openai_compatible" {
127        return Err(ProviderError::Configuration(format!(
128            "unsupported provider family for connection {}",
129            connection.id
130        )));
131    }
132    let base_url = std::env::var(&connection.base_url_env).map_err(|_| {
133        ProviderError::Configuration(format!(
134            "provider endpoint environment variable is unavailable for {}",
135            connection.id
136        ))
137    })?;
138    let credential = connection
139        .api_key_env
140        .as_ref()
141        .map_or(CredentialSource::None, |name| {
142            CredentialSource::Environment(name.clone())
143        });
144    let config = ProviderConfig::new(
145        connection.id.clone(),
146        base_url,
147        fallback_model.to_owned(),
148        credential,
149    );
150    let mut provider =
151        scriberig_agent_runtime::openai_compatible::OpenAiCompatibleProvider::new(config)?;
152    for model in &catalog.library.models {
153        let uses_connection = catalog
154            .library
155            .profiles
156            .iter()
157            .any(|profile| profile.model == model.id && profile.connection == connection.id);
158        if !uses_connection {
159            continue;
160        }
161        for (provider_model, level, parameters) in &catalog.thinking_parameters {
162            if provider_model == &model.provider_model_id {
163                provider = provider.with_model_thinking_parameters(
164                    provider_model.clone(),
165                    level.clone(),
166                    parameters.clone(),
167                );
168            }
169        }
170    }
171    Ok(NativeProvider::OpenAiCompatible(Box::new(provider)))
172}

174/// Selects a registered deterministic provider scenario.
175///
176/// Configured deterministic scenario.
177///
178/// Returns a configuration error when the scenario identifier is unknown.
179/// # Returns
180/// The operation result.
181/// # Errors
182/// Returns an error when the operation fails.
183fn deterministic_provider() -> Result<Option<NativeProvider>, ProviderError> {
184    let scenario =
185        std::env::var("SCRIBE_AGENT_SCENARIO").unwrap_or_else(|_| "bootstrap".to_owned());
186    let provider = match scenario.as_str() {
187        "arithmetic" => {
188            NativeProvider::Arithmetic(crate::deterministic_provider::ArithmeticProvider)
189        }
190        "bootstrap" => {
191            NativeProvider::Deterministic(crate::deterministic_provider::BootstrapProvider)
192        }
193        "bootstrap_import_failure" => NativeProvider::BootstrapImportFailure(
194            crate::bootstrap_failure_provider::BootstrapImportFailureProvider,
195        ),
196        "self_hosting_php" => self_hosting(ValidationMode::Normal, None),
197        "self_hosting_php_validation_failure" => self_hosting(ValidationMode::FailFirst, None),
198        "self_hosting_php_validation_repair" => self_hosting(ValidationMode::RepairFirst, None),
199        "self_hosting_php_cancellation" => {
200            self_hosting(ValidationMode::Normal, Some(CancellationPhase::Onboarding))
201        }
202        "self_hosting_php_discovery_cancellation" => {
203            self_hosting(ValidationMode::Normal, Some(CancellationPhase::Discovery))
204        }
205        "self_hosting_php_implementation_cancellation" => self_hosting(
206            ValidationMode::Normal,
207            Some(CancellationPhase::Implementation),
208        ),
209        unknown => {
210            return Err(ProviderError::Configuration(format!(
211                "unsupported SCRIBE_AGENT_SCENARIO: {unknown}"
212            )));
213        }
214    };
215    Ok(Some(provider))
216}

218/// Builds a PHP self-hosting scenario with explicit validation and cancellation behavior.
219///
220/// - `validation`: First-candidate validation behavior.
221/// - `cancellation`: Optional delayed operation phase.
222///
223/// Provider-neutral deterministic PHP scenario.
224/// # Returns
225/// The operation result.
226fn self_hosting(
227    validation: ValidationMode,
228    cancellation: Option<CancellationPhase>,
229) -> NativeProvider {
230    NativeProvider::SelfHostingPhp(crate::self_hosting_provider::PhpSelfHostingProvider {
231        validation,
232        cancellation,
233    })
234}

237mod tests {
238    use super::configured_provider_config;
239    use scriberig_agent_runtime::provider::{CredentialSource, ProviderConfig};

241    /// Uses shared doctor configuration when no TUI-specific endpoint is set.
242    #[test]
243    fn shared_provider_configuration_is_used_by_tui() {
244        let shared = ProviderConfig::new(
245            "llm-strata",
246            "http://localhost:1234/v1",
247            "configured-model",
248            CredentialSource::Environment("API_TOKEN".to_owned()),
249        );
250
251        let selected = configured_provider_config(None, None, "configured-model", Some(shared))
252            .expect("shared provider settings");
253
254        assert_eq!(selected.profile_id, "llm-strata");
255        assert_eq!(selected.base_url, "http://localhost:1234/v1");
256        assert_eq!(selected.model, "configured-model");
257        assert!(matches!(
258            selected.credential_source,
259            CredentialSource::Environment(ref name) if name == "API_TOKEN"
260        ));
261    }
262
263    /// Keeps an explicit TUI endpoint ahead of shared provider configuration.
264    #[test]
265    fn explicit_tui_provider_configuration_remains_an_override() {
266        let selected = configured_provider_config(
267            Some("http://localhost:4321/v1".to_owned()),
268            Some("TUI_TOKEN".to_owned()),
269            "tui-model",
270            Some(ProviderConfig::new(
271                "shared",
272                "http://localhost:1234/v1",
273                "shared-model",
274                CredentialSource::None,
275            )),
276        )
277        .expect("TUI override");
278
279        assert_eq!(selected.profile_id, "local");
280        assert_eq!(selected.base_url, "http://localhost:4321/v1");
281        assert_eq!(selected.model, "tui-model");
282        assert!(matches!(
283            selected.credential_source,
284            CredentialSource::Environment(ref name) if name == "TUI_TOKEN"
285        ));
286    }

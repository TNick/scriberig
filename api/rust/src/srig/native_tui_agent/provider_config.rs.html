10/// Builds one configured provider adapter for the TUI runtime.
11///
12/// - `provider_kind`: Trusted provider selection from process configuration.
13/// - `model`: Configured model name.
14/// - `catalog`: Safe model descriptors and provider-private thinking mappings.
15/// - `super`: Parent native agent runtime module.
16///
17/// Configured provider, or `None` when remote configuration is incomplete.
18///
19/// Returns a structured error for an unsupported provider or scenario.
20/// # Returns
21/// The operation result.
22/// # Errors
23/// Returns an error when the operation fails.
24pub(super) fn from_environment(
25    provider_kind: &str,
26    model: &str,
27    catalog: &catalog::ProviderCatalog,
28) -> Result<Option<NativeProvider>, ProviderError> {
29    match provider_kind {
30        "deterministic" => deterministic_provider(),
31        "openai_compatible" => std::env::var("SCRIBE_AGENT_BASE_URL")
32            .ok()
33            .map(|base_url| {
34                let credential = std::env::var("SCRIBE_AGENT_API_KEY_ENV")
35                    .map_or(CredentialSource::None, CredentialSource::Environment);
36                let config = ProviderConfig::new("local", base_url, model.to_owned(), credential);
37                let mut provider =
38                    scriberig_agent_runtime::openai_compatible::OpenAiCompatibleProvider::new(
39                        config,
40                    )?;
41                for (model, level, parameters) in &catalog.thinking_parameters {
42                    provider = provider.with_model_thinking_parameters(
43                        model.clone(),
44                        level.clone(),
45                        parameters.clone(),
46                    );
47                }
48                Ok(NativeProvider::OpenAiCompatible(Box::new(provider)))
49            })
50            .transpose(),
51        unknown => Err(ProviderError::Configuration(format!(
52            "unsupported SCRIBE_AGENT_PROVIDER: {unknown}"
53        ))),
54    }
55}

57/// Builds one named OpenAI-compatible adapter from credential-free references.
58///
59/// - `connection`: Safe adapter ID and environment variable names.
60/// - `catalog`: Model descriptors and adapter-only request mappings.
61/// - `fallback_model`: Model name used for provider configuration defaults.
62///
63/// A configured provider adapter for that connection.
64///
65/// Returns a configuration failure when the endpoint is missing or unsupported.
66/// # Returns
67/// The configured provider adapter.
68/// # Errors
69/// Returns an error when the operation fails.
70pub(super) fn from_connection(
71    connection: &catalog::ProviderConnection,
72    catalog: &catalog::ProviderCatalog,
73    fallback_model: &str,
74) -> Result<NativeProvider, ProviderError> {
75    if connection.provider != "openai_compatible" {
76        return Err(ProviderError::Configuration(format!(
77            "unsupported provider family for connection {}",
78            connection.id
79        )));
80    }
81    let base_url = std::env::var(&connection.base_url_env).map_err(|_| {
82        ProviderError::Configuration(format!(
83            "provider endpoint environment variable is unavailable for {}",
84            connection.id
85        ))
86    })?;
87    let credential = connection
88        .api_key_env
89        .as_ref()
90        .map_or(CredentialSource::None, |name| {
91            CredentialSource::Environment(name.clone())
92        });
93    let config = ProviderConfig::new(
94        connection.id.clone(),
95        base_url,
96        fallback_model.to_owned(),
97        credential,
98    );
99    let mut provider =
100        scriberig_agent_runtime::openai_compatible::OpenAiCompatibleProvider::new(config)?;
101    for model in &catalog.library.models {
102        let uses_connection = catalog
103            .library
104            .profiles
105            .iter()
106            .any(|profile| profile.model == model.id && profile.connection == connection.id);
107        if !uses_connection {
108            continue;
109        }
110        for (provider_model, level, parameters) in &catalog.thinking_parameters {
111            if provider_model == &model.provider_model_id {
112                provider = provider.with_model_thinking_parameters(
113                    provider_model.clone(),
114                    level.clone(),
115                    parameters.clone(),
116                );
117            }
118        }
119    }
120    Ok(NativeProvider::OpenAiCompatible(Box::new(provider)))
121}

123/// Selects a registered deterministic provider scenario.
124///
125/// Configured deterministic scenario.
126///
127/// Returns a configuration error when the scenario identifier is unknown.
128/// # Returns
129/// The operation result.
130/// # Errors
131/// Returns an error when the operation fails.
132fn deterministic_provider() -> Result<Option<NativeProvider>, ProviderError> {
133    let scenario =
134        std::env::var("SCRIBE_AGENT_SCENARIO").unwrap_or_else(|_| "bootstrap".to_owned());
135    let provider = match scenario.as_str() {
136        "bootstrap" => {
137            NativeProvider::Deterministic(crate::deterministic_provider::BootstrapProvider)
138        }
139        "bootstrap_import_failure" => NativeProvider::BootstrapImportFailure(
140            crate::bootstrap_failure_provider::BootstrapImportFailureProvider,
141        ),
142        "self_hosting_php" => self_hosting(ValidationMode::Normal, None),
143        "self_hosting_php_validation_failure" => self_hosting(ValidationMode::FailFirst, None),
144        "self_hosting_php_validation_repair" => self_hosting(ValidationMode::RepairFirst, None),
145        "self_hosting_php_cancellation" => {
146            self_hosting(ValidationMode::Normal, Some(CancellationPhase::Onboarding))
147        }
148        "self_hosting_php_discovery_cancellation" => {
149            self_hosting(ValidationMode::Normal, Some(CancellationPhase::Discovery))
150        }
151        "self_hosting_php_implementation_cancellation" => self_hosting(
152            ValidationMode::Normal,
153            Some(CancellationPhase::Implementation),
154        ),
155        unknown => {
156            return Err(ProviderError::Configuration(format!(
157                "unsupported SCRIBE_AGENT_SCENARIO: {unknown}"
158            )));
159        }
160    };
161    Ok(Some(provider))
162}

164/// Builds a PHP self-hosting scenario with explicit validation and cancellation behavior.
165///
166/// - `validation`: First-candidate validation behavior.
167/// - `cancellation`: Optional delayed operation phase.
168///
169/// Provider-neutral deterministic PHP scenario.
170/// # Returns
171/// The operation result.
172fn self_hosting(
173    validation: ValidationMode,
174    cancellation: Option<CancellationPhase>,
175) -> NativeProvider {
176    NativeProvider::SelfHostingPhp(crate::self_hosting_provider::PhpSelfHostingProvider {
177        validation,
178        cancellation,
179    })

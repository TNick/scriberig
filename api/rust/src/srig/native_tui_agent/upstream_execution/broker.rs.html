21/// Signals the cancellation monitor when semantic dispatch exits or unwinds.
22struct CancellationMonitorGuard {
23    /// Shared completion flag polled by the monitor thread.
24    finished: Arc<std::sync::atomic::AtomicBool>,
25}
26
27impl Drop for CancellationMonitorGuard {
28    /// Stops the monitor thread during ordinary return or panic unwinding.
29    fn drop(&mut self) {
30        self.finished
31            .store(true, std::sync::atomic::Ordering::Release);
32    }
33}

35/// Dispatches branch tool requests and settles recursive child invocations.
36///
37/// # Parameters
38/// - `service`: Borrowed host application boundary.
39/// - `context`: Trusted caller context for application operations.
40/// - `session`: Canonical session identity for durable children.
41/// - `scope`: Current `ScribeRig` branch scope.
42/// - `project_store`: Durable project store for application tools.
43/// - `invocation`: Provider/model values inherited by child branches.
44/// - `catalog`: Trusted tool catalog.
45/// - `descriptors`: Branch-visible tool metadata.
46/// - `receiver`: Scoped host broker request queue.
47/// - `evidence`: Shared ordered evidence sink.
48/// - `cancellation`: Branch cancellation token.
49/// - `output_callback`: Provider-visible assistant text sink.
50///
51/// # Returns
52/// Success after all upstream tools have been dropped and the queue closes.
53///
54/// # Errors
55/// Returns a safe broker settlement error.
56#[allow(clippy::too_many_arguments)]
57pub(super) async fn dispatch_tools<B: ApplicationBackend + Sync>(
58    service: &ApplicationService<B>,
59    context: &ApplicationContext,
60    session: EntityId,
61    scope: ChildExecutionScope,
62    project_store: Arc<SqliteStore>,
63    invocation: UpstreamBranchInvocation,
64    descriptors: Vec<ToolDescriptor>,
65    mut receiver: mpsc::UnboundedReceiver<ToolRequest>,
66    evidence: Arc<Mutex<Vec<ToolExecutionEvidence>>>,
67    cancellation: CancellationToken,
68    output_callback: Option<scriberig_agent_runtime::OutputCallback>,
69) -> Result<(), String> {
70    let local_tool = LocalAgentTool::new(service).with_project_store(Arc::clone(&project_store));
71    let mut ordinal = 0_u64;
72    let mut tool_calls = 0_usize;
73    let mut child_branches = 0_usize;
74    while let Some(request) = receiver.recv().await {
75        tool_calls = tool_calls.saturating_add(1);
76        if tool_calls
77            > scriberig_agent_runtime::orchestration::RuntimeLimits::default().max_tool_calls
78        {
79            let _ = request
80                .response
81                .send(Err(AgentError::tool("tool call limit reached")));
82            continue;
83        }
84        if serde_json::to_vec(&request.arguments)
85            .is_ok_and(|arguments| arguments.len() > MAX_TOOL_ARGUMENT_BYTES)
86        {
87            let _ = request.response.send(Err(AgentError::tool(
88                "tool arguments exceed the branch limit",
89            )));
90            continue;
91        }
92        if cancellation.is_cancelled() || request.cancellation.is_cancelled() {
93            let _ = request
94                .response
95                .send(Err(AgentError::tool("semantic operation cancelled")));
96            continue;
97        }
98        let descriptor = descriptors
99            .iter()
100            .find(|descriptor| descriptor.definition.name == request.name)
101            .ok_or_else(|| "tool descriptor disappeared from branch catalog".to_owned())?;
102        let result = if request.name == "delegate" {
103            child_branches = child_branches.saturating_add(1);
104            if child_branches
105                > scriberig_agent_runtime::orchestration::RuntimeLimits::default()
106                    .max_child_branches
107            {
108                let _ = request
109                    .response
110                    .send(Err(AgentError::tool("child branch limit reached")));
111                continue;
112            }
113            delegation::dispatch_child(
114                service,
115                context,
116                session,
117                &scope,
118                &invocation,
119                &request,
120                Arc::clone(&project_store),
121                cancellation.child_token(),
122                output_callback.clone(),
123            )
124            .await
125        } else {
126            invoke_semantic_tool(&local_tool, context, &request)
127        };
128        let failed = result.is_err();
129        let evidence_record = evidence::tool_evidence(
130            ordinal,
131            descriptor,
132            &request.arguments,
133            result.as_ref().ok(),
134            failed,
135        );
136        ordinal = ordinal.saturating_add(1);
137        if let Ok(mut records) = evidence.lock() {
138            records.push(evidence_record);
139        }
140        let response = result.map(|(value, _)| AgentToolResult::text(value));
141        let _ = request.response.send(response);
142    }
143    Ok(())
144}

146/// Invokes an authorized non-delegation semantic operation synchronously.
147///
148/// # Parameters
149/// - `tool`: Local shared application operation adapter.
150/// - `context`: Trusted caller context.
151/// - `request`: Validated upstream tool call.
152///
153/// # Returns
154/// Compact safe JSON output and an optional child branch identity.
155fn invoke_semantic_tool<B: ApplicationBackend + Sync>(
156    tool: &LocalAgentTool<'_, B>,
157    context: &ApplicationContext,
158    request: &ToolRequest,
159) -> Result<(String, Option<EntityId>), AgentError> {
160    let cancelled = Arc::new(std::sync::atomic::AtomicBool::new(
161        request.cancellation.is_cancelled(),
162    ));
163    let finished = Arc::new(std::sync::atomic::AtomicBool::new(false));
164    let token = request.cancellation.clone();
165    let monitor_cancelled = Arc::clone(&cancelled);
166    let monitor_finished = Arc::clone(&finished);
167    let result = std::thread::scope(|scope| {
168        let monitor = scope.spawn(move || {
169            while !monitor_finished.load(std::sync::atomic::Ordering::Acquire) {
170                if token.is_cancelled() {
171                    monitor_cancelled.store(true, std::sync::atomic::Ordering::Release);
172                    break;
173                }
174                std::thread::sleep(std::time::Duration::from_millis(10));
175            }
176        });
177        let _completion_guard = CancellationMonitorGuard {
178            finished: Arc::clone(&finished),
179        };
180        let result = (|| -> Result<_, AgentError> {
181            let operation = if request.name == "inspect" {
182                let entity = request.arguments["entity"]
183                    .as_str()
184                    .ok_or_else(|| AgentError::tool("invalid entity argument"))?
185                    .parse::<EntityId>()
186                    .map_err(|_| AgentError::tool("invalid entity argument"))?;
187                let snapshot = request.arguments["snapshot"]
188                    .as_str()
189                    .map(str::parse::<scriberig_core::SnapshotId>)
190                    .transpose()
191                    .map_err(|_| AgentError::tool("invalid snapshot argument"))?;
192                let detail = match request.arguments["detail"].as_str() {
193                    Some("detailed") => scriberig_agent_core::ToolDetail::Detailed,
194                    _ => scriberig_agent_core::ToolDetail::Compact,
195                };
196                scriberig_agent_core::Tool::invoke(
197                    tool,
198                    context,
199                    scriberig_agent_core::ToolInput {
200                        entity,
201                        snapshot,
202                        detail,
203                    },
204                )
205            } else {
206                scriberig_agent_core::Tool::invoke_named_cancellable(
207                    tool,
208                    context,
209                    &request.name,
210                    &request.arguments,
211                    cancelled.as_ref(),
212                )
213            };
214            operation.map_err(|_| AgentError::tool("semantic operation failed"))
215        })();
216        finished.store(true, std::sync::atomic::Ordering::Release);
217        let _ = monitor.join();
218        result
219    })?;
220    let mut handles = result.handles;
221    if handles.len() > 32 {
222        if let Some(index) = handles
223            .iter()
224            .position(|handle| handle.starts_with("next_offset:"))
225        {
226            let continuation = handles.remove(index);
227            handles.truncate(31);
228            handles.push(continuation);
229        } else {
230            handles.truncate(32);
231        }
232    }
233    let output = serde_json::json!({
234        "entity": result.entity.to_string(),
235        "revision": result.revision.map(|revision| revision.to_string()),
236        "snapshot": result.snapshot.map(|snapshot| snapshot.to_string()),
237        "object_type": result.object_type,
238        "summary": result.summary.map(|summary| evidence::bounded_text(&summary, 2_048)),
239        "handles": handles,
240    });
241    Ok((output.to_string(), None))
242}

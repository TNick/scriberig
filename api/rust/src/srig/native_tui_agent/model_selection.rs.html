1//! Resolves persisted model selections for native agent invocations.

3use scriberig_app::{
4    AgentConsoleBackend, ApplicationBackend, ApplicationContext, ApplicationError,
5    ApplicationService, ExecutionSelectionSources, ModelLibrary, SessionBackend, SessionRequest,
6    resolve_execution_selection,
7};
8use scriberig_core::EntityId;
9
10/// Resolves a session selection into native provider request settings.
11///
12/// - `service`: Shared application service owning model and session state.
13/// - `context`: Authorized caller context.
14/// - `session`: Session whose persisted selection applies to this invocation.
15/// - `fallback_model`: Runtime model used when no profile is selected.
16///
17/// Provider model identifier and the selected thinking id when available.
18///
19/// Returns a safe application error for invalid selections or an unsupported
20/// provider connection.
21pub(super) fn selected_runtime_model<B: ApplicationBackend>(
22    service: &ApplicationService<B>,
23    context: &ApplicationContext,
24    session: EntityId,
25    fallback_model: &str,
26) -> Result<(String, Option<String>, String), ApplicationError> {
27    let saved = SessionBackend::read_session(service, context, &SessionRequest { session })?;
28    let selection = saved.execution;
29    if selection.model_profile.is_none() {
30        return Ok((fallback_model.to_owned(), None, "local".to_owned()));
31    }
32    let library = AgentConsoleBackend::agent_model_library(service, context)?;
33    resolve_selected_runtime_model(&library, &selection, fallback_model)
34}
35
36/// Resolves a validated model-library selection for the native provider.
37///
38/// - `library`: Safe model and profile descriptors.
39/// - `selection`: Immutable session execution selection for this invocation.
40/// - `fallback_model`: Runtime model used when no profile is selected.
41///
42/// Provider model identifier and model-advertised thinking id, when selected.
43///
44/// Returns an application error for invalid selections or an unsupported
45/// provider connection.
46/// # Returns
47/// The operation result.
48/// # Errors
49/// Returns an error when the operation fails.
50pub(super) fn resolve_selected_runtime_model(
51    library: &ModelLibrary,
52    selection: &scriberig_app::SessionExecutionSelection,
53    fallback_model: &str,
54) -> Result<(String, Option<String>, String), ApplicationError> {
55    let (provider_model, thinking_level) = resolve_execution_selection(
56        ExecutionSelectionSources {
57            invocation: None,
58            plan_node: None,
59            branch: None,
60            session: Some(selection),
61            project: None,
62            user_profile: None,
63        },
64        library,
65        fallback_model,
66    )?;
67    let Some(profile_id) = selection.model_profile.as_deref() else {
68        return Ok((provider_model, thinking_level, "local".to_owned()));
69    };
70    let profile = library
71        .profile(profile_id)
72        .filter(|profile| profile.enabled)
73        .ok_or_else(|| ApplicationError::InvalidRequest {
74            operation: "agent_start".to_owned(),
75            message: "selected model profile is unavailable".to_owned(),
76        })?;
77    let model = library
78        .model(&profile.model)
79        .ok_or_else(|| ApplicationError::Internal {
80            message: "selected model profile has no model descriptor".to_owned(),
81        })?;
82    if selection
83        .model
84        .as_deref()
85        .is_some_and(|selected| selected != model.id.as_str())
86    {
87        return Err(ApplicationError::InvalidRequest {
88            operation: "agent_start".to_owned(),
89            message: "selected model profile no longer matches the session model".to_owned(),
90        });
91    }
92    Ok((provider_model, thinking_level, profile.connection.clone()))
93}

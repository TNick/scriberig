1//! Resolves persisted model selections for native agent invocations.
2
3use scriberig_app::{
4    ApplicationBackend, ApplicationContext, ApplicationError, ApplicationService,
5    ExecutionSelectionSources, SessionBackend, SessionRequest, resolve_execution_selection_typed,
6};
7use scriberig_core::EntityId;
8
9/// Holds the exact upstream model and product selection derived together.
10pub(super) struct ResolvedNativeModel {
11    /// Exact catalog model selected for provider construction.
12    pub model: rpi_ai::Model,
13    /// Upstream-clamped `ScribeRig` thinking assignment.
14    pub thinking_level: Option<rpi_ai::ThinkingLevel>,
15    /// Host connection identity for credentials and compatibility paths.
16    pub connection: String,
17}
18
19/// Resolves a session selection into native provider request settings.
20///
21/// - `service`: Shared application service owning model and session state.
22/// - `context`: Authorized caller context.
23/// - `session`: Session whose persisted selection applies to this invocation.
24/// - `fallback_model`: Runtime model used when no profile is selected.
25///
26/// Upstream provider/model identity and the selected thinking id.
27///
28/// Returns a safe application error for invalid selections or an unsupported
29/// provider connection.
30pub(super) fn selected_runtime_model<B: ApplicationBackend>(
31    service: &ApplicationService<B>,
32    context: &ApplicationContext,
33    session: EntityId,
34    fallback_model: &str,
35) -> Result<ResolvedNativeModel, ApplicationError> {
36    let saved = SessionBackend::read_session(service, context, &SessionRequest { session })?;
37    let selection = saved.execution;
38    let upstream_models = service.upstream_models();
39    resolve_selected_runtime_model_with_model(&upstream_models, &selection, fallback_model)
40}

42/// Resolves an upstream model selection for the native provider.
43///
44/// - `upstream_models`: Upstream models used to validate the selection.
45/// - `selection`: Immutable session execution selection for this invocation.
46/// - `fallback_model`: Runtime model used when no profile is selected.
47///
48/// Upstream provider/model identity, thinking id, and native adapter identity.
49///
50/// Returns an application error for invalid selections or an unsupported
51/// provider connection.
52/// # Returns
53/// The operation result.
54/// # Errors
55/// Returns an error when the operation fails.
56#[cfg(test)]
57pub(super) fn resolve_selected_runtime_model(
58    upstream_models: &[rpi_ai::Model],
59    selection: &scriberig_app::SessionExecutionSelection,
60    fallback_model: &str,
61) -> Result<(String, String, Option<String>, String), ApplicationError> {
62    let resolved =
63        resolve_selected_runtime_model_with_model(upstream_models, selection, fallback_model)?;
64    Ok((
65        resolved.model.provider,
66        resolved.model.id,
67        resolved
68            .thinking_level
69            .map(|level| level.as_str().to_owned()),
70        resolved.connection,
71    ))
72}

74/// Resolves a model and its session settings from one catalog snapshot.
75///
76/// - `upstream_models`: Upstream models used to validate the selection.
77/// - `selection`: Immutable session execution selection for this invocation.
78/// - `fallback_model`: Runtime model used when no profile is selected.
79///
80/// The exact resolved model, thinking assignment, and connection identity.
81///
82/// # Returns
83/// The exact upstream model and effective session selections.
84///
85/// # Errors
86/// Returns an application error for invalid selections.
87pub(super) fn resolve_selected_runtime_model_with_model(
88    upstream_models: &[rpi_ai::Model],
89    selection: &scriberig_app::SessionExecutionSelection,
90    fallback_model: &str,
91) -> Result<ResolvedNativeModel, ApplicationError> {
92    let (provider_id, model_id, thinking_level) = resolve_execution_selection_typed(
93        ExecutionSelectionSources {
94            invocation: None,
95            plan_node: None,
96            branch: None,
97            session: Some(selection),
98            project: None,
99            user_profile: None,
100        },
101        upstream_models,
102        fallback_model,
103    )?;
104    let model = upstream_models
105        .iter()
106        .find(|model| model.provider == provider_id && model.id == model_id)
107        .cloned()
108        .ok_or_else(|| ApplicationError::InvalidRequest {
109            operation: "agent_start".to_owned(),
110            message: "resolved model is missing from the upstream catalog".to_owned(),
111        })?;
112    let connection = if selection.provider_id.is_some() {
113        super::provider_config::catalog::connection_identity(&provider_id, &model_id)
114    } else {
115        "local".to_owned()
116    };
117    Ok(ResolvedNativeModel {
118        model,
119        thinking_level,
120        connection,
121    })
122}

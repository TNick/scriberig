4use scriberig_agent_runtime::host_configuration::HostConfigurationError;

8impl NativeTuiAgentRunner {
9    /// Reads upstream descriptors for a persisted provider model.
10    ///
11    /// # Parameters
12    /// - `provider_id`: Exact upstream provider identity.
13    /// - `model_id`: Provider-native model identity.
14    ///
15    /// # Returns
16    /// Safe upstream property descriptors for an editable config row.
17    ///
18    /// # Errors
19    /// Returns an error when the row is read-only or has an invalid schema.
20    pub(super) fn model_properties(
21        &self,
22        provider_id: &str,
23        model_id: &str,
24    ) -> Result<Vec<rpi_model_runtime::ModelPropertyDescriptor>, HostConfigurationError> {
25        {
26            let configuration = self.model_config.lock().map_err(|_| {
27                HostConfigurationError::Configuration(
28                    "model configuration is unavailable".to_owned(),
29                )
30            })?;
31            require_editable_row(
32                &configuration.0,
33                configuration.1.is_some(),
34                provider_id,
35                model_id,
36            )?;
37        }
38        let model = self
39            .model_catalog
40            .snapshot()
41            .models
42            .into_iter()
43            .find(|model| model.provider == provider_id && model.id == model_id)
44            .ok_or_else(|| {
45                HostConfigurationError::Configuration("configured model is unavailable".to_owned())
46            })?;
47        rpi_model_runtime::model_property_descriptors(&model).map_err(|_| {
48            HostConfigurationError::Configuration(
49                "upstream model property schema is invalid".to_owned(),
50            )
51        })
52    }
53
54    /// Validates, persists, and publishes one upstream model property edit.
55    ///
56    /// # Parameters
57    /// - `provider_id`: Exact upstream provider identity.
58    /// - `model_id`: Provider-native model identity.
59    /// - `key`: Provider-declared property key.
60    /// - `value`: User value passed to upstream validation.
61    ///
62    /// # Errors
63    /// Returns an error when the row is read-only or upstream rejects the edit.
64    ///
65    /// # Returns
66    /// `Ok(())` after the validated property is persisted.
67    pub(super) fn set_model_property(
68        &self,
69        provider_id: &str,
70        model_id: &str,
71        key: &str,
72        value: serde_json::Value,
73    ) -> Result<(), HostConfigurationError> {
74        let model = self
75            .model_catalog
76            .snapshot()
77            .models
78            .into_iter()
79            .find(|model| model.provider == provider_id && model.id == model_id)
80            .ok_or_else(|| {
81                HostConfigurationError::Configuration("configured model is unavailable".to_owned())
82            })?;
83        let safe_property = rpi_model_runtime::model_property_descriptors(&model)
84            .map_err(|_| {
85                HostConfigurationError::Configuration(
86                    "upstream model property schema is invalid".to_owned(),
87                )
88            })?
89            .into_iter()
90            .find(|property| property.key == key)
91            .is_some_and(|property| property.is_safe_for_generic_editor());
92        if !safe_property {
93            return Err(HostConfigurationError::Configuration(
94                "model property is not available in the generic editor".to_owned(),
95            ));
96        }
97
98        let mut configuration = self.model_config.lock().map_err(|_| {
99            HostConfigurationError::Configuration("model configuration is unavailable".to_owned())
100        })?;
101        let (config, path) = &mut *configuration;
102        require_editable_row(config, path.is_some(), provider_id, model_id)?;
103        let path = path.as_deref().ok_or_else(read_only_configuration)?;
104
105        let mut updated_config = config.clone();
106        rpi_model_runtime::update_model_config_property(
107            &mut updated_config,
108            provider_id,
109            model_id,
110            key,
111            value,
112        )
113        .map_err(|_| {
114            HostConfigurationError::Configuration(
115                "upstream model configuration update failed".to_owned(),
116            )
117        })?;
118        let endpoint_config = updated_config.clone();
119        let previous_models = self.model_catalog.snapshot().models;
120        let resolver = super::provider_config::catalog::EnvironmentResolver;
121        self.model_catalog.refresh_from_configuration_with_commit(
122            &mut updated_config,
123            &[],
124            &resolver,
125            |models| {
126                rpi_model_runtime::resolve_model_endpoints(models, &endpoint_config, &resolver)
127                    .map_err(|_| {
128                        HostConfigurationError::Configuration(
129                            "configured model endpoint could not be resolved".to_owned(),
130                        )
131                    })?;
132                preserve_host_endpoints(models, &endpoint_config, &previous_models);
133                for model in models {
134                    model.headers = None;
135                }
136                Ok(())
137            },
138            |candidate| {
139                rpi_model_runtime::ModelsConfigFile::save(path, candidate).map_err(|_| {
140                    HostConfigurationError::Configuration(
141                        "upstream model configuration persistence failed".to_owned(),
142                    )
143                })
144            },
145        )?;
146        *config = updated_config;
147        Ok(())
148    }
149}

151/// Retains resolved host endpoint overrides when config has no endpoint value.
152///
153/// The fallback endpoint override is process configuration rather than a
154/// provider model property, so a property edit must not reset it while
155/// rebuilding the upstream catalog.
156///
157/// # Parameters
158/// - `models`: Candidate upstream catalog rows being refreshed.
159/// - `config`: Candidate upstream model config.
160/// - `previous_models`: Previously published rows with resolved host endpoints.
161fn preserve_host_endpoints(
162    models: &mut [rpi_ai::Model],
163    config: &ModelsConfig,
164    previous_models: &[rpi_ai::Model],
165) {
166    for model in models {
167        let configured_endpoint = config.providers.get(&model.provider).and_then(|provider| {
168            provider
169                .models
170                .iter()
171                .find(|definition| definition.id == model.id)
172                .and_then(|definition| definition.base_url.as_deref())
173                .or(provider.base_url.as_deref())
174        });
175        if configured_endpoint.is_some() {
176            continue;
177        }
178        if let Some(previous) = previous_models
179            .iter()
180            .find(|previous| previous.provider == model.provider && previous.id == model.id)
181        {
182            model.base_url.clone_from(&previous.base_url);
183        }
184    }
185}

187/// Requires a file-backed config row before displaying or saving properties.
188///
189/// # Parameters
190/// - `config`: Upstream configuration selected by the host.
191/// - `has_persistent_path`: Whether the host selected a writable config file.
192/// - `provider_id`: Exact upstream provider identity.
193/// - `model_id`: Provider-native model identity.
194///
195/// # Errors
196/// Returns a generic read-only error when this source cannot be edited.
197///
198/// # Returns
199/// `Ok(())` when the model row is editable.
200fn require_editable_row(
201    config: &ModelsConfig,
202    has_persistent_path: bool,
203    provider_id: &str,
204    model_id: &str,
205) -> Result<(), HostConfigurationError> {
206    let has_row = config
207        .providers
208        .get(provider_id)
209        .is_some_and(|provider| provider.models.iter().any(|model| model.id == model_id));
210    if has_persistent_path && has_row {
211        Ok(())
212    } else {
213        Err(read_only_configuration())
214    }
215}

217/// Creates a safe error for sources that are not file-backed config rows.
218///
219/// # Returns
220/// A generic provider configuration error.
221fn read_only_configuration() -> HostConfigurationError {
222    HostConfigurationError::Configuration("model properties are read-only".to_owned())

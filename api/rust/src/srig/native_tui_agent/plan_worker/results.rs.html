12/// Builds the durable result for a completed model-backed plan worker.
13///
14/// - `branch`: Exact worker branch that produced the result.
15/// - `execution`: Resolved provider, model, and thinking assignment.
16/// - `result`: Runtime output and captured model request projections.
17///
18/// # Returns
19/// A completed branch result with safe model provenance handles.
20///
21/// # Errors
22/// This helper is infallible; `PlanWorkerExecutionResult` is a value type.
23pub(super) fn completed_worker_result(
24    branch: EntityId,
25    execution: &scriberig_app::ResolvedPlanExecution,
26    result: scriberig_agent_runtime::AgentResult,
27) -> PlanWorkerExecutionResult {
28    let mut result_handles = compact_result_handles(&result);
29    result_handles.push(format!("provider:{}", execution.provider_id));
30    result_handles.push(format!("model:{}", execution.model_id));
31    if let Some(thinking) = execution.thinking_level {
32        result_handles.push(format!("thinking:{}", thinking.as_str()));
33    }
34    PlanWorkerExecutionResult {
35        result: BranchResult {
36            id: EntityId::new(),
37            branch,
38            status: BranchStatus::Completed,
39            summary: Some(result.output),
40            result_kind: Some("plan-worker".to_owned()),
41            candidate_context: None,
42            references: Vec::new(),
43            result_handles,
44            touched: Vec::new(),
45            produced_changes: Vec::new(),
46            validations: Vec::new(),
47            unresolved_questions: Vec::new(),
48        },
49        model_requests: result.model_requests,
50        tool_evidence: result.tool_evidence,
51    }
52}

54/// Creates the bounded failed result used when captured requests precede error.
55///
56/// - `branch`: Exact claimed branch that executed the failed request.
57///
58/// # Returns
59/// An abandoned worker result that cannot satisfy plan dependencies.
60///
61/// # Errors
62/// This helper is infallible; failure is represented in the returned result.
63pub(super) fn failed_plan_worker_result(branch: EntityId) -> BranchResult {
64    BranchResult {
65        id: EntityId::new(),
66        branch,
67        status: BranchStatus::Abandoned,
68        summary: Some("worker execution failed".to_owned()),
69        result_kind: Some("worker_failure".to_owned()),
70        candidate_context: None,
71        references: Vec::new(),
72        result_handles: vec!["failure-code:worker_failed".to_owned()],
73        touched: Vec::new(),
74        produced_changes: Vec::new(),
75        validations: Vec::new(),
76        unresolved_questions: Vec::new(),
77    }
78}

80/// Merges the exact candidate handles admitted from fulfilled dependencies.
81///
82/// This operation creates a reviewable candidate at the shared base snapshot;
83/// validation and explicit project application remain separate operations.
84///
85/// - `project_store`: Durable project and candidate owner.
86/// - `claim`: Exclusive worker branch claim.
87/// - `worker_context`: Exact dependency results compiled for this node.
88///
89/// # Returns
90/// A completed branch result containing the newly merged candidate handle.
91///
92/// # Errors
93/// Returns an error when merge inputs are invalid or conflict.
94pub(super) fn integrate_dependency_candidates(
95    project_store: &Arc<SqliteStore>,
96    claim: &WorkerClaim,
97    worker_context: &scriberig_agent_runtime::plan_context::PlanWorkerContext,
98) -> Result<BranchResult, String> {
99    let candidates = worker_context
100        .inputs
101        .dependency_results
102        .iter()
103        .flat_map(|result| &result.semantic_handles)
104        .filter_map(|handle| handle.strip_prefix("candidate:"))
105        .collect::<Vec<_>>();
106    let [left, right] = candidates.as_slice() else {
107        return Err(format!(
108            "candidate_merge_invalid: integration node requires exactly two dependency candidate handles; received {}",
109            candidates.len()
110        ));
111    };
112    let merged = scriberig_project::ProjectService::new(Arc::clone(project_store))
113        .merge_project_candidates(left, right)
114        .map_err(|error| format!("{}: {}", error.code, error.message))?;
115    let summary = format!(
116        "integrated candidate {} combines {} and {} at exact base {}",
117        merged.id, left, right, merged.base_snapshot
118    );
119    Ok(BranchResult {
120        id: EntityId::new(),
121        branch: claim.branch,
122        status: BranchStatus::Completed,
123        summary: Some(summary),
124        result_kind: Some("project_candidate_merge".to_owned()),
125        candidate_context: None,
126        references: Vec::new(),
127        result_handles: vec![
128            format!("candidate:{}", merged.id),
129            format!("candidate_parent:{left}"),
130            format!("candidate_parent:{right}"),
131            format!("project:{}", merged.project),
132            format!("snapshot:{}", merged.base_snapshot),
133        ],
134        touched: Vec::new(),
135        produced_changes: Vec::new(),
136        validations: Vec::new(),
137        unresolved_questions: Vec::new(),
138    })
139}

1//! Publishes provider-visible reasoning-summary activity for one TUI root branch.
2
3use std::sync::atomic::{AtomicU64, Ordering};
4use std::sync::{Arc, Mutex};
5
6use scriberig_app::OperationRegistry;
7use scriberig_core::OperationId;
8use scriberig_session::{AgentActivityKind, AgentActivityState};
9
10/// Maximum summary characters retained for transient presentation.
11const MAX_SUMMARY_CHARS: usize = 256;
12
13/// Host sink retaining one invocation's bounded reasoning-summary progress.
14///
15/// The runtime only forwards upstream `ReasoningSummaryDelta` text here; raw
16/// thinking deltas and hidden chain-of-thought never reach this sink.
17///
18/// # Public members
19/// None.
20///
21/// # Private members
22/// - `registry`: Shared operation event registry.
23/// - `operation`: Active operation identity.
24/// - `branch`: Durable branch identity associated with the activity.
25/// - `text`: Accumulated safe summary text.
26/// - `ordinal`: Last published per-invocation arrival ordinal.
27pub(super) struct ReasoningSummaryActivity {
28    /// Registry receiving bounded transient activity events.
29    registry: OperationRegistry,
30    /// Operation that owns the activity event.
31    operation: OperationId,
32    /// Durable branch identity for the activity.
33    branch: String,
34    /// Accumulated provider-visible summary text.
35    text: Mutex<String>,
36    /// Highest observed per-invocation ordinal.
37    ordinal: AtomicU64,
38}
39
40impl ReasoningSummaryActivity {
41    /// Creates a summary sink for one accepted root invocation.
42    ///
43    /// # Parameters
44    /// - `registry`: Operation registry receiving activity events.
45    /// - `operation`: Exact active operation identity.
46    /// - `branch`: Exact durable root branch identity.
47    ///
48    /// # Returns
49    /// A shared sink suitable for the runtime summary callback.
50    pub(super) fn new(
51        registry: OperationRegistry,
52        operation: OperationId,
53        branch: String,
54    ) -> Arc<Self> {
55        Arc::new(Self {
56            registry,
57            operation,
58            branch,
59            text: Mutex::new(String::new()),
60            ordinal: AtomicU64::new(0),
61        })
62    }
63
64    /// Returns a runtime callback that records safe summary fragments in order.
65    ///
66    /// # Returns
67    /// A callback accepting an invocation ordinal and safe summary delta.
68    pub(super) fn callback(self: &Arc<Self>) -> scriberig_agent_runtime::ReasoningSummaryCallback {
69        let sink = Arc::clone(self);
70        Arc::new(move |ordinal, delta| sink.observe(ordinal, delta))
71    }
72
73    /// Publishes the settled summary once the invocation returns.
74    pub(super) fn settle(&self) {
75        let Ok(text) = self.text.lock() else {
76            return;
77        };
78        if text.is_empty() {
79            return;
80        }
81        let summary = text.clone();
82        drop(text);
83        let ordinal = self
84            .ordinal
85            .fetch_add(1, Ordering::Relaxed)
86            .saturating_add(1);
87        let _ = self.registry.publish_agent_activity(
88            &self.operation,
89            self.branch.clone(),
90            AgentActivityKind::ReasoningSummary,
91            AgentActivityState::Completed,
92            summary,
93            ordinal,
94        );
95    }
96
97    /// Appends one safe fragment and publishes bounded progress.
98    ///
99    /// # Parameters
100    /// - `ordinal`: Monotonic per-invocation arrival ordinal.
101    /// - `delta`: Safe provider-visible summary fragment.
102    fn observe(&self, ordinal: u64, delta: &str) {
103        let Ok(mut text) = self.text.lock() else {
104            return;
105        };
106        text.push_str(delta);
107        if text.chars().count() > MAX_SUMMARY_CHARS {
108            *text = text.chars().take(MAX_SUMMARY_CHARS).collect();
109        }
110        let summary = text.clone();
111        drop(text);
112        self.ordinal.fetch_max(ordinal, Ordering::Relaxed);
113        let _ = self.registry.publish_agent_activity(
114            &self.operation,
115            self.branch.clone(),
116            AgentActivityKind::ReasoningSummary,
117            AgentActivityState::Running,
118            summary,
119            ordinal,
120        );
121    }
122}

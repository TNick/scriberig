1//! Persists successful upstream branch results as `ScribeRig` activity.
2
3use scriberig_app::{ApplicationBackend, ApplicationError, ApplicationService};
4
5use super::super::{selection_handles, workspace_from_result};
6
7/// Persists the successful root-branch result and returns its TUI handle.
8///
9/// - `service`: Application service owning durable agent activity.
10/// - `handle`: Accepted operation and root branch identity.
11/// - `session`: Session that owns the completed branch.
12/// - `selection`: Workspace selection used to build the execution context.
13/// - `result`: Completed agent output and safe execution evidence.
14///
15/// # Returns
16/// The stable operation, session, branch, and workspace handles.
17///
18/// # Errors
19/// Returns an error when result projection or durable completion fails.
20pub(super) fn complete_accepted_result<B: ApplicationBackend>(
21    service: &ApplicationService<B>,
22    handle: &scriberig_app::LocalAgentHandle,
23    session: scriberig_core::EntityId,
24    selection: Option<&scriberig_app::AgentSelection>,
25    result: scriberig_agent_runtime::AgentResult,
26) -> Result<scriberig_tui::AgentHandle, ApplicationError> {
27    let workspace = workspace_from_result(&result)?;
28    let mut result_handles = crate::child_runtime::compact_result_handles(&result);
29    let workspace_handles = workspace.as_ref().map_or_else(
30        || selection_handles(selection),
31        |workspace| {
32            vec![
33                format!("project:{}", workspace.project),
34                format!("snapshot:{}", workspace.snapshot),
35                format!("root:{}", workspace.root),
36                format!("manifest:{}", workspace.manifest_id),
37            ]
38        },
39    );
40    for handle in workspace_handles {
41        if !result_handles.contains(&handle) {
42            result_handles.push(handle);
43        }
44    }
45    service.complete_local_agent_with_evidence(
46        handle,
47        result.output,
48        result_handles,
49        result.tool_evidence,
50        result.model_requests,
51    )?;
52    Ok(scriberig_tui::AgentHandle {
53        operation: handle.operation.clone(),
54        session,
55        branch: Some(handle.root_branch),
56        workspace,
57        failure: None,
58    })
59}

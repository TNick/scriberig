13/// Holds authoritative upstream models and a credential-free display projection.

17///   to the native composition module.

19/// # Private members

21pub(in crate::native_tui_agent) struct ProviderCatalog {

46/// Resolves literal values and environment references without running commands.
47struct EnvironmentResolver;
48
49impl ConfigValueResolver for EnvironmentResolver {
50    /// Resolves literals or `$NAME` and `${NAME}` expressions from the process.
51    ///
52    /// # Parameters
53    /// - `value`: Literal or environment variable reference.
54    /// - `environment`: Optional scoped environment overlay.
55    ///
56    /// # Returns
57    /// Resolved value, or `None` when the variable is unavailable.
58    fn resolve(
59        &self,
60        value: &str,
61        environment: Option<&BTreeMap<String, String>>,
62    ) -> Option<String> {
63        if let Some(name) = value
64            .strip_prefix("${")
65            .and_then(|value| value.strip_suffix('}'))
66        {
67            return environment
68                .and_then(|values| values.get(name).cloned())
69                .or_else(|| std::env::var(name).ok());
70        }
71        if let Some(name) = value.strip_prefix('$') {
72            return environment
73                .and_then(|values| values.get(name).cloned())
74                .or_else(|| std::env::var(name).ok());
75        }
76        Some(value.to_owned())
77    }
78
79    /// Resolves values without using cached host state.
80    ///
81    /// # Parameters
82    /// - `value`: Literal or environment variable reference.
83    /// - `environment`: Optional scoped environment overlay.
84    ///
85    /// # Returns
86    /// Resolved value, or `None` when the variable is unavailable.
87    fn resolve_uncached(
88        &self,
89        value: &str,
90        environment: Option<&BTreeMap<String, String>>,
91    ) -> Option<String> {
92        self.resolve(value, environment)
93    }
94
95    /// Reads one host environment variable.
96    ///
97    /// # Parameters
98    /// - `name`: Environment variable name.
99    ///
100    /// # Returns
101    /// The value when the variable is available.
102    fn environment_value(&self, name: &str) -> Option<String> {
103        std::env::var(name).ok()
104    }
105}

107/// Adds the local fallback model when no provider is configured.
108///
109/// - `config`: Upstream catalog to update.
110/// - `fallback_model`: Model ID used for the fallback entry.
111fn insert_fallback_provider(config: &mut ModelsConfig, fallback_model: &str) {
112    if !config.providers.is_empty() {
113        return;
114    }
115    config.providers.insert(
116        "openai".to_owned(),
117        rpi_model_runtime::ProviderConfig {
118            name: Some("OpenAI".to_owned()),
119            base_url: None,
120            api_key: None,
121            api: Some("openai-completions".to_owned()),
122            headers: None,
123            auth_header: None,
124            models: vec![rpi_model_runtime::ModelDefinition {
125                id: fallback_model.to_owned(),
126                name: None,
127                base_url: None,
128                reasoning: None,
129                thinking_level_map: None,
130                context_window: None,
131                max_tokens: None,
132                cost: None,
133                sampling_params: None,
134                provider_options: BTreeMap::new(),
135                input: None,
136                headers: None,
137                compat: None,
138            }],
139        },
140    );
141}

143/// Removes typed credentials and rejects catalog-supplied auth material.
144///
145/// - `config`: Upstream catalog whose auth fields are inspected and cleared.
146///
147/// Returns an error when typed credentials or request headers are configured.
148/// # Returns
149/// `Ok(())` when the catalog contains no unsupported auth material.
150/// # Errors
151/// Returns an error when catalog auth is unsupported by the host boundary.
152fn strip_unsupported_catalog_auth(config: &mut ModelsConfig) -> Result<(), ProviderError> {
153    let has_auth = config.providers.values().any(|provider| {
154        provider.api_key.is_some()
155            || provider.auth_header.is_some()
156            || provider
157                .headers
158                .as_ref()
159                .is_some_and(|headers| !headers.is_empty())
160            || provider.models.iter().any(|model| {
161                model
162                    .headers
163                    .as_ref()
164                    .is_some_and(|headers| !headers.is_empty())
165            })
166    });
167    for provider in config.providers.values_mut() {
168        provider.api_key = None;
169        provider.auth_header = None;
170        provider.headers = None;
171        for model in &mut provider.models {
172            model.headers = None;
173        }
174    }
175    if has_auth {
176        Err(unsupported_catalog_auth_error())
177    } else {
178        Ok(())
179    }
180}

182/// Loads upstream models from `SCRIBE_AGENT_MODEL_CATALOG`.
183///
184/// The variable contains upstream `ModelsConfig` JSON. Only `OpenAI` Completions
185/// entries can currently be invoked by Scriberig's native provider adapter.
186///
187/// - `fallback_model`: Model used when no provider configuration is present.
188///
189/// Upstream model authority and a safe display projection.
190///
191/// Returns a configuration error for malformed or unsupported configuration.
192/// # Returns
193/// The operation result.
194/// # Errors
195/// Returns an error when catalog construction or display projection fails.
196pub(in crate::native_tui_agent) fn from_environment(
197    fallback_model: &str,
198) -> Result<ProviderCatalog, ProviderError> {
199    let json = std::env::var("SCRIBE_AGENT_MODEL_CATALOG").ok();
200    let mut config = match json.as_deref() {
201        Some(json) => {
202            let value =
203                serde_json::from_str::<serde_json::Value>(json).map_err(|_| invalid_catalog())?;
204            if !value
205                .as_object()
206                .is_some_and(|object| object.contains_key("providers"))
207            {
208                return Err(invalid_catalog());
209            }
210            serde_json::from_value::<ModelsConfig>(value).map_err(|_| invalid_catalog())?
211        }
212        None => ModelsConfig::default(),
213    };
214    insert_fallback_provider(&mut config, fallback_model);
215    strip_unsupported_catalog_auth(&mut config)?;
216
217    let resolver = EnvironmentResolver;
218    let mut models = build_catalog(&mut config, &[], &resolver);
219    models.retain(|model| model.api == Api::OpenaiCompletions);
220    if models.is_empty() {
221        return Err(invalid_catalog());
222    }
223    for model in &mut models {
224        model.headers = None;
225    }
226    let library = display_projection(&models)?;
227    let connections = models
228        .iter()
229        .map(|model| {
230            let provider = config
231                .providers
232                .get(&model.provider)
233                .ok_or_else(invalid_catalog)?;
234            let model_config = provider.models.iter().find(|item| item.id == model.id);
235            let base_url = model_config
236                .and_then(|item| item.base_url.as_deref())
237                .or(provider.base_url.as_deref())
238                .and_then(|value| resolver.resolve(value, None))
239                .unwrap_or_else(|| model.base_url.clone());
240            Ok(ProviderConnection {
241                id: connection_identity(&model.provider, &model.id),
242                provider: "openai_compatible".to_owned(),
243                base_url,
244            })
245        })
246        .collect::<Result<Vec<_>, ProviderError>>()?;
247    Ok(ProviderCatalog {
248        models,
249        library,
250        connections,
251    })
252}

254/// Builds the current protocol's safe model/profile projection.
255///
256/// # Parameters
257/// - `models`: Upstream models selected for the current native adapter.
258///
259/// # Returns
260/// A protocol-compatible projection without request-only model fields.
261///
262/// # Errors
263/// Returns a configuration error when a model identity or library is invalid.
264fn display_projection(models: &[Model]) -> Result<ModelLibrary, ProviderError> {
265    let mut descriptors = Vec::with_capacity(models.len());
266    let mut profiles = Vec::with_capacity(models.len());
267    for model in models {
268        let identity = connection_identity(&model.provider, &model.id);
269        let id = ModelId::new(identity.clone()).map_err(|_| invalid_catalog())?;
270        let levels = model
271            .supported_thinking_levels()
272            .into_iter()
273            .enumerate()
274            .map(|(index, level)| ThinkingLevelDescriptor {
275                id: ThinkingLevelId::new(level.as_str())
276                    .expect("upstream thinking level names are canonical"),
277                label: level.as_str().to_owned(),
278                description: None,
279                order: u32::try_from(index).unwrap_or(u32::MAX),
280            })
281            .collect::<Vec<_>>();
282        descriptors.push(ModelDescriptor {
283            id: id.clone(),
284            display_name: model.name.clone(),
285            provider_profile: model.provider.clone(),
286            provider_model_id: model.id.clone(),
287            capabilities: ModelCapabilities::default(),
288            default_thinking_level: None,
289            thinking_levels: levels,
290        });
291        profiles.push(ModelProfile {
292            id: format!("{identity}-profile"),
293            display_name: model.name.clone(),
294            model: id,
295            connection: connection_identity(&model.provider, &model.id),
296            credential: None,
297            default_thinking_level: None,
298            enabled: true,
299        });
300    }
301    ModelLibrary::new(descriptors, profiles).map_err(|_| invalid_catalog())
302}

304/// Creates the same generic failure for invalid catalog data.
305///
306/// # Returns
307/// A configuration error that contains no provider values.
308fn invalid_catalog() -> ProviderError {
309    ProviderError::Configuration("invalid upstream model catalog".to_owned())
310}

312/// Rejects auth configuration that cannot use Scriberig's authorized host path.
313///
314/// # Returns
315/// A configuration error with no credential or header values.
316fn unsupported_catalog_auth_error() -> ProviderError {
317    ProviderError::Configuration(
318        "catalog credentials and headers require Scriberig host configuration".to_owned(),
319    )

322/// Encodes provider and model identities without delimiter collisions.
323///
324/// # Parameters
325/// - `provider_id`: Upstream provider identity.
326/// - `model_id`: Upstream model identity.
327///
328/// # Returns
329/// A versioned, byte-length-prefixed identity for projections and adapter lookup.
330pub(in crate::native_tui_agent) fn connection_identity(
331    provider_id: &str,
332    model_id: &str,
333) -> String {
334    format!(
335        "rpi1:{}:{}{}:{}",
336        provider_id.len(),
337        provider_id,
338        model_id.len(),
339        model_id
340    )

3use scriberig_core::{ProjectId, SnapshotId};
4use scriberig_store::{RevisionStore, SnapshotStore};
5use scriberig_store_sqlite::{ProjectFileRecord, SqliteStore};

8use crate::support::adapter_for;

10/// Loads project source artifacts from one exact immutable snapshot.
11///
12/// - `store`: Shared project object and ownership store.
13/// - `project`: Project whose sources are reconstructed.
14/// - `snapshot`: Exact source snapshot.
15/// - `super`: Parent candidate module visibility.
16///
17/// Deterministically ordered source file records represented by the snapshot.
18///
19/// Returns an error when source ownership or immutable revisions are inconsistent.
20/// # Returns
21/// The operation result.
22/// # Errors
23/// Returns an error when the operation fails.
24pub(super) fn project_files_at(
25    store: &SqliteStore,
26    project: ProjectId,
27    snapshot: SnapshotId,
28) -> Result<Vec<ProjectFileRecord>, ProjectError> {
29    let entities = store
30        .project_entities(project)
31        .map_err(crate::errors::storage_error)?;
32    let mut files = Vec::new();
33    for entity in entities
34        .into_iter()
35        .filter(|entity| entity.kind == "source")
36    {
37        let revision_id = store
38            .resolve(snapshot, entity.entity)
39            .map_err(crate::errors::storage_debug_error)?
40            .ok_or_else(|| invalid_source(&entity.relative_path))?;
41        let revision = store
42            .get(revision_id)
43            .map_err(crate::errors::storage_debug_error)?
44            .ok_or_else(|| invalid_source(&entity.relative_path))?;
45        if revision.object_type().as_str() != "project.source" {
46            return Err(invalid_source(&entity.relative_path));
47        }
48        let bytes = revision.payload().to_vec();
49        let adapter = adapter_for(std::path::Path::new(&entity.relative_path), &bytes);
50        files.push(ProjectFileRecord {
51            project_id: project,
52            relative_path: entity.relative_path,
53            entity: entity.entity,
54            digest: *blake3::hash(&bytes).as_bytes(),
55            bytes,
56            adapter: adapter.to_owned(),
57            status: if adapter == "unsupported" {
58                "unsupported"
59            } else {
60                "ready"
61            }
62            .to_owned(),
63        });
64    }
65    files.sort_by(|left, right| left.relative_path.cmp(&right.relative_path));
66    Ok(files)
67}

69/// Creates a safe corruption error for an unavailable source artifact.
70///
71/// - `path`: Project-relative source path.
72///
73/// Structured project source error.
74/// # Returns
75/// The operation result.
76fn invalid_source(path: &str) -> ProjectError {
77    ProjectError {
78        code: "project_snapshot_corrupt",
79        message: format!("snapshot source is unavailable: {path}"),
80        diagnostics: Vec::new(),
81    }

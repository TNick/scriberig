1//! Serializes reuse of exact candidate validation workspaces.
2
3use std::sync::Mutex;
4use std::sync::atomic::{AtomicBool, Ordering};
5use std::thread;
6use std::time::{Duration, Instant};
7
8/// Describes cancellation or failure while acquiring the workspace lock.
9pub(super) enum ValidationLockError {
10    /// The owning validation operation was cancelled.
11    Cancelled,
12    /// Lock acquisition failed or exceeded its bounded wait.
13    Failed(String),
14}
15
16/// Acquires the process-wide candidate workspace lock while observing cancellation.
17///
18/// # Parameters
19/// - `cancelled`: Owning operation's cancellation state.
20/// - `super`: Parent candidate validation module.
21///
22/// # Returns
23/// A guard protecting deterministic candidate workspace reuse.
24///
25/// # Errors
26/// Returns cancellation or a bounded lock timeout.
27pub(super) fn acquire(
28    cancelled: &AtomicBool,
29) -> Result<std::sync::MutexGuard<'static, ()>, ValidationLockError> {
30    static VALIDATION_LOCK: Mutex<()> = Mutex::new(());
31    let deadline = Instant::now() + Duration::from_secs(600);
32    loop {
33        if cancelled.load(Ordering::Acquire) {
34            return Err(ValidationLockError::Cancelled);
35        }
36        match VALIDATION_LOCK.try_lock() {
37            Ok(guard) => return Ok(guard),
38            Err(std::sync::TryLockError::Poisoned(_)) => {
39                return Err(ValidationLockError::Failed(
40                    "candidate validation workspace lock poisoned".to_owned(),
41                ));
42            }
43            Err(std::sync::TryLockError::WouldBlock) if Instant::now() >= deadline => {
44                return Err(ValidationLockError::Failed(
45                    "candidate validation workspace lock timeout".to_owned(),
46                ));
47            }
48            Err(std::sync::TryLockError::WouldBlock) => thread::sleep(Duration::from_millis(10)),
49        }
50    }
51}

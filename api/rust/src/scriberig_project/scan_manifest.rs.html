1//! Persists source scans selected by an explicit project-relative manifest.
2
3use std::collections::BTreeSet;
4use std::path::Path;
5
6use scriberig_store_sqlite::{ProjectFileRecord, ProjectRecord, ProjectStatus};
7
8use crate::{ProjectError, ProjectService, ScanReport, errors, support};
9
10impl ProjectService {
11    /// Scans only an explicit set of project-relative files.
12    ///
13    /// This preserves an exact source-selection policy during candidate refresh
14    /// while still including candidate-created paths.
15    ///
16    /// - `name`: Optional display name for a new project.
17    /// - `path`: Filesystem project root.
18    /// - `paths`: Relative file manifest selected for this scan.
19    /// - `crate`: Internal project scan module.
20    ///
21    /// The registered project and selected deterministic scan files.
22    ///
23    /// Returns a structured filesystem or storage failure.
24    /// # Returns
25    /// The operation result.
26    /// # Errors
27    /// Returns an error when the operation fails.
28    pub(crate) fn scan_with_paths(
29        &self,
30        name: Option<String>,
31        path: impl AsRef<Path>,
32        paths: &BTreeSet<String>,
33    ) -> Result<ScanReport, ProjectError> {
34        let path = path.as_ref();
35        let display_name = name
36            .or_else(|| {
37                path.file_name()
38                    .and_then(|value| value.to_str())
39                    .map(str::to_owned)
40            })
41            .unwrap_or_else(|| "project".to_owned());
42        let project = self
43            .store
44            .register_project(display_name, path)
45            .map_err(errors::storage_error)?;
46        self.persist_scan(&project, Some(paths), false)
47    }
48
49    /// Scans and durably stores a registered project's selected files.
50    ///
51    /// - `project`: Registered project identity and path.
52    /// - `paths`: Optional exact relative-path filter.
53    /// - `git_tracked`: Whether the selection came from the Git index.
54    /// - `super`: Parent project service module.
55    ///
56    /// Persisted project scan report.
57    ///
58    /// Returns a structured filesystem or storage failure.
59    /// # Returns
60    /// The operation result.
61    /// # Errors
62    /// Returns an error when the operation fails.
63    pub(super) fn persist_scan(
64        &self,
65        project: &ProjectRecord,
66        paths: Option<&BTreeSet<String>>,
67        git_tracked: bool,
68    ) -> Result<ScanReport, ProjectError> {
69        let files = support::scan_directory(&project.path, paths)
70            .map_err(|error| errors::filesystem_error(&error))?;
71        let records = files
72            .iter()
73            .map(|file| ProjectFileRecord {
74                project_id: project.id,
75                relative_path: file.relative_path.clone(),
76                entity: support::source_entity(project.id, &file.relative_path),
77                digest: *blake3::hash(&file.bytes).as_bytes(),
78                bytes: file.bytes.clone(),
79                adapter: file.adapter.clone(),
80                status: file.status.clone(),
81            })
82            .collect::<Vec<_>>();
83        self.store
84            .replace_project_files(project.id, &records)
85            .map_err(errors::storage_error)?;
86        self.store
87            .set_project_status(
88                project.id,
89                ProjectStatus::ScanReady,
90                project.current_snapshot,
91            )
92            .map_err(errors::storage_error)?;
93        let project = self
94            .store
95            .project(project.id)
96            .map_err(errors::storage_error)?
97            .ok_or_else(|| errors::missing_project(project.id))?;
98        Ok(ScanReport {
99            project,
100            files,
101            git_tracked,
102        })
103    }

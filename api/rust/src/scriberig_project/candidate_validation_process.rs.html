1//! Runs bounded validation child processes and captures their output.
2
3use std::io::Read;
4use std::process::{Child, Command, ExitStatus, Output, Stdio};
5use std::sync::atomic::{AtomicBool, Ordering};
6use std::thread;
7use std::time::{Duration, Instant};
8
9/// Represents cancellation or failure while running candidate validation.
10pub(super) enum ProcessFailure {
11    /// Validation was cancelled by its owning operation.
12    Cancelled,
13    /// Validation could not complete and has a diagnostic.
14    Failed(String),
15}
16
17/// Retains the diagnostic tail within a character limit.
18///
19/// - `value`: Complete process output.
20/// - `maximum`: Maximum number of Unicode scalar values to retain.
21///
22/// The most recent bounded portion of the output.
23/// # Returns
24/// The output suffix bounded by `maximum` Unicode scalar values.
25pub(super) fn bounded_tail(value: &str, maximum: usize) -> String {
26    let mut tail = value.chars().rev().take(maximum).collect::<Vec<_>>();
27    tail.reverse();
28    tail.into_iter().collect()
29}

31/// Runs a validation command with a deadline and bounded captured output.
32///
33/// - `command`: Configured validation process.
34/// - `timeout`: Maximum process duration.
35/// - `cancelled`: Shared cancellation flag for the owning operation.
36/// - `super`: Parent candidate validation module.
37///
38/// Process output with each stream limited to 64 KiB.
39///
40/// Returns a structured message for startup, timeout, or output-reader failure.
41/// # Returns
42/// The operation result.
43/// # Errors
44/// Returns an error when the operation fails.
45pub(super) fn run_bounded(
46    mut command: Command,
47    timeout: Duration,
48    cancelled: &AtomicBool,
49) -> Result<Output, ProcessFailure> {
50    command.stdout(Stdio::piped()).stderr(Stdio::piped());
51    let mut child = command
52        .spawn()
53        .map_err(|error| ProcessFailure::Failed(format!("start failure: {error}")))?;
54    let stdout = child
55        .stdout
56        .take()
57        .ok_or_else(|| ProcessFailure::Failed("stdout pipe unavailable".to_owned()))?;
58    let stderr = child
59        .stderr
60        .take()
61        .ok_or_else(|| ProcessFailure::Failed("stderr pipe unavailable".to_owned()))?;
62    let stdout_reader = thread::spawn(move || read_bounded(stdout));
63    let stderr_reader = thread::spawn(move || read_bounded(stderr));
64    let status = wait_bounded(&mut child, timeout, cancelled)?;
65    let stdout = stdout_reader
66        .join()
67        .map_err(|_| ProcessFailure::Failed("stdout reader failed".to_owned()))?
68        .map_err(ProcessFailure::Failed)?;
69    let stderr = stderr_reader
70        .join()
71        .map_err(|_| ProcessFailure::Failed("stderr reader failed".to_owned()))?
72        .map_err(ProcessFailure::Failed)?;
73    Ok(Output {
74        status,
75        stdout,
76        stderr,
77    })
78}

80/// Waits for a child process and terminates it when its deadline expires.
81///
82/// - `child`: Running validation process.
83/// - `timeout`: Maximum process duration.
84/// - `cancelled`: Shared cancellation flag for the owning operation.
85///
86/// The completed process status.
87///
88/// Returns a timeout or process polling failure.
89/// # Returns
90/// The operation result.
91/// # Errors
92/// Returns an error when the operation fails.
93fn wait_bounded(
94    child: &mut Child,
95    timeout: Duration,
96    cancelled: &AtomicBool,
97) -> Result<ExitStatus, ProcessFailure> {
98    let deadline = Instant::now() + timeout;
99    loop {
100        if cancelled.load(Ordering::Acquire) {
101            let _ = child.kill();
102            let _ = child.wait();
103            return Err(ProcessFailure::Cancelled);
104        }
105        if let Some(status) = child
106            .try_wait()
107            .map_err(|error| ProcessFailure::Failed(error.to_string()))?
108        {
109            return Ok(status);
110        }
111        if Instant::now() >= deadline {
112            let _ = child.kill();
113            let _ = child.wait();
114            return Err(ProcessFailure::Failed(format!(
115                "timeout after {} seconds",
116                timeout.as_secs()
117            )));
118        }
119        thread::sleep(Duration::from_millis(50));
120    }
121}

123/// Drains a process stream while retaining only a bounded prefix.
124///
125/// - `reader`: Child process output stream.
126///
127/// At most 64 KiB of output.
128///
129/// Returns a stream read failure.
130/// # Returns
131/// The operation result.
132/// # Errors
133/// Returns an error when the operation fails.
134fn read_bounded(mut reader: impl Read) -> Result<Vec<u8>, String> {
135    let mut retained = Vec::new();
136    let mut buffer = [0_u8; 8192];
137    loop {
138        let count = reader
139            .read(&mut buffer)
140            .map_err(|error| error.to_string())?;
141        if count == 0 {
142            return Ok(retained);
143        }
144        let remaining = 65_536_usize.saturating_sub(retained.len());
145        retained.extend_from_slice(&buffer[..count.min(remaining)]);
146    }
147}

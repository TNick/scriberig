1//! Runs bounded validation child processes and captures their output.
2
3use std::io::Read;
4use std::process::{Child, Command, ExitStatus, Output, Stdio};
5use std::sync::atomic::{AtomicBool, Ordering};
6use std::thread;
7use std::time::{Duration, Instant};
8
9/// Represents cancellation or failure while running candidate validation.
10pub(super) enum ProcessFailure {
11    /// Validation was cancelled by its owning operation.
12    Cancelled,
13    /// Validation could not complete and has a diagnostic.
14    Failed(String),
15}
16
17/// Runs a validation command with a deadline and bounded captured output.
18///
19/// # Parameters
20/// - `command`: Configured validation process.
21/// - `timeout`: Maximum process duration.
22/// - `cancelled`: Shared cancellation flag for the owning operation.
23/// - `super`: Parent candidate validation module.
24///
25/// # Returns
26/// Process output with each stream limited to 64 KiB.
27///
28/// # Errors
29/// Returns a structured message for startup, timeout, or output-reader failure.
30pub(super) fn run_bounded(
31    mut command: Command,
32    timeout: Duration,
33    cancelled: &AtomicBool,
34) -> Result<Output, ProcessFailure> {
35    command.stdout(Stdio::piped()).stderr(Stdio::piped());
36    let mut child = command
37        .spawn()
38        .map_err(|error| ProcessFailure::Failed(format!("start failure: {error}")))?;
39    let stdout = child
40        .stdout
41        .take()
42        .ok_or_else(|| ProcessFailure::Failed("stdout pipe unavailable".to_owned()))?;
43    let stderr = child
44        .stderr
45        .take()
46        .ok_or_else(|| ProcessFailure::Failed("stderr pipe unavailable".to_owned()))?;
47    let stdout_reader = thread::spawn(move || read_bounded(stdout));
48    let stderr_reader = thread::spawn(move || read_bounded(stderr));
49    let status = wait_bounded(&mut child, timeout, cancelled)?;
50    let stdout = stdout_reader
51        .join()
52        .map_err(|_| ProcessFailure::Failed("stdout reader failed".to_owned()))?
53        .map_err(ProcessFailure::Failed)?;
54    let stderr = stderr_reader
55        .join()
56        .map_err(|_| ProcessFailure::Failed("stderr reader failed".to_owned()))?
57        .map_err(ProcessFailure::Failed)?;
58    Ok(Output {
59        status,
60        stdout,
61        stderr,
62    })
63}

65/// Waits for a child process and terminates it when its deadline expires.
66///
67/// # Parameters
68/// - `child`: Running validation process.
69/// - `timeout`: Maximum process duration.
70/// - `cancelled`: Shared cancellation flag for the owning operation.
71///
72/// # Returns
73/// The completed process status.
74///
75/// # Errors
76/// Returns a timeout or process polling failure.
77fn wait_bounded(
78    child: &mut Child,
79    timeout: Duration,
80    cancelled: &AtomicBool,
81) -> Result<ExitStatus, ProcessFailure> {
82    let deadline = Instant::now() + timeout;
83    loop {
84        if cancelled.load(Ordering::Acquire) {
85            let _ = child.kill();
86            let _ = child.wait();
87            return Err(ProcessFailure::Cancelled);
88        }
89        if let Some(status) = child
90            .try_wait()
91            .map_err(|error| ProcessFailure::Failed(error.to_string()))?
92        {
93            return Ok(status);
94        }
95        if Instant::now() >= deadline {
96            let _ = child.kill();
97            let _ = child.wait();
98            return Err(ProcessFailure::Failed(format!(
99                "timeout after {} seconds",
100                timeout.as_secs()
101            )));
102        }
103        thread::sleep(Duration::from_millis(50));
104    }
105}

107/// Drains a process stream while retaining only a bounded prefix.
108///
109/// # Parameters
110/// - `reader`: Child process output stream.
111///
112/// # Returns
113/// At most 64 KiB of output.
114///
115/// # Errors
116/// Returns a stream read failure.
117fn read_bounded(mut reader: impl Read) -> Result<Vec<u8>, String> {
118    let mut retained = Vec::new();
119    let mut buffer = [0_u8; 8192];
120    loop {
121        let count = reader
122            .read(&mut buffer)
123            .map_err(|error| error.to_string())?;
124        if count == 0 {
125            return Ok(retained);
126        }
127        let remaining = 65_536_usize.saturating_sub(retained.len());
128        retained.extend_from_slice(&buffer[..count.min(remaining)]);
129    }
130}

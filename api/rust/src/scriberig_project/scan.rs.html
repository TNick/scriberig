6use crate::ScanFile;

8/// Scans supported and unknown files below one project root.
9///
10/// - `root`: Project root to scan.
11/// - `tracked`: Optional set of Git-indexed paths to retain.
12/// - `crate`: Internal visibility marker.
13///
14/// Deterministically ordered scan files.
15///
16/// Returns a filesystem traversal or read error.
17/// # Returns
18/// The operation result.
19/// # Errors
20/// Returns an error when the operation fails.
21pub(crate) fn scan_directory(
22    root: &Path,
23    tracked: Option<&BTreeSet<String>>,
24) -> std::io::Result<Vec<ScanFile>> {
25    let mut files = Vec::new();
26    let sensitive = scriberig_secrets::sensitive::SensitivePaths::from_process()
27        .map_err(std::io::Error::other)?;
28    visit_directory(root, root, tracked, &mut files, &sensitive)?;
29    files.sort_by(|left, right| left.relative_path.cmp(&right.relative_path));
30    Ok(files)
31}

33/// Visits a directory recursively without following symlinks or generated trees.
34///
35/// - `root`: Project root used for relative paths.
36/// - `directory`: Directory currently being visited.
37/// - `tracked`: Optional set of paths selected by Git.
38/// - `files`: Scan results being collected.
39/// - `sensitive`: Credential paths excluded before reading any source bytes.
40///
41/// Unit after the directory has been visited.
42///
43/// Returns a filesystem traversal or read error.
44/// # Returns
45/// The operation result.
46/// # Errors
47/// Returns an error when the operation fails.
48fn visit_directory(
49    root: &Path,
50    directory: &Path,
51    tracked: Option<&BTreeSet<String>>,
52    files: &mut Vec<ScanFile>,
53    sensitive: &scriberig_secrets::sensitive::SensitivePaths,
54) -> std::io::Result<()> {
55    for entry in std::fs::read_dir(directory)? {
56        let entry = entry?;
57        let path = entry.path();
58        let relative = path.strip_prefix(root).unwrap_or(&path);
59        if is_excluded(relative) || sensitive.contains(&path) || entry.file_type()?.is_symlink() {
60            continue;
61        }
62        let file_type = entry.file_type()?;
63        if file_type.is_dir() {
64            visit_directory(root, &path, tracked, files, sensitive)?;
65        } else if file_type.is_file() {
66            let relative_path = relative.to_string_lossy().replace('\\', "/");
67            if tracked.is_some_and(|paths| !paths.contains(&relative_path)) {
68                continue;
69            }
70            let bytes = std::fs::read(&path)?;
71            let adapter = adapter_for(&path, &bytes).to_owned();
72            files.push(ScanFile {
73                relative_path,
74                bytes,
75                status: if adapter == "unsupported" {
76                    "unsupported".to_owned()
77                } else {
78                    "ready".to_owned()
79                },
80                adapter,
81            });
82        }
83    }
84    Ok(())
85}

87/// Returns whether a project-relative path is excluded by default.
88///
89/// - `path`: Relative path to classify.
90///
91/// Whether the path belongs to a generated or hidden tree.
92/// # Returns
93/// The operation result.
94fn is_excluded(path: &Path) -> bool {
95    path.components().any(|component| {
96        matches!(
97            component.as_os_str().to_str(),
98            Some(
99                ".git"
100                    | "target"
101                    | "node_modules"
102                    | "dist"
103                    | "build"
104                    | ".pages"
105                    | "__pycache__"
106                    | ".venv"
107                    | ".cache"
108            )
109        )
110    })
111}

113/// Selects a supported source adapter by file extension.
114///
115/// - `path`: File path to classify.
116/// - `bytes`: File bytes used to distinguish unknown text from binary data.
117/// - `crate`: Internal visibility marker.
118///
119/// The selected adapter name.
120/// # Returns
121/// The operation result.
122pub(crate) fn adapter_for(path: &Path, bytes: &[u8]) -> &'static str {
123    let name = path.file_name().and_then(|value| value.to_str());
124    if matches!(
125        name,
126        Some(
127            "Dockerfile"
128                | "Makefile"
129                | "justfile"
130                | "Taskfile"
131                | "LICENSE"
132                | "README"
133                | ".gitignore"
134                | ".gitattributes"
135                | ".editorconfig"
136                | ".gitmodules"
137                | ".npmrc"
138                | ".yarnrc"
139                | ".prettierrc"
140                | ".eslintrc"
141                | ".babelrc"
142                | ".flake8"
143                | "gradlew"
144                | "Gemfile"
145                | "Rakefile"
146                | "Brewfile"
147                | "Vagrantfile"
148                | "Procfile"
149        )
150    ) {
151        return "text";
152    }
153    match path
154        .extension()
155        .and_then(|value| value.to_str())
156        .map(str::to_ascii_lowercase)
157        .as_deref()
158    {
159        Some("html" | "htm") => "html",
160        Some("c" | "h") => "c",
161        Some("cc" | "cpp" | "cxx" | "hpp" | "hh" | "hxx") => "cpp",
162        Some("js" | "mjs" | "cjs" | "jsx") => "javascript",
163        Some("ts" | "mts" | "cts" | "tsx") => "typescript",
164        Some("md" | "markdown") => "markdown",
165        Some("py" | "pyi" | "pyw") => "python",
166        Some("rs") => "rust",
167        Some("pdf") => "pdf",
168        Some(
169            "sh" | "bash" | "zsh" | "fish" | "ps1" | "psm1" | "bat" | "cmd" | "mmd" | "mermaid"
170            | "svg" | "txt" | "log" | "env" | "xml" | "ini" | "cfg" | "properties" | "csv" | "tsv"
171            | "rst" | "adoc" | "tex" | "graphql" | "gql" | "proto" | "sql" | "go" | "java" | "kt"
172            | "kts" | "cs" | "swift" | "php" | "rb" | "lua" | "dart" | "scala" | "m" | "mm" | "hs"
173            | "ex" | "exs" | "erl" | "zig" | "lock" | "sum" | "mod" | "gradle",
174        ) => "text",
175        Some("json" | "jsonc" | "ipynb") => "json",
176        Some("yaml" | "yml") => "yaml",
177        Some("toml") => "toml",
178        Some(
179            "png" | "jpg" | "jpeg" | "gif" | "webp" | "ico" | "bmp" | "woff" | "woff2" | "ttf"
180            | "otf" | "wasm" | "zip" | "gz" | "tar" | "7z" | "jar" | "exe" | "dll" | "o" | "a"
181            | "so",
182        ) => "binary",
183        _ if std::str::from_utf8(bytes).is_ok() => "text",
184        _ => "binary",
185    }
186}

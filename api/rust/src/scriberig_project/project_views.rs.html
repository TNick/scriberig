1//! Exposes snapshot-bound project symbols and source projections.
2
3use scriberig_core::{EntityId, ProjectId, SnapshotId};
4
5use crate::{ProjectError, ProjectService};
6
7/// A named semantic symbol owned by a project source file.
8#[derive(Clone, Debug, Eq, PartialEq)]
9pub struct ProjectSymbol {
10    /// Stable semantic entity identity.
11    pub entity: EntityId,
12    /// Exact snapshot containing the symbol.
13    pub snapshot: SnapshotId,
14    /// Project-relative source path.
15    pub path: String,
16    /// Semantic kind such as `rust.method`.
17    pub kind: String,
18    /// Human-readable semantic name when available.
19    pub name: Option<String>,
20}
21
22/// An exact source projection associated with one semantic entity.
23#[derive(Clone, Debug, Eq, PartialEq)]
24pub struct ProjectSourceProjection {
25    /// Project identity owning the projection.
26    pub project: ProjectId,
27    /// Exact snapshot rendered by the projection.
28    pub snapshot: SnapshotId,
29    /// Selected semantic entity.
30    pub entity: EntityId,
31    /// Selected semantic name.
32    pub name: Option<String>,
33    /// Semantic kind.
34    pub kind: String,
35    /// Project-relative source path.
36    pub path: String,
37    /// Source bytes rendered by the projection.
38    pub text: Vec<u8>,
39    /// Whether this semantic projection supports editing.
40    pub editable: bool,
41}
42
43impl ProjectService {
44    /// Lists named semantic symbols in the current project snapshot.
45    ///
46    /// # Parameters
47    /// - `project`: Project whose symbols are requested.
48    ///
49    /// # Returns
50    /// Snapshot-bound named symbols and source entities.
51    ///
52    /// # Errors
53    /// Returns a structured error when the project or symbol store is unavailable.
54    pub fn symbols(&self, project: ProjectId) -> Result<Vec<ProjectSymbol>, ProjectError> {
55        let workspace = self.workspace(project)?;
56        self.store
57            .project_entities(project)
58            .map_err(crate::errors::storage_error)
59            .map(|entities| {
60                entities
61                    .into_iter()
62                    .filter(|entity| entity.name.is_some() || entity.kind == "source")
63                    .map(|entity| ProjectSymbol {
64                        entity: entity.entity,
65                        snapshot: workspace.snapshot,
66                        path: entity.relative_path,
67                        kind: entity.kind,
68                        name: entity.name,
69                    })
70                    .collect()
71            })
72    }
73
74    /// Renders an exact source projection for one semantic entity.
75    ///
76    /// # Parameters
77    /// - `project`: Project owning the entity.
78    /// - `entity`: Semantic entity to render.
79    ///
80    /// # Returns
81    /// Snapshot-bound source projection.
82    ///
83    /// # Errors
84    /// Returns a structured error when the entity or source file is missing.
85    pub fn projection(
86        &self,
87        project: ProjectId,
88        entity: EntityId,
89    ) -> Result<ProjectSourceProjection, ProjectError> {
90        let workspace = self.workspace(project)?;
91        let entity_record = self
92            .store
93            .project_entities(project)
94            .map_err(crate::errors::storage_error)?
95            .into_iter()
96            .find(|record| record.entity == entity)
97            .ok_or_else(|| ProjectError {
98                code: "entity_not_found",
99                message: "semantic entity is not owned by the project".to_owned(),
100                diagnostics: Vec::new(),
101            })?;
102        let file = self
103            .store
104            .project_files(project)
105            .map_err(crate::errors::storage_error)?
106            .into_iter()
107            .find(|record| record.relative_path == entity_record.relative_path)
108            .ok_or_else(|| ProjectError {
109                code: "source_not_found",
110                message: "semantic entity has no source projection".to_owned(),
111                diagnostics: Vec::new(),
112            })?;
113        Ok(ProjectSourceProjection {
114            project,
115            snapshot: workspace.snapshot,
116            entity,
117            name: entity_record.name,
118            kind: entity_record.kind.clone(),
119            path: entity_record.relative_path,
120            text: file.bytes,
121            editable: entity_record.kind == "rust.method",
122        })
123    }
124}

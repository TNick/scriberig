22/// Runs the interactive shell until a local quit event is received.
23///
24/// # Errors
25/// Returns an I/O error when terminal setup, drawing, or event reading fails.
26pub fn run<S: TuiService + Sync>(
27    service: &S,
28    context: &ApplicationContext,
29    config: &TuiConfig,
30) -> io::Result<()> {
31    if !config.interactive {
32        run_headless(service, context, config);
33        return Ok(());
34    }
35    validate_interactive_terminal(
36        config.interactive,
37        io::stdin().is_terminal(),
38        io::stdout().is_terminal(),
39    )?;
40    let _guard = TerminalGuard::enter()?;
41    let backend = CrosstermBackend::new(io::stdout());
42    let mut terminal = Terminal::new(backend)?;
43    let mut state = TuiState {
44        project: config.project.clone(),
45        ..TuiState::default()
46    };
47    std::thread::scope(|scope| {
48        let (work_sender, work_receiver) = mpsc::sync_channel(32);
49        let (message_sender, message_receiver) = mpsc::sync_channel(32);
50        let worker_scope = scope;
51        let worker_context = context.clone();
52        let worker = scope.spawn(move || {
53            worker_loop(
54                worker_scope,
55                service,
56                worker_context,
57                work_receiver,
58                message_sender,
59            );
60        });
61        let mut redraw = true;
62        try_queue(&work_sender, Work::Effect(Effect::RequestCapabilities));
63        if let Some(entity) = config.initial_entity {
64            let request_id = state.next_request_id;
65            state.next_request_id += 1;
66            let effects = reduce(
67                &mut state,
68                crate::action::Action::InspectRequested {
69                    request_id,
70                    entity,
71                    snapshot: config.snapshot,
72                },
73            );
74            queue_effects(&work_sender, effects);
75        }
76        let mut event_cursor = 0_usize;
77        let mut poll_in_flight = false;
78        while !state.should_quit {
79            drain_messages(
80                &message_receiver,
81                &mut state,
82                &mut event_cursor,
83                &mut poll_in_flight,
84                &work_sender,
85                &mut redraw,
86            );
87            if !poll_in_flight {
88                poll_in_flight = try_queue(&work_sender, Work::PollApplicationEvents);
89            }
90            if redraw {
91                terminal.draw(|frame| render::render(frame, &state))?;
92                redraw = false;
93            }
94            let input = event::read_event(Duration::from_millis(250))?;
95            if matches!(input, TuiEvent::Tick) {
96                continue;
97            }
98            queue_effects(&work_sender, state.apply_event(&input));
99            terminal.draw(|frame| render::render(frame, &state))?;
100        }
101        drop(work_sender);
102        drop(message_receiver);
103        worker
104            .join()
105            .map_err(|_| io::Error::other("TUI worker panicked"))?;
106        Ok(())
107    })
108}

110/// Describes one unit of work executed outside the terminal thread.
111enum Work {
112    /// Execute one reducer-produced semantic effect.
113    Effect(Effect),
114    /// Poll the application event boundary.
115    PollApplicationEvents,
116}
117
118/// Carries one completed worker operation back to the terminal thread.
119enum RuntimeMessage {
120    /// A reducer action produced by semantic work.
121    Action(Box<Action>),
122    /// Retained application events returned by a poll.
123    ApplicationEvents(Vec<scriberig_app::ApplicationEvent>),
124    /// A failed application event poll.
125    ApplicationEventsFailed { category: String, message: String },
126}
127
128/// Runs semantic work until the terminal loop drops its work channel.
129#[allow(clippy::needless_pass_by_value)] // The scoped worker owns these values for its full run.
130fn worker_loop<'scope, S: TuiService + Sync>(
131    scope: &'scope std::thread::Scope<'scope, '_>,
132    service: &'scope S,
133    context: ApplicationContext,
134    receiver: Receiver<Work>,
135    sender: SyncSender<RuntimeMessage>,
136) {
137    while let Ok(work) = receiver.recv() {
138        if let Work::Effect(effect @ Effect::SubmitAgent { .. }) = work {
139            let sender = sender.clone();
140            let context = context.clone();
141            scope.spawn(move || {
142                let action = execute_effect(service, &context, effect);
143                let _ = sender.send(RuntimeMessage::Action(Box::new(action)));
144            });
145            continue;
146        }
147        let message = match work {
148            Work::Effect(effect) => {
149                RuntimeMessage::Action(Box::new(execute_effect(service, &context, effect)))
150            }
151            Work::PollApplicationEvents => match service.application_events(&context) {
152                Ok(events) => RuntimeMessage::ApplicationEvents(events),
153                Err(error) => RuntimeMessage::ApplicationEventsFailed {
154                    category: error.category().to_owned(),
155                    message: crate::inspector::sanitize(&error.to_string()),
156                },
157            },
158        };
159        if sender.send(message).is_err() {
160            break;
161        }
162    }
163}

165/// Queues one unit of work without allowing the UI thread to block.
166///
167/// # Parameters
168/// - `sender`: Bounded worker queue.
169/// - `work`: Work item to enqueue.
170///
171/// # Returns
172/// Whether the work item was accepted.
173fn try_queue(sender: &SyncSender<Work>, work: Work) -> bool {
174    match sender.try_send(work) {
175        Ok(()) => true,
176        Err(TrySendError::Full(_) | TrySendError::Disconnected(_)) => false,
177    }
178}
179
180/// Queues all reducer effects that fit in the bounded worker queue.
181///
182/// # Parameters
183/// - `sender`: Bounded worker queue.
184/// - `effects`: Reducer effects to enqueue.
185fn queue_effects(sender: &SyncSender<Work>, effects: Vec<Effect>) {
186    for effect in effects {
187        if !try_queue(sender, Work::Effect(effect)) {
188            break;
189        }
190    }
191}

193/// Applies completed worker messages to the state and schedules follow-up work.
194///
195/// # Parameters
196/// - `receiver`: Completed worker message queue.
197/// - `state`: Mutable TUI state.
198/// - `event_cursor`: Cursor into retained application events.
199/// - `poll_in_flight`: Whether an event poll is active.
200/// - `sender`: Bounded worker queue.
201/// - `redraw`: Whether the terminal needs repainting.
202fn drain_messages(
203    receiver: &Receiver<RuntimeMessage>,
204    state: &mut TuiState,
205    event_cursor: &mut usize,
206    poll_in_flight: &mut bool,
207    sender: &SyncSender<Work>,
208    redraw: &mut bool,
209) {
210    while let Ok(message) = receiver.try_recv() {
211        match message {
212            RuntimeMessage::Action(action) => {
213                *redraw = true;
214                queue_effects(sender, reduce(state, *action));
215            }
216            RuntimeMessage::ApplicationEvents(events) => {
217                *poll_in_flight = false;
218                if state.connection == crate::state::ConnectionState::Disconnected {
219                    *redraw = true;
220                    queue_effects(sender, reduce(state, Action::ConnectionRestored));
221                }
222                let start = (*event_cursor).min(events.len());
223                let end = (start + 32).min(events.len());
224                for application_event in &events[start..end] {
225                    *redraw = true;
226                    let event = TuiEvent::Application(application_event.clone());
227                    queue_effects(sender, state.apply_event(&event));
228                }
229                *event_cursor = end;
230            }
231            RuntimeMessage::ApplicationEventsFailed { category, message } => {
232                *poll_in_flight = false;
233                *redraw = true;
234                reduce(
235                    state,
236                    Action::OperationFailed {
237                        request_id: 0,
238                        category,
239                        message,
240                    },
241                );
242            }
243        }
244    }
245}

247/// Executes initial semantic work without taking ownership of a terminal.
248///
249/// # Parameters
250/// - `service`: Application service used for semantic work.
251/// - `context`: Caller capability and snapshot context.
252/// - `config`: Initial-selection configuration.
253///
254/// # Returns
255/// Success after the requested headless work has been dispatched.
256///
257/// # Errors
258/// Returns an I/O error if the headless execution boundary fails.
259fn run_headless<S: TuiService>(service: &S, context: &ApplicationContext, config: &TuiConfig) {
260    let mut state = TuiState::default();
261    let action = execute_effect(service, context, crate::effect::Effect::RequestCapabilities);
262    reduce(&mut state, action);
263    if let Some(entity) = config.initial_entity {
264        let request_id = state.next_request_id;
265        state.next_request_id += 1;
266        let effects = reduce(
267            &mut state,
268            crate::action::Action::InspectRequested {
269                request_id,
270                entity,
271                snapshot: config.snapshot,
272            },
273        );
274        for effect in effects {
275            let action = execute_effect(service, context, effect);
276            reduce(&mut state, action);
277        }
278    }
279}

281/// Rejects interactive startup when the caller does not have a terminal.
282/// - `interactive`: Argument named `interactive`.
283/// - `stdin_is_terminal`: Argument named `stdin_is_terminal`.
284/// - `stdout_is_terminal`: Argument named `stdout_is_terminal`.
285///
286/// # Returns
287/// The value produced by the declaration.
288///
289/// # Errors
290/// Returns an error when the operation cannot complete.
291fn validate_interactive_terminal(
292    interactive: bool,
293    stdin_is_terminal: bool,
294    stdout_is_terminal: bool,
295) -> io::Result<()> {
296    if interactive && (!stdin_is_terminal || !stdout_is_terminal) {
297        return Err(io::Error::new(
298            io::ErrorKind::NotConnected,
299            "interactive TUI requires a terminal on stdin and stdout",
300        ));
301    }
302    Ok(())
303}

26/// Runs the interactive shell until a local quit event is received.
27///
28/// Returns an I/O error when terminal setup, drawing, or event reading fails.
29///
30/// # Errors
31/// Returns an I/O error when terminal setup, event handling, or rendering fails.
32pub fn run<S: TuiService + Sync>(
33    service: &S,
34    context: &ApplicationContext,
35    config: &TuiConfig,
36) -> io::Result<()> {
37    if !config.interactive {
38        run_headless(service, context, config);
39        return Ok(());
40    }
41    validate_interactive_terminal(
42        config.interactive,
43        io::stdin().is_terminal(),
44        io::stdout().is_terminal(),
45    )?;
46    let mut debug_log = config
47        .debug
48        .then(|| TuiDebugLog::open(config.debug_log_file.as_deref()))
49        .transpose()?;
50    let state = TuiState {
51        project: config.project.clone(),
52        agent_selection: config.agent_selection.clone(),
53        ..TuiState::default()
54    };
55    if let Some(log) = debug_log.as_mut() {
56        log.record_message("runtime-start", &state)?;
57    }
58    upstream::run(service, context, config, state, debug_log.as_mut())
59}

89/// Describes one unit of work executed outside the terminal thread.
90enum Work {
91    /// Execute one reducer-produced semantic effect.
92    Effect(Box<Effect>),
93    /// Poll the application event boundary.
94    PollApplicationEvents,
95}
96
97/// Carries one completed worker operation back to the terminal thread.
98enum RuntimeMessage {
99    /// A reducer action produced by semantic work.
100    Action(Box<Action>),
101    /// Retained application events returned by a poll.
102    ApplicationEvents(Vec<scriberig_app::ApplicationEvent>),
103    /// A failed application event poll.
104    ApplicationEventsFailed { category: String, message: String },
105}
106
107/// Runs semantic work until the terminal loop drops its work channel.
108#[allow(clippy::needless_pass_by_value)] // The scoped worker owns these values for its full run.
109fn worker_loop<'scope, S: TuiService + Sync>(
110    scope: &'scope std::thread::Scope<'scope, '_>,
111    service: &'scope S,
112    context: ApplicationContext,
113    receiver: Receiver<Work>,
114    sender: SyncSender<RuntimeMessage>,
115) {
116    while let Ok(work) = receiver.recv() {
117        if let Work::Effect(effect) = &work
118            && matches!(
119                effect.as_ref(),
120                Effect::SubmitAgent { .. }
121                    | Effect::StartNewConversation { .. }
122                    | Effect::ContinueConversation { .. }
123                    | Effect::PlanLifecycle { .. }
124                    | Effect::RetryPlanNode(_)
125                    | Effect::AddPlanNextStep(_)
126                    | Effect::DecomposePlanNode(_)
127                    | Effect::MutatePlanSemantics(_)
128            )
129        {
130            let sender = sender.clone();
131            let context = context.clone();
132            let effect = (**effect).clone();
133            scope.spawn(move || {
134                let action = execute_effect(service, &context, effect);
135                let _ = sender.send(RuntimeMessage::Action(Box::new(action)));
136            });
137            continue;
138        }
139        let message = match work {
140            Work::Effect(effect) => {
141                RuntimeMessage::Action(Box::new(execute_effect(service, &context, *effect)))
142            }
143            Work::PollApplicationEvents => match service.application_events(&context) {
144                Ok(events) => RuntimeMessage::ApplicationEvents(events),
145                Err(error) => RuntimeMessage::ApplicationEventsFailed {
146                    category: error.category().to_owned(),
147                    message: crate::inspector::sanitize(&error.to_string()),
148                },
149            },
150        };
151        if sender.send(message).is_err() {
152            break;
153        }
154    }
155}

157/// Queues one unit of work without allowing the UI thread to block.
158///
159/// - `sender`: Bounded worker queue.
160/// - `work`: Work item to enqueue.
161///
162/// Whether the work item was accepted.
163/// # Returns
164/// The operation result.
165fn try_queue(sender: &SyncSender<Work>, work: Work) -> bool {
166    match sender.try_send(work) {
167        Ok(()) => true,
168        Err(TrySendError::Full(_) | TrySendError::Disconnected(_)) => false,
169    }
170}

172/// Retains reducer effects and queues as many as the bounded worker accepts.
173///
174/// - `sender`: Bounded worker queue.
175/// - `pending`: Effects retained until worker capacity returns.
176/// - `effects`: Reducer effects to enqueue.
177fn queue_effects(sender: &SyncSender<Work>, pending: &mut VecDeque<Effect>, effects: Vec<Effect>) {
178    pending.extend(effects);
179    flush_effects(sender, pending);
180}
181
182/// Queues retained effects until the bounded worker queue is full.
183///
184/// - `sender`: Bounded worker queue.
185/// - `pending`: Effects retained until worker capacity returns.
186fn flush_effects(sender: &SyncSender<Work>, pending: &mut VecDeque<Effect>) {
187    while let Some(effect) = pending.pop_front() {
188        if matches!(&effect, Effect::None) {
189            continue;
190        }
191        match sender.try_send(Work::Effect(Box::new(effect))) {
192            Ok(()) => {}
193            Err(TrySendError::Full(Work::Effect(effect))) => {
194                pending.push_front(*effect);
195                break;
196            }
197            Err(TrySendError::Disconnected(_)) => break,
198            Err(TrySendError::Full(Work::PollApplicationEvents)) => {
199                unreachable!("only effects are queued through this helper")
200            }
201        }
202    }
203}

205/// Applies one bounded batch of retained application lifecycle events.
206///
207/// - `state`: TUI state receiving event reductions.
208/// - `pending_events`: Events waiting for reducer processing.
209/// - `sender`: Bounded worker queue for resulting semantic effects.
210/// - `pending_effects`: Effects retained until worker capacity returns.
211/// - `redraw`: Whether visible state changed.
212fn dispatch_pending_events(
213    state: &mut TuiState,
214    pending_events: &mut VecDeque<scriberig_app::ApplicationEvent>,
215    sender: &SyncSender<Work>,
216    pending_effects: &mut VecDeque<Effect>,
217    redraw: &mut bool,
218) {
219    for _ in 0..APPLICATION_EVENT_BATCH_SIZE {
220        let Some(application_event) = pending_events.pop_front() else {
221            break;
222        };
223        *redraw = true;
224        queue_effects(
225            sender,
226            pending_effects,
227            state.apply_application_event(&application_event),
228        );
229    }
230}

232/// Executes initial semantic work without taking ownership of a terminal.
233///
234/// - `service`: Application service used for semantic work.
235/// - `context`: Caller capability and snapshot context.
236/// - `config`: Initial-selection configuration.
237///
238/// Success after the requested headless work has been dispatched.
239///
240/// Returns an I/O error if the headless execution boundary fails.
241fn run_headless<S: TuiService>(service: &S, context: &ApplicationContext, config: &TuiConfig) {
242    let mut state = TuiState::default();
243    let action = execute_effect(service, context, crate::effect::Effect::RequestCapabilities);
244    reduce(&mut state, action);
245    if let Some(entity) = config.initial_entity {
246        let request_id = state.next_request_id;
247        state.next_request_id += 1;
248        let effects = reduce(
249            &mut state,
250            crate::action::Action::InspectRequested {
251                request_id,
252                entity,
253                snapshot: config.snapshot,
254                switch_view: false,
255            },
256        );
257        for effect in effects {
258            if matches!(&effect, crate::effect::Effect::None) {
259                continue;
260            }
261            let action = execute_effect(service, context, effect);
262            reduce(&mut state, action);
263        }
264    }
265}

267/// Rejects interactive startup when the caller does not have a terminal.
268/// - `interactive`: Argument named `interactive`.
269/// - `stdin_is_terminal`: Argument named `stdin_is_terminal`.
270/// - `stdout_is_terminal`: Argument named `stdout_is_terminal`.
271///
272///
273/// Returns an error when the operation cannot complete.
274/// # Returns
275/// The operation result.
276/// # Errors
277/// Returns an error when the operation fails.
278fn validate_interactive_terminal(
279    interactive: bool,
280    stdin_is_terminal: bool,
281    stdout_is_terminal: bool,
282) -> io::Result<()> {
283    if interactive && (!stdin_is_terminal || !stdout_is_terminal) {
284        return Err(io::Error::new(
285            io::ErrorKind::NotConnected,
286            "interactive TUI requires a terminal on stdin and stdout",
287        ));
288    }
289    Ok(())
290}

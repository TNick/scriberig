85/// Returns bindings relevant to the current state for contextual hints.
86///
87/// # Parameters
88/// - `state`: Current terminal state.
89///
90/// # Returns
91/// Keyboard hints for the active context.
92#[must_use]
93pub fn hints(state: &TuiState) -> Vec<(&'static str, &'static str)> {
94    let context = if state.is_input_active() {
95        Context::Input
96    } else if state.overlay.is_some() {
97        Context::Overlay
98    } else {
99        Context::Browse
100    };
101    let mut hints = bindings()
102        .iter()
103        .filter(|binding| binding.context == context)
104        .filter_map(|binding| {
105            metadata()
106                .iter()
107                .find(|item| item.id == binding.action)
108                .map(|item| (key_label(binding.chord), item.label))
109        })
110        .collect::<Vec<_>>();
111    if !hints
112        .iter()
113        .any(|(key, label)| *key == "Ctrl-C" && *label == "quit")
114    {
115        hints.insert(0, ("Ctrl-C", "quit"));
116    }
117    hints
118}

120/// Formats a key for help and hints.
121/// - `chord`: Key chord to format.
122///
123/// # Returns
124/// Human-readable key label.
125#[must_use]
126pub fn key_label(chord: KeyChord) -> &'static str {
127    match (chord.code, chord.modifiers) {
128        (
129            KeyCode::Char('c'),
130            Modifiers {
131                control: true,
132                alt: false,
133                shift: false,
134            },
135        ) => "Ctrl-C",
136        (KeyCode::Esc, _) => "Esc",
137        (
138            KeyCode::Tab,
139            Modifiers {
140                control: false,
141                alt: false,
142                shift: true,
143            },
144        ) => "Shift-Tab",
145        (KeyCode::Tab, _) => "Tab",
146        (KeyCode::Enter, _) => "Enter",
147        (KeyCode::Backspace, _) => "Backspace",
148        (KeyCode::Help, _) => "?",
149        (KeyCode::Char(':'), _) => ":",
150        (KeyCode::Char('/'), _) => "/",
151        (KeyCode::Char('r'), _) => "r",
152        (KeyCode::Char('q'), _) => "q",
153        (KeyCode::Up, _) => "Up",
154        (KeyCode::Down, _) => "Down",
155        (KeyCode::Left, _) => "Left",
156        (KeyCode::Right, _) => "Right",
157        _ => "key",
158    }

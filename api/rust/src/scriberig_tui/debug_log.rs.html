12/// Owns the optional TUI diagnostic log file.
13///
14/// Public members: None.
15///
16/// Private members
17/// - `file`: Append-only diagnostic destination.
18/// - `path`: Destination recorded at startup.
19pub(crate) struct TuiDebugLog {
20    /// Append-only diagnostic destination.
21    file: File,
22    /// Destination recorded at startup.
23    path: PathBuf,
24}
25
26impl TuiDebugLog {
27    /// Opens an explicit log path or a timestamped path under the TUI log folder.
28    ///
29    /// - `requested_path`: Optional caller-selected log destination.
30    ///
31    /// The initialized log writer.
32    ///
33    /// # Returns
34    /// The initialized diagnostic writer.
35    ///
36    /// # Errors
37    /// Returns an I/O error when its parent directory or file cannot be created.
38    pub(crate) fn open(requested_path: Option<&Path>) -> io::Result<Self> {
39        let path = requested_path.map_or_else(default_path, Path::to_path_buf);
40        if let Some(parent) = path.parent() {
41            if !parent.as_os_str().is_empty() {
42                fs::create_dir_all(parent)?;
43            }
44        }
45        let file = OpenOptions::new().create(true).append(true).open(&path)?;
46        let mut log = Self { file, path };
47        writeln!(
48            log.file,
49            "time_ms={} event=debug-log-start path={}",
50            timestamp_millis(),
51            log.path.display()
52        )?;
53        log.file.flush()?;
54        Ok(log)
55    }
56
57    /// Records state after an asynchronous application result is reduced.
58    ///
59    /// - `message`: Safe description of the returned runtime message.
60    /// - `state`: Resulting TUI state, summarized without conversation content.
61    ///
62    /// # Returns
63    /// Unit after the message has been recorded.
64    ///
65    /// # Errors
66    /// Returns an I/O error when the diagnostic destination cannot be written.
67    pub(crate) fn record_message(&mut self, message: &str, state: &TuiState) -> io::Result<()> {
68        writeln!(
69            self.file,
70            "time_ms={} event=runtime-message message={message} draft_bytes={} pending={} overlay={} model_library={} operation_error={} error_present={}",
71            timestamp_millis(),
72            state.conversation_composer.draft.len(),
73            state.conversation_composer.pending,
74            overlay_label(state),
75            model_library_label(&state.model_library),
76            operation_error_label(state),
77            state.conversation_composer.error.is_some(),
78        )?;
79        self.file.flush()
80    }
81}

83/// Returns the current Unix timestamp in milliseconds.
84///
85/// Unix milliseconds, or zero when the system clock predates the Unix epoch.
86/// # Returns
87/// The operation result.
88fn timestamp_millis() -> u128 {
89    SystemTime::now()
90        .duration_since(UNIX_EPOCH)
91        .unwrap_or_default()
92        .as_millis()
93}

95/// Resolves a timestamped log path using UTC calendar time.
96///
97/// Timestamped path under `playground/tui-logs`.
98/// # Returns
99/// The operation result.
100fn default_path() -> PathBuf {
101    let seconds = SystemTime::now()
102        .duration_since(UNIX_EPOCH)
103        .unwrap_or_default()
104        .as_secs();
105    let days = seconds / 86_400;
106    let day_seconds = seconds % 86_400;
107    let (year, month, day) = civil_date(i64::try_from(days).unwrap_or(i64::MAX));
108    let hour = day_seconds / 3_600;
109    let minute = day_seconds % 3_600 / 60;
110    let second = day_seconds % 60;
111    PathBuf::from(format!(
112        "playground/tui-logs/{year:04}{month:02}{day:02}-{hour:02}{minute:02}{second:02}.log"
113    ))
114}

116/// Converts days since the Unix epoch to a Gregorian calendar date.
117///
118/// - `days`: Whole UTC days since 1970-01-01.
119///
120/// Year, month, and day components.
121/// # Returns
122/// The operation result.
123fn civil_date(days: i64) -> (i64, i64, i64) {
124    let shifted = days + 719_468;
125    let era = shifted.div_euclid(146_097);
126    let day_of_era = shifted - era * 146_097;
127    let year_of_era =
128        (day_of_era - day_of_era / 1_460 + day_of_era / 36_524 - day_of_era / 146_096) / 365;
129    let mut year = year_of_era + era * 400;
130    let day_of_year = day_of_era - (365 * year_of_era + year_of_era / 4 - year_of_era / 100);
131    let month_prime = (5 * day_of_year + 2) / 153;
132    let day = day_of_year - (153 * month_prime + 2) / 5 + 1;
133    let month = month_prime + if month_prime < 10 { 3 } else { -9 };
134    year += i64::from(month <= 2);
135    (year, month, day)
136}

138/// Describes the current overlay without exposing its query text.
139///
140/// - `state`: Current TUI state.
141///
142/// Overlay name or `none`.
143/// # Returns
144/// The operation result.
145fn overlay_label(state: &TuiState) -> &'static str {
146    match state.overlay {
147        None => "none",
148        Some(crate::overlay::Overlay::Help) => "help",
149        Some(crate::overlay::Overlay::SessionList(_)) => "session-list",
150        Some(crate::overlay::Overlay::Palette(_)) => "palette",
151        Some(crate::overlay::Overlay::Input(_)) => "input",
152        Some(crate::overlay::Overlay::Confirm(_)) => "confirm",
153        Some(crate::overlay::Overlay::Composer(_)) => "composer",
154    }
155}

157/// Describes model-library state and enabled profile counts.
158///
159/// - `state`: Credential-safe model-library loading state.
160///
161/// Safe loading-state label.
162/// # Returns
163/// The operation result.
164fn model_library_label(
165    state: &crate::inspector::LoadState<scriberig_protocol::ModelLibraryResponse>,
166) -> String {
167    match state {
168        crate::inspector::LoadState::NotRequested => "not-requested".to_owned(),
169        crate::inspector::LoadState::Loading { .. } => "loading".to_owned(),
170        crate::inspector::LoadState::Failed { category, .. } => format!("failed:{category}"),
171        crate::inspector::LoadState::Ready(library) => {
172            format!("ready:models={}", library.models.len())
173        }
174    }
175}

177/// Returns the latest bounded, sanitized runtime error diagnostic.
178///
179/// - `state`: Current TUI state.
180///
181/// User-facing diagnostic or `none`.
182/// # Returns
183/// The operation result.
184fn operation_error_label(state: &TuiState) -> String {
185    let diagnostic = state
186        .operation_error
187        .as_ref()
188        .map(|(category, message)| format!("{category}: {message}"))
189        .or_else(|| state.conversation_composer.error.clone());
190    diagnostic.map_or_else(
191        || "none".to_owned(),
192        |message| {
193            crate::inspector::sanitize(&message)
194                .chars()
195                .take(512)
196                .collect()
197        },
198    )
199}

201#[cfg(test)]
202mod tests {
203    use std::{fs, path::PathBuf};
204
205    use crate::state::TuiState;
206
207    use super::{TuiDebugLog, default_path, timestamp_millis};
208
209    /// Verifies nested log directories are created and text content is redacted.
210    #[test]
211    fn creates_log_parent_and_records_content_safe_runtime_diagnostics() {
212        let timestamp_path = default_path();
213        let timestamp = timestamp_path
214            .file_name()
215            .and_then(|name| name.to_str())
216            .and_then(|name| name.strip_suffix(".log"))
217            .expect("timestamped log name");
218        let directory = PathBuf::from("playground/agentic-work")
219            .join(format!("{timestamp} tui log {}", std::process::id()));
220        let log_path = directory.join("nested/session.log");
221        let mut state = TuiState::default();
222        state.conversation_composer.draft = "private prompt text".to_owned();
223        state.conversation_composer.error = Some("provider request failed".to_owned());
224        {
225            let mut log = TuiDebugLog::open(Some(&log_path)).expect("open nested log");
226            log.record_message("composer-submit", &state)
227                .expect("record runtime message");
228        }
229
230        let contents = fs::read_to_string(&log_path).expect("read log");
231        assert!(contents.contains("event=runtime-message message=composer-submit"));
232        assert!(contents.contains("draft_bytes=19"));
233        assert!(contents.contains("operation_error=provider request failed"));
234        assert!(!contents.contains("private prompt text"));
235        assert!(timestamp_millis() > 0);
236        fs::remove_dir_all(directory).expect("remove test log directory");
237    }

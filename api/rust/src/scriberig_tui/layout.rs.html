1//! Computes responsive shell regions without performing rendering or I/O.
2
3use ratatui::layout::Rect;
4
5/// Responsive body layout selected from the terminal width.
6#[derive(Clone, Copy, Debug, Eq, PartialEq)]
7pub enum LayoutClass {
8    /// Explorer, primary, and inspector are visible.
9    Wide,
10    /// Explorer and primary are visible.
11    Medium,
12    /// Only the primary view is visible.
13    Narrow,
14    /// The terminal is too small for the shell and gets a clear message.
15    TooSmall,
16}
17
18/// All shell regions for one frame.
19#[derive(Clone, Copy, Debug, Eq, PartialEq)]
20pub struct ShellLayout {
21    /// The selected responsive width class.
22    pub class: LayoutClass,
23    /// The context bar rectangle.
24    pub context: Rect,
25    /// The explorer rectangle, when visible.
26    pub explorer: Option<Rect>,
27    /// The primary rectangle.
28    pub primary: Rect,
29    /// The inspector rectangle, when visible.
30    pub inspector: Option<Rect>,
31    /// The status rectangle.
32    pub status: Rect,
33}
34
35/// Computes shell regions for a terminal rectangle.
36#[must_use]
37/// - `area`: Argument named `area`.
38///
39/// # Returns
40/// The value produced by the declaration.
41pub fn shell_layout(area: Rect) -> ShellLayout {
42    let vertical = ratatui::layout::Layout::vertical([
43        ratatui::layout::Constraint::Length(2),
44        ratatui::layout::Constraint::Min(1),
45        ratatui::layout::Constraint::Length(1),
46    ])
47    .split(area);
48    let class = match area.width {
49        120.. => LayoutClass::Wide,
50        80..=119 => LayoutClass::Medium,
51        20..=79 => LayoutClass::Narrow,
52        _ => LayoutClass::TooSmall,
53    };
54    let body = vertical[1];
55    let (explorer, primary, inspector) = match class {
56        LayoutClass::Wide => {
57            let columns = ratatui::layout::Layout::horizontal([
58                ratatui::layout::Constraint::Percentage(25),
59                ratatui::layout::Constraint::Percentage(50),
60                ratatui::layout::Constraint::Percentage(25),
61            ])
62            .split(body);
63            (Some(columns[0]), columns[1], Some(columns[2]))
64        }
65        LayoutClass::Medium => {
66            let columns = ratatui::layout::Layout::horizontal([
67                ratatui::layout::Constraint::Percentage(30),
68                ratatui::layout::Constraint::Percentage(70),
69            ])
70            .split(body);
71            (Some(columns[0]), columns[1], None)
72        }
73        LayoutClass::Narrow | LayoutClass::TooSmall => (None, body, None),
74    };
75    ShellLayout {
76        class,
77        context: vertical[0],
78        explorer,
79        primary,
80        inspector,
81        status: vertical[2],
82    }
83}

85#[cfg(test)]
86mod tests {
87    use ratatui::layout::Rect;
88
89    use super::{LayoutClass, shell_layout};
90
91    /// Verifies the three documented responsive width classes.
92    #[test]
93    fn selects_required_width_classes() {
94        assert_eq!(
95            shell_layout(Rect::new(0, 0, 140, 40)).class,
96            LayoutClass::Wide
97        );
98        assert_eq!(
99            shell_layout(Rect::new(0, 0, 100, 30)).class,
100            LayoutClass::Medium
101        );
102        assert_eq!(
103            shell_layout(Rect::new(0, 0, 72, 24)).class,
104            LayoutClass::Narrow
105        );
106    }
107
108    /// Verifies very small dimensions remain bounded and safe.
109    #[test]
110    fn handles_small_terminal() {
111        let layout = shell_layout(Rect::new(0, 0, 10, 3));
112        assert_eq!(layout.class, LayoutClass::TooSmall);
113    }

6use scriberig_agent_runtime::AgentRuntime;

8/// Runs one semantic project agent and stores its normal candidate review state.
9///
10/// # Parameters
11/// - `adapter`: Application operation adapter holding project workflow state.
12/// - `context`: Trusted runtime context for semantic tools.
13/// - `entity`: Selected source entity.
14/// - `snapshot`: Exact candidate base snapshot.
15/// - `prompt`: User-authored change instruction.
16///
17/// # Returns
18/// The normal application candidate review summary.
19///
20/// # Errors
21/// Returns a structured agent, validation, or storage failure.
22pub(super) fn project_agent<B: ApplicationBackend>(
23    adapter: &ApplicationOperationAdapter<B>,
24    context: &ApplicationContext,
25    entity: EntityId,
26    snapshot: SnapshotId,
27    prompt: &str,
28) -> Result<ApplicationResult, ApplicationError> {
29    let projection = adapter.project_projection.as_ref().ok_or_else(|| {
30        ApplicationError::UnsupportedOperation {
31            operation: "project agent".to_owned(),
32        }
33    })?;
34    let (path, bytes) =
35        projection
36            .sources
37            .get(&entity)
38            .ok_or_else(|| ApplicationError::NotFound {
39                subject: entity.to_string(),
40            })?;
41    let replacement =
42        String::from_utf8_lossy(bytes).replace("left + right", "left.saturating_add(right)");
43    let tool = ProjectAgentTool {
44        projection,
45        entity,
46        snapshot,
47    };
48    let mut runtime = AgentRuntime::new(golden_provider(entity, snapshot), tool, 4);
49    let result = runtime
50        .run(context, prompt, &std::sync::atomic::AtomicBool::new(false))
51        .map_err(|error| ApplicationError::OperationFailed {
52            operation: "project_agent".to_owned(),
53            message: error.to_string(),
54        })?;
55    let candidate = result
56        .messages
57        .iter()
58        .rev()
59        .find_map(|message| match message {
60            scriberig_agent_runtime::AgentMessage::Tool(value)
61                if value.object_type.as_deref() == Some("migration_candidate") =>
62            {
63                value.handles.first().cloned()
64            }
65            _ => None,
66        })
67        .ok_or_else(|| ApplicationError::OperationFailed {
68            operation: "project_agent".to_owned(),
69            message: "agent did not create a candidate".to_owned(),
70        })?;
71    let candidate_value = projection
72        .workflow
73        .propose_candidate(projection.project, entity, replacement)
74        .map_err(|error| ApplicationError::ValidationFailure {
75            message: error.to_string(),
76        })?;
77    if candidate != candidate_value.id {
78        return Err(ApplicationError::Internal {
79            message: "agent candidate identity changed between tool calls".to_owned(),
80        });
81    }
82    adapter
83        .project_candidates
84        .lock()
85        .map_err(|_| ApplicationError::Internal {
86            message: "project candidate lock poisoned".to_owned(),
87        })?
88        .insert(candidate_value.id.clone(), candidate_value.clone());
89    let validation = match &candidate_value.validation {
90        scriberig_project::CandidateValidation::Pending => scriberig_app::ValidationStatus::Skipped,
91        scriberig_project::CandidateValidation::Passed => scriberig_app::ValidationStatus::Passed,
92        scriberig_project::CandidateValidation::Failed { .. } => {
93            scriberig_app::ValidationStatus::Failed
94        }
95    };
96    Ok(ApplicationResult::Apply(scriberig_app::OperationSummary {
97        handle: candidate_value.id,
98        snapshot: Some(snapshot),
99        revision: None,
100        branch: None,
101        diff: None,
102        history: None,
103        rendered: None,
104        validation: Some(validation),
105        validation_profile: None,
106        validation_diagnostics: Vec::new(),
107        conflicts: vec![format!(
108            "scope: entity={entity} snapshot={snapshot}; {path}: add | old: left + right | new: left.saturating_add(right)"
109        )],
110    }))
111}

28/// Runs one semantic project agent and stores its normal candidate review state.
29///
30/// - `adapter`: Application operation adapter holding project workflow state.
31/// - `context`: Trusted runtime context for semantic tools.
32/// - `entity`: Selected source entity.
33/// - `snapshot`: Exact candidate base snapshot.
34/// - `prompt`: User-authored change instruction.
35///
36/// The normal application candidate review summary.
37///
38/// Returns a structured agent, validation, or storage failure.
39pub(super) fn project_agent<B: ApplicationBackend>(
40    adapter: &ApplicationOperationAdapter<B>,
41    context: &ApplicationContext,
42    entity: EntityId,
43    snapshot: SnapshotId,
44    prompt: &str,
45) -> Result<ApplicationResult, ApplicationError> {
46    let projection = adapter.project_projection.as_ref().ok_or_else(|| {
47        ApplicationError::UnsupportedOperation {
48            operation: "project agent".to_owned(),
49        }
50    })?;
51    let (path, bytes) =
52        projection
53            .sources
54            .get(&entity)
55            .ok_or_else(|| ApplicationError::NotFound {
56                subject: entity.to_string(),
57            })?;
58    let replacement =
59        String::from_utf8_lossy(bytes).replace("left + right", "left.saturating_add(right)");
60    let branch = EntityId::new();
61    let project_context = BranchExecutionContext::new(
62        projection.project.to_string(),
63        branch.to_string(),
64        "project candidate proposal",
65        prompt,
66    )
67    .with_project_scope(ProjectSemanticScope {
68        project: projection.project.to_string(),
69        snapshot: snapshot.to_string(),
70        root: projection.project.to_string(),
71        selection: Some(entity.to_string()),
72    });
73    let compiled_context = ContextCompiler::new(ContextPolicy::default()).compile(&project_context);
74    let (provider, model) = faux_provider(entity, snapshot, &replacement);
75    let candidate_id: Arc<Mutex<Option<String>>> = Arc::new(Mutex::new(None));
76    let tools = project_tools(
77        projection.clone(),
78        entity,
79        snapshot,
80        branch,
81        Arc::clone(&adapter.project_candidates),
82        Arc::clone(&candidate_id),
83        context.clone(),
84    )?;
85    let invocation = UpstreamBranchInvocation {
86        branch,
87        provider,
88        model,
89        request_secret: None,
90        bearer: false,
91        applied_settings: None,
92        provider_profile: Some("scriberig-deterministic-project-agent".to_owned()),
93        thinking_level: rpi_ai::ThinkingLevel::Off,
94        context: compiled_context,
95        tools,
96    };
97    run_upstream_branch_invocation_blocking(invocation, prompt).map_err(|error| {
98        ApplicationError::OperationFailed {
99            operation: "project_agent".to_owned(),
100            message: error.error.to_string(),
101        }
102    })?;
103    let candidate_id = candidate_id
104        .lock()
105        .map_err(|_| ApplicationError::Internal {
106            message: "project candidate identity lock poisoned".to_owned(),
107        })?
108        .clone()
109        .ok_or_else(|| ApplicationError::OperationFailed {
110            operation: "project_agent".to_owned(),
111            message: "agent did not create a candidate".to_owned(),
112        })?;
113    let candidate_value = projection
114        .workflow
115        .candidate(&candidate_id)
116        .map_err(|error| ApplicationError::ValidationFailure {
117            message: error.to_string(),
118        })?
119        .ok_or_else(|| ApplicationError::Internal {
120            message: "created project candidate disappeared".to_owned(),
121        })?;
122    let validation = match &candidate_value.validation {
123        scriberig_project::CandidateValidation::Pending => scriberig_app::ValidationStatus::Skipped,
124        scriberig_project::CandidateValidation::Passed => scriberig_app::ValidationStatus::Passed,
125        scriberig_project::CandidateValidation::Failed { .. } => {
126            scriberig_app::ValidationStatus::Failed
127        }
128    };
129    Ok(ApplicationResult::Apply(scriberig_app::OperationSummary {
130        handle: candidate_value.id,
131        snapshot: Some(snapshot),
132        revision: None,
133        branch: None,
134        diff: None,
135        history: None,
136        rendered: None,
137        validation: Some(validation),
138        validation_profile: None,
139        validation_diagnostics: Vec::new(),
140        review_details: vec![format!(
141            "scope: entity={entity} snapshot={snapshot}; {path}: add | old: left + right | new: left.saturating_add(right)"
142        )],
143        conflicts: Vec::new(),
144    }))
145}

147/// Creates branch-scoped inspect and candidate tools for the project flow.
148///
149/// - `projection`: Snapshot-bound project source projection.
150/// - `entity`: Selected source entity.
151/// - `snapshot`: Exact candidate base snapshot.
152/// - `branch`: Canonical branch identity for this execution.
153/// - `candidate_store`: TUI-owned candidate review cache.
154/// - `candidate_id`: Sink receiving the created candidate identity.
155/// - `context`: Trusted `ScribeRig` application request context.
156///
157/// The upstream semantic tools authorized for this branch.
158///
159/// Returns a safe application error when tool schemas cannot be bound.
160/// # Returns
161/// Branch-scoped inspect and candidate tools.
162///
163/// # Errors
164/// Returns an application error when schemas cannot be bound.
165fn project_tools(
166    projection: ProjectProjection,
167    entity: EntityId,
168    snapshot: SnapshotId,
169    branch: EntityId,
170    candidate_store: Arc<
171        Mutex<std::collections::HashMap<String, scriberig_project::ProjectCandidate>>,
172    >,
173    candidate_id: Arc<Mutex<Option<String>>>,
174    context: ApplicationContext,
175) -> Result<Vec<Arc<SemanticAgentTool>>, ApplicationError> {
176    let inspect = definition(
177        "inspect",
178        "Inspect the selected semantic entity.",
179        json!({"type": "object", "required": ["entity", "snapshot"], "properties": {
180            "entity": {"type": "string"}, "snapshot": {"type": "string"},
181            "detail": {"type": "string"}
182        }}),
183    );
184    let propose = definition(
185        "project_propose",
186        "Create a reviewed source candidate.",
187        json!({"type": "object", "required": ["text"], "properties": {
188            "text": {"type": "string"}
189        }}),
190    );
191    let catalog = ToolCatalog::new(vec![
192        descriptor(inspect.clone(), ToolCategory::SemanticRead, false),
193        descriptor(propose.clone(), ToolCategory::SemanticMutation, true),
194    ]);
195    let tool = Arc::new(ProjectAgentTool {
196        projection,
197        entity,
198        snapshot,
199    });
200    let inspect_tool = inspect_tool(&inspect, branch, &catalog, &tool, &context)?;
201    let propose_tool = candidate_tool(
202        &propose,
203        branch,
204        catalog,
205        &tool,
206        candidate_store,
207        candidate_id,
208        context,
209    )?;
210    Ok(vec![Arc::new(inspect_tool), Arc::new(propose_tool)])
211}

213/// Creates the snapshot-bound inspect operation for a project branch.
214///
215/// - `definition`: Provider-visible inspect schema.
216/// - `branch`: Canonical branch identity.
217/// - `catalog`: Trusted branch tool catalog.
218/// - `tool`: Project-backed inspect adapter.
219/// - `context`: Trusted application request context.
220///
221/// The validated inspect tool.
222///
223/// # Returns
224/// An inspect tool bound to the exact branch and snapshot.
225///
226/// # Errors
227/// Returns an application error when the inspect tool cannot be bound.
228fn inspect_tool(
229    definition: &ProviderTool,
230    branch: EntityId,
231    catalog: &ToolCatalog,
232    tool: &Arc<ProjectAgentTool>,
233    context: &ApplicationContext,
234) -> Result<SemanticAgentTool, ApplicationError> {
235    SemanticAgentTool::new(
236        definition,
237        "Inspect selected source",
238        branch,
239        catalog.clone(),
240        CapabilityContext::default(),
241        {
242            let tool = Arc::clone(tool);
243            let context = context.clone();
244            move |_id, arguments, _signal, _update| {
245                let tool = Arc::clone(&tool);
246                let context = context.clone();
247                async move {
248                    let entity = parse_id::<EntityId>(&arguments, "entity")?;
249                    let snapshot = parse_id::<SnapshotId>(&arguments, "snapshot")?;
250                    let result = tool
251                        .invoke(
252                            &context,
253                            ToolInput {
254                                entity,
255                                snapshot: Some(snapshot),
256                                detail: ToolDetail::Detailed,
257                            },
258                        )
259                        .map_err(|_| AgentError::tool("semantic inspect failed"))?;
260                    Ok(AgentToolResult::text(format!(
261                        "{}: {} {}",
262                        result.entity,
263                        result.summary.unwrap_or_default(),
264                        result.handles.join(", ")
265                    )))
266                }
267            }
268        },
269    )
270    .map_err(|_| ApplicationError::Internal {
271        message: "project inspect tool is not visible to the branch".to_owned(),
272    })
273}

275/// Creates the candidate-writing operation and records its review identity.
276///
277/// - `definition`: Provider-visible proposal schema.
278/// - `branch`: Canonical branch identity.
279/// - `catalog`: Trusted branch tool catalog.
280/// - `tool`: Project-backed proposal adapter.
281/// - `candidate_store`: TUI cache for normal candidate review.
282/// - `candidate_id`: Sink receiving the created candidate identity.
283/// - `context`: Trusted application request context.
284///
285/// The validated candidate proposal tool.
286///
287/// # Returns
288/// A candidate tool bound to the exact branch and review cache.
289///
290/// # Errors
291/// Returns an application error when the proposal tool cannot be bound.
292fn candidate_tool(
293    definition: &ProviderTool,
294    branch: EntityId,
295    catalog: ToolCatalog,
296    tool: &Arc<ProjectAgentTool>,
297    candidate_store: Arc<
298        Mutex<std::collections::HashMap<String, scriberig_project::ProjectCandidate>>,
299    >,
300    candidate_id: Arc<Mutex<Option<String>>>,
301    context: ApplicationContext,
302) -> Result<SemanticAgentTool, ApplicationError> {
303    SemanticAgentTool::new(
304        definition,
305        "Create project candidate",
306        branch,
307        catalog,
308        CapabilityContext::default(),
309        {
310            let tool = Arc::clone(tool);
311            move |_id, arguments, _signal, _update| {
312                let tool = Arc::clone(&tool);
313                let candidate_store = Arc::clone(&candidate_store);
314                let candidate_id = Arc::clone(&candidate_id);
315                let context = context.clone();
316                async move {
317                    let result = tool
318                        .invoke_named(&context, "project_propose", &arguments)
319                        .map_err(|_| AgentError::tool("candidate proposal failed"))?;
320                    let id = result.handles.first().cloned().ok_or_else(|| {
321                        AgentError::tool("candidate operation returned no handle")
322                    })?;
323                    let candidate = tool
324                        .projection
325                        .workflow
326                        .candidate(&id)
327                        .map_err(|_| AgentError::tool("candidate lookup failed"))?
328                        .ok_or_else(|| AgentError::tool("candidate disappeared"))?;
329                    candidate_store
330                        .lock()
331                        .map_err(|_| AgentError::tool("candidate cache unavailable"))?
332                        .insert(id.clone(), candidate);
333                    candidate_id
334                        .lock()
335                        .map_err(|_| AgentError::tool("candidate result unavailable"))?
336                        .replace(id.clone());
337                    Ok(AgentToolResult::text(format!("candidate created: {id}")))
338                }
339            }
340        },
341    )
342    .map_err(|_| ApplicationError::Internal {
343        message: "project candidate tool is not visible to the branch".to_owned(),
344    })
345}

347/// Creates a stable schema descriptor for the deterministic project tools.
348///
349/// - `name`: Stable provider-visible operation name.
350/// - `description`: Human-readable operation purpose.
351/// - `parameters`: JSON schema validated by the upstream agent.
352///
353/// The trusted provider-facing tool definition.
354/// # Returns
355/// A provider schema for one trusted semantic operation.
356fn definition(name: &str, description: &str, parameters: Value) -> ProviderTool {
357    ProviderTool {
358        name: name.to_owned(),
359        description: description.to_owned(),
360        parameters: parameters.into(),
361        constrained_sampling: None,
362    }
363}

365/// Adds one trusted operation definition to the branch catalog.
366///
367/// - `definition`: Provider-facing operation schema.
368/// - `category`: Semantic operation category.
369/// - `state_changing`: Whether the operation writes a candidate.
370///
371/// The full branch-visible tool descriptor.
372/// # Returns
373/// A trusted descriptor with `ScribeRig` semantic category and policy.
374fn descriptor(
375    definition: ProviderTool,
376    category: ToolCategory,
377    state_changing: bool,
378) -> ToolDescriptor {
379    ToolDescriptor {
380        definition,
381        category,
382        result_shape: "bounded semantic result".to_owned(),
383        invocation_label: None,
384        state_changing,
385        execution_mode: ToolExecutionMode::Atomic,
386        required_capability: None,
387    }
388}

390/// Parses one provider argument as a stable semantic identifier.
391///
392/// - `arguments`: Provider-supplied JSON object.
393/// - `name`: Required string field.
394///
395/// The parsed identifier.
396///
397/// # Errors
398/// Returns an upstream safe tool error when the value is invalid.
399fn parse_id<T: std::str::FromStr>(arguments: &Value, name: &str) -> Result<T, AgentError> {
400    arguments
401        .get(name)
402        .and_then(Value::as_str)
403        .and_then(|value| value.parse().ok())
404        .ok_or_else(|| AgentError::tool("invalid semantic tool identifier"))
405}

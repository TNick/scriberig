68/// Lists remote snapshot entities for the TUI explorer.
69///
70/// - `client`: Remote protocol client.
71/// - `entity`: Collection owner to exclude from the result.
72/// - `snapshot`: Exact snapshot context.
73/// - `continuation`: Optional page continuation.
74///
75/// A bounded Explorer page.
76///
77/// Returns a sanitized remote application error.
78pub(crate) fn list_explorer<T: scriberig_protocol::ProtocolTransport>(
79    client: &scriberig_client::Client<T>,
80    entity: EntityId,
81    snapshot: scriberig_core::SnapshotId,
82    continuation: Option<scriberig_query::ContinuationToken>,
83) -> Result<crate::explorer::ExplorerPage, ApplicationError> {
84    let result = client
85        .list(scriberig_query::ListRequest {
86            subject: scriberig_query::EntityInSnapshot { entity, snapshot },
87            collection: scriberig_query::CollectionRef {
88                owner: entity,
89                collection_type: "snapshot.entities".to_owned(),
90            },
91            page: scriberig_query::PageRequest {
92                continuation,
93                ..scriberig_query::PageRequest::default()
94            },
95            detail: scriberig_query::DetailLevel::Compact,
96        })
97        .map_err(remote_error)?;
98    let rows = result
99        .page
100        .items
101        .into_iter()
102        .filter_map(|entry| match entry.target {
103            scriberig_query::TargetRef::Snapshot(target) => Some(crate::explorer::ExplorerRow {
104                selection: crate::selection::EntitySelection {
105                    entity: target.entity,
106                    revision: None,
107                    snapshot: Some(target.snapshot),
108                },
109                label: entry.label.unwrap_or_else(|| target.entity.to_string()),
110                kind: entry.entity_type.map(|value| value.to_string()),
111                inspectable: true,
112            }),
113            scriberig_query::TargetRef::Revision(_) => None,
114        })
115        .filter(|row| row.selection.entity != entity)
116        .collect();
117    Ok(crate::explorer::ExplorerPage {
118        rows,
119        next: result.page.next,
120    })
121}

123/// Finds remote entities in an exact snapshot.
124///
125/// - `client`: Remote protocol client.
126/// - `snapshot`: Exact discovery scope.
127/// - `query`: Search text.
128///
129/// A bounded TUI discovery model.
130///
131/// Returns a sanitized remote application error.
132/// Converts a compact protocol session response to the shared session model.
133///
134/// - `response`: Protocol session tree response.
135///
136/// Session value accepted by the existing TUI projection.
137///
138/// Returns an internal error when a remote reference is malformed.
139/// - `crate`: Internal visibility marker.
140/// # Returns
141/// The operation result.
142/// # Errors
143/// Returns an error when the operation fails.
144pub(crate) fn session_from_response(
145    response: scriberig_protocol::SessionReadResponse,
146) -> Result<scriberig_session::Session, ApplicationError> {
147    let id = parse_entity(&response.session.id)?;
148    let branches = response
149        .branches
150        .into_iter()
151        .map(|value| {
152            let branch_id = parse_entity(&value.id)?;
153            let mut branch =
154                scriberig_session::Branch::new(branch_id, value.title).map_err(|error| {
155                    ApplicationError::Internal {
156                        message: format!("remote branch was invalid: {error}"),
157                    }
158                })?;
159            branch.parent = value.parent.as_deref().map(parse_entity).transpose()?;
160            branch.children = value
161                .children
162                .iter()
163                .map(|child| parse_entity(child))
164                .collect::<Result<Vec<_>, _>>()?;
165            let status = match value.status {
166                scriberig_protocol::SessionBranchStatus::Active => {
167                    scriberig_session::BranchStatus::Active
168                }
169                scriberig_protocol::SessionBranchStatus::Completed => {
170                    scriberig_session::BranchStatus::Completed
171                }
172                scriberig_protocol::SessionBranchStatus::Abandoned => {
173                    scriberig_session::BranchStatus::Abandoned
174                }
175            };
176            branch
177                .transition(status)
178                .map_err(|error| ApplicationError::Internal {
179                    message: format!("remote branch status was invalid: {error}"),
180                })?;
181            branch.turns = value
182                .turns
183                .into_iter()
184                .map(|turn| {
185                    Ok(scriberig_session::Turn {
186                        id: parse_entity(&turn.id)?,
187                        ordinal: turn.ordinal,
188                        human: turn.has_human_input.then(|| scriberig_session::HumanInput {
189                            text: String::new(),
190                        }),
191                        agent: turn
192                            .has_agent_output
193                            .then(|| scriberig_session::AgentOutput {
194                                text: turn.agent_output.unwrap_or_default(),
195                            }),
196                        tool_invocation: turn
197                            .tool_invocation
198                            .as_deref()
199                            .map(parse_entity)
200                            .transpose()?,
201                        tool_result: turn.tool_result.as_deref().map(parse_entity).transpose()?,
202                    })
203                })
204                .collect::<Result<Vec<_>, ApplicationError>>()?;
205            branch.result = value
206                .result
207                .map(|result| {
208                    Ok(scriberig_session::BranchResult {
209                        id: parse_entity(&result.id)?,
210                        branch: branch_id,
211                        status: scriberig_session::BranchStatus::Completed,
212                        summary: None,
213                        result_kind: None,
214                        candidate_context: None,
215                        references: result
216                            .references
217                            .iter()
218                            .map(|reference| parse_entity(reference))
219                            .collect::<Result<Vec<_>, _>>()?,
220                        result_handles: result.result_handles,
221                        touched: Vec::new(),
222                        produced_changes: Vec::new(),
223                        validations: Vec::new(),
224                        unresolved_questions: Vec::new(),
225                    })
226                })
227                .transpose()?;
228            remote_activity::restore_activity(&mut branch, value.activity)?;
229            Ok(branch)
230        })
231        .collect::<Result<Vec<_>, ApplicationError>>()?;
232    let plans = response
233        .plans
234        .into_iter()
235        .map(plan_mapping::from_response)
236        .collect::<Result<Vec<_>, ApplicationError>>()?;
237    Ok(scriberig_session::Session {
238        id,
239        branches,
240        goals: Vec::new(),
241        tasks: Vec::new(),
242        plans,
243        execution: execution_from_response(response.execution),
244    })
245}

247/// Parses the structural parent identity carried by the current remote DTO.
248///
249/// - `value`: Tagged `branch:`/`node:` parent or legacy branch identity.
250///
251/// The typed structural parent, or `None` when the field is empty.
252///
253/// Returns an internal error when the tagged identity is malformed.
254/// # Returns
255/// The operation result.
256/// # Errors
257/// Returns an error when the operation fails.
258fn parse_plan_parent(
259    value: &str,
260) -> Result<Option<scriberig_session::PlanNodeParent>, ApplicationError> {
261    if let Some(branch) = value.strip_prefix("branch:") {
262        return parse_entity(branch)
263            .map(scriberig_session::PlanNodeParent::Branch)
264            .map(Some);
265    }
266    if let Some(node) = value.strip_prefix("node:") {
267        return parse_entity(node)
268            .map(scriberig_session::PlanNodeParent::Node)
269            .map(Some);
270    }
271    if value.is_empty() {
272        return Ok(None);
273    }
274    parse_entity(value)
275        .map(scriberig_session::PlanNodeParent::Branch)
276        .map(Some)
277}

279/// Converts the public execution selection and history into the session owner.
280///
281/// - `selection`: Protocol selection carried by the session response.
282///
283/// Equivalent session-domain execution selection.
284/// # Returns
285/// The operation result.
286fn execution_from_response(
287    selection: scriberig_protocol::SessionExecutionSelection,
288) -> scriberig_session::SessionExecutionSelection {
289    scriberig_session::SessionExecutionSelection {
290        revision: selection.revision,
291        model_profile: selection.model_profile,
292        model: selection.model,
293        thinking_level: selection.thinking_level,
294        history: selection
295            .history
296            .into_iter()
297            .map(|change| scriberig_session::SessionExecutionChange {
298                revision: change.revision,
299                kind: change.kind,
300                summary: change.summary,
301                model: change.model,
302                thinking_level: change.thinking_level,
303            })
304            .collect(),
305    }
306}

344/// Converts client/protocol failures into safe application-boundary errors.
345///
346/// - `error`: Client or protocol failure.
347///
348/// A sanitized application error.
349/// - `crate`: Internal visibility marker.
350/// # Returns
351/// The operation result.
352pub(crate) fn remote_error(error: scriberig_client::ClientError) -> ApplicationError {
353    match error {
354        scriberig_client::ClientError::Transport(message) => {
355            ApplicationError::TransportUnavailable {
356                message: crate::inspector::sanitize(&message),
357            }
358        }
359        scriberig_client::ClientError::Encoding(message) => ApplicationError::ProtocolFailure {
360            category: "request_encoding".to_owned(),
361            message: crate::inspector::sanitize(&message),
362        },
363        scriberig_client::ClientError::Protocol(error) => ApplicationError::ProtocolFailure {
364            category: protocol_category(&error.code).to_owned(),
365            message: crate::inspector::sanitize(&error.message),
366        },
367        scriberig_client::ClientError::UnexpectedResponse => ApplicationError::ProtocolFailure {
368            category: "unexpected_response".to_owned(),
369            message: "remote response did not match the requested operation".to_owned(),
370        },
371    }
372}

374/// Converts a protocol error code to the stable presentation category.
375///
376/// - `code`: Protocol error code to classify.
377///
378/// Stable snake-case category text.
379/// - `crate`: Internal visibility marker.
380/// # Returns
381/// The operation result.
382pub(crate) fn protocol_category(code: &scriberig_protocol::ProtocolErrorCode) -> &'static str {
383    match code {
384        scriberig_protocol::ProtocolErrorCode::InvalidRequest => "invalid_request",
385        scriberig_protocol::ProtocolErrorCode::NotFound => "not_found",
386        scriberig_protocol::ProtocolErrorCode::SchemaViolation => "schema_violation",
387        scriberig_protocol::ProtocolErrorCode::StaleBaseSnapshot => "stale_base_snapshot",
388        scriberig_protocol::ProtocolErrorCode::SemanticConflict => "semantic_conflict",
389        scriberig_protocol::ProtocolErrorCode::ValidationFailure => "validation_failure",
390        scriberig_protocol::ProtocolErrorCode::PermissionDenied => "permission_denied",
391        scriberig_protocol::ProtocolErrorCode::UnsupportedOperation => "unsupported_operation",
392        scriberig_protocol::ProtocolErrorCode::UnsupportedCapability => "unsupported_capability",
393        scriberig_protocol::ProtocolErrorCode::Conflict => "conflict",
394        scriberig_protocol::ProtocolErrorCode::Ambiguous => "ambiguous",
395        scriberig_protocol::ProtocolErrorCode::SnapshotUnavailable => "snapshot_unavailable",
396        scriberig_protocol::ProtocolErrorCode::OperationFailed => "operation_failed",
397        scriberig_protocol::ProtocolErrorCode::ProjectionOutOfSync => "projection_out_of_sync",
398        scriberig_protocol::ProtocolErrorCode::OperationCancelled => "operation_cancelled",
399        scriberig_protocol::ProtocolErrorCode::Internal => "internal",
400        scriberig_protocol::ProtocolErrorCode::UnsupportedVersion => "unsupported_version",
401        scriberig_protocol::ProtocolErrorCode::CursorTooOld => "cursor_too_old",
402    }

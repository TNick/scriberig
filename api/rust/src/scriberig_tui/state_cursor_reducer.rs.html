8/// Reduces one cursor movement action.
9/// - `super`: Parent-module imports used by this helper.
10/// - `state`: TUI state to update.
11/// - `direction`: Cursor movement direction.
12///
13/// Effects produced by the movement.
14/// # Returns
15/// The operation result.
16pub(super) fn reduce_cursor(state: &mut TuiState, direction: Direction) -> Vec<Effect> {
17    if let Some(Overlay::Palette(palette)) = state.overlay.as_mut() {
18        let last = palette.filtered().len().saturating_sub(1);
19        palette.selected = match direction {
20            Direction::Previous => palette.selected.saturating_sub(1),
21            Direction::Next => palette.selected.saturating_add(1).min(last),
22        };
23    } else {
24        let last = cursor_limit(state);
25        state.cursor = match direction {
26            Direction::Previous => state.cursor.saturating_sub(1),
27            Direction::Next => state.cursor.saturating_add(1).min(last),
28        };
29        if state.primary_view == crate::selection::PrimaryView::Conversation {
30            state.conversation_cursor = state
31                .conversation
32                .as_ref()
33                .and_then(|model| model.rows.get(state.cursor))
34                .map(|row| row.source);
35            if state.conversation_browsing {
36                state.follow_coordinator = false;
37            }
38        }
39    }
40    vec![Effect::None]
41}

43/// Returns the last selectable row in the focused presentation.
44///
45/// - `state`: Current TUI presentation state.
46///
47/// The zero-based upper cursor bound for the active pane.
48/// # Returns
49/// The operation result.
50fn cursor_limit(state: &TuiState) -> usize {
51    match state.focus {
52        Focus::Explorer => state
53            .explorer
54            .root
55            .iter()
56            .count()
57            .saturating_add(state.explorer.rows.len())
58            .saturating_sub(1),
59        Focus::Primary => match state.primary_view {
60            crate::selection::PrimaryView::Conversation => state
61                .conversation
62                .as_ref()
63                .map_or(0, |model| model.rows.len().saturating_sub(1)),
64            crate::selection::PrimaryView::Explorer => state
65                .explorer
66                .rows
67                .len()
68                .saturating_add(usize::from(state.explorer.root.is_some()))
69                .saturating_sub(1),
70            crate::selection::PrimaryView::History => match &state.history {
71                LoadState::Ready(model) => model.rows.len().saturating_sub(1),
72                _ => 0,
73            },
74            crate::selection::PrimaryView::Find => match &state.find {
75                LoadState::Ready(model) => model.rows.len().saturating_sub(1),
76                _ => 0,
77            },
78            crate::selection::PrimaryView::Diff => match &state.diff {
79                LoadState::Ready(model) => model.rows.len().saturating_sub(1),
80                _ => 0,
81            },
82            crate::selection::PrimaryView::Session => match &state.session {
83                LoadState::Ready(model) => model.rows.len().saturating_sub(1),
84                _ => 0,
85            },
86            crate::selection::PrimaryView::Operations => state.operations.len().saturating_sub(1),
87            crate::selection::PrimaryView::Changes => match &state.review {
88                LoadState::Ready(model) => model.conflicts.len().saturating_sub(1),
89                _ => 0,
90            },
91            _ => 0,
92        },
93        Focus::Inspector => 0,
94    }
95}

97/// Advances focus in the requested direction when no overlay is active.
98/// - `super`: Parent-module imports used by this helper.
99/// - `state`: TUI state to update.
100/// - `reverse`: Whether to move backward.
101///
102/// No-op effect after updating focus.
103/// # Returns
104/// The operation result.
105pub(super) fn reduce_focus(state: &mut TuiState, reverse: bool) -> Vec<Effect> {
106    if state.overlay.is_none() {
107        state.focus = state.focus.next(reverse);
108    }
109    vec![Effect::None]

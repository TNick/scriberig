1//! Refreshes `ScribeRig` state after an upstream connection is restored.
2
3use crate::{action::Action, effect::Effect, state::ConnectionState};
4
5use super::super::TuiState;
6
7/// Refreshes active semantic, operation, and session projections.
8///
9/// - `state`: TUI state receiving restored connection status.
10///
11/// # Returns
12/// Refresh effects for the active selection and live agent handles.
13#[must_use]
14pub(super) fn reduce(state: &mut TuiState) -> Vec<Effect> {
15    state.connection = ConnectionState::Connected;
16    state.stale = true;
17    let mut effects = if state.selection.is_some() {
18        super::reduce(state, Action::RefreshRequested)
19    } else {
20        Vec::new()
21    };
22    for handle in state.agent_handles.clone() {
23        if !state.pending_operation_refresh.contains(&handle.operation) {
24            state
25                .pending_operation_refresh
26                .push(handle.operation.clone());
27            effects.push(Effect::OperationStatus {
28                operation: handle.operation,
29            });
30        }
31        let request_id = state.next_request_id;
32        state.next_request_id += 1;
33        effects.push(Effect::ReadSession {
34            request_id,
35            session: handle.session,
36            branch: handle.branch,
37        });
38    }
39    if effects.is_empty() {
40        effects.push(Effect::None);
41    }
42    effects
43}

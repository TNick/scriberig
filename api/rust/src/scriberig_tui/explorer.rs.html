1//! Defines semantic explorer rows without inventing unsupported relation trees.
2
3use crate::selection::EntitySelection;
4
5/// One explorer row backed by a stable semantic selection.
6#[derive(Clone, Debug, Eq, PartialEq)]
7pub struct ExplorerRow {
8    /// Stable row selection.
9    pub selection: EntitySelection,
10    /// Safe compact label.
11    pub label: String,
12    /// Whether the row can be activated for inspection.
13    pub inspectable: bool,
14}
15
16/// Explorer model containing only explicitly known semantic rows.
17#[derive(Clone, Debug, Default, Eq, PartialEq)]
18pub struct ExplorerModel {
19    /// Optional semantic root row.
20    pub root: Option<ExplorerRow>,
21    /// Rows returned by a future public list/follow operation.
22    pub rows: Vec<ExplorerRow>,
23}
24
25impl ExplorerModel {
26    /// Creates an explorer rooted at an explicitly entered entity.
27    #[must_use]
28    pub fn from_selection(selection: EntitySelection) -> Self {
29        let label = selection.entity.to_string();
30        Self {
31            root: Some(ExplorerRow {
32                selection,
33                label,
34                inspectable: true,
35            }),
36            rows: Vec::new(),
37        }
38    }
39}

27/// Configures the first application-backed TUI shell.
28///
29/// # Public members
30/// - `interactive`: Whether to take over an interactive terminal.
31/// - `debug`: Whether to write content-redacted runtime diagnostics.
32/// - `debug_log_file`: Optional diagnostic destination override.
33/// - `initial_entity`: Optional initial semantic entity.
34/// - `snapshot`: Optional exact initial snapshot.
35/// - `project`: Optional active project label.
36/// - `agent_selection`: Exact semantic scope for the first conversation turn.
37///
38/// # Private members
39/// None.
40#[derive(Clone, Debug, Default, Eq, PartialEq)]
41pub struct TuiConfig {
42    /// Runs a line-oriented interactive shell when true.
43    pub interactive: bool,
44    /// Writes content-redacted runtime diagnostics under `playground/tui-logs`.
45    pub debug: bool,
46    /// Optional path overriding the generated debug log path.
47    pub debug_log_file: Option<std::path::PathBuf>,
48    /// Optional entity inspected when the shell starts.
49    pub initial_entity: Option<EntityId>,
50    /// Optional exact snapshot for the initial inspection.
51    pub snapshot: Option<SnapshotId>,
52    /// Optional active project identity shown in the shell context.
53    pub project: Option<String>,
54    /// Exact semantic scope inherited by the first conversation turn.
55    pub agent_selection: Option<scriberig_app::AgentSelection>,
56}

58/// Describes why the TUI runtime stopped.
59#[derive(Clone, Copy, Debug, Eq, PartialEq)]
60pub enum RunOutcome {
61    /// The user requested normal termination.
62    Quit,
63}

65/// Structured TUI runtime failure.
66#[derive(Debug)]
67pub enum TuiError {
68    /// The application returned a structured semantic failure.
69    Application(ApplicationError),
70    /// Terminal input or output failed.
71    Io(io::Error),
72    /// A command line contained invalid semantic input.
73    InvalidCommand(String),
74}
75
76impl std::fmt::Display for TuiError {
77    /// Formats a safe terminal runtime error.
78    /// - `formatter`: Argument named `formatter`.
79    ///
80    ///
81    /// Returns an error when the operation cannot complete.
82    /// # Returns
83    /// The operation result.
84    /// # Errors
85    /// Returns an error when the operation fails.
86    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
87        match self {
88            Self::Application(error) => error.fmt(formatter),
89            Self::Io(error) => error.fmt(formatter),
90            Self::InvalidCommand(message) => formatter.write_str(message),
91        }
92    }
93}

97impl From<io::Error> for TuiError {

103    fn from(error: io::Error) -> Self {
104        Self::Io(error)
105    }

115/// Runs the line-oriented TUI shell over an application service.
116///
117/// - `service`: Application service used for semantic inspection.
118/// - `context`: Caller capability and snapshot context.
119/// - `config`: TUI interaction and initial-selection configuration.
120///
121/// The terminal session outcome.
122///
123/// Returns terminal I/O, invalid-command, or application failures.
124///
125/// # Errors
126/// Returns terminal I/O or application failures from the selected runtime.
127pub fn run<S: TuiService + Sync>(
128    service: &S,
129    context: &ApplicationContext,
130    config: &TuiConfig,
131) -> Result<RunOutcome, TuiError> {
132    runtime::run(service, context, config).map_err(TuiError::Io)?;
133    Ok(RunOutcome::Quit)
134}

136/// Executes one inspection without taking over the caller's terminal.
137///
138/// - `service`: Application service used for semantic inspection.
139/// - `context`: Caller capability and snapshot context.
140/// - `entity`: Entity to inspect.
141/// - `snapshot`: Optional exact snapshot selector.
142///
143/// Snapshot-bound inspection result.
144///
145/// Returns an application or TUI boundary failure.
146///
147/// # Errors
148/// Returns application failures from snapshot-bound inspection.
149pub fn inspect_headless<S: TuiService>(
150    service: &S,
151    context: &ApplicationContext,
152    entity: EntityId,
153    snapshot: Option<SnapshotId>,
154) -> Result<InspectResult, TuiError> {
155    service
156        .inspect_entity(context, entity, snapshot)
157        .map_err(TuiError::Application)
158}

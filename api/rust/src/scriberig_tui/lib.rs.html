11/// Configures the first application-backed TUI shell.
12#[derive(Clone, Debug, Default, Eq, PartialEq)]
13pub struct TuiConfig {
14    /// Runs a line-oriented interactive shell when true.
15    pub interactive: bool,
16    /// Optional entity inspected when the shell starts.
17    pub initial_entity: Option<EntityId>,
18    /// Optional exact snapshot for the initial inspection.
19    pub snapshot: Option<SnapshotId>,
20}
21
22/// Describes why the TUI runtime stopped.
23#[derive(Clone, Copy, Debug, Eq, PartialEq)]
24pub enum RunOutcome {
25    /// The user requested normal termination.
26    Quit,
27}
28
29/// Structured TUI runtime failure.
30#[derive(Debug)]
31pub enum TuiError {
32    /// The application returned a structured semantic failure.
33    Application(ApplicationError),
34    /// Terminal input or output failed.
35    Io(io::Error),
36    /// A command line contained invalid semantic input.
37    InvalidCommand(String),
38}
39
40impl std::fmt::Display for TuiError {
41    /// Formats a safe terminal runtime error.
42    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
43        match self {
44            Self::Application(error) => error.fmt(formatter),
45            Self::Io(error) => error.fmt(formatter),
46            Self::InvalidCommand(message) => formatter.write_str(message),
47        }
48    }
49}
50
51impl std::error::Error for TuiError {}
52
53impl From<io::Error> for TuiError {
54    /// Converts terminal I/O failures into the TUI error model.
55    fn from(error: io::Error) -> Self {
56        Self::Io(error)
57    }
58}

60/// Provides semantic operations required by the TUI shell.
61pub trait TuiService {
62    /// Inspects one entity through the shared application boundary.
63    ///
64    /// # Parameters
65    /// - `context`: Trusted application request context.
66    /// - `entity`: Entity identity to inspect.
67    /// - `snapshot`: Optional exact snapshot.
68    ///
69    /// # Returns
70    /// Snapshot-aware inspection result.
71    ///
72    /// # Errors
73    /// Returns the structured application failure.
74    fn inspect_entity(
75        &self,
76        context: &ApplicationContext,
77        entity: EntityId,
78        snapshot: Option<SnapshotId>,
79    ) -> Result<InspectResult, ApplicationError>;
80}
81
82impl<S: QueryService> TuiService for S {
83    /// Delegates TUI inspection to the shared query service.
84    fn inspect_entity(
85        &self,
86        context: &ApplicationContext,
87        entity: EntityId,
88        snapshot: Option<SnapshotId>,
89    ) -> Result<InspectResult, ApplicationError> {
90        self.inspect(
91            context,
92            &InspectRequest {
93                entity,
94                snapshot: snapshot.map_or(SnapshotResolution::Current, SnapshotResolution::At),
95                detail: DetailLevel::Detailed,
96            },
97        )
98    }
99}

101/// Runs the line-oriented TUI shell over an application service.
102///
103/// # Parameters
104/// - `service`: Application service used for semantic inspection.
105/// - `context`: Caller capability and snapshot context.
106/// - `config`: TUI interaction and initial-selection configuration.
107///
108/// # Returns
109/// The terminal session outcome.
110///
111/// # Errors
112/// Returns terminal I/O, invalid-command, or application failures.
113pub fn run<S: TuiService>(
114    service: &S,
115    context: &ApplicationContext,
116    config: &TuiConfig,
117) -> Result<RunOutcome, TuiError> {
118    let stdout = io::stdout();
119    let mut output = stdout.lock();
120    writeln!(
121        output,
122        "scriberig tui — type `inspect <entity>` or `q` to quit"
123    )?;
124    if let Some(entity) = config.initial_entity {
125        render_inspection(
126            &mut output,
127            service.inspect_entity(context, entity, config.snapshot),
128        )?;
129    }
130    if !config.interactive {
131        return Ok(RunOutcome::Quit);
132    }
133    for line in io::stdin().lock().lines() {
134        let line = line?;
135        if line.trim().eq_ignore_ascii_case("q") {
136            break;
137        }
138        let Some(entity_text) = line.strip_prefix("inspect ") else {
139            writeln!(output, "commands: inspect <entity>, q")?;
140            continue;
141        };
142        let entity = entity_text
143            .trim()
144            .parse::<EntityId>()
145            .map_err(|_| TuiError::InvalidCommand("invalid entity identifier".to_owned()))?;
146        render_inspection(&mut output, service.inspect_entity(context, entity, None))?;
147    }
148    Ok(RunOutcome::Quit)
149}

151/// Executes one inspection without taking over the caller's terminal.
152///
153/// # Parameters
154/// - `service`: Application service used for semantic inspection.
155/// - `context`: Caller capability and snapshot context.
156/// - `entity`: Entity to inspect.
157/// - `snapshot`: Optional exact snapshot selector.
158///
159/// # Returns
160/// Snapshot-bound inspection result.
161///
162/// # Errors
163/// Returns an application or TUI boundary failure.
164pub fn inspect_headless<S: TuiService>(
165    service: &S,
166    context: &ApplicationContext,
167    entity: EntityId,
168    snapshot: Option<SnapshotId>,
169) -> Result<InspectResult, TuiError> {
170    service
171        .inspect_entity(context, entity, snapshot)
172        .map_err(TuiError::Application)
173}

175/// Renders one structured inspection in the shell's compact text form.
176fn render_inspection(
177    output: &mut impl Write,
178    result: Result<InspectResult, ApplicationError>,
179) -> Result<(), TuiError> {
180    match result {
181        Ok(result) => {
182            writeln!(output, "entity: {}", result.entity)?;
183            writeln!(output, "snapshot: {:?}", result.snapshot)?;
184            writeln!(output, "revision: {:?}", result.revision)?;
185            writeln!(output, "type: {:?}", result.object_type)?;
186            writeln!(output, "summary: {:?}", result.description.or(result.label))?;
187        }
188        Err(error) => writeln!(output, "error [{}]: {}", error.category(), error)?,
189    }
190    Ok(())
191}

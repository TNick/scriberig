8/// Routes one plan command to shared durable plan operations.
9///
10/// - `state`: Current `ScribeRig` session and presentation state.
11/// - `argument`: Text following `/plan`.
12///
13/// Effects for the requested plan operation or local validation.
14/// # Returns
15/// Effects for a recognized plan command.
16pub(super) fn plan_command(state: &mut TuiState, argument: &str) -> Vec<Effect> {
17    if argument.is_empty() {
18        return invalid_plan_command(
19            state,
20            "Use /plan <goal>, /plan show, or /plan start|pause|resume|stop|complete.",
21        );
22    }
23    if argument == "show" {
24        return show_plan(state);
25    }
26    if let Some(reason) = argument.strip_prefix("replan ").map(str::trim)
27        && !reason.is_empty()
28    {
29        return replan(state, reason);
30    }
31    if let Some(action) = lifecycle_action(argument) {
32        return change_lifecycle(state, action);
33    }
34    create_plan(state, argument)
35}

37/// Shows a selected durable plan summary.
38///
39/// - `state`: Current session and presentation state.
40///
41/// A no-op effect after updating the shell message or validation message.
42/// # Returns
43/// Effects for the plan summary request.
44fn show_plan(state: &mut TuiState) -> Vec<Effect> {
45    let Some(session) = state.session_snapshot.as_ref() else {
46        return invalid_plan_command(state, "Load a session before showing a plan.");
47    };
48    let Some(plan) = selected_plan(state, session, true) else {
49        return invalid_plan_command(state, "This session has no plan yet.");
50    };
51    let completed = plan
52        .nodes
53        .iter()
54        .filter(|node| node.result.is_some())
55        .count();
56    state.shell_message = format!(
57        "Plan: {} · {:?} · revision {} · {}/{} results · {} dependencies",
58        plan.goal,
59        plan.status,
60        plan.revision,
61        completed,
62        plan.nodes.len(),
63        plan.dependencies.len()
64    );
65    vec![Effect::None]
66}

68/// Creates one additive plan revision from a replan reason.
69///
70/// - `state`: Current session and selected plan state.
71/// - `reason`: User-provided purpose for the addition.
72///
73/// A plan replan effect or local validation response.
74/// # Returns
75/// Effects for the requested plan revision.
76fn replan(state: &mut TuiState, reason: &str) -> Vec<Effect> {
77    let Some(session) = state.session_snapshot.as_ref() else {
78        return invalid_plan_command(state, "Load a session before revising a plan.");
79    };
80    let Some(plan) = selected_plan(state, session, false) else {
81        return invalid_plan_command(state, "Create a plan before revising it.");
82    };
83    let title = reason.lines().next().unwrap_or("Replanned work");
84    let title = title.chars().take(80).collect::<String>();
85    vec![Effect::Replan(scriberig_app::PlanReplanRequest {
86        session: session.id,
87        plan: plan.id,
88        expected_revision: plan.revision,
89        additions: vec![scriberig_app::PlanNodeDraft {
90            key: "replan-addition".to_owned(),
91            parent_index: None,
92            title,
93            purpose: reason.to_owned(),
94            expected_result: Some(scriberig_session::ExpectedResult {
95                kind: "summary".to_owned(),
96                requires_handle: false,
97            }),
98            semantic_scope: Vec::new(),
99            assignment: None,
100        }],
101        dependencies: Vec::new(),
102        removed_dependencies: Vec::new(),
103        superseded: Vec::new(),
104        assignment_changes: Vec::new(),
105        preserved_terminal_nodes: plan
106            .nodes
107            .iter()
108            .filter(|node| node.result.is_some())
109            .map(|node| node.id)
110            .collect(),
111        reason: reason.to_owned(),
112    })]
113}

115/// Applies a plan lifecycle action to the selected durable plan.
116///
117/// - `state`: Current session and selected plan state.
118/// - `action`: Requested lifecycle transition.
119///
120/// A lifecycle effect or local validation response.
121/// # Returns
122/// Effects for the requested lifecycle transition.
123fn change_lifecycle(
124    state: &mut TuiState,
125    action: scriberig_app::PlanLifecycleAction,
126) -> Vec<Effect> {
127    let Some(session) = state.session_snapshot.as_ref() else {
128        return invalid_plan_command(state, "Load a session before controlling a plan.");
129    };
130    let Some(plan) = selected_plan(state, session, false) else {
131        return invalid_plan_command(state, "Create a plan before changing its lifecycle.");
132    };
133    vec![Effect::PlanLifecycle {
134        request: scriberig_app::PlanLifecycleRequest {
135            session: session.id,
136            plan: plan.id,
137            expected_revision: plan.revision,
138        },
139        action,
140    }]
141}

143/// Creates a durable plan from the active conversation target.
144///
145/// - `state`: Current conversation target.
146/// - `argument`: User-provided goal and node draft.
147///
148/// A plan creation effect or local validation response.
149/// # Returns
150/// Effects for the requested plan creation.
151fn create_plan(state: &mut TuiState, argument: &str) -> Vec<Effect> {
152    let crate::state::ConversationTarget::Existing {
153        session,
154        reply_anchor,
155    } = state.conversation_composer.target
156    else {
157        return invalid_plan_command(state, "Start a conversation before creating a plan.");
158    };
159    let draft = match plan::parse_plan_argument(argument) {
160        Ok(draft) => draft,
161        Err(message) => {
162            state.conversation_composer.error = Some(message);
163            return vec![Effect::None];
164        }
165    };
166    vec![Effect::CreatePlan(scriberig_app::PlanCreateRequest {
167        session,
168        anchor: reply_anchor,
169        goal: draft.goal,
170        nodes: draft.nodes,
171        dependencies: draft.dependencies,
172    })]
173}

175/// Finds the active lifecycle action encoded by a plan command.
176///
177/// - `argument`: Text following `/plan`.
178///
179/// The requested lifecycle action, if the argument names one.
180/// # Returns
181/// A supported plan lifecycle action.
182fn lifecycle_action(argument: &str) -> Option<scriberig_app::PlanLifecycleAction> {
183    match argument {
184        "start" => Some(scriberig_app::PlanLifecycleAction::Start),
185        "pause" => Some(scriberig_app::PlanLifecycleAction::Pause),
186        "resume" => Some(scriberig_app::PlanLifecycleAction::Resume),
187        "stop" => Some(scriberig_app::PlanLifecycleAction::Stop),
188        "complete" => Some(scriberig_app::PlanLifecycleAction::Complete),
189        _ => None,
190    }
191}

193/// Finds the selected plan or the most recent plan in a session.
194///
195/// - `state`: Current plan and conversation cursor.
196/// - `session`: Durable session snapshot containing plans.
197/// - `use_plan_cursor`: Whether the plan view cursor takes precedence.
198///
199/// The selected plan, falling back to the latest plan.
200/// # Returns
201/// A plan in the session when one exists.
202fn selected_plan<'a>(
203    state: &TuiState,
204    session: &'a scriberig_session::Session,
205    use_plan_cursor: bool,
206) -> Option<&'a scriberig_session::Plan> {
207    let selected = if use_plan_cursor && state.primary_view == crate::selection::PrimaryView::Plan {
208        state.plan_cursor.map(|selection| selection.plan)
209    } else {
210        match state.conversation_cursor {
211            Some(crate::view::conversation::ConversationSource::Plan(plan)) => Some(plan),
212            Some(crate::view::conversation::ConversationSource::PlanNode(node)) => session
213                .plans
214                .iter()
215                .find(|plan| plan.nodes.iter().any(|item| item.id == node))
216                .map(|plan| plan.id),
217            _ => None,
218        }
219    };
220    selected
221        .and_then(|id| session.plans.iter().find(|plan| plan.id == id))
222        .or_else(|| session.plans.last())
223}

225/// Stores a plan validation message and produces a no-op effect.
226///
227/// - `state`: Presentation state receiving the message.
228/// - `message`: User-facing validation detail.
229///
230/// A no-op effect.
231/// # Returns
232/// A single no-op effect for local command validation.
233fn invalid_plan_command(state: &mut TuiState, message: &str) -> Vec<Effect> {
234    state.conversation_composer.error = Some(message.to_owned());
235    vec![Effect::None]

1//! Parses local plan outlines into shared plan creation requests.
2
3/// One structured plan draft parsed from composer input.
4///
5/// # Public members
6/// - `goal`: Plan goal from the first nonempty line.
7/// - `nodes`: Ordered node drafts referenced by dependency indices.
8/// - `dependencies`: One-based dependency references converted to zero-based
9///   node indices.
10///
11/// # Private members
12/// None.
13pub(super) struct PlanDraftInput {
14    /// Plan goal from the first nonempty line.
15    pub(super) goal: String,
16    /// Ordered node drafts referenced by dependency indices.
17    pub(super) nodes: Vec<scriberig_app::PlanNodeDraft>,
18    /// Zero-based prerequisite and dependent node indices.
19    pub(super) dependencies: Vec<(usize, usize)>,
20}
21
22/// Parses a goal and optional bullet outline with explicit dependency edges.
23///
24/// The first nonempty line is the goal. Outline nodes use `- Title: purpose`.
25/// Dependencies use `depends: 1 -> 3`, with one-based node numbers.
26///
27/// - `argument`: Text following the local `/plan` command.
28///
29/// A structured plan draft suitable for the shared application operation.
30///
31/// Returns an input error for malformed outline or dependency lines.
32/// # Returns
33/// The operation result.
34/// # Errors
35/// Returns an error when the operation fails.
36pub(super) fn parse_plan_argument(argument: &str) -> Result<PlanDraftInput, String> {
37    let mut lines = argument
38        .lines()
39        .map(str::trim)
40        .filter(|line| !line.is_empty());
41    let goal = lines
42        .next()
43        .filter(|line| !line.starts_with('-') && !line.starts_with("depends:"))
44        .ok_or_else(|| "Start /plan with a goal on the first line.".to_owned())?
45        .to_owned();
46    let mut nodes = Vec::new();
47    let mut dependencies = Vec::new();
48    for line in lines {
49        if let Some(edge) = line.strip_prefix("depends:") {
50            dependencies.push(parse_dependency(edge.trim())?);
51        } else if let Some(node) = line.strip_prefix("- ") {
52            nodes.push(parse_node(node, nodes.len())?);
53        } else {
54            return Err(
55                "Use '- Title: purpose' for plan nodes and 'depends: 1 -> 2' for dependencies."
56                    .to_owned(),
57            );
58        }
59    }
60    if nodes.is_empty() {
61        let title = goal.chars().take(80).collect::<String>();
62        nodes.push(node_draft("goal".to_owned(), title, goal.clone()));
63    }
64    validate_dependencies(&dependencies, nodes.len())?;
65    Ok(PlanDraftInput {
66        goal,
67        nodes,
68        dependencies,
69    })
70}

72/// Parses one outline line into a stable ordered node draft.
73///
74/// - `value`: Text following the outline marker.
75/// - `index`: Zero-based position used to create a stable request key.
76///
77/// A node draft with bounded title and full purpose.
78///
79/// Returns an input error when title or purpose is empty.
80/// # Returns
81/// The operation result.
82/// # Errors
83/// Returns an error when the operation fails.
84fn parse_node(value: &str, index: usize) -> Result<scriberig_app::PlanNodeDraft, String> {
85    let (title, purpose) = value
86        .split_once(':')
87        .ok_or_else(|| "Plan nodes use '- Title: purpose'.".to_owned())?;
88    let title = title.trim();
89    let purpose = purpose.trim();
90    if title.is_empty() || purpose.is_empty() {
91        return Err("Plan node title and purpose must not be empty.".to_owned());
92    }
93    Ok(node_draft(
94        format!("node-{}", index + 1),
95        title.chars().take(80).collect(),
96        purpose.to_owned(),
97    ))
98}

100/// Creates one node draft with the shared default result contract.
101///
102/// - `key`: Stable request-local node key.
103/// - `title`: Bounded tree label.
104/// - `purpose`: Worker instruction for the node.
105///
106/// A node draft attached beneath the plan anchor.
107/// # Returns
108/// The operation result.
109fn node_draft(key: String, title: String, purpose: String) -> scriberig_app::PlanNodeDraft {
110    scriberig_app::PlanNodeDraft {
111        key,
112        parent_index: None,
113        title,
114        purpose,
115        expected_result: Some(scriberig_session::ExpectedResult {
116            kind: "summary".to_owned(),
117            requires_handle: false,
118        }),
119        semantic_scope: Vec::new(),
120        assignment: None,
121    }
122}

124/// Parses one one-based prerequisite to dependent node reference.
125///
126/// - `value`: Dependency edge in `N -> M` form.
127///
128/// Zero-based prerequisite and dependent node indices.
129///
130/// Returns an input error for malformed or zero indices.
131/// # Returns
132/// The operation result.
133/// # Errors
134/// Returns an error when the operation fails.
135fn parse_dependency(value: &str) -> Result<(usize, usize), String> {
136    let (prerequisite, dependent) = value
137        .split_once("->")
138        .ok_or_else(|| "Dependencies use 'depends: 1 -> 2'.".to_owned())?;
139    let prerequisite = parse_node_number(prerequisite.trim())?;
140    let dependent = parse_node_number(dependent.trim())?;
141    Ok((prerequisite - 1, dependent - 1))
142}
143
144/// Parses one positive one-based plan node number.
145///
146/// - `value`: Node number text.
147///
148/// Positive node number.
149///
150/// Returns an input error when the value is not a positive integer.
151/// # Returns
152/// The operation result.
153/// # Errors
154/// Returns an error when the operation fails.
155fn parse_node_number(value: &str) -> Result<usize, String> {
156    value
157        .parse::<usize>()
158        .ok()
159        .filter(|number| *number > 0)
160        .ok_or_else(|| "Dependency node numbers start at 1.".to_owned())
161}

163/// Ensures every dependency endpoint names an existing distinct plan node.
164///
165/// - `dependencies`: Parsed zero-based edges.
166/// - `node_count`: Number of nodes in the draft.
167///
168/// Unit when all references are in range and non-reflexive.
169///
170/// Returns an input error when an endpoint is invalid.
171/// # Returns
172/// The operation result.
173/// # Errors
174/// Returns an error when the operation fails.
175fn validate_dependencies(dependencies: &[(usize, usize)], node_count: usize) -> Result<(), String> {
176    if dependencies.iter().any(|(prerequisite, dependent)| {
177        *prerequisite >= node_count || *dependent >= node_count || prerequisite == dependent
178    }) {
179        return Err("Dependency edges must reference two different existing nodes.".to_owned());
180    }
181    Ok(())
182}

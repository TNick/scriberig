5/// Validation state exposed by a change review.
6#[derive(Clone, Debug, Eq, PartialEq)]
7pub enum ReviewValidation {
8    /// No validation evidence was returned by the application operation.
9    Unknown,
10    /// The candidate was returned successfully by the application.
11    Passed,
12    /// The application returned a validation diagnostic.
13    Failed(String),
14}
15
16/// Read-only review model for an application-owned change or candidate.
17#[derive(Clone, Debug, Eq, PartialEq)]
18pub struct ChangeReviewModel {
19    /// Exact base snapshot reviewed by the user.
20    pub base_snapshot: SnapshotId,
21    /// Candidate result snapshot, when returned by the operation.
22    pub candidate_snapshot: Option<SnapshotId>,
23    /// Stable proposed change identity.
24    pub change: String,
25    /// Optional branch receiving the reviewed candidate.
26    pub branch: Option<RootName>,
27    /// Affected semantic entities, when the result carries a diff.
28    pub affected_entities: Vec<EntityId>,
29    /// Exact shared entity and revision transitions returned by the diff.
30    pub deltas: Vec<sr_change::EntityDelta>,
31    /// Non-blocking candidate and materialization details.
32    pub details: Vec<String>,
33    /// Structured conflict summaries surfaced by the application.
34    pub conflicts: Vec<String>,
35    /// Validation state and safe diagnostics.
36    pub validation: ReviewValidation,
37    /// Registered policy that produced the current validation result.
38    pub validation_profile: Option<String>,
39    /// Whether the model represents a candidate rather than a promoted state.
40    pub requires_confirmation: bool,
41}
42
43impl ChangeReviewModel {
44    /// Builds a review model from a public candidate summary.
45    #[must_use]
46    /// - `base_snapshot`: Argument named `base_snapshot`.
47    /// - `change`: Argument named `change`.
48    /// - `branch`: Argument named `branch`.
49    /// - `summary`: Argument named `summary`.
50    ///
51    /// # Returns
52    /// The operation result.
53    pub fn from_summary(
54        base_snapshot: SnapshotId,
55        change: impl Into<String>,
56        branch: Option<RootName>,
57        summary: &scriberig_app::OperationSummary,
58    ) -> Self {
59        let validation = match summary.validation {
60            Some(scriberig_app::ValidationStatus::Passed) => ReviewValidation::Passed,
61            Some(
62                scriberig_app::ValidationStatus::Failed | scriberig_app::ValidationStatus::Error,
63            ) => ReviewValidation::Failed(if summary.validation_diagnostics.is_empty() {
64                "validation failed".to_owned()
65            } else {
66                summary.validation_diagnostics.join("; ")
67            }),
68            Some(
69                scriberig_app::ValidationStatus::Skipped | scriberig_app::ValidationStatus::Unknown,
70            )
71            | None => ReviewValidation::Unknown,
72        };
73        Self {
74            base_snapshot,
75            candidate_snapshot: summary.snapshot,
76            change: change.into(),
77            branch,
78            affected_entities: summary
79                .diff
80                .as_ref()
81                .map(|diff| diff.deltas.iter().map(|delta| delta.entity).collect())
82                .unwrap_or_default(),
83            deltas: summary
84                .diff
85                .as_ref()
86                .map_or_else(Vec::new, |diff| diff.deltas.clone()),
87            details: summary.review_details.clone(),
88            conflicts: summary.conflicts.clone(),
89            validation,
90            validation_profile: summary.validation_profile.clone(),
91            requires_confirmation: true,
92        }
93    }
94}

96/// Explicit user intent to invoke canonical application commit.
97#[derive(Clone, Debug, Eq, PartialEq)]
98pub struct ConfirmedCommit {
99    /// Exact reviewed base snapshot.
100    pub base_snapshot: SnapshotId,
101    /// Exact candidate result snapshot shown during review.
102    pub candidate_snapshot: SnapshotId,
103    /// Stable change identity.
104    pub change: String,
105    /// Reviewed branch context.
106    pub branch: Option<RootName>,
107}

109#[cfg(test)]
110mod tests {
111    use super::{ChangeReviewModel, ReviewValidation};
112    use scriberig_app::{OperationSummary, ValidationStatus};
113    use scriberig_core::{EntityId, RevisionId, SnapshotId};
114
115    /// Verifies review status remains unknown without application evidence.
116    #[test]
117    fn review_preserves_validation_and_conflict_evidence() {
118        let summary = OperationSummary {
119            handle: "candidate".to_owned(),
120            snapshot: None,
121            revision: None,
122            branch: None,
123            diff: None,
124            history: None,
125            rendered: None,
126            validation: Some(ValidationStatus::Failed),
127            validation_profile: Some("rust-workspace".to_owned()),
128            validation_diagnostics: vec!["invalid relation".to_owned()],
129            review_details: Vec::new(),
130            conflicts: vec!["entity conflict".to_owned()],
131        };
132        let model =
133            ChangeReviewModel::from_summary(SnapshotId::new([1; 32]), "candidate", None, &summary);
134        assert_eq!(
135            model.validation,
136            ReviewValidation::Failed("invalid relation".to_owned())
137        );
138        assert_eq!(model.validation_profile.as_deref(), Some("rust-workspace"));
139        assert_eq!(model.conflicts, vec!["entity conflict"]);
140    }
141
142    /// Verifies the exact candidate snapshot survives adaptation for review.
143    #[test]
144    fn review_preserves_candidate_snapshot_identity() {
145        let candidate = SnapshotId::new([2; 32]);
146        let summary = OperationSummary {
147            handle: "candidate".to_owned(),
148            snapshot: Some(candidate),
149            revision: None,
150            branch: None,
151            diff: None,
152            history: None,
153            rendered: None,
154            validation: Some(ValidationStatus::Passed),
155            validation_profile: None,
156            validation_diagnostics: Vec::new(),
157            review_details: Vec::new(),
158            conflicts: Vec::new(),
159        };
160        let model =
161            ChangeReviewModel::from_summary(SnapshotId::new([1; 32]), "candidate", None, &summary);
162
163        assert_eq!(model.candidate_snapshot, Some(candidate));
164    }

166    /// Verifies exact shared before/after revision references survive review.
167    #[test]
168    fn review_preserves_semantic_revision_deltas() {
169        let base = SnapshotId::new([3; 32]);
170        let candidate = SnapshotId::new([4; 32]);
171        let entity = EntityId::from_bytes([5; 16]);
172        let before = RevisionId::new([6; 32]);
173        let after = RevisionId::new([7; 32]);
174        let summary = OperationSummary {
175            handle: "candidate".to_owned(),
176            snapshot: Some(candidate),
177            revision: None,
178            branch: None,
179            diff: Some(sr_change::SemanticDiff::new(
180                base,
181                candidate,
182                vec![
183                    sr_change::EntityDelta::new(entity, Some(before), Some(after))
184                        .expect("changed revision"),
185                ],
186            )),
187            history: None,
188            rendered: None,
189            validation: Some(ValidationStatus::Passed),
190            validation_profile: None,
191            validation_diagnostics: Vec::new(),
192            review_details: Vec::new(),
193            conflicts: Vec::new(),
194        };
195        let model = ChangeReviewModel::from_summary(base, "candidate", None, &summary);
196
197        assert_eq!(model.deltas.len(), 1);
198        assert_eq!(model.deltas[0].entity, entity);
199        assert_eq!(model.deltas[0].before, Some(before));
200        assert_eq!(model.deltas[0].after, Some(after));
201    }

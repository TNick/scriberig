3use super::{SessionRow, SessionRowKind};

8/// Appends turn and tool rows for one branch.
9///
10/// # Parameters
11/// - `source`: Branch whose turns are projected.
12/// - `depth`: Parent branch row depth.
13/// - `transient`: Whether the owning branch is still active.
14/// - `rows`: Output rows being built.
15/// - `tool_invocation_details`: Safe details indexed by invocation identity.
16pub(super) fn append(
17    source: &Branch,
18    depth: usize,
19    transient: bool,
20    rows: &mut Vec<SessionRow>,
21    tool_invocation_details: &mut BTreeMap<EntityId, String>,
22) {
23    for turn in SessionViews::turns(source) {
24        let agent_output = source
25            .turns
26            .iter()
27            .find(|stored| stored.id == turn.id)
28            .and_then(|stored| stored.agent.as_ref());
29        let label = agent_output.map_or_else(
30            || format!("Turn {}", turn.ordinal),
31            |agent| {
32                format!(
33                    "Turn {}: {}",
34                    turn.ordinal,
35                    crate::inspector::sanitize(&agent.text)
36                )
37            },
38        );
39        rows.push(SessionRow {
40            reference: turn.id,
41            depth: depth + 1,
42            label,
43            kind: SessionRowKind::Turn,
44            candidate: None,
45            snapshot: None,
46            transient,
47        });
48        append_turn_tools(
49            turn.tool_invocation,
50            turn.tool_result,
51            depth,
52            transient,
53            rows,
54            tool_invocation_details,
55            source,
56        );
57    }
58}

60/// Appends a turn's tool references and safe argument detail.
61///
62/// # Parameters
63/// - `invocation`: Stable tool invocation referenced by the turn.
64/// - `result`: Stable tool result referenced by the turn.
65/// - `depth`: Parent branch row depth.
66/// - `transient`: Whether the owning branch is still active.
67/// - `rows`: Output rows being built.
68/// - `tool_invocation_details`: Safe details indexed by invocation identity.
69/// - `source`: Branch containing the invocation records.
70fn append_turn_tools(
71    invocation: Option<EntityId>,
72    result: Option<EntityId>,
73    depth: usize,
74    transient: bool,
75    rows: &mut Vec<SessionRow>,
76    tool_invocation_details: &mut BTreeMap<EntityId, String>,
77    source: &Branch,
78) {
79    for reference in [invocation, result].into_iter().flatten() {
80        rows.push(SessionRow {
81            reference,
82            depth: depth + 2,
83            label: format!("Tool {reference}"),
84            kind: SessionRowKind::Tool,
85            candidate: None,
86            snapshot: None,
87            transient,
88        });
89    }
90    if let Some(reference) = invocation
91        && let Some(invocation) = source
92            .tool_invocations
93            .iter()
94            .find(|item| item.id == reference)
95        && let Some(detail) = safe_invocation_detail(invocation.presentation_input.as_deref())
96    {
97        tool_invocation_details.insert(reference, detail);
98    }
99}

101/// Parses and bounds schema-redacted invocation arguments for detailed views.
102///
103/// # Parameters
104/// - `input`: Safe structured arguments captured with the invocation.
105///
106/// # Returns
107/// A sanitized pretty-printed projection capped at 4096 characters.
108fn safe_invocation_detail(input: Option<&[u8]>) -> Option<String> {
109    let value = serde_json::from_slice::<serde_json::Value>(input?).ok()?;
110    let text = serde_json::to_string_pretty(&value).ok()?;
111    Some(
112        text.chars()
113            .filter(|character| *character == '\n' || !character.is_control())
114            .take(4096)
115            .collect(),
116    )
117}

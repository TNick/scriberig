1//! Projects structured semantic diff results into stable terminal rows.
2
3use scriberig_core::{EntityId, RevisionId, SnapshotId};
4
5/// One semantic entity transition in a diff.
6#[derive(Clone, Debug, Eq, PartialEq)]
7pub struct DiffRow {
8    /// Stable entity identity.
9    pub entity: EntityId,
10    /// Revision on the left/base side.
11    pub before: Option<RevisionId>,
12    /// Revision on the right/target side.
13    pub after: Option<RevisionId>,
14}
15
16/// Explicit two-sided semantic diff view model.
17#[derive(Clone, Debug, Eq, PartialEq)]
18pub struct DiffModel {
19    /// Explicit left/base snapshot.
20    pub left: SnapshotId,
21    /// Explicit right/target snapshot.
22    pub right: SnapshotId,
23    /// Structured changed entity rows.
24    pub rows: Vec<DiffRow>,
25}
26
27impl DiffModel {
28    /// Builds a TUI model without comparing rendered text.
29    #[must_use]
30    /// - `diff`: Argument named `diff`.
31    ///
32    /// # Returns
33    /// The value produced by the declaration.
34    pub fn from_diff(diff: &sr_change::SemanticDiff) -> Self {
35        Self {
36            left: diff.left,
37            right: diff.right,
38            rows: diff
39                .deltas
40                .iter()
41                .map(|delta| DiffRow {
42                    entity: delta.entity,
43                    before: delta.before,
44                    after: delta.after,
45                })
46                .collect(),
47        }
48    }
49}

51#[cfg(test)]
52mod tests {
53    use super::DiffModel;
54    use scriberig_core::{EntityId, RevisionId, SnapshotId};
55    use sr_change::{EntityDelta, SemanticDiff};
56
57    /// Verifies both exact diff sides and stable row references are retained.
58    #[test]
59    fn preserves_explicit_sides_and_entity_reference() {
60        let entity = EntityId::from_bytes([1; 16]);
61        let diff = SemanticDiff::new(
62            SnapshotId::new([2; 32]),
63            SnapshotId::new([3; 32]),
64            vec![
65                EntityDelta::new(
66                    entity,
67                    Some(RevisionId::new([4; 32])),
68                    Some(RevisionId::new([5; 32])),
69                )
70                .expect("delta"),
71            ],
72        );
73        let model = DiffModel::from_diff(&diff);
74        assert_eq!(model.left, SnapshotId::new([2; 32]));
75        assert_eq!(model.right, SnapshotId::new([3; 32]));
76        assert_eq!(model.rows[0].entity, entity);
77    }
78}

8/// Builds Markdown content for the selected `ScribeRig` semantic view.
9///
10/// - `state`: Current navigation location and read-model projections.
11///
12/// # Returns
13/// Bounded, terminal-safe content for the upstream Markdown component.
14pub(super) fn render_view(state: &TuiState) -> String {
15    match state.primary_view {
16        PrimaryView::Conversation => conversation(state),
17        PrimaryView::Overview => model_views::overview(state),
18        PrimaryView::Explorer => model_views::explorer(state),
19        PrimaryView::History => model_views::history(state),
20        PrimaryView::Find => model_views::find(state),
21        PrimaryView::Diff => model_views::diff(state),
22        PrimaryView::Projection => model_views::projection(state),
23        PrimaryView::Changes => model_views::changes(state),
24        PrimaryView::Session => runtime_views::session(state),
25        PrimaryView::Operations => runtime_views::operations(state),
26        PrimaryView::Plan => model_views::plan(state),
27    }
28}

30/// Renders recursive session and plan rows with their stable ancestry order.
31///
32/// - `state`: Current conversation and transient operation projections.
33///
34/// # Returns
35/// Bounded conversation content.
36fn conversation(state: &TuiState) -> String {
37    let mut output = String::from("# Conversation\n\n");
38    if let Some(model) = &state.conversation {
39        for (index, row) in model.rows.iter().take(MAX_ROWS).enumerate() {
40            let kind = format!("{:?}", row.kind);
41            let cursor = if state.coordinator_cursor.is_some_and(|(_, node)| {
42                row.source == crate::view::conversation::ConversationSource::PlanNode(node)
43            }) {
44                " · coordinator cursor"
45            } else {
46                ""
47            };
48            append_value(
49                &mut output,
50                &format!(
51                    "{}{}[{kind}] {}{cursor}",
52                    if index == state.cursor { "› " } else { "  " },
53                    "  ".repeat(row.depth.min(8)),
54                    row.label
55                ),
56            );
57            if let crate::view::conversation::ConversationSource::Activity(invocation) = row.source
58                && state.prompt_visibility == crate::state::PromptVisibility::Shown
59                && let Some(detail) = state.model_request_details.get(&invocation)
60            {
61                append_model_request(&mut output, detail);
62            }
63        }
64        if model.rows.len() > MAX_ROWS {
65            append_value(&mut output, "… additional conversation rows omitted");
66        }
67    }
68    if state.reasoning_visibility == crate::state::ReasoningVisibility::Shown {
69        for activity in state.transient_agent_activities.iter().take(MAX_ROWS) {
70            append_value(
71                &mut output,
72                &format!(
73                    "{} · {:?}: {}",
74                    activity.branch, activity.state, activity.summary
75                ),
76            );
77        }
78    }
79    for response in state.transient_agent_outputs.iter().take(MAX_ROWS) {
80        append_value(&mut output, &response.text);
81    }
82    if let Some(prompt) = state.conversation_composer.pending_prompt.as_deref() {
83        append_value(&mut output, &format!("You: {prompt}"));
84    }
85    if let Some(error) = state.conversation_composer.error.as_deref() {
86        append_status(&mut output, "Error", error, SemanticRole::Error);
87    }
88    if state.conversation.is_none() && state.conversation_composer.pending_prompt.is_none() {
89        append_value(&mut output, "Type a message to start a conversation.");
90    }
91    append_plan_graphs(&mut output, state);
92    output
93}

95/// Renders the exact `ScribeRig` plan DAG separately from the conversation tree.
96///
97/// - `output`: Conversation content receiving an independent plan section.
98/// - `state`: Exact session snapshot containing durable plan definitions.
99fn append_plan_graphs(output: &mut String, state: &TuiState) {
100    let Some(session) = &state.session_snapshot else {
101        return;
102    };
103    if session.plans.is_empty() {
104        return;
105    }
106    append_value(output, "## Durable plan dependency graphs");
107    for plan in session.plans.iter().take(32) {
108        append_value(
109            output,
110            &format!(
111                "Plan {} · revision {} · {:?}",
112                plan.id, plan.revision, plan.status
113            ),
114        );
115        append_value(output, &format!("Goal: {}", plan.goal));
116        if let Some(claim) = &plan.coordinator_claim {
117            append_value(output, &format!("Coordinator claim: {}", claim.id));
118        }
119        if let Some(cursor) = plan.coordinator_cursor {
120            append_value(output, &format!("Coordinator cursor: {cursor}"));
121        }
122        for node in plan.nodes.iter().take(MAX_ROWS) {
123            let parent = match node.parent {
124                Some(scriberig_session::PlanNodeParent::Branch(branch)) => {
125                    format!("session branch {branch}")
126                }
127                Some(scriberig_session::PlanNodeParent::Node(parent)) => {
128                    format!("plan node {parent}")
129                }
130                None => "plan root".to_owned(),
131            };
132            append_value(
133                output,
134                &format!(
135                    "{} · {:?} · {} · parent: {parent}",
136                    node.title, node.status, node.id
137                ),
138            );
139            if let Some(branch) = node.branch {
140                append_value(output, &format!("  execution branch: {branch}"));
141            }
142            if let Some(result) = node.result {
143                append_value(output, &format!("  BranchResult: {result}"));
144            }
145            if let Some(assignment) = &node.assignment {
146                append_value(
147                    output,
148                    &format!(
149                        "  upstream execution: {}/{} · thinking {}",
150                        assignment.provider_id,
151                        assignment.model_id,
152                        assignment.thinking_level.as_deref().unwrap_or("default")
153                    ),
154                );
155            }
156        }
157        for edge in plan.dependencies.iter().take(MAX_ROWS) {
158            append_value(
159                output,
160                &format!(
161                    "Dependency edge: {} → {} · fulfilled by {}",
162                    edge.prerequisite,
163                    edge.dependent,
164                    edge.fulfilled_by
165                        .map_or_else(|| "pending".to_owned(), |id| id.to_string())
166                ),
167            );
168        }
169    }
170}

172/// Shows a safe provider-neutral request only when prompt visibility is enabled.
173///
174/// - `output`: Conversation content receiving the captured request details.
175/// - `detail`: Safe request projection associated with a stable invocation ID.
176fn append_model_request(
177    output: &mut String,
178    detail: &scriberig_protocol::ModelRequestDetailResponse,
179) {
180    append_value(
181        output,
182        &format!(
183            "Effective request · {} · {}",
184            detail.invocation, detail.model
185        ),
186    );
187    if let Some(provider) = &detail.provider_id {
188        append_value(output, &format!("Provider: {provider}"));
189    }
190    if let Some(thinking) = &detail.thinking_level {
191        append_value(output, &format!("Thinking level: {thinking}"));
192    }
193    if let Some(profile) = &detail.profile {
194        append_value(
195            output,
196            &format!(
197                "Profile revision: {} @ {}",
198                profile.identity, profile.revision
199            ),
200        );
201    }
202    for component in detail.active_components.iter().take(64) {
203        append_value(
204            output,
205            &format!(
206                "Context component: {} @ {}",
207                component.identity, component.revision
208            ),
209        );
210    }
211    for message in detail.messages.iter().take(64) {
212        append_value(output, &format!("{}: {}", message.role, message.content));
213    }
214    for tool in detail.tools.iter().take(64) {
215        append_value(
216            output,
217            &format!("Tool: {} · {}", tool.name, tool.description),
218        );
219        if let Some(parameters) = &tool.parameters {
220            append_value(output, &format!("Tool parameters: {parameters}"));
221        }
222    }
223    if let Some(report) = &detail.applied_settings {
224        let summary = serde_json::to_string(report).unwrap_or_else(|_| "{}".to_owned());
225        append_value(output, &format!("Applied settings: {summary}"));
226    }
227    if detail.truncated {
228        append_value(output, "Effective request detail is truncated.");
229    }

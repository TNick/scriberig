8/// Builds reusable tree items and details for the active semantic view.
9///
10/// - `state`: Current `ScribeRig` view and cursor.
11///
12/// Stable upstream items, the semantic selection, and supplementary details.
13/// # Returns
14/// A tree projection when the active view has ready semantic rows.
15pub(super) fn projection(state: &TuiState) -> Option<TreeProjection> {
16    let rows = semantic_rows(state)?;
17    let details = match state.primary_view {
18        crate::selection::PrimaryView::Conversation => super::details::conversation_details(state),
19        crate::selection::PrimaryView::Session => super::details::session_details(state),
20        _ => return None,
21    };
22    Some(TreeProjection {
23        items: make_tree_items(&rows),
24        selected_id: selected_tree_id(state, &rows),
25        details,
26    })
27}

29/// Projects current visible conversation or session rows to stable tree IDs.
30///
31/// - `state`: `ScribeRig` view state supplying the visible rows.
32///
33/// The ordered semantic rows with their source depth.
34/// # Returns
35/// Projected rows for supported conversation and session views.
36fn semantic_rows(state: &TuiState) -> Option<Vec<SemanticTreeRow>> {
37    match state.primary_view {
38        crate::selection::PrimaryView::Conversation => Some(
39            state
40                .conversation
41                .as_ref()?
42                .rows
43                .iter()
44                .map(|row| SemanticTreeRow {
45                    id: conversation_source_id(row.source),
46                    label: format!("{:?} · {}", row.kind, bounded_label(&row.label)),
47                    depth: row.depth,
48                })
49                .collect(),
50        ),
51        crate::selection::PrimaryView::Session => match &state.session {
52            crate::inspector::LoadState::Ready(model) => Some(
53                model
54                    .rows
55                    .iter()
56                    .map(|row| SemanticTreeRow {
57                        id: session_row_id(row.identity()),
58                        label: format!(
59                            "{:?} · {} · {}",
60                            row.kind,
61                            bounded_label(&row.label),
62                            row.reference
63                        ),
64                        depth: row.depth,
65                    })
66                    .collect(),
67            ),
68            _ => None,
69        },
70        _ => None,
71    }
72}

74/// Creates tree items from semantic preorder while preserving projected depth.
75///
76/// - `rows`: Ordered `ScribeRig` semantic rows.
77///
78/// Upstream tree items with stable IDs and nearest preceding parents.
79/// # Returns
80/// Tree items preserving the semantic preorder.
81fn make_tree_items(rows: &[SemanticTreeRow]) -> Vec<TreeItem> {
82    let mut ancestors = Vec::<(usize, String)>::new();
83    rows.iter()
84        .map(|row| {
85            while ancestors
86                .last()
87                .is_some_and(|(depth, _)| *depth >= row.depth)
88            {
89                ancestors.pop();
90            }
91            let parent = ancestors.last().map(|(_, id)| id.clone());
92            ancestors.push((row.depth, row.id.clone()));
93            TreeItem {
94                id: row.id.clone(),
95                label: row.label.clone(),
96                parent_id: parent,
97            }
98        })
99        .collect()
100}

102/// Resolves current selection to a stable upstream item ID.
103///
104/// - `state`: `ScribeRig`'s selected row.
105/// - `rows`: Stable row IDs in the active tree.
106///
107/// The matching ID, when the current row remains projected.
108/// # Returns
109/// Stable item ID when the selected row is visible.
110fn selected_tree_id(state: &TuiState, rows: &[SemanticTreeRow]) -> Option<String> {
111    let selected = match state.primary_view {
112        crate::selection::PrimaryView::Conversation => {
113            state.conversation_cursor.map(conversation_source_id)
114        }
115        crate::selection::PrimaryView::Session => state.session_cursor.map(session_row_id),
116        _ => None,
117    };
118
119    // A retained semantic identity must never be replaced by an unrelated row
120    // that happens to occupy its old flattened cursor after projection changes.
121    if let Some(selected) = selected {
122        return rows
123            .iter()
124            .find(|row| row.id == selected)
125            .map(|row| row.id.clone());
126    }
127
128    rows.get(state.cursor).map(|row| row.id.clone())
129}

131/// Converts a typed stable session-row identity to its upstream tree key.
132///
133/// - `identity`: `ScribeRig` row identity retained across projection refreshes.
134///
135/// A namespaced upstream tree item ID.
136/// # Returns
137/// Stable item ID namespaced for session rows.
138fn session_row_id(identity: crate::view::session::SessionRowIdentity) -> String {
139    format!("session-row:{:?}:{}", identity.kind, identity.reference)
140}

142/// Converts a typed conversation source to a namespaced tree item ID.
143///
144/// - `source`: Stable `ScribeRig` source identity.
145///
146/// The stable ID used by the upstream tree.
147/// # Returns
148/// Stable item ID namespaced for conversation sources.
149fn conversation_source_id(source: crate::view::conversation::ConversationSource) -> String {
150    use crate::view::conversation::ConversationSource as Source;
151    match source {
152        Source::Session(id) => format!("session:{id}"),
153        Source::Branch(id) => format!("branch:{id}"),
154        Source::Activity(id) => format!("activity:{id}"),
155        Source::BranchResult(id) => format!("branch-result:{id}"),
156        Source::Plan(id) => format!("plan:{id}"),
157        Source::PlanNode(id) => format!("plan-node:{id}"),
158        Source::TransientActivity(id) => format!("transient:{id}"),
159    }
160}

162/// Sanitizes and bounds a semantic label before it reaches `TreeList` rendering.
163///
164/// - `label`: Projected `ScribeRig` row label.
165///
166/// A terminal-safe label limited to 800 characters.
167/// # Returns
168/// Sanitized and bounded semantic label.
169fn bounded_label(label: &str) -> String {
170    crate::inspector::sanitize(label)
171        .chars()
172        .take(800)
173        .collect()
174}

176/// Carries one semantic tree row between projections and the upstream widget.

181    label: String,

183    depth: usize,

14/// Opens the upstream model selector for the current plan-node assignment.
15///
16/// # Parameters
17/// - `context`: Shared UI state and services used by assignment selection.
18/// - `pending_selector`: Selector waiting for a model-library response.
19/// - `prompt`: Submitted assignment command.
20///
21/// # Returns
22/// Success after preparing the semantic target and opening or scheduling the
23/// upstream selector.
24///
25/// # Errors
26/// Returns an error when the shared presentation state lock is poisoned.
27pub(super) fn open<S>(context: &mut InputContext<'_, '_, S>, prompt: &str) -> io::Result<()> {
28    let mut state = context
29        .state
30        .lock()
31        .map_err(|_| io::Error::other("TUI state lock poisoned"))?;
32    prompt.clone_into(&mut state.conversation_composer.draft);
33    let effects = crate::state::reduce(&mut state, Action::InputSubmitted);
34    queue_effects(context.work_sender, context.pending_effects, effects);
35    let target_exists = state.plan_assignment_target.is_some();
36    let library = match &state.model_library {
37        crate::inspector::LoadState::Ready(library) => Some(library.clone()),
38        _ => None,
39    };
40    drop(state);
41    if !target_exists {
42        return Ok(());
43    }
44    if let Some(library) = library {
45        let previous_focus = context.screen.get_focus();
46        if open_control_selector(
47            context.screen,
48            context.shared_state,
49            ControlKind::PlanAssignment,
50            &library,
51            None,
52            context.input_sender,
53        ) {
54            if context.focus_before_overlay.is_none() {
55                *context.focus_before_overlay = previous_focus;
56            }
57            context.editor.set_focused(false);
58        }
59    } else {
60        *context.pending_selector = Some(ControlKind::PlanAssignment);
61    }
62    Ok(())
63}

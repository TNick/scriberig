8/// Borrows runtime state required to process one composer input event.
9///
10/// # Public members
11/// None.
12///
13/// # Private members
14/// - `service`: Application operations for model properties.
15/// - `context`: Authorization context for application operations.
16/// - `state`: Shared `ScribeRig` presentation state.
17/// - `screen`: Upstream terminal screen.
18/// - `editor`: Upstream composer editor.
19/// - `composer`: `ScribeRig` composer adapter.
20/// - `semantic_view`: `ScribeRig` semantic navigation adapter.
21/// - `input_sender`: Channel for follow-up composer inputs.
22/// - `work_sender`: Worker command channel.
23/// - `pending_selector`: Model selector waiting for a model library response.
24/// - `pending_effects`: Effects waiting for worker capacity.
25/// - `focus_before_overlay`: Focus target to restore after closing overlays.
26/// - `shared_state`: Shared state used by upstream selectors.
27/// - `debug_log`: Optional redacted input diagnostics.
28/// - `should_quit`: Event loop termination flag.
29/// - `redraw`: Whether the upstream screen needs a render request.
30/// - `suppress_composer_restore`: Whether a just-submitted draft must not be
31///   pushed back into the editor before its cleared draft change arrives.
32pub struct InputContext<'a, 'log, S> {
33    /// Application operations for model properties.
34    pub service: &'a S,
35    /// Authorization context for application operations.
36    pub context: &'a scriberig_app::ApplicationContext,
37    /// Shared `ScribeRig` presentation state.
38    pub state: &'a Arc<Mutex<TuiState>>,
39    /// Upstream terminal screen.
40    pub screen: &'a Arc<TuiAltScreen>,
41    /// Upstream composer editor.
42    pub editor: &'a Editor,
43    /// `ScribeRig` composer adapter.
44    pub composer: &'a Arc<super::super::super::components::ComposerAdapter>,
45    /// `ScribeRig` semantic navigation adapter.
46    pub semantic_view: &'a Arc<super::super::super::semantic_views::SemanticViewPanel>,
47    /// Channel for follow-up composer inputs.
48    pub input_sender: &'a std::sync::mpsc::Sender<ComposerInput>,
49    /// Worker command channel.
50    pub work_sender: &'a std::sync::mpsc::SyncSender<Work>,
51    /// Model selector waiting for a model library response.
52    pub pending_selector: &'a mut Option<ControlKind>,
53    /// Effects waiting for worker capacity.
54    pub pending_effects: &'a mut std::collections::VecDeque<Effect>,
55    /// Focus target to restore after closing overlays.
56    pub focus_before_overlay: &'a mut Option<Arc<dyn Component>>,
57    /// Shared state used by upstream selectors.
58    pub shared_state: &'a Arc<Mutex<TuiState>>,
59    /// Optional redacted input diagnostics.
60    pub debug_log: &'a mut Option<&'log mut TuiDebugLog>,
61    /// Event loop termination flag.
62    pub should_quit: &'a mut bool,
63    /// Whether the upstream screen needs a render request.
64    pub redraw: &'a mut bool,
65    /// Whether a just-submitted draft must not be restored into the editor.
66    pub suppress_composer_restore: &'a mut bool,
67}

12/// Applies one selected upstream control result to `ScribeRig` state.
13///
14/// - `context`: Borrowed terminal and application state.
15/// - `choice`: Stable selected control identity.
16///
17/// Success after updating the selected control or showing its follow-up selector.
18/// # Returns
19/// Completion status for the selector action.
20///
21/// # Errors
22/// Returns an error when shared state locking or diagnostics fail.
23pub(super) fn handle<S: TuiService + Sync>(
24    context: InputContext<'_, '_, S>,
25    choice: SelectorChoice,
26) -> io::Result<()> {
27    let InputContext {
28        state,
29        screen,
30        editor,
31        input_sender,
32        work_sender,
33        pending_effects,
34        focus_before_overlay,
35        shared_state,
36        debug_log,
37        ..
38    } = context;
39    let provider_choice = matches!(&choice, SelectorChoice::Provider(_));
40    if provider_choice {
41        screen.hide_overlay();
42    } else {
43        restore_overlay_focus(screen, editor, focus_before_overlay);
44    }
45    let mut state = state
46        .lock()
47        .map_err(|_| io::Error::other("TUI state lock poisoned"))?;
48    state.overlay = None;
49    match choice {
50        SelectorChoice::Provider(provider_id) => {
51            let library = match &state.model_library {
52                crate::inspector::LoadState::Ready(library) => Some(library.clone()),
53                _ => None,
54            };
55            drop(state);
56            if let Some(library) = library {
57                let previous_focus = screen.get_focus();
58                let opened = open_control_selector(
59                    screen,
60                    shared_state,
61                    ControlKind::Model,
62                    &library,
63                    Some(&provider_id),
64                    input_sender,
65                );
66                if opened {
67                    if focus_before_overlay.is_none() {
68                        *focus_before_overlay = previous_focus;
69                    }
70                    editor.set_focused(false);
71                } else {
72                    restore_overlay_focus(screen, editor, focus_before_overlay);
73                }
74            } else {
75                restore_overlay_focus(screen, editor, focus_before_overlay);
76            }
77        }
78        SelectorChoice::Model(model_id) => {
79            let effects = crate::state::select_model(&mut state, &model_id);
80            queue_effects(work_sender, pending_effects, effects);
81            if let Some(log) = debug_log.as_deref_mut() {
82                log.record_message("model-selected", &state)?;
83            }
84        }
85        SelectorChoice::PlanModel(model_id) => {
86            let item_id = format!("assign-plan-model:{model_id}");
87            if let Some(effects) = crate::state::activate_palette_control(&mut state, &item_id) {
88                queue_effects(work_sender, pending_effects, effects);
89            }
90            if let Some(log) = debug_log.as_deref_mut() {
91                log.record_message("plan-model-assigned", &state)?;
92            }
93        }
94        SelectorChoice::Thinking(level_id) => {
95            let effects = match &state.model_library {
96                crate::inspector::LoadState::Ready(library)
97                    if current_model_advertises_thinking_level(&state, library, &level_id) =>
98                {
99                    crate::state::select_thinking_level(&mut state, &level_id)
100                }
101                _ => {
102                    state.overlay = None;
103                    "That thinking level is no longer available for the selected model."
104                        .clone_into(&mut state.shell_message);
105                    Vec::new()
106                }
107            };
108            queue_effects(work_sender, pending_effects, effects);
109            if let Some(log) = debug_log.as_deref_mut() {
110                log.record_message("thinking-selected", &state)?;
111            }
112        }
113        SelectorChoice::Theme(theme_id) => match rpi_tui::ThemePreset::from_id(&theme_id) {
114            Some(preset) => rpi_tui::apply_theme_preset(preset),
115            None => {
116                "Upstream returned an unknown theme preset.".clone_into(&mut state.shell_message);
117            }
118        },
119    }
120    Ok(())
121}

123/// Checks a selected thinking ID against the currently assigned upstream model.
124///
125/// - `state`: Current `ScribeRig` session state.
126/// - `library`: Credential-safe projection of the upstream model catalog.
127/// - `level_id`: Selected scalar thinking ID.
128///
129/// Whether the current provider/model pair advertises the selected level.
130/// # Returns
131/// `true` only when the selection remains valid for the active model.
132fn current_model_advertises_thinking_level(
133    state: &crate::state::TuiState,
134    library: &scriberig_protocol::ModelLibraryResponse,
135    level_id: &str,
136) -> bool {
137    let crate::inspector::LoadState::Ready(session) = &state.session else {
138        return false;
139    };
140    let Some(provider_id) = session.execution.provider_id.as_deref() else {
141        return false;
142    };
143    let Some(model_id) = session.execution.model_id.as_deref() else {
144        return false;
145    };
146    library.models.iter().any(|model| {
147        model.provider_id == provider_id
148            && model.provider_model_id == model_id
149            && model.thinking_levels.iter().any(|level| level == level_id)
150    })
151}

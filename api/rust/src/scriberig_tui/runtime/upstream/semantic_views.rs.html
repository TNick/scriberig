37/// Adapts selected `ScribeRig` semantic views to upstream tree and Markdown APIs.
38///
39/// # Public members
40/// None.
41///
42/// # Private members
43/// - `state`: Shared `ScribeRig` navigation and semantic view projections.
44/// - `markdown`: Upstream Markdown renderer for supplementary details.
45/// - `tree`: Upstream tree of the already-filtered visible `ScribeRig` rows.
46/// - `select_list`: Upstream list of stable non-tree semantic references.
47/// - `keybindings`: Upstream keybinding manager shared with the active runtime.
48/// - `input_sender`: Channel for `ScribeRig` navigation intents.
49pub(super) struct SemanticViewPanel {
50    /// Shared `ScribeRig` navigation and semantic view projections.
51    state: Arc<Mutex<TuiState>>,
52    /// Upstream Markdown renderer for supplementary details.
53    markdown: Markdown,
54    /// Upstream tree of the current visible semantic rows.
55    tree: Mutex<Option<RetainedTree>>,
56    /// Upstream selectable list for stable non-tree semantic rows.
57    select_list: Mutex<Option<SelectList>>,
58    /// Upstream keybinding manager shared with the active runtime.
59    keybindings: Arc<Keybindings>,
60    /// Channel carrying `ScribeRig` navigation intents to its reducer.
61    input_sender: Sender<super::ComposerInput>,
62}

64impl SemanticViewPanel {
65    /// Creates the semantic projection component.
66    ///
67    /// - `state`: Shared TUI state supplying the active semantic view.
68    /// - `input_sender`: Channel for forwarding `ScribeRig` navigation intents.
69    /// - `keybindings`: Upstream keybinding manager shared with the runtime.
70    ///
71    /// # Returns
72    /// A component rendered by upstream semantic-view components.
73    pub(super) fn new(
74        state: Arc<Mutex<TuiState>>,
75        input_sender: Sender<super::ComposerInput>,
76        keybindings: Arc<Keybindings>,
77    ) -> Self {
78        Self {
79            state,
80            markdown: Markdown::new("# Conversation\n\nType a message to start.", 1, 0),
81            tree: Mutex::new(None),
82            select_list: Mutex::new(None),
83            keybindings,
84            input_sender,
85        }
86    }
87}

89impl Component for SemanticViewPanel {
90    /// Renders conversation/session rows with `TreeList` and other details with Markdown.
91    ///
92    /// - `width`: Available render width in terminal columns.
93    ///
94    /// # Returns
95    /// Rendered lines from the active upstream component.
96    fn render(&self, width: usize) -> Vec<String> {
97        let projection = match self.state.lock() {
98            Ok(state) => Self::projection_for(&state),
99            Err(_) => None,
100        };
101        if let Some(projection) = projection {
102            self.markdown.set_content(projection.details.clone());
103            let mut rows = self.render_tree(&projection, width);
104            rows.extend(self.markdown.render(width));
105            return rows;
106        }
107        if let Ok(state) = self.state.lock() {
108            if let Some((list, details)) = selection::semantic_list(&state) {
109                if let Ok(mut current) = self.select_list.lock() {
110                    *current = Some(list);
111                }
112                self.markdown.set_content(details);
113                let mut rows = self
114                    .select_list
115                    .lock()
116                    .ok()
117                    .and_then(|list| list.as_ref().map(|list| list.render(width)))
118                    .unwrap_or_default();
119                rows.extend(self.markdown.render(width));
120                return rows;
121            }
122            let content = content::render_view(&state);
123            if self.markdown.get_content() != content {
124                self.markdown.set_content(content);
125            }
126            if let Ok(mut list) = self.select_list.lock() {
127                *list = None;
128            }
129        }
130        self.markdown.render(width)
131    }
132
133    /// Converts focused upstream navigation keys into `ScribeRig` intents.
134    ///
135    /// - `event`: Normalized event delivered by the upstream focus router.
136    ///
137    /// # Returns
138    /// `true` when the event requests a `ScribeRig` navigation action.
139    fn handle_event(&self, event: &InputEvent) -> bool {
140        let InputEvent::Key(key) = event else {
141            return false;
142        };
143        let direction = if self.keybindings.matches(key, keys::SELECT_UP) {
144            Some(crate::action::Direction::Previous)
145        } else if self.keybindings.matches(key, keys::SELECT_DOWN) {
146            Some(crate::action::Direction::Next)
147        } else {
148            None
149        };
150        if self.keybindings.matches(key, keys::SELECT_CANCEL) {
151            let _ = self
152                .input_sender
153                .send(super::ComposerInput::TogglePresentationFocus(false));
154            return true;
155        }
156        if let Some(direction) = direction {
157            let tree_view = self.state.lock().is_ok_and(|state| {
158                matches!(
159                    state.primary_view,
160                    crate::selection::PrimaryView::Conversation
161                        | crate::selection::PrimaryView::Session
162                        | crate::selection::PrimaryView::Plan
163                )
164            });
165            if tree_view {
166                return self.handle_tree_movement(event, direction);
167            }
168            let list_view = self
169                .state
170                .lock()
171                .is_ok_and(|state| selection::supports(&state));
172            if list_view {
173                return self.handle_list_movement(event, direction);
174            }
175            return self
176                .input_sender
177                .send(super::ComposerInput::PresentationAction(Box::new(
178                    crate::action::Action::CursorMoved(direction),
179                )))
180                .is_ok();
181        }
182        if self.keybindings.matches(key, keys::CURSOR_LEFT)
183            || self.keybindings.matches(key, keys::CURSOR_RIGHT)
184        {
185            let Ok(state) = self.state.lock() else {
186                return false;
187            };
188            if state.primary_view != crate::selection::PrimaryView::Conversation
189                || !state.conversation_browsing
190            {
191                return false;
192            }
193            drop(state);
194            return self
195                .input_sender
196                .send(super::ComposerInput::PresentationAction(Box::new(
197                    crate::action::Action::ConversationToggleExpansion,
198                )))
199                .is_ok();
200        }
201        if self.keybindings.matches(key, keys::SELECT_CONFIRM) {
202            return self
203                .input_sender
204                .send(super::ComposerInput::PresentationAction(Box::new(
205                    crate::action::Action::Activated,
206                )))
207                .is_ok();
208        }
209        false
210    }
211
212    /// Invalidates upstream Markdown's cached frame.
213    fn invalidate(&self) {
214        self.markdown.invalidate();
215        if let Ok(tree) = self.tree.lock()
216            && let Some(tree) = tree.as_ref()
217        {
218            tree.tree.invalidate();
219        }
220        if let Ok(list) = self.select_list.lock()
221            && let Some(list) = list.as_ref()
222        {
223            list.invalidate();
224        }
225    }
226
227    /// Supports upstream layout-node inspection.
228    ///
229    /// # Returns
230    /// A type-erased reference to this component.
231    fn as_any(&self) -> &dyn Any {
232        self
233    }
234}

236impl SemanticViewPanel {
237    /// Selects the reusable tree projection for the active semantic view.
238    ///
239    /// - `state`: Current `ScribeRig` presentation state.
240    ///
241    /// Tree items, selection, and details when the view is tree-backed.
242    /// # Returns
243    /// A projection for conversation, session, and plan views.
244    fn projection_for(state: &TuiState) -> Option<TreeProjection> {
245        match state.primary_view {
246            crate::selection::PrimaryView::Conversation
247            | crate::selection::PrimaryView::Session => tree::projection(state),
248            crate::selection::PrimaryView::Plan => {
249                plan::projection(state).map(|(projection, _)| projection)
250            }
251            _ => None,
252        }
253    }
254
255    /// Reuses the retained tree and renders the current projection.
256    ///
257    /// - `projection`: Current semantic tree projection.
258    /// - `width`: Available render width in terminal columns.
259    ///
260    /// # Returns
261    /// Rendered tree rows, or no rows when validation fails.
262    fn render_tree(&self, projection: &TreeProjection, width: usize) -> Vec<String> {
263        let Ok(mut slot) = self.tree.lock() else {
264            return Vec::new();
265        };
266        if sync_tree(&mut slot, projection).is_none() {
267            return Vec::new();
268        }
269        slot.as_ref()
270            .map(|tree| tree.tree.render(width))
271            .unwrap_or_default()
272    }
273
274    /// Moves the upstream selectable list and forwards `ScribeRig` cursor intent.
275    ///
276    /// - `event`: Upstream key event to apply to the list.
277    /// - `direction`: Semantic movement direction selected by upstream keys.
278    ///
279    /// # Returns
280    /// `true` when the active semantic list handles the key.
281    fn handle_list_movement(
282        &self,
283        event: &InputEvent,
284        direction: crate::action::Direction,
285    ) -> bool {
286        let Ok(state) = self.state.lock() else {
287            return false;
288        };
289        let Some((list, rows)) = selection::list_from_state(&state) else {
290            drop(state);
291            return self.send_cursor_action(direction);
292        };
293        let before = list.get_selected_item().map(|item| item.value);
294        let _ = list.handle_event(event);
295        let after = list.get_selected_item().map(|item| item.value);
296        let old_index = before.and_then(|id| rows.iter().position(|row| row.id == id));
297        let new_index = after.and_then(|id| rows.iter().position(|row| row.id == id));
298        drop(state);
299        if let (Some(old), Some(new)) = (old_index, new_index)
300            && old != new
301        {
302            let actual_direction = if new < old {
303                crate::action::Direction::Previous
304            } else {
305                crate::action::Direction::Next
306            };
307            return self
308                .input_sender
309                .send(super::ComposerInput::PresentationAction(Box::new(
310                    crate::action::Action::CursorMoved(actual_direction),
311                )))
312                .is_ok();
313        }
314        true
315    }
316
317    /// Moves the upstream tree cursor and forwards the matching `ScribeRig` action.
318    ///
319    /// - `event`: Upstream key event to apply to the tree.
320    /// - `direction`: Semantic movement direction selected by upstream keys.
321    ///
322    /// # Returns
323    /// `true` when the active semantic tree handles the key.
324    fn handle_tree_movement(
325        &self,
326        event: &InputEvent,
327        direction: crate::action::Direction,
328    ) -> bool {
329        let projection = match self.state.lock() {
330            Ok(state) => Self::projection_for(&state),
331            Err(_) => None,
332        };
333        let Some(projection) = projection else {
334            return self.send_cursor_action(direction);
335        };
336        let Ok(mut slot) = self.tree.lock() else {
337            return false;
338        };
339        if sync_tree(&mut slot, &projection).is_none() {
340            drop(slot);
341            return self.send_cursor_action(direction);
342        }
343        let Some(tree) = slot.as_ref() else {
344            drop(slot);
345            return self.send_cursor_action(direction);
346        };
347        let before = tree.tree.selected_id();
348        let _ = tree.tree.handle_event(event);
349        let after = tree.tree.selected_id();
350        drop(slot);
351        let old_index =
352            before.and_then(|id| projection.items.iter().position(|item| item.id == id));
353        let new_index = after.and_then(|id| projection.items.iter().position(|item| item.id == id));
354        if let (Some(old), Some(new)) = (old_index, new_index)
355            && old != new
356        {
357            let actual_direction = if new < old {
358                crate::action::Direction::Previous
359            } else {
360                crate::action::Direction::Next
361            };
362            return self
363                .input_sender
364                .send(super::ComposerInput::PresentationAction(Box::new(
365                    crate::action::Action::CursorMoved(actual_direction),
366                )))
367                .is_ok();
368        }
369        true
370    }
371
372    /// Sends the current non-tree cursor movement action.
373    ///
374    /// - `direction`: Semantic movement direction selected by upstream keys.
375    ///
376    /// # Returns
377    /// `true` when the action was queued.
378    fn send_cursor_action(&self, direction: crate::action::Direction) -> bool {
379        self.input_sender
380            .send(super::ComposerInput::PresentationAction(Box::new(
381                crate::action::Action::CursorMoved(direction),
382            )))
383            .is_ok()
384    }

1//! Owns terminal setup and restoration for the interactive runtime.

3use std::io::{self, Write};
4use std::panic::{self, PanicHookInfo};
5use std::sync::Arc;
6
7use crossterm::{
8    cursor::{Hide, Show},
9    event::{DisableBracketedPaste, EnableBracketedPaste},
10    execute,
11    terminal::{EnterAlternateScreen, LeaveAlternateScreen, disable_raw_mode, enable_raw_mode},
12};

14/// A panic hook that can be restored after terminal teardown.
15type PanicHook = Arc<dyn Fn(&PanicHookInfo<'_>) + Send + Sync + 'static>;
16
17/// Restores terminal state and composes with the previous panic hook.
18pub struct TerminalGuard {
19    previous_hook: Option<PanicHook>,
20    restored: bool,
21}
22
23/// Tracks terminal capabilities changed by the guard.
24#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
25#[cfg(test)]
26#[allow(clippy::struct_excessive_bools)]
27struct LifecycleState {
28    raw_mode: bool,
29    alternate_screen: bool,
30    cursor_hidden: bool,
31    bracketed_paste: bool,
32}
33
34impl TerminalGuard {
35    /// Enters raw/alternate-screen mode and installs panic restoration.
36    ///
37    /// # Errors
38    /// Returns an I/O error when raw mode or alternate-screen setup fails.
39    pub fn enter() -> io::Result<Self> {
40        enable_raw_mode()?;
41        let mut stdout = io::stdout();
42        if let Err(error) = execute!(stdout, EnterAlternateScreen, Hide, EnableBracketedPaste) {
43            let _ = disable_raw_mode();
44            let _ = Self::restore_terminal();
45            return Err(error);
46        }
47        let previous_hook: Arc<dyn Fn(&PanicHookInfo<'_>) + Send + Sync> =
48            Arc::from(panic::take_hook());
49        let panic_hook = Arc::clone(&previous_hook);
50        panic::set_hook(Box::new(move |panic| {
51            let _ = Self::restore_terminal();
52            panic_hook(panic);
53        }));
54        Ok(Self {
55            previous_hook: Some(previous_hook),
56            restored: false,
57        })
58    }
59
60    /// Restores terminal state once and reinstalls the previous panic hook.
61    ///
62    /// # Errors
63    /// Returns an I/O error when terminal restoration fails.
64    pub fn restore(&mut self) -> io::Result<()> {
65        if self.restored {
66            return Ok(());
67        }
68        let result = Self::restore_terminal();
69        self.restored = true;
70        if let Some(previous_hook) = self.previous_hook.take() {
71            panic::set_hook(Box::new(move |panic| previous_hook(panic)));
72        }
73        result
74    }
75
76    /// Performs the backend commands needed to leave interactive mode.
77    fn restore_terminal() -> io::Result<()> {
78        let mut first_error = disable_raw_mode().err();
79        let mut stdout = io::stdout();
80        for command in [
81            TerminalCommand::DisableBracketedPaste,
82            TerminalCommand::Show,
83            TerminalCommand::LeaveAlternateScreen,
84        ] {
85            let result = match command {
86                TerminalCommand::DisableBracketedPaste => {
87                    execute!(stdout, DisableBracketedPaste)
88                }
89                TerminalCommand::Show => execute!(stdout, Show),
90                TerminalCommand::LeaveAlternateScreen => execute!(stdout, LeaveAlternateScreen),
91            };
92            if first_error.is_none() {
93                first_error = result.err();
94            }
95        }
96        first_error.map_or(Ok(()), Err)
97    }
98}
99
100/// Identifies one independently attempted terminal restoration command.
101enum TerminalCommand {
102    /// Disables bracketed paste mode.
103    DisableBracketedPaste,
104    /// Shows the cursor.
105    Show,
106    /// Leaves the alternate screen.
107    LeaveAlternateScreen,
108}
109
110impl Drop for TerminalGuard {
111    /// Attempts restoration if the owner did not explicitly restore first.
112    fn drop(&mut self) {
113        let _ = self.restore();
114    }
115}
116
117/// Flushes terminal output after a lifecycle command in tests or adapters.
118///
119/// # Errors
120/// Returns the writer's flush error.
121pub fn flush(output: &mut impl Write) -> io::Result<()> {
122    output.flush()
123}

5/// Converts the application model library into credential-safe wire descriptors.
6///
7/// - `library`: Validated application model library.
8///
9/// Protocol descriptors containing references but no resolved credentials.
10/// # Returns
11/// The operation result.
12pub(in crate::service) fn model_library_response(
13    library: scriberig_app::ModelLibrary,
14) -> scriberig_protocol::ModelLibraryResponse {
15    scriberig_protocol::ModelLibraryResponse {
16        models: library
17            .models
18            .into_iter()
19            .map(|model| scriberig_protocol::PublicModelDescriptor {
20                id: model.id.as_str().to_owned(),
21                display_name: model.display_name,
22                provider_profile: model.provider_profile,
23                provider_model_id: model.provider_model_id,
24                capabilities: scriberig_protocol::PublicModelCapabilities {
25                    streaming: model.capabilities.streaming,
26                    tool_calls: model.capabilities.tool_calls,
27                    structured_output: model.capabilities.structured_output,
28                },
29                thinking_levels: model
30                    .thinking_levels
31                    .into_iter()
32                    .map(|level| scriberig_protocol::PublicThinkingLevelDescriptor {
33                        id: level.id.as_str().to_owned(),
34                        label: level.label,
35                        description: level.description,
36                        order: level.order,
37                    })
38                    .collect(),
39                default_thinking_level: model
40                    .default_thinking_level
41                    .map(|level| level.as_str().to_owned()),
42            })
43            .collect(),
44        profiles: library
45            .profiles
46            .into_iter()
47            .map(|profile| scriberig_protocol::PublicModelProfile {
48                id: profile.id,
49                display_name: profile.display_name,
50                model: profile.model.as_str().to_owned(),
51                connection: profile.connection,
52                credential: profile
53                    .credential
54                    .map(|credential| scriberig_protocol::ModelCredentialRef { id: credential.id }),
55                default_thinking_level: profile
56                    .default_thinking_level
57                    .map(|level| level.as_str().to_owned()),
58                enabled: profile.enabled,
59            })
60            .collect(),
61    }
62}

64impl<S: QueryService> TuiService for S {
65    /// Delegates inspection through a public query service.
66    ///
67    /// - `context`: Application context for the request.
68    /// - `entity`: Entity to inspect.
69    /// - `snapshot`: Optional exact snapshot.
70    ///
71    /// The inspection result.
72    ///
73    /// Returns the query service error.
74    /// # Returns
75    /// The operation result.
76    /// # Errors
77    /// Returns an error when the operation fails.
78    fn inspect_entity(
79        &self,
80        context: &ApplicationContext,
81        entity: EntityId,
82        snapshot: Option<SnapshotId>,
83    ) -> Result<InspectResult, ApplicationError> {
84        self.inspect(
85            context,
86            &InspectRequest {
87                entity,
88                snapshot: snapshot.map_or(SnapshotResolution::Current, SnapshotResolution::At),
89                detail: scriberig_app::DetailLevel::Detailed,
90            },
91        )
92    }
93}

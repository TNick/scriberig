13/// Adapts a protocol transport to TUI service operations.
14pub struct RemoteAdapter<T> {
15    /// Protocol client used for semantic requests.
16    pub(super) client: scriberig_client::Client<T>,
17    /// Last remote event cursor delivered to the TUI runtime.
18    event_cursor: std::sync::Mutex<Option<u64>>,
19    /// Session identities whose durable state can be refreshed after a cursor gap.
20    known_sessions: std::sync::Mutex<BTreeSet<EntityId>>,
21}
22
23impl<T> RemoteAdapter<T> {
24    /// Creates a remote TUI adapter over a public protocol client.
25    ///
26    /// # Parameters
27    /// - `client`: Protocol client used for remote TUI requests.
28    ///
29    /// # Returns
30    /// A remote adapter backed by the client.
31    #[must_use]
32    pub fn new(client: scriberig_client::Client<T>) -> Self {
33        Self {
34            client,
35            event_cursor: std::sync::Mutex::new(None),
36            known_sessions: std::sync::Mutex::new(BTreeSet::new()),
37        }
38    }
39}

42    /// Inspects an entity through the public protocol client.
43    ///
44    /// # Parameters
45    /// - `_context`: Application context for the request.
46    /// - `entity`: Entity to inspect.
47    /// - `snapshot`: Optional exact snapshot.
48    ///
49    /// # Returns
50    /// The decoded inspection result.
51    ///
52    /// # Errors
53    /// Returns a protocol or response-decoding error.
54    fn inspect_entity(
55        &self,
56        _context: &ApplicationContext,
57        entity: EntityId,
58        snapshot: Option<SnapshotId>,
59    ) -> Result<InspectResult, ApplicationError> {
60        let selector = snapshot.map_or(scriberig_protocol::SnapshotSelector::Current, |value| {
61            scriberig_protocol::SnapshotSelector::Exact(scriberig_protocol::SnapshotReference {
62                id: value.to_string(),
63            })
64        });
65        let response = self
66            .client
67            .inspect(entity.to_string(), selector, true)
68            .map_err(remote_error)?;
69        let resolved_entity = EntityId::from_str(&response.entity.id).map_err(|error| {
70            ApplicationError::Internal {
71                message: format!("remote response contained invalid entity identity: {error}"),
72            }
73        })?;
74        let resolved_snapshot = SnapshotId::from_str(&response.snapshot.id).map_err(|error| {
75            ApplicationError::Internal {
76                message: format!("remote response contained invalid snapshot identity: {error}"),
77            }
78        })?;
79        let revision = RevisionId::from_str(&response.revision.id).ok();
80        Ok(InspectResult {
81            entity: resolved_entity,
82            revision,
83            snapshot: Some(resolved_snapshot),
84            object_type: response
85                .object_type
86                .and_then(|value| scriberig_core::ObjectType::new(value).ok()),
87            label: response.summary,
88            description: None,
89            capabilities: Vec::new(),
90            trust: None,
91        })
92    }

220    /// Reads and reconstructs the compact remote session view.
221    ///
222    /// # Parameters
223    /// - `_context`: Authorized application context.
224    /// - `session`: Stable session identity.
225    ///
226    /// # Returns
227    /// A provider-neutral session value usable by the existing TUI view model.
228    ///
229    /// # Errors
230    /// Returns a sanitized remote or identity-decoding error.
231    fn read_session(
232        &self,
233        _context: &ApplicationContext,
234        session: EntityId,
235    ) -> Result<scriberig_session::Session, ApplicationError> {
236        let response = self
237            .client
238            .read_session(session.to_string())
239            .map_err(remote_error)?;
240        if let Ok(mut known_sessions) = self.known_sessions.lock() {
241            known_sessions.insert(session);
242        }
243        session_from_response(response)
244    }

6/// Reads runtime capability descriptors from the remote client.
7///
8/// - `adapter`: Remote service adapter holding the protocol client.
9///
10/// Decoded runtime capabilities.
11///
12/// Returns a protocol capability-discovery error.
13pub(super) fn runtime_capabilities<T: ProtocolTransport>(
14    adapter: &RemoteAdapter<T>,
15) -> Result<Vec<super::RuntimeCapability>, ApplicationError> {
16    adapter
17        .client
18        .capabilities()
19        .map_err(super::service_remote_mapping::remote_error)
20        .map(|response| {
21            response
22                .capabilities
23                .into_iter()
24                .filter_map(|capability| {
25                    Some(super::RuntimeCapability {
26                        id: scriberig_app::CapabilityId::new(capability.id).ok()?,
27                        description: capability.description,
28                        available: capability.available,
29                        scope: capability.domain,
30                    })
31                })
32                .collect()
33        })
34}

100/// Applies a revision-checked provider/model selection remotely.
101///
102/// - `adapter`: Remote service adapter holding the protocol client.
103/// - `request`: Session and observed selection revision.
104///
105/// The authoritative updated execution selection.
106///
107/// Returns a sanitized remote protocol error.
108pub(super) fn set_session_model<T: ProtocolTransport>(
109    adapter: &RemoteAdapter<T>,
110    request: &scriberig_app::SessionModelSelectionRequest,
111) -> Result<scriberig_session::SessionExecutionSelection, ApplicationError> {
112    adapter
113        .client
114        .set_session_model(
115            request.session.to_string(),
116            request.expected_revision,
117            request.provider_id.clone(),
118            request.model_id.clone(),
119        )
120        .map_err(super::service_remote_mapping::remote_error)
121        .map(to_session_selection)
122}

147/// Converts a protocol selection into the session-domain value.
148///
149/// - `selection`: Protocol selection returned by the server.
150///
151/// Equivalent provider-neutral session selection.
152/// # Returns
153/// The operation result.
154fn to_session_selection(
155    selection: scriberig_protocol::SessionExecutionSelection,
156) -> scriberig_session::SessionExecutionSelection {
157    scriberig_session::SessionExecutionSelection {
158        revision: selection.revision,
159        provider_id: selection.provider_id,
160        model_id: selection.model_id,
161        thinking_level: selection.thinking_level,
162        history: selection
163            .history
164            .into_iter()
165            .map(|change| scriberig_session::SessionExecutionChange {
166                revision: change.revision,
167                kind: change.kind,
168                summary: change.summary,
169                provider_id: change.provider_id,
170                model_id: change.model_id,
171                thinking_level: change.thinking_level,
172            })
173            .collect(),
174    }

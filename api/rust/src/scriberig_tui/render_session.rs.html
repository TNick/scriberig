11/// Maximum untrusted row-label characters presented in one terminal frame.
12const MAX_SESSION_LABEL_CHARS: usize = 512;
13
14/// Renders session activity with visible rails and finite depth colors.
15///
16/// - `state`: Cached session and presentation state.
17/// - `available_width`: Width available for the session rows.
18/// - `available_rows`: Number of rows available for session presentation.
19///
20/// Session activity text with hierarchy preserved independently of color.
21/// # Returns
22/// The operation result.
23pub(super) fn session_text(
24    state: &TuiState,
25    available_width: usize,
26    available_rows: usize,
27) -> Text<'static> {
28    let LoadState::Ready(model) = &state.session else {
29        return Text::from(super::render_primary::primary_text(state));
30    };
31    let visible_rows = visible_session_rows(
32        model.rows.len(),
33        state.cursor,
34        state.focus == crate::state::Focus::Primary,
35        available_rows,
36    );
37    let mut lines = model
38        .rows
39        .iter()
40        .enumerate()
41        .skip(visible_rows.start)
42        .take(visible_rows.len())
43        .flat_map(|(index, row)| {
44            let label = super::render_primary::session_row_label(state, row);
45            let label = scriberig_tui_kit::ExternalText::new(&bounded_label(&label));
46            let mut spans = vec![Span::raw(
47                if state.focus == crate::state::Focus::Primary && state.cursor == index {
48                    "> "
49                } else {
50                    "  "
51                },
52            )];
53            let rail_capacity = available_width.saturating_sub(12) / 2;
54            let visible_rails = row.depth.min(rail_capacity);
55            let first_visible_depth = row.depth.saturating_sub(visible_rails);
56            if first_visible_depth > 0 {
57                spans.push(Span::raw("…"));
58            }
59            spans.extend(
60                (first_visible_depth..row.depth)
61                    .map(|depth| Span::styled("▏ ", Style::default().fg(depth_color(depth)))),
62            );
63            if row.transient {
64                spans.push(Span::raw("~ "));
65            }
66            let mut row_lines = label.as_str().lines();
67            let first = row_lines.next().unwrap_or_default();
68            spans.push(Span::raw(first.to_owned()));
69            let mut rendered = vec![Line::from(spans)];
70            let continuation_indent = format!(
71                "  {}{}",
72                if first_visible_depth > 0 { "…" } else { "" },
73                "  ".repeat(visible_rails)
74            );
75            rendered
76                .extend(row_lines.map(|line| Line::from(format!("{continuation_indent}{line}"))));
77            rendered
78        })
79        .collect::<Vec<_>>();
80    lines.extend(state.transient_agent_outputs.iter().map(|output| {
81        Line::from(format!(
82            "[stream · {}] {}",
83            output.operation,
84            crate::inspector::sanitize(&output.text)
85        ))
86    }));
87    lines.extend(model.execution.history.iter().map(|change| {
88        let label = match change.kind {
89            scriberig_session::AgentActivityKind::ModelSelectionChanged => "Model selection",
90            scriberig_session::AgentActivityKind::ThinkingLevelChanged => "Thinking selection",
91            _ => "Execution selection",
92        };
93        Line::from(format!(
94            "[{label} · revision {}] {}",
95            change.revision,
96            crate::inspector::sanitize(&change.summary)
97        ))
98    }));
99    if state.reasoning_visibility == crate::state::ReasoningVisibility::Shown {
100        lines.extend(
101            state
102                .transient_agent_activities
103                .iter()
104                .map(|activity| {
105                    let label = if activity.activity_kind
106                        == scriberig_session::AgentActivityKind::ReasoningSummary
107                    {
108                        "Reasoning summary"
109                    } else {
110                        "Live activity"
111                    };
112                    Line::from(format!(
113                        "[{label}] {}: {} ({:?})",
114                        activity.branch,
115                        crate::inspector::sanitize(&activity.summary),
116                        activity.state
117                    ))
118                })
119                .collect::<Vec<_>>(),
120        );
121    }
122    Text::from(lines)
123}

125/// Bounds a row label before terminal sanitization allocates its display form.
126///
127/// - `label`: User or runtime supplied session row text.
128///
129/// A bounded prefix with a truncation marker when content was clipped.
130/// # Returns
131/// The bounded safe label.
132fn bounded_label(label: &str) -> String {
133    let mut characters = label.chars();
134    let mut bounded = characters
135        .by_ref()
136        .take(MAX_SESSION_LABEL_CHARS)
137        .collect::<String>();
138    if characters.next().is_some() {
139        bounded.push('…');
140    }
141    bounded
142}

144/// Selects the session rows that fit the viewport around the active row.
145///
146/// - `row_count`: Number of rows in the complete semantic session tree.
147/// - `cursor`: Selected row index.
148/// - `focused`: Whether the session tree owns keyboard focus.
149/// - `available_rows`: Number of rows available for presentation.
150///
151/// A bounded row range that keeps the selected row visible when focused.
152/// # Returns
153/// The visible row indices.
154fn visible_session_rows(
155    row_count: usize,
156    cursor: usize,
157    focused: bool,
158    available_rows: usize,
159) -> Range<usize> {
160    let visible_count = row_count.min(available_rows.max(1));
161    let selected = if focused {
162        cursor.min(row_count.saturating_sub(1))
163    } else {
164        0
165    };
166    let start = selected.saturating_add(1).saturating_sub(visible_count);
167    start..start.saturating_add(visible_count).min(row_count)
168}

170#[cfg(test)]
171mod tests {
172    use super::session_text;
173    use crate::{inspector::LoadState, state::TuiState, view::session::SessionRow};
174
175    /// Verifies large session trees render only rows that fit the viewport.
176    #[test]
177    fn large_session_tree_renders_a_bounded_selected_window() {
178        let rows = (0..10_000)
179            .map(|index| SessionRow {
180                reference: scriberig_core::EntityId::from_bytes(
181                    [u8::try_from(index % 256).expect("bounded entity byte"); 16],
182                ),
183                depth: index % 8,
184                label: if index == 9_999 {
185                    "x".repeat(100_000)
186                } else {
187                    format!("branch-{index}")
188                },
189                kind: crate::view::session::SessionRowKind::Branch,
190                candidate: None,
191                snapshot: None,
192                transient: false,
193            })
194            .collect();
195        let mut state = TuiState::default();
196        state.session = LoadState::Ready(crate::view::session::SessionModel {
197            rows,
198            ..crate::view::session::SessionModel::default()
199        });
200        state.cursor = 9_999;
201        state.focus = crate::state::Focus::Primary;
202
203        let rendered = session_text(&state, 80, 12);
204        let text = rendered
205            .lines
206            .iter()
207            .map(ToString::to_string)
208            .collect::<Vec<_>>();
209
210        assert_eq!(text.len(), 12);
211        assert!(text.last().is_some_and(|row| row.ends_with('…')));
212        assert!(text.last().is_some_and(|row| row.chars().count() <= 540));
213        assert!(!text.iter().any(|row| row.contains("branch-0")));
214    }

113/// Finds command suggestions from the shared descriptor table.
114///
115/// - `prefix`: Slash-prefixed composer content.
116///
117/// Matching command descriptors in registry order.
118/// # Returns
119/// The operation result.
120#[must_use]
121pub fn suggestions(prefix: &str) -> Vec<&'static SlashCommandDescriptor> {
122    if !prefix.starts_with('/') || prefix.starts_with("//") || prefix.contains(' ') {
123        return Vec::new();
124    }
125    COMMANDS
126        .iter()
127        .filter(|command| command.name.starts_with(prefix))
128        .collect()
129}
130
131/// Finds a descriptor for an exact command name.
132///
133/// - `name`: Canonical slash command spelling.
134///
135/// The registered descriptor when the command is recognized.
136/// # Returns
137/// The operation result.
138#[must_use]
139pub fn find(name: &str) -> Option<&'static SlashCommandDescriptor> {
140    COMMANDS.iter().find(|command| command.name == name)
141}

143#[cfg(test)]
144mod tests {
145    use super::{SlashCommandDescriptor, suggestions};
146
147    /// Keeps completion examples tied to the same registry used by parsing.
148    #[test]
149    fn command_prefixes_suggest_model_and_plan() {
150        assert!(suggestions("/mo").iter().any(|item| item.name == "/model"));
151        assert!(suggestions("/pl").iter().any(|item| item.name == "/plan"));
152        assert_eq!(suggestions("//pl"), [] as [&SlashCommandDescriptor; 0]);
153    }

3/// Direction for cursor movement in a list-like presentation.
4#[derive(Clone, Copy, Debug, Eq, PartialEq)]
5pub enum Direction {
6    /// Move toward the previous item.
7    Previous,
8    /// Move toward the next item.
9    Next,
10}

12/// A typed presentation action independent of terminal or service handles.
13#[derive(Clone, Debug, Eq, PartialEq)]
14#[allow(clippy::large_enum_variant)]
15pub enum Action {
16    /// Records that an effect had no stateful result.
17    Noop,
18    /// Request normal runtime termination.
19    QuitRequested,
20    /// Close the active overlay, if any.
21    OverlayClosed,
22    /// Move focus to the next visible pane.
23    FocusNext,
24    /// Move focus to the previous visible pane.
25    FocusPrevious,
26    /// Move a local cursor.
27    CursorMoved(Direction),
28    /// Activate the focused row or overlay item.
29    Activated,
30    /// Navigate to the previous semantic presentation location.
31    Back,
32    /// Navigate to the next semantic presentation location.
33    Forward,
34    /// Request a presentation refresh.
35    RefreshRequested,
36    /// Open the help overlay.
37    HelpOpened,
38    /// Open the local command palette.
39    PaletteOpened,
40    /// Open the find input overlay.
41    FindOpened,
42    /// Open the exact snapshot chooser for semantic comparison.
43    DiffChooserOpened,
44    /// Open a source projection for the exact selected snapshot.
45    ProjectionOpened,
46    /// Update the active text input.
47    InputChanged(String),
48    /// Submit the active text input.
49    InputSubmitted,
50    /// Applies an accepted conversation submission with authoritative handles.
51    ConversationAgentAccepted(crate::service::AgentHandle),
52    /// Enters or leaves conversation-tree browse mode.
53    ConversationBrowsing(bool),
54    /// Pins the current navigable conversation source as the reply target.
55    ConversationPinCursor,
56    /// Resumes local follow of the shared plan coordinator cursor.
57    ConversationFollowCoordinator,
58    /// Expands or collapses the currently selected branch in the local tree.
59    ConversationToggleExpansion,
60    /// Record a terminal resize.
61    TerminalResized {
62        /// New terminal width in columns.
63        width: u16,
64        /// New terminal height in rows.
65        height: u16,
66    },
67    /// Insert one paste payload into the active input.
68    Paste(String),
69    /// Activate a concrete semantic location.
70    ActivateLocation(crate::selection::NavigationLocation),
71    /// Mark an inspect request as outstanding.
72    InspectRequested {
73        /// Monotonic request identifier.
74        request_id: u64,
75        /// Entity being inspected.
76        entity: scriberig_core::EntityId,
77        /// Optional exact snapshot.
78        snapshot: Option<scriberig_core::SnapshotId>,
79    },
80    /// Apply a successful inspect result.
81    InspectSucceeded {
82        /// Request that produced the result.
83        request_id: u64,

87    /// Apply a failed inspect result.
88    InspectFailed {
89        /// Request that failed.

107    /// Replace observed lifecycle state with a current application snapshot.
108    OperationsLoaded(Vec<scriberig_app::OperationRecord>),
109    /// Switches the primary view without changing semantic selection.
110    ViewSelected(crate::selection::PrimaryView),
111    /// Applies a history operation result.
112    HistorySucceeded {
113        /// Request generation.
114        request_id: u64,
115        /// Projected history.

118    /// Applies a diff operation result.

120        /// Request generation.

122        /// Projected diff.

124    },

127        /// Request generation.

129        /// Projected rendered content.
130        model: crate::view::projection::ProjectionModel,
131    },
132    /// Applies a candidate review result.
133    ReviewSucceeded {
134        /// Request generation.
135        request_id: u64,
136        /// Projected review.
137        model: crate::view::changes::ChangeReviewModel,

141        /// Request generation.

143        /// Reviewed candidate result.
144        model: crate::view::changes::ChangeReviewModel,
145        /// Authoritative successor snapshot returned by the application.
146        snapshot: scriberig_core::SnapshotId,
147    },
148    /// Applies a failed operation result without retrying it.
149    OperationFailed {
150        /// Request generation.
151        request_id: u64,
152        /// Stable application error category.

154        /// Safe error message.

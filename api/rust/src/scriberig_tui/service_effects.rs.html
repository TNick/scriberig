10/// Supplies shared application operations to semantic TUI views.
11pub trait TuiOperationService {
12    /// Executes one public application operation without exposing storage.
13    ///
14    /// # Parameters
15    /// - `context`: Authorized application context.
16    /// - `request`: Public universal application request.
17    ///
18    /// # Returns
19    /// Structured application result owned by the application layer.
20    ///
21    /// # Errors
22    /// Returns the structured application error from the owning subsystem.
23    fn execute_operation(
24        &self,
25        context: &ApplicationContext,
26        request: &ApplicationRequest,
27    ) -> Result<scriberig_app::ApplicationResult, ApplicationError>;
28}

30/// Adapts the complete public application facade for operation-backed views.
31#[allow(clippy::too_many_lines)]
32pub fn execute_effect<S: TuiService>(
33    service: &S,
34    context: &ApplicationContext,
35    effect: Effect,
36) -> Action {
37    match effect {
38        Effect::InspectEntity {
39            request_id,
40            entity,
41            snapshot,
42        } => match service.inspect_entity(context, entity, snapshot) {
43            Ok(result) => Action::InspectSucceeded {
44                request_id,
45                model: InspectorModel::from_result(result),
46            },
47            Err(error) => Action::InspectFailed {
48                request_id,
49                category: error.category().to_owned(),
50                message: crate::inspector::sanitize(&error.to_string()),
51            },
52        },
53        Effect::ListExplorer {
54            request_id,
55            entity,
56            snapshot,
57            continuation,
58        } => service
59            .list_explorer(context, entity, snapshot, continuation)
60            .map(|page| Action::ExplorerSucceeded {
61                request_id,
62                rows: page.rows,
63                next: page.next,
64            })
65            .unwrap_or_else(|error| Action::OperationFailed {
66                request_id,
67                category: error.category().to_owned(),
68                message: crate::inspector::sanitize(&error.to_string()),
69            }),
70        Effect::RequestCapabilities => match service.runtime_capabilities(context) {
71            Ok(capabilities) => Action::CapabilitiesLoaded(capabilities),
72            Err(error) => Action::InspectFailed {
73                request_id: 0,
74                category: error.category().to_owned(),
75                message: crate::inspector::sanitize(&error.to_string()),
76            },
77        },
78        Effect::History { .. }
79        | Effect::Diff { .. }
80        | Effect::Render { .. }
81        | Effect::Apply { .. }
82        | Effect::Commit { .. } => execute_operation_effect(service, context, effect),
83        Effect::OperationStatus { operation } => service
84            .operation_status(context, &operation)
85            .map(|record| {
86                Action::OperationObserved(crate::view::operations::OperationModel::from_record(
87                    &record,
88                ))
89            })
90            .unwrap_or_else(|error| Action::OperationFailed {
91                request_id: 0,
92                category: error.category().to_owned(),
93                message: crate::inspector::sanitize(&error.to_string()),
94            }),
95        Effect::CancelOperation { operation } => service
96            .cancel_operation(context, &operation)
97            .map(|record| {
98                Action::OperationObserved(crate::view::operations::OperationModel::from_record(
99                    &record,
100                ))
101            })
102            .unwrap_or_else(|error| Action::OperationFailed {
103                request_id: 0,
104                category: error.category().to_owned(),
105                message: crate::inspector::sanitize(&error.to_string()),
106            }),
107        Effect::ReadSession {
108            request_id,
109            session,
110            branch,
111        } => branch
112            .map_or_else(
113                || service.read_session(context, session),
114                |branch| service.read_session_branch(context, session, branch),
115            )
116            .map(|value| Action::SessionSucceeded {
117                request_id,
118                model: crate::view::session::SessionModel::from_session(&value),
119            })
120            .unwrap_or_else(|error| Action::OperationFailed {
121                request_id,
122                category: error.category().to_owned(),
123                message: crate::inspector::sanitize(&error.to_string()),
124            }),
125        Effect::SubmitAgent {
126            session,
127            branch,
128            prompt,
129        } => service
130            .submit_agent_handle(context, session, branch, &prompt)
131            .map(Action::AgentSubmitted)
132            .unwrap_or_else(|error| Action::OperationFailed {
133                request_id: 0,
134                category: error.category().to_owned(),
135                message: crate::inspector::sanitize(&error.to_string()),
136            }),
137        Effect::RequestRefresh => Action::RefreshRequested,
138        Effect::RequestQuit => Action::QuitRequested,
139        Effect::RequestFind {
140            request_id,
141            text,
142            snapshot,
143        } => service
144            .find_entities(context, snapshot, &text)
145            .map(|model| Action::FindSucceeded { request_id, model })
146            .unwrap_or_else(|error| Action::OperationFailed {
147                request_id: 0,
148                category: error.category().to_owned(),
149                message: crate::inspector::sanitize(&error.to_string()),
150            }),
151        Effect::RequestSemanticAction { id, .. } => Action::OperationFailed {
152            request_id: 0,
153            category: "unsupported".to_owned(),
154            message: format!("semantic action is not available through this TUI service: {id}"),
155        },
156        Effect::None => Action::Noop,
157    }
158}

160/// Selects one branch and all descendants from an authoritative session.
161///
162/// # Parameters
163/// - `session`: Session graph to filter.
164/// - `branch`: Selected branch identity.
165/// - `crate`: Internal visibility marker.
166///
167/// # Returns
168/// A session containing the selected branch and descendants.
169pub(crate) fn select_branch_subtree(mut session: Session, branch: EntityId) -> Session {
170    let mut selected = BTreeSet::from([branch]);
171    let mut changed = true;
172    while changed {
173        changed = false;
174        for candidate in &session.branches {
175            if candidate
176                .parent
177                .is_some_and(|parent| selected.contains(&parent))
178                && selected.insert(candidate.id)
179            {
180                changed = true;
181            }
182        }
183    }
184    session
185        .branches
186        .retain(|candidate| selected.contains(&candidate.id));
187    session
188}

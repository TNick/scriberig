20/// Opens the modal composer for one session branch.
21///
22/// - `state`: TUI state receiving the composer overlay.
23/// - `session`: Stable session identity.
24/// - `branch`: Optional branch receiving the submission.
25///
26/// A no-op effect after opening the composer.
27/// # Returns
28/// The operation result.
29pub(super) fn composer_opened(
30    state: &mut TuiState,
31    session: scriberig_core::EntityId,
32    branch: Option<scriberig_core::EntityId>,
33) -> Vec<Effect> {
34    state.overlay = Some(Overlay::Composer(crate::overlay::ComposerState {
35        session,
36        branch,
37        text: String::new(),
38        error: None,
39    }));
40    vec![Effect::None]
41}

43/// Opens and focuses the persistent session composer while refreshing rows.
44///
45/// - `state`: TUI state receiving the persistent composer.
46/// - `session`: Stable session identity.
47/// - `branch`: Optional branch receiving the submission.
48///
49/// A session-read effect.
50/// # Returns
51/// The operation result.
52pub(super) fn session_composer_opened(
53    state: &mut TuiState,
54    session: scriberig_core::EntityId,
55    branch: Option<scriberig_core::EntityId>,
56) -> Vec<Effect> {
57    state.primary_view = crate::selection::PrimaryView::Session;
58    state.overlay = None;
59    let effects = if let Some(branch) = branch {
60        let request_id = state.next_request_id;
61        state.next_request_id += 1;
62        state.session = LoadState::Loading { request_id };
63        vec![Effect::ReadSession {
64            request_id,
65            session,
66            branch: Some(branch),
67        }]
68    } else {
69        state.session = LoadState::NotRequested;
70        vec![Effect::None]
71    };
72    state.transient_agent_outputs.clear();
73    state.session_composer = Some(crate::state::SessionComposerState {
74        session,
75        branch,
76        text: String::new(),
77        pending_prompt: None,
78        error: None,
79    });
80    state.session_composer_focused = true;
81    effects
82}

9/// Applies a current session read and reveals its pending branch target.
10///
11/// - `state`: TUI state receiving the session projection.
12/// - `request_id`: Request token returned with the projection.
13/// - `model`: Loaded session projection.
14///
15/// A no-op effect after applying or rejecting the response.
16/// # Parameters
17/// - `conversation`: Input for this operation.
18/// - `session_snapshot`: Input for this operation.
19/// # Returns
20/// The operation result.
21pub(super) fn session_succeeded(
22    state: &mut TuiState,
23    request_id: u64,
24    model: crate::view::session::SessionModel,
25    conversation: crate::view::conversation::ConversationModel,
26    session_snapshot: Option<scriberig_session::Session>,
27) -> Vec<Effect> {
28    if matches!(state.session, LoadState::Loading { request_id: current } if current == request_id)
29    {
30        if let Some(branch) = state.session_branch_focus.take()
31            && let Some(index) = model.rows.iter().position(|row| {
32                row.reference == branch && row.kind == crate::view::session::SessionRowKind::Branch
33            })
34        {
35            state.cursor = index;
36        }
37        state.session = LoadState::Ready(model);
38        state.session_snapshot = session_snapshot;
39        state.conversation = Some(state.session_snapshot.as_ref().map_or_else(
40            || conversation,
41            |session| crate::state::conversation_projection(state, session),
42        ));
43        if state.follow_coordinator {
44            state.coordinator_cursor = state
45                .conversation
46                .as_ref()
47                .and_then(|model| model.coordinator_cursor);
48        }
49        if let Some(source) = state.conversation_cursor
50            && let Some(index) = state
51                .conversation
52                .as_ref()
53                .and_then(|model| model.index_of(source))
54        {
55            state.cursor = index;
56        } else if let Some(row) = state
57            .conversation
58            .as_ref()
59            .and_then(|model| model.rows.first())
60        {
61            state.conversation_cursor = Some(row.source);
62            state.cursor = 0;
63        }
64        if state.primary_view == crate::selection::PrimaryView::Changes {
65            return super::state_actions::review_latest_candidate(state);
66        }
67    }
68    vec![Effect::None]
69}

71/// Applies a shared plan response to the cached durable session projection.
72///
73/// - `state`: TUI state receiving the authoritative plan.
74/// - `plan`: Persisted plan returned by the shared application service.
75///
76/// A no-op effect after rebuilding the conversation projection.
77/// # Returns
78/// The operation result.
79pub(super) fn plan_updated(state: &mut TuiState, mut plan: scriberig_session::Plan) -> Vec<Effect> {
80    let Some(mut session) = state.session_snapshot.clone() else {
81        state.conversation_composer.error = Some("Load a session before editing plans.".to_owned());
82        return vec![Effect::None];
83    };
84    plan.session = Some(session.id);
85    if let Some(existing) = session.plans.iter_mut().find(|item| item.id == plan.id) {
86        *existing = plan.clone();
87    } else {
88        session.plans.push(plan);
89    }
90    state.conversation = Some(crate::state::conversation_projection(state, &session));
91    state.coordinator_cursor = state
92        .conversation
93        .as_ref()
94        .and_then(|model| model.coordinator_cursor);
95    state.session_snapshot = Some(session);
96    vec![Effect::None]
97}

99/// Applies accepted conversation work using only returned durable handles.
100///
101/// - `state`: TUI presentation receiving the accepted operation.
102/// - `handle`: Application allocated operation, session, and branch.
103///
104/// A durable session refresh effect.
105/// # Returns
106/// The operation result.
107pub(super) fn conversation_agent_accepted(
108    state: &mut TuiState,
109    handle: &crate::service::AgentHandle,
110) -> Vec<Effect> {
111    let Some(branch) = handle.branch else {
112        state.conversation_composer.pending = false;
113        state.conversation_composer.error = Some("accepted work has no branch handle".to_owned());
114        return vec![Effect::None];
115    };
116    state.conversation_composer.target = crate::state::ConversationTarget::Existing {
117        session: handle.session,
118        reply_anchor: scriberig_session::ContinuationAnchor::BranchHead { branch },
119    };
120    state.conversation_composer.pending = false;
121    state.conversation_composer.draft.clear();
122    state.conversation_composer.error = handle
123        .failure
124        .as_ref()
125        .map(|failure| failure.message.clone());
126    state.conversation_browsing = false;
127    state.conversation_cursor = Some(crate::view::conversation::ConversationSource::Branch(
128        branch,
129    ));
130    state.agent_handles.push(handle.clone());
131    state.session_branch_focus = Some(branch);
132    let request_id = state.next_request_id;
133    state.next_request_id = state.next_request_id.saturating_add(1);
134    state.session = LoadState::Loading { request_id };
135    vec![Effect::ReadSession {
136        request_id,
137        session: handle.session,
138        branch: None,
139    }]
140}

142/// Pins the selected replyable source without changing coordinator ownership.
143///
144/// - `state`: TUI state with a semantic navigation cursor.
145///
146/// A no-op effect after updating local reply presentation state.
147/// # Returns
148/// The operation result.
149pub(super) fn conversation_pin_cursor(state: &mut TuiState) -> Vec<Effect> {
150    let selected = state
151        .conversation
152        .as_ref()
153        .and_then(|model| model.rows.get(state.cursor));
154    let Some((session, anchor)) = selected.and_then(|row| {
155        let session =
156            state
157                .conversation
158                .as_ref()?
159                .rows
160                .iter()
161                .find_map(|item| match item.source {
162                    crate::view::conversation::ConversationSource::Session(session) => {
163                        Some(session)
164                    }
165                    _ => None,
166                })?;
167        Some((session, row.reply_anchor?))
168    }) else {
169        state.shell_message = "This conversation row cannot receive a reply.".to_owned();
170        return vec![Effect::None];
171    };
172    state.conversation_composer.target = crate::state::ConversationTarget::Existing {
173        session,
174        reply_anchor: anchor,
175    };
176    state.follow_coordinator = false;
177    state.conversation_browsing = false;
178    state.shell_message = "Reply target pinned to the selected conversation node.".to_owned();
179    vec![Effect::None]
180}

182/// Toggles local child-history expansion for the selected stable branch.
183///
184/// - `state`: TUI state with a selected conversation branch.
185///
186/// A no-op effect after rebuilding the presentation projection.
187/// # Returns
188/// The operation result.
189pub(super) fn conversation_toggle_expansion(state: &mut TuiState) -> Vec<Effect> {
190    let Some(crate::view::conversation::ConversationSource::Branch(branch)) =
191        state.conversation_cursor
192    else {
193        return vec![Effect::None];
194    };
195    if !state.expanded_branches.remove(&branch) {
196        state.expanded_branches.insert(branch);
197    }
198    if let Some(session) = &state.session_snapshot {
199        state.conversation = Some(crate::state::conversation_projection(state, session));
200    }
201    if let Some(source) = state.conversation_cursor
202        && let Some(index) = state
203            .conversation
204            .as_ref()
205            .and_then(|model| model.index_of(source))
206    {
207        state.cursor = index;
208    }
209    vec![Effect::None]
210}

212/// Applies a loaded model library to the pending model or thinking picker.
213///
214/// - `state`: TUI state receiving the model library.
215/// - `request_id`: Request token returned with the library.
216/// - `library`: Credential-safe model descriptors.
217///
218/// A no-op effect after applying or rejecting the response.
219/// # Returns
220/// The operation result.
221pub(super) fn model_library_loaded(
222    state: &mut TuiState,
223    request_id: u64,
224    library: &scriberig_protocol::ModelLibraryResponse,
225) -> Vec<Effect> {
226    if matches!(state.model_library, LoadState::Loading { request_id: current } if current == request_id)
227    {
228        state.model_library = LoadState::Ready(library.clone());
229        state.shell_message = "Model library loaded.".to_owned();
230        if state.thinking_picker_pending.is_some() {
231            state.thinking_picker_pending = None;
232            let _ = super::show_thinking_picker(state);
233        } else if state.plan_assignment_target.is_some() {
234            super::state_model_picker::show_plan_assignment_picker(state, library);
235        } else {
236            super::show_model_picker(state, library);
237        }
238    }
239    vec![Effect::None]
240}

242/// Applies an updated model and thinking selection to the session header.
243///
244/// - `state`: TUI state containing the active session.
245/// - `selection`: Authoritative session execution selection.
246///
247/// A no-op effect after updating the local projection.
248/// # Returns
249/// The operation result.
250pub(super) fn session_selection_updated(
251    state: &mut TuiState,
252    selection: scriberig_session::SessionExecutionSelection,
253) -> Vec<Effect> {
254    if let Some(session) = state.session_snapshot.as_mut() {
255        session.execution = selection.clone();
256    }
257    if let LoadState::Ready(session) = &mut state.session {
258        let previous_model = session.execution.model.clone();
259        let next_model = selection.model.clone();
260        let previous_thinking = session.execution.thinking_level.clone();
261        let next_thinking = selection.thinking_level.clone();
262        session.set_execution(selection);
263        if previous_model != next_model && previous_thinking != next_thinking {
264            state.shell_message = format!(
265                "Model changed to {}; thinking level adjusted to {}.",
266                next_model.as_deref().unwrap_or("unspecified"),
267                next_thinking.as_deref().unwrap_or("unspecified"),
268            );
269        } else if previous_thinking != next_thinking {
270            state.shell_message = match (previous_thinking, next_thinking) {
271                (Some(previous), Some(next)) => {
272                    format!("Thinking level changed from {previous} to {next}.")
273                }
274                (None, Some(next)) => format!("Thinking level set to {next}."),
275                (Some(previous), None) => {
276                    format!("Thinking level {previous} is no longer selected.")
277                }
278                (None, None) => "Thinking level selection cleared.".to_owned(),
279            };
280        } else {
281            state.shell_message = "Future model selection updated.".to_owned();
282        }
283    } else {
284        state.shell_message = "Future model selection updated.".to_owned();
285    }
286    vec![Effect::None]
287}

289/// Retains one sanitized live output observation for the session view.
290///
291/// - `state`: TUI state retaining bounded transient output.
292/// - `output`: Provider-neutral output observation.
293///
294/// A no-op effect after updating the transient observation list.
295/// # Returns
296/// The operation result.
297pub(super) fn agent_output_observed(state: &mut TuiState, output: &str) -> Vec<Effect> {
298    state
299        .agent_output_observations
300        .push(crate::inspector::sanitize(output));
301    if state.agent_output_observations.len() > 32 {
302        state.agent_output_observations.remove(0);
303    }
304    state.shell_message = "agent output observed (non-canonical)".to_owned();
305    vec![Effect::None]
306}

308/// Starts an authoritative read for one selected session.
309///
310/// - `state`: TUI state receiving the session read.
311/// - `session`: Stable session identity.
312/// - `branch`: Optional branch scope for the read.
313///
314/// A session-read effect for an existing branch, or no effect for a new session.
315/// # Returns
316/// The operation result.
317pub(super) fn session_requested(
318    state: &mut TuiState,
319    session: scriberig_core::EntityId,
320    branch: Option<scriberig_core::EntityId>,
321) -> Vec<Effect> {
322    let request_id = state.next_request_id;
323    state.next_request_id += 1;
324    state.agent_output_observations.clear();
325    state.session = LoadState::Loading { request_id };
326    vec![Effect::ReadSession {
327        request_id,
328        session,
329        branch,
330    }]
331}

356/// Opens and focuses the persistent session composer while refreshing rows.
357///
358/// - `state`: TUI state receiving the persistent composer.
359/// - `session`: Stable session identity.
360/// - `branch`: Optional branch receiving the submission.
361///
362/// A session-read effect.
363/// # Returns
364/// The operation result.
365pub(super) fn session_composer_opened(
366    state: &mut TuiState,
367    session: scriberig_core::EntityId,
368    branch: Option<scriberig_core::EntityId>,
369) -> Vec<Effect> {
370    state.primary_view = crate::selection::PrimaryView::Session;
371    state.overlay = None;
372    let effects = if let Some(branch) = branch {
373        let request_id = state.next_request_id;
374        state.next_request_id += 1;
375        state.session = LoadState::Loading { request_id };
376        vec![Effect::ReadSession {
377            request_id,
378            session,
379            branch: Some(branch),
380        }]
381    } else {
382        state.session = LoadState::NotRequested;
383        vec![Effect::None]
384    };
385    state.agent_output_observations.clear();
386    state.session_composer = Some(crate::state::SessionComposerState {
387        session,
388        branch,
389        text: String::new(),
390        pending_prompt: None,
391        error: None,
392    });
393    state.session_composer_focused = true;
394    effects
395}

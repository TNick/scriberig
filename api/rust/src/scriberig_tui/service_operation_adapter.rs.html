7/// Optional local project labels keyed by stable entity identity.
8pub type ExplorerLabels = std::collections::HashMap<EntityId, String>;
9
10/// Adapts the application backend to operation-oriented TUI services.
11pub struct ApplicationOperationAdapter<B> {
12    pub(crate) service: ApplicationService<B>,
13    pub(crate) explorer_labels: ExplorerLabels,
14    pub(crate) project_projection: Option<ProjectProjection>,
15    /// Shared durable project workflow available even before project selection.
16    pub(crate) project_workflow: Option<scriberig_project::ProjectService>,
17    pub(crate) project_candidates: std::sync::Arc<
18        std::sync::Mutex<std::collections::HashMap<String, scriberig_project::ProjectCandidate>>,
19    >,
20    pub(crate) agent_runner: std::sync::Arc<dyn super::AgentCommandRunner<B>>,
21}
22
23/// Adapts the public protocol client to the TUI inspection boundary.
24impl<B: ApplicationBackend> ApplicationOperationAdapter<B> {
25    /// Creates an operation adapter around an application service.
26    ///
27    /// # Parameters
28    /// - `service`: Application service used by the adapter.
29    ///
30    /// # Returns
31    /// An adapter with no project-specific explorer labels.
32    #[must_use]
33    pub fn new(service: ApplicationService<B>) -> Self {
34        Self {
35            service,
36            explorer_labels: std::collections::HashMap::new(),
37            project_projection: None,
38            project_workflow: None,
39            project_candidates: std::sync::Arc::new(std::sync::Mutex::new(
40                std::collections::HashMap::new(),
41            )),
42            agent_runner: std::sync::Arc::new(super::SharedAgentCommandRunner),
43        }
44    }
45
46    /// Supplies the runtime that executes agent commands for this TUI client.
47    ///
48    /// # Parameters
49    /// - `self`: Adapter receiving the runtime.
50    /// - `runner`: Product-composed provider and orchestration adapter.
51    ///
52    /// # Returns
53    /// The configured application operation adapter.
54    #[must_use]
55    pub fn with_agent_runner<R>(mut self, runner: R) -> Self
56    where
57        R: super::AgentCommandRunner<B> + 'static,
58    {
59        self.agent_runner = std::sync::Arc::new(runner);
60        self
61    }
62
63    /// Supplies project-relative labels for local explorer navigation.
64    ///
65    /// # Parameters
66    /// - `self`: Adapter receiving the project labels.
67    /// - `labels`: Human-readable labels keyed by stable entity identity.
68    ///
69    /// # Returns
70    /// The adapter configured with the supplied labels.
71    #[must_use]
72    pub fn with_explorer_labels(mut self, labels: ExplorerLabels) -> Self {
73        self.explorer_labels = labels;
74        self
75    }
76
77    /// Supplies the active project's exact source projection data.
78    ///
79    /// # Parameters
80    /// - `self`: Adapter receiving project projection data.
81    /// - `projection`: Snapshot-bound source projection.
82    ///
83    /// # Returns
84    /// The configured adapter.
85    #[must_use]
86    pub fn with_project_projection(mut self, projection: ProjectProjection) -> Self {
87        self.project_workflow = Some(projection.workflow.clone());
88        self.project_projection = Some(projection);
89        self
90    }
91
92    /// Supplies durable project operations for candidates created after onboarding.
93    ///
94    /// # Parameters
95    /// - `self`: Adapter receiving the shared project workflow.
96    /// - `workflow`: Project service using the product's durable store.
97    ///
98    /// # Returns
99    /// The configured application operation adapter.
100    #[must_use]
101    pub fn with_project_workflow(mut self, workflow: scriberig_project::ProjectService) -> Self {
102        self.project_workflow = Some(workflow);
103        self
104    }
105}

107impl<B: ApplicationBackend> TuiService for ApplicationOperationAdapter<B> {
108    /// Submits local agent work through the runtime composed by the product.
109    ///
110    /// # Parameters
111    /// - `self`: Adapter with the injected shared runtime boundary.
112    /// - `context`: Application context for the operation.
113    /// - `session`: Stable session identity.
114    /// - `branch`: Optional continuation branch.
115    /// - `prompt`: User-authored instruction.
116    ///
117    /// # Returns
118    /// Stable operation, session, and branch handles.
119    ///
120    /// # Errors
121    /// Returns an application submission or session-read error.
122    fn submit_agent_handle(
123        &self,
124        context: &ApplicationContext,
125        session: EntityId,
126        branch: Option<EntityId>,
127        prompt: &str,
128    ) -> Result<super::AgentHandle, ApplicationError> {
129        self.agent_runner
130            .submit(&self.service, context, session, branch, prompt)
131    }
132
133    /// Reports the semantic operations exposed by the embedded application.
134    ///
135    /// # Parameters
136    /// - `self`: Embedded application adapter.
137    /// - `_context`: Authorization context used by the surrounding runtime.
138    ///
139    /// # Returns
140    /// Capability descriptors used to keep the command palette honest.
141    ///
142    /// # Errors
143    /// This adapter has no separate capability-discovery failure path.
144    fn runtime_capabilities(
145        &self,
146        _context: &ApplicationContext,
147    ) -> Result<Vec<RuntimeCapability>, ApplicationError> {
148        service_operation_helpers::runtime_capabilities(self)
149    }
150
151    /// Runs the deterministic provider through the selected semantic scope.
152    ///
153    /// # Parameters
154    /// - `context`: Trusted application context.
155    /// - `entity`: Selected semantic entity.
156    /// - `snapshot`: Exact candidate base snapshot.
157    /// - `prompt`: User instruction.
158    ///
159    /// # Returns
160    /// The normal candidate review operation result.
161    ///
162    /// # Errors
163    /// Returns a structured agent, validation, or storage error.
164    fn project_agent(
165        &self,
166        context: &ApplicationContext,
167        entity: EntityId,
168        snapshot: SnapshotId,
169        prompt: &str,
170    ) -> Result<ApplicationResult, ApplicationError> {
171        service_operation_agent::project_agent(self, context, entity, snapshot, prompt)
172    }
173
174    /// Delegates inspection through the public application facade.
175    ///
176    /// # Parameters
177    /// - `context`: Application context for the request.
178    /// - `entity`: Entity to inspect.
179    /// - `snapshot`: Optional exact snapshot.
180    ///
181    /// # Returns
182    /// The inspection result.
183    ///
184    /// # Errors
185    /// Returns the application inspection error.
186    fn inspect_entity(
187        &self,
188        context: &ApplicationContext,
189        entity: EntityId,
190        snapshot: Option<SnapshotId>,
191    ) -> Result<InspectResult, ApplicationError> {
192        service_operation_helpers::inspect_entity(self, context, entity, snapshot)
193    }
194
195    /// Lists the shared snapshot entity collection for Explorer navigation.
196    ///
197    /// # Parameters
198    /// - `context`: Application context.
199    /// - `entity`: Entity owning the collection.
200    /// - `snapshot`: Exact snapshot context.
201    /// - `continuation`: Optional continuation token.
202    ///
203    /// # Returns
204    /// A bounded Explorer page.
205    ///
206    /// # Errors
207    /// Returns the application listing error.
208    fn list_explorer(
209        &self,
210        context: &ApplicationContext,
211        entity: EntityId,
212        snapshot: SnapshotId,
213        continuation: Option<scriberig_query::ContinuationToken>,
214    ) -> Result<crate::explorer::ExplorerPage, ApplicationError> {
215        service_operation_helpers::list_explorer(self, context, entity, snapshot, continuation)
216    }
217
218    /// Executes exact-text discovery through the shared application boundary.
219    ///
220    /// # Parameters
221    /// - `context`: Application context.
222    /// - `snapshot`: Exact discovery scope.
223    /// - `query`: Search text.
224    ///
225    /// # Returns
226    /// A bounded find model.
227    ///
228    /// # Errors
229    /// Returns the application discovery error.

231        &self,
232        context: &ApplicationContext,
233        snapshot: SnapshotId,
234        query: &str,
235    ) -> Result<crate::view::find::FindModel, ApplicationError> {

237    }
238
239    /// Delegates shared operations through the public application facade.
240    ///
241    /// # Parameters
242    /// - `context`: Application context for the request.
243    /// - `request`: Public operation request.
244    ///
245    /// # Returns
246    /// The operation result.
247    ///
248    /// # Errors
249    /// Returns the application operation error.
250    fn execute_operation(
251        &self,
252        context: &ApplicationContext,
253        request: &ApplicationRequest,
254    ) -> Result<ApplicationResult, ApplicationError> {
255        if let Some(result) = service_operation_helpers::execute_operation(self, context, request)?
256        {
257            return Ok(result);
258        }
259        self.service.execute(context, request)
260    }

262    /// Reads an operation record from the shared application registry.
263    ///
264    /// # Parameters

267    ///
268    /// # Returns

270    ///
271    /// # Errors
272    /// Returns a lookup error when the operation is unknown.

274        &self,

287    /// # Returns

289    ///
290    /// # Errors
291    /// Returns a cancellation error when the operation is unknown.
292    fn cancel_operation(
293        &self,
294        _context: &ApplicationContext,
295        operation: &scriberig_core::OperationId,

300    /// Returns lifecycle events retained by the shared application registry.

303    /// - `_context`: Application context for event access.
304    ///

307    ///
308    /// # Errors

312        _context: &ApplicationContext,

315    }

317    /// Reads a session through the shared application boundary.

323    /// # Returns

329        &self,

333        service_operation_helpers::read_session(self, context, session)

336    /// Submits an agent command through the shared application boundary.
337    ///
338    /// # Parameters
339    /// - `context`: Application context for the request.

341    /// - `branch`: Optional branch identity.
342    /// - `prompt`: Agent prompt to submit.
343    ///
344    /// # Returns
345    /// The accepted operation identity.
346    ///
347    /// # Errors
348    /// Returns the application agent error.
349    fn submit_agent_command(
350        &self,
351        context: &ApplicationContext,
352        session: EntityId,
353        branch: Option<EntityId>,
354        prompt: &str,
355    ) -> Result<scriberig_core::OperationId, ApplicationError> {
356        service_operation_helpers::submit_agent_command(self, context, session, branch, prompt)

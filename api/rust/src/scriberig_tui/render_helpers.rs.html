7/// Returns the stable title for the selected auxiliary primary view.

42/// Builds a panel border with a visible focus state.
43///
44/// - `title`: Panel title displayed in the border.
45/// - `focused`: Whether this panel currently owns keyboard focus.
46/// - `super`: Parent-module visibility marker.
47///
48/// A styled panel block.
49/// # Returns
50/// The operation result.
51pub(super) fn panel_block(title: &str, focused: bool) -> Block<'_> {
52    let color = if focused {
53        Color::Cyan
54    } else {
55        Color::DarkGray
56    };
57    let title = if focused {
58        format!("{title}[ACTIVE]")
59    } else {
60        title.to_owned()
61    };
62    Block::default()
63        .title(title)
64        .borders(Borders::ALL)
65        .border_style(Style::default().fg(color))
66}

68/// Resolves one branch-depth role through the shared TUI theme facade.
69///
70/// - `depth`: Zero-based branch ancestry depth.
71///
72/// A finite palette color that cycles for deeper branches.
73/// # Returns
74/// The Ratatui color for the ancestry level.
75pub(super) fn depth_color(depth: usize) -> Color {
76    match scriberig_tui_kit::style_for(scriberig_tui_kit::Tone::BranchDepth(depth)).foreground {
77        scriberig_tui_kit::PaletteColor::Default => Color::Reset,
78        scriberig_tui_kit::PaletteColor::Ansi256(index) => Color::Indexed(index),
79    }
80}

82/// Formats inspector loading, error, and ready states with exact identifiers.
83///
84/// - `state`: Cached presentation state.
85/// - `super`: Parent-module visibility marker.
86///
87/// Text representing inspector state.
88/// # Returns
89/// The operation result.
90pub(super) fn inspector_text(state: &TuiState) -> String {
91    match &state.inspector {
92        crate::inspector::LoadState::NotRequested => {
93            "Shell ready\n\nEnter an entity with : or /.".to_owned()
94        }
95        crate::inspector::LoadState::Loading { request_id } => {
96            format!("Inspecting entity (request {request_id})...")
97        }
98        crate::inspector::LoadState::Failed {
99            category,
100            message,
101            stale,
102        } => format!(
103            "Error [{category}]{}\n{message}",
104            if *stale { " (stale data retained)" } else { "" }
105        ),
106        crate::inspector::LoadState::Ready(model) => format!(
107            "Entity: {}\nRevision: {}\nSnapshot: {}\nKind: {}\nLabel: {}\n{}",
108            model.entity,
109            model
110                .revision
111                .map_or_else(|| "none".to_owned(), |value| value.to_string()),
112            model
113                .snapshot
114                .map_or_else(|| "none".to_owned(), |value| value.to_string()),
115            model.kind.as_deref().unwrap_or("unknown"),
116            model.label.as_deref().unwrap_or("none"),
117            model.description.as_deref().unwrap_or("No description"),
118        ),
119    }
120}

122/// Renders the runtime status line.
123///
124/// - `frame`: Terminal frame to draw.
125/// - `area`: Status-line rectangle.
126/// - `state`: Cached presentation state.
127/// - `super`: Parent-module visibility marker.
128pub(super) fn render_status(frame: &mut Frame<'_>, area: Rect, state: &TuiState) {
129    let hints = keymap::hints(state)
130        .into_iter()
131        .map(|(key, label)| format!("{key} {label}"))
132        .collect::<Vec<_>>()
133        .join(" | ");
134    frame.render_widget(Paragraph::new(format!(" {hints} ")), area);
135}

3use crate::state::{Focus, TuiState};

5/// Formats the bounded semantic explorer model.
6///
7/// - `state`: Cached presentation state.
8/// - `super`: Parent module visibility marker.
9///
10/// Text representing the explorer pane.
11/// # Returns
12/// The operation result.
13pub(super) fn explorer_text(state: &TuiState) -> String {
14    let Some(root) = state.explorer.root.as_ref() else {
15        return "Explorer\n  No semantic root entered".to_owned();
16    };
17    let mut rows = vec![display_row(
18        root,
19        &if state.cursor == 0 && state.focus == Focus::Explorer {
20            "> ".to_owned()
21        } else {
22            "  ".to_owned()
23        },
24    )];
25    rows.extend(state.explorer.rows.iter().enumerate().map(|(index, row)| {
26        let cursor = index + 1;
27        display_row(
28            row,
29            &if state.cursor == cursor && state.focus == Focus::Explorer {
30                "> ".to_owned()
31            } else {
32                "  ".to_owned()
33            },
34        )
35    }));
36    if state.explorer.next.is_some() {
37        rows.push("  … more entities; press Enter on the root".to_owned());
38    }
39    format!("Explorer\n{}", rows.join("\n"))
40}

42/// Formats one explorer row with its readable label and semantic kind.
43///
44/// - `row`: Explorer row to format.
45/// - `prefix`: Selection marker and indentation prefix.
46///
47/// A compact human-readable explorer line.
48/// # Returns
49/// The operation result.
50fn display_row(row: &crate::explorer::ExplorerRow, prefix: &str) -> String {
51    let kind = row
52        .kind
53        .as_deref()
54        .map(|value| format!(" [{value}]"))
55        .unwrap_or_default();
56    format!("{prefix}{}{kind}", row.label)
57}

59/// Computes the explorer text offset needed to keep the cursor in view.
60///
61/// - `state`: Current TUI presentation state.
62/// - `height`: Outer explorer pane height.
63/// - `super`: Parent module visibility marker.
64///
65/// The number of text rows to skip from the top of the explorer.
66/// # Returns
67/// The operation result.
68pub(super) fn explorer_scroll(state: &TuiState, height: u16) -> u16 {
69    let visible = height.saturating_sub(2).max(1);
70    let cursor_line = state.cursor.saturating_add(1);
71    cursor_line
72        .saturating_sub(visible as usize - 1)
73        .try_into()
74        .unwrap_or(u16::MAX)
75}

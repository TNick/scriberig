1//! Converts typed remote protocol events to shared TUI application events.
2
3use super::parse_entity;
4
5/// Converts one protocol event into the shared application event model.
6///
7/// - `event`: Typed remote event envelope.
8///
9/// The equivalent application event when required fields are valid.
10/// # Returns
11/// The operation result.
12pub(crate) fn map_remote_event(
13    event: &scriberig_protocol::Event,
14) -> Option<scriberig_app::ApplicationEvent> {
15    if event.kind == scriberig_protocol::EventKind::SessionChanged {
16        let session = parse_entity(&event.entity.as_ref()?.id).ok()?;
17        return Some(scriberig_app::ApplicationEvent::SessionChanged { session });
18    }
19    if event.kind == scriberig_protocol::EventKind::SessionSelectionChanged {
20        let session = parse_entity(&event.entity.as_ref()?.id).ok()?;
21        return Some(scriberig_app::ApplicationEvent::SessionSelectionChanged {
22            session,
23            activity_kind: *event.activity_kind.as_ref()?,
24            summary: metadata_value(event, "summary")?,
25        });
26    }
27    if event.kind == scriberig_protocol::EventKind::PlanActivity {
28        let plan = parse_entity(&event.entity.as_ref()?.id).ok()?;
29        let session = parse_entity(&metadata_value(event, "session")?).ok()?;
30        let node = metadata_value(event, "node").and_then(|value| parse_entity(&value).ok());
31        return Some(scriberig_app::ApplicationEvent::PlanActivity {
32            session,
33            plan,
34            node,
35            activity_kind: *event.activity_kind.as_ref()?,
36            state: *event.activity_state.as_ref()?,
37            summary: metadata_value(event, "summary")?,
38        });
39    }
40    let operation = event.operation.as_ref()?.id.parse().ok()?;
41    match event.kind {
42        scriberig_protocol::EventKind::OperationStarted => {
43            Some(scriberig_app::ApplicationEvent::OperationStarted { operation })
44        }
45        scriberig_protocol::EventKind::OperationProgressed => {
46            let percent = metadata_value(event, "percent")?.parse::<u8>().ok()?;
47            let progress = scriberig_app::OperationProgress {
48                status: scriberig_app::OperationStatus::Running,
49                completed: Some(u64::from(percent)),
50                total: Some(100),
51                phase: None,
52            };
53            Some(scriberig_app::ApplicationEvent::OperationProgressed {
54                operation,
55                progress,
56            })
57        }
58        scriberig_protocol::EventKind::OperationCompleted => {
59            Some(scriberig_app::ApplicationEvent::OperationCompleted { operation })
60        }
61        scriberig_protocol::EventKind::OperationFailed => {
62            let error =
63                metadata_value(event, "error").unwrap_or_else(|| "operation failed".to_owned());
64            Some(scriberig_app::ApplicationEvent::OperationFailed { operation, error })
65        }
66        scriberig_protocol::EventKind::OperationCancelled => {
67            Some(scriberig_app::ApplicationEvent::OperationCancelled { operation })
68        }
69        scriberig_protocol::EventKind::AgentOutputProduced => {
70            let output = metadata_value(event, "output")?;
71            Some(scriberig_app::ApplicationEvent::AgentOutputProduced { operation, output })
72        }
73        scriberig_protocol::EventKind::AgentOutputDelta => {
74            let text = metadata_value(event, "text")?;
75            Some(scriberig_app::ApplicationEvent::AgentOutputDelta { operation, text })
76        }
77        scriberig_protocol::EventKind::AgentActivity => {
78            let branch = metadata_value(event, "branch")?;
79            let summary = metadata_value(event, "summary")?;
80            let ordinal = metadata_value(event, "ordinal")
81                .and_then(|value| value.parse().ok())
82                .unwrap_or_default();
83            Some(scriberig_app::ApplicationEvent::AgentActivity {
84                operation,
85                branch,
86                activity_kind: *event.activity_kind.as_ref()?,
87                state: *event.activity_state.as_ref()?,
88                summary,
89                ordinal,
90            })
91        }
92        _ => None,
93    }
94}

96/// Reads one string value from safe event metadata.
97///
98/// - `event`: Protocol event containing the metadata.
99/// - `key`: Stable metadata key.
100///
101/// The metadata value when present.
102/// # Returns
103/// The operation result.
104fn metadata_value(event: &scriberig_protocol::Event, key: &str) -> Option<String> {
105    event
106        .metadata
107        .iter()
108        .find(|(name, _)| name == key)
109        .map(|(_, value)| value.clone())
110}

10/// Executes an accepted agent request through the composed product runtime.
11pub trait AgentCommandRunner<B: ApplicationBackend>: Send + Sync {
12    /// Applies a plan lifecycle transition and advances it when supported.
13    ///
14    /// Product runners may run the shared durable coordinator after accepting
15    /// a start or resume transition. Provider-free adapters only persist the
16    /// lifecycle change.
17    ///
18    /// - `self`: Runtime adapter selected by the product composition.
19    /// - `service`: Shared application service owning plan state.
20    /// - `context`: Trusted caller capabilities and scope.
21    /// - `request`: Plan identity and observed revision.
22    /// - `action`: Requested plan transition.
23    ///
24    /// The latest authoritative plan after lifecycle processing.
25    ///
26    /// Returns an application or worker execution failure.
27    /// # Returns
28    /// The authoritative plan.
29    /// # Errors
30    /// Returns an error when the transition or execution fails.
31    fn plan_lifecycle(
32        &self,
33        service: &ApplicationService<B>,
34        context: &ApplicationContext,
35        request: &scriberig_app::PlanLifecycleRequest,
36        action: scriberig_app::PlanLifecycleAction,
37    ) -> Result<scriberig_session::Plan, ApplicationError>
38    where
39        ApplicationService<B>: SessionBackend,
40    {
41        scriberig_app::SessionBackend::plan_lifecycle(service, context, request, action)
42    }
43
44    /// Submits agent work and returns its durable application handles.
45    ///
46    /// - `self`: Runtime adapter selected by the product composition.
47    /// - `service`: Shared application service owning agent/session state.
48    /// - `context`: Trusted caller capabilities and semantic scope.
49    /// - `session`: Session receiving the user instruction.
50    /// - `branch`: Optional branch receiving the instruction.
51    /// - `prompt`: User-authored instruction.
52    ///
53    /// Stable operation, session, and branch handles.
54    ///
55    /// Returns an application failure when submission or execution fails.
56    /// # Returns
57    /// The operation result.
58    /// # Errors
59    /// Returns an error when the operation fails.
60    fn submit(
61        &self,
62        service: &ApplicationService<B>,
63        context: &ApplicationContext,
64        session: EntityId,
65        branch: Option<EntityId>,
66        prompt: &str,
67    ) -> Result<AgentHandle, ApplicationError>;
68
69    /// Executes an operation whose session turn and identity are already durable.
70    ///
71    /// Product runners should continue the accepted operation without creating
72    /// another user turn. Provider-free adapters return the accepted handles.
73    ///
74    /// - `self`: Runtime adapter selected by the product composition.
75    /// - `service`: Shared application service owning agent/session state.
76    /// - `context`: Trusted caller capabilities and semantic scope.
77    /// - `handle`: Durable operation, branch, and original user prompt.
78    ///
79    /// Stable operation, session, and branch handles after execution.
80    ///
81    /// Returns an application failure when execution fails.
82    /// # Parameters
83    /// - `_service`: Input for this operation.
84    /// - `_context`: Input for this operation.
85    /// # Returns
86    /// The operation result.
87    /// # Errors
88    /// Returns an error when the operation fails.
89    fn execute_accepted(
90        &self,
91        _service: &ApplicationService<B>,
92        _context: &ApplicationContext,
93        handle: &LocalAgentHandle,
94    ) -> Result<AgentHandle, ApplicationError> {
95        Ok(AgentHandle {
96            operation: handle.operation.clone(),
97            session: handle.session,
98            branch: Some(handle.root_branch),
99            workspace: None,
100        })
101    }
102}

104/// Uses the shared application session contract without adding a provider.
105#[derive(Clone, Copy, Debug, Default)]
106pub struct SharedAgentCommandRunner;
107
108impl<B: ApplicationBackend> AgentCommandRunner<B> for SharedAgentCommandRunner {
109    /// Accepts agent work through the shared session contract.
110    ///
111    /// - `self`: Provider-free fallback adapter.
112    /// - `service`: Shared application service owning agent/session state.
113    /// - `context`: Trusted caller capabilities and semantic scope.
114    /// - `session`: Session receiving the user instruction.
115    /// - `branch`: Optional branch receiving the instruction.
116    /// - `prompt`: User-authored instruction.
117    ///
118    /// Stable operation, session, and branch handles.
119    ///
120    /// Returns an application failure when the shared operation is rejected.
121    /// # Returns
122    /// The operation result.
123    /// # Errors
124    /// Returns an error when the operation fails.
125    fn submit(
126        &self,
127        service: &ApplicationService<B>,
128        context: &ApplicationContext,
129        session: EntityId,
130        branch: Option<EntityId>,
131        prompt: &str,
132    ) -> Result<AgentHandle, ApplicationError> {
133        let operation = SessionBackend::submit_agent_command(
134            service,
135            context,
136            &AgentCommandRequest {
137                session,
138                branch,
139                anchor: None,
140                prompt: prompt.to_owned(),
141                selection: None,
142            },
143        )?;
144        let branch = branch.or_else(|| {
145            SessionBackend::read_session(service, context, &SessionRequest { session })
146                .ok()
147                .and_then(|value| value.branches.last().map(|value| value.id))
148        });
149        Ok(AgentHandle {
150            operation,
151            session,
152            branch,
153            workspace: None,
154        })
155    }
156}

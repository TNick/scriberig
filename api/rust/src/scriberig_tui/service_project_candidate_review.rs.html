9/// Loads an exact-base candidate for the shared change review operation.
10///
11/// - `workflow`: Shared durable project workflow.
12/// - `request`: Candidate ID and exact base snapshot from the application request.
13/// - `super`: Parent TUI service module.
14///
15/// Candidate review data when the change handle is a durable project candidate.
16///
17/// Returns storage, stale-base, or candidate retrieval errors.
18/// # Returns
19/// The operation result.
20/// # Errors
21/// Returns an error when the operation fails.
22pub(super) fn review_candidate(
23    workflow: &ProjectService,
24    request: &scriberig_app::ApplyRequest,
25) -> Result<Option<ApplicationResult>, ApplicationError> {
26    let Some(candidate) =
27        workflow
28            .candidate(&request.change)
29            .map_err(|error| ApplicationError::OperationFailed {
30                operation: error.code.to_owned(),
31                message: error.message,
32            })?
33    else {
34        return Ok(None);
35    };
36    if candidate.base_snapshot != request.base_snapshot {
37        return Err(ApplicationError::StaleBaseSnapshot {
38            requested: request.base_snapshot,
39            current: candidate.base_snapshot,
40        });
41    }
42    let (validation, diagnostics) = match &candidate.validation {
43        scriberig_project::CandidateValidation::Pending => {
44            (scriberig_app::ValidationStatus::Skipped, Vec::new())
45        }
46        scriberig_project::CandidateValidation::Passed => {
47            (scriberig_app::ValidationStatus::Passed, Vec::new())
48        }
49        scriberig_project::CandidateValidation::Failed { diagnostics } => {
50            (scriberig_app::ValidationStatus::Failed, diagnostics.clone())
51        }
52    };
53    let mut review_details = vec![format!(
54        "scope: entity={} snapshot={}",
55        candidate.entity, candidate.base_snapshot
56    )];
57    review_details.extend(
58        candidate
59            .effects
60            .iter()
61            .map(|effect| {
62                format!(
63                    "{} {}\nold:\n{}\nnew:\n{}",
64                    effect_kind(effect.kind),
65                    effect.path,
66                    review_bytes(effect.old.as_deref()),
67                    review_bytes(effect.new.as_deref())
68                )
69            })
70            .collect::<Vec<_>>(),
71    );
72    Ok(Some(ApplicationResult::Apply(OperationSummary {
73        handle: candidate.id,
74        snapshot: Some(candidate.base_snapshot),
75        revision: None,
76        branch: None,
77        diff: None,
78        history: None,
79        rendered: None,
80        validation: Some(validation),
81        validation_profile: candidate.validation_profile.clone(),
82        validation_diagnostics: diagnostics,
83        review_details,
84        conflicts: Vec::new(),
85    })))
86}

88/// Applies one durable project candidate after explicit application confirmation.
89///
90/// - `workflow`: Shared project candidate service.
91/// - `request`: Candidate ID and exact base snapshot from the commit operation.
92/// - `super`: Parent TUI service module.
93///
94/// Canonical successor and materialization receipt in the shared commit result.
95///
96/// Returns missing, stale, validation, safety, or effect-recovery errors.
97/// # Returns
98/// The operation result.
99/// # Errors
100/// Returns an error when the operation fails.
101pub(super) fn apply_candidate(
102    workflow: &ProjectService,
103    request: &CommitRequest,
104) -> Result<ApplicationResult, ApplicationError> {
105    let candidate = workflow
106        .candidate(&request.change)
107        .map_err(|error| ApplicationError::OperationFailed {
108            operation: error.code.to_owned(),
109            message: error.message,
110        })?
111        .ok_or_else(|| ApplicationError::NotFound {
112            subject: request.change.clone(),
113        })?;
114    if candidate.base_snapshot != request.base_snapshot {
115        return Err(ApplicationError::StaleBaseSnapshot {
116            requested: request.base_snapshot,
117            current: candidate.base_snapshot,
118        });
119    }
120    let report = workflow.apply_candidate(&candidate).map_err(|error| {
121        ApplicationError::OperationFailed {
122            operation: error.code.to_owned(),
123            message: error.message,
124        }
125    })?;
126    Ok(ApplicationResult::Commit(OperationSummary {
127        handle: report.candidate,
128        snapshot: Some(report.successor_snapshot),
129        revision: None,
130        branch: None,
131        diff: None,
132        history: None,
133        rendered: None,
134        validation: Some(ValidationStatus::Passed),
135        validation_profile: candidate.validation_profile,
136        validation_diagnostics: Vec::new(),
137        review_details: vec![format!("materialized {}", report.path)],
138        conflicts: Vec::new(),
139    }))
140}

142/// Returns a bounded, readable preview for candidate source bytes.
143///
144/// - `bytes`: Optional exact source bytes.
145///
146/// UTF-8 preview or a stable marker for absent source data.
147/// # Returns
148/// The operation result.
149fn review_bytes(bytes: Option<&[u8]>) -> String {
150    bytes.map_or_else(
151        || "<absent>".to_owned(),
152        |bytes| {
153            let text = String::from_utf8_lossy(bytes);
154            let mut preview = text.chars().take(500).collect::<String>();
155            if text.chars().count() > 500 {
156                preview.push_str("\n… truncated …");
157            }
158            preview
159        },
160    )
161}

163/// Returns the stable source effect label used in review content.
164///
165/// - `kind`: Candidate source effect kind.
166///
167/// Lowercase effect label.
168/// # Returns
169/// The operation result.
170fn effect_kind(kind: scriberig_project::ProjectSourceEffectKind) -> &'static str {
171    match kind {
172        scriberig_project::ProjectSourceEffectKind::Create => "create",
173        scriberig_project::ProjectSourceEffectKind::Update => "update",
174        scriberig_project::ProjectSourceEffectKind::Delete => "delete",
175    }

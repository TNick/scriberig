3use super::{ApplicationError, EntityId};
4use scriberig_protocol::{SessionActivityItem, SessionActivityKind, SessionActivityState};
5use scriberig_session::{
6    Branch, CacheDisposition, ExecutionActivity, ModelInvocation, ModelOutput, ToolInvocation,
7    ToolResult,

10/// Restores safe activity references and records on a remote branch.
11///
12/// - `branch`: Branch being hydrated from the remote response.
13/// - `items`: Ordered safe activity projections from the server.
14///
15/// Unit after activity records and references have been restored.
16///
17/// Returns an internal error when a relationship contains a malformed entity.
18/// # Returns
19/// The operation result.
20/// # Errors
21/// Returns an error when the operation fails.
22pub(super) fn restore_activity(
23    branch: &mut Branch,
24    items: Vec<SessionActivityItem>,
25) -> Result<(), ApplicationError> {
26    for item in items {
27        restore_activity_item(branch, item)?;
28    }
29    Ok(())
30}
31
32/// Restores one safe activity record and its causal references.
33///
34/// - `branch`: Branch being hydrated from the remote response.
35/// - `item`: Safe activity projection from the server.
36///
37/// Unit after the activity item has been restored.
38///
39/// Returns an internal error when a relationship contains a malformed entity.
40/// # Returns
41/// The operation result.
42/// # Errors
43/// Returns an error when the operation fails.
44fn restore_activity_item(
45    branch: &mut Branch,
46    item: SessionActivityItem,
47) -> Result<(), ApplicationError> {
48    let id = activity_identity(&item)?;
49    restore_activity_record(branch, item, id)
50}
51
52/// Resolves the durable identity used to restore one activity item.
53///
54/// - `item`: Safe activity projection from the server.
55///
56/// Parsed activity identity using durable references where required.
57///
58/// Returns an internal error when the activity identity is malformed.
59/// # Returns
60/// The operation result.
61/// # Errors
62/// Returns an error when the operation fails.
63fn activity_identity(item: &SessionActivityItem) -> Result<EntityId, ApplicationError> {
64    let durable_id = item.durable_ref.as_deref().unwrap_or(&item.id);
65    match item.kind {
66        SessionActivityKind::HumanInput => parse_activity_entity(durable_id),
67        SessionActivityKind::AssistantOutput
68            if item
69                .parent
70                .as_deref()
71                .is_none_or(|parent| parent.ends_with(":human")) =>
72        {
73            parse_activity_entity(durable_id)
74        }
75        _ => parse_activity_entity(&item.id),
76    }
77}

79/// Adds the projected activity record and relationship to the branch.
80///
81/// - `branch`: Branch being hydrated from the remote response.
82/// - `item`: Safe activity projection from the server.
83/// - `id`: Parsed identity for this activity item.
84///
85/// Unit after the activity record has been restored.
86///
87/// Returns an internal error when a relationship contains a malformed entity.
88/// # Returns
89/// The operation result.
90/// # Errors
91/// Returns an error when the operation fails.
92fn restore_activity_record(
93    branch: &mut Branch,
94    item: SessionActivityItem,
95    id: EntityId,
96) -> Result<(), ApplicationError> {
97    match item.kind {
98        SessionActivityKind::HumanInput => {
99            branch.activity.push(ExecutionActivity::HumanInput(id));
100        }
101        SessionActivityKind::ModelInvocation => {
102            branch.model_invocations.push(ModelInvocation {
103                id,
104                branch: branch.id,
105                ordinal: item.sequence,
106                agent_profile: None,
107                context_fingerprint: None,
108                provider_profile: None,
109                model: item.model,
110                model_profile: None,
111                provider_model_id: None,
112                thinking_level: item.thinking_level,
113                request_fingerprint: None,
114                cache: CacheDisposition::NotUsed,
115                output: None,
116                request: None,
117            });
118            branch.activity.push(ExecutionActivity::ModelInvocation(id));
119        }
120        SessionActivityKind::AssistantOutput => {
121            if let Some(parent) = item.parent.filter(|parent| !parent.ends_with(":human")) {
122                let invocation = parse_activity_entity(&parent)?;
123                branch.model_outputs.push(ModelOutput {
124                    id,
125                    invocation,
126                    text: Some(item.summary),
127                    structured_output: None,
128                    finish_reason: None,
129                    usage: None,
130                });
131                if let Some(record) = branch
132                    .model_invocations
133                    .iter_mut()
134                    .find(|record| record.id == invocation)
135                {
136                    record.output = Some(id);
137                }
138                branch.activity.push(ExecutionActivity::ModelOutput(id));
139            } else {
140                branch.activity.push(ExecutionActivity::AgentOutput(id));
141            }
142        }
143        SessionActivityKind::ToolInvocation => {
144            let presentation_label = item.summary.clone();
145            let parent_model_invocation = item
146                .parent
147                .as_deref()
148                .map(parse_activity_entity)
149                .transpose()?;
150            branch.tool_invocations.push(ToolInvocation {
151                id,
152                name: item.tool_name.unwrap_or(item.summary),
153                input: Vec::new(),
154                parent_model_invocation,
155                presentation_input: item
156                    .tool_arguments
157                    .map(|arguments| serde_json::to_vec(&arguments).unwrap_or_default()),
158                presentation_label: Some(presentation_label),
159                category: item.tool_category,
160            });
161            branch.activity.push(ExecutionActivity::ToolInvocation(id));
162        }
163        SessionActivityKind::ToolResult => {
164            let invocation = item
165                .parent
166                .as_deref()
167                .map(parse_activity_entity)
168                .transpose()?
169                .ok_or_else(|| malformed_activity("tool result has no invocation parent"))?;
170            branch.tool_results.push(ToolResult {
171                id,
172                invocation,
173                output: Vec::new(),
174                failed: item.state == SessionActivityState::Failed,
175                presentation_preview: item
176                    .tool_result_preview
177                    .and_then(|preview| serde_json::to_vec(&preview).ok()),
178            });
179            branch.activity.push(ExecutionActivity::ToolResult(id));
180        }
181        SessionActivityKind::BranchStarted => {
182            if let Some(parent) = item.parent.as_deref() {
183                branch.activity.push(ExecutionActivity::ChildBranchSpawned {
184                    branch: id,
185                    parent_tool_invocation: parse_activity_entity(parent)?,
186                });
187            } else {
188                branch.activity.push(ExecutionActivity::ChildBranch(id));
189            }
190        }
191        SessionActivityKind::BranchCompleted => {
192            branch.activity.push(ExecutionActivity::BranchResult(id));
193        }
194        _ => {}
195    }
196    Ok(())
197}

199/// Parses a stable entity reference carried by activity.
200///
201/// - `value`: Remote activity identity.
202///
203/// Parsed entity identity.
204///
205/// Returns an internal error for malformed entity values.
206/// # Returns
207/// The operation result.
208/// # Errors
209/// Returns an error when the operation fails.
210fn parse_activity_entity(value: &str) -> Result<EntityId, ApplicationError> {
211    value.parse().map_err(|error| ApplicationError::Internal {
212        message: format!("remote activity contained invalid entity identity: {error}"),
213    })
214}

216/// Creates a sanitized malformed-activity error.
217///
218/// - `message`: Safe validation detail.
219///
220/// An internal application error describing the invalid activity.
221/// # Returns
222/// The operation result.
223fn malformed_activity(message: &str) -> ApplicationError {
224    ApplicationError::Internal {
225        message: message.to_owned(),
226    }
227}

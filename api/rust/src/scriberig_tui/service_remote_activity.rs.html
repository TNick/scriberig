10/// Restores safe activity references and records on a remote branch.
11///
12/// # Parameters
13/// - `branch`: Branch being hydrated from the remote response.
14/// - `items`: Ordered safe activity projections from the server.
15///
16/// # Returns
17/// Unit after activity records and references have been restored.
18///
19/// # Errors
20/// Returns an internal error when a relationship contains a malformed entity.
21pub(super) fn restore_activity(
22    branch: &mut Branch,
23    items: Vec<SessionActivityItem>,
24) -> Result<(), ApplicationError> {
25    for item in items {
26        restore_activity_item(branch, item)?;
27    }
28    Ok(())
29}

31/// Restores one safe activity record and its causal references.
32///
33/// # Parameters
34/// - `branch`: Branch being hydrated from the remote response.
35/// - `item`: Safe activity projection from the server.
36///
37/// # Returns
38/// Unit after the activity item has been restored.
39///
40/// # Errors
41/// Returns an internal error when a relationship contains a malformed entity.
42fn restore_activity_item(
43    branch: &mut Branch,
44    item: SessionActivityItem,
45) -> Result<(), ApplicationError> {
46    let id = activity_identity(&item)?;
47    restore_activity_record(branch, item, id)
48}

50/// Resolves the durable identity used to restore one activity item.
51///
52/// # Parameters
53/// - `item`: Safe activity projection from the server.
54///
55/// # Returns
56/// Parsed activity identity using durable references where required.
57///
58/// # Errors
59/// Returns an internal error when the activity identity is malformed.
60fn activity_identity(item: &SessionActivityItem) -> Result<EntityId, ApplicationError> {
61    let durable_id = item.durable_ref.as_deref().unwrap_or(&item.id);
62    match item.kind {
63        SessionActivityKind::HumanInput => parse_activity_entity(durable_id),
64        SessionActivityKind::AssistantOutput
65            if item
66                .parent
67                .as_deref()
68                .is_none_or(|parent| parent.ends_with(":human")) =>
69        {
70            parse_activity_entity(durable_id)
71        }
72        _ => parse_activity_entity(&item.id),
73    }
74}

76/// Adds the projected activity record and relationship to the branch.
77///
78/// # Parameters
79/// - `branch`: Branch being hydrated from the remote response.
80/// - `item`: Safe activity projection from the server.
81/// - `id`: Parsed identity for this activity item.
82///
83/// # Returns
84/// Unit after the activity record has been restored.
85///
86/// # Errors
87/// Returns an internal error when a relationship contains a malformed entity.
88fn restore_activity_record(
89    branch: &mut Branch,
90    item: SessionActivityItem,
91    id: EntityId,
92) -> Result<(), ApplicationError> {
93    match item.kind {
94        SessionActivityKind::HumanInput => {
95            branch.activity.push(ExecutionActivity::HumanInput(id));
96        }
97        SessionActivityKind::ModelInvocation => {
98            branch.model_invocations.push(ModelInvocation {
99                id,
100                branch: branch.id,
101                ordinal: item.sequence,
102                agent_profile: None,
103                context_fingerprint: None,
104                provider_profile: None,
105                model: item.model,
106                model_profile: None,
107                provider_model_id: None,
108                thinking_level: item.thinking_level,
109                request_fingerprint: None,
110                cache: CacheDisposition::NotUsed,
111                output: None,
112                request: None,
113            });
114            branch.activity.push(ExecutionActivity::ModelInvocation(id));
115        }
116        SessionActivityKind::AssistantOutput => {
117            if let Some(parent) = item.parent.filter(|parent| !parent.ends_with(":human")) {
118                let invocation = parse_activity_entity(&parent)?;
119                branch.model_outputs.push(ModelOutput {
120                    id,
121                    invocation,
122                    text: Some(item.summary),
123                    structured_output: None,
124                    finish_reason: None,
125                    usage: None,
126                });
127                if let Some(record) = branch
128                    .model_invocations
129                    .iter_mut()
130                    .find(|record| record.id == invocation)
131                {
132                    record.output = Some(id);
133                }
134                branch.activity.push(ExecutionActivity::ModelOutput(id));
135            } else {
136                branch.activity.push(ExecutionActivity::AgentOutput(id));
137            }
138        }
139        SessionActivityKind::ToolInvocation => {
140            let presentation_label = item.summary.clone();
141            let parent_model_invocation = item
142                .parent
143                .as_deref()
144                .map(parse_activity_entity)
145                .transpose()?;
146            branch.tool_invocations.push(ToolInvocation {
147                id,
148                name: item.tool_name.unwrap_or(item.summary),
149                input: Vec::new(),
150                parent_model_invocation,
151                presentation_input: item
152                    .tool_arguments
153                    .map(|arguments| serde_json::to_vec(&arguments).unwrap_or_default()),
154                presentation_label: Some(presentation_label),
155            });
156            branch.activity.push(ExecutionActivity::ToolInvocation(id));
157        }
158        SessionActivityKind::ToolResult => {
159            let invocation = item
160                .parent
161                .as_deref()
162                .map(parse_activity_entity)
163                .transpose()?
164                .ok_or_else(|| malformed_activity("tool result has no invocation parent"))?;
165            branch.tool_results.push(ToolResult {
166                id,
167                invocation,
168                output: Vec::new(),
169                failed: item.state == SessionActivityState::Failed,
170                presentation_preview: item
171                    .tool_result_preview
172                    .and_then(|preview| serde_json::to_vec(&preview).ok()),
173            });
174            branch.activity.push(ExecutionActivity::ToolResult(id));
175        }
176        SessionActivityKind::ChildBranch => {
177            if let Some(parent) = item.parent.as_deref() {
178                branch.activity.push(ExecutionActivity::ChildBranchSpawned {
179                    branch: id,
180                    parent_tool_invocation: parse_activity_entity(parent)?,
181                });
182            } else {
183                branch.activity.push(ExecutionActivity::ChildBranch(id));
184            }
185        }
186        SessionActivityKind::BranchResult => {
187            branch.activity.push(ExecutionActivity::BranchResult(id));
188        }
189    }
190    Ok(())
191}

193/// Parses a stable entity reference carried by activity.
194///
195/// # Parameters
196/// - `value`: Remote activity identity.
197///
198/// # Returns
199/// Parsed entity identity.
200///
201/// # Errors
202/// Returns an internal error for malformed entity values.
203fn parse_activity_entity(value: &str) -> Result<EntityId, ApplicationError> {
204    value.parse().map_err(|error| ApplicationError::Internal {
205        message: format!("remote activity contained invalid entity identity: {error}"),
206    })
207}

209/// Creates a sanitized malformed-activity error.
210///
211/// # Parameters
212/// - `message`: Safe validation detail.
213///
214/// # Returns
215/// An internal application error describing the invalid activity.
216fn malformed_activity(message: &str) -> ApplicationError {
217    ApplicationError::Internal {
218        message: message.to_owned(),
219    }
220}

329    fn shared_fixture_preserves_causal_activity_identity() {
330        let fixture: serde_json::Value = serde_json::from_str(include_str!(
331            "../../../tests/fixtures/agent_console_sensitive_activity.json"
332        ))
333        .expect("cross-surface fixture");
334        let items = serde_json::from_value(fixture["protocolActivity"].clone())
335            .expect("protocol activity items");
336        let branch_id = EntityId::from_bytes([0x52; 16]);
337        let mut branch = Branch::new(branch_id, "root").expect("branch");
338
339        restore_activity(&mut branch, items).expect("restore shared activity");
340
341        assert_eq!(
342            branch.model_invocations[0].id,
343            EntityId::from_bytes([0x53; 16])
344        );
345        assert_eq!(
346            branch.tool_invocations[0].id,
347            EntityId::from_bytes([0x54; 16])
348        );
349        assert_eq!(branch.tool_results[0].id, EntityId::from_bytes([0x55; 16]));
350        assert_eq!(
351            branch.tool_invocations[0].parent_model_invocation,
352            Some(branch.model_invocations[0].id)
353        );
354        assert_eq!(
355            branch.tool_results[0].invocation,
356            branch.tool_invocations[0].id
357        );
358        assert_eq!(
359            branch.tool_invocations[0].presentation_input.as_deref(),
360            Some(br#"{"query":"find documents","token":"[REDACTED]"}"#.as_slice())
361        );
362        let session_id = EntityId::from_bytes([0x51; 16]);
363        let session_model = SessionModel::from_session(&Session {
364            id: session_id,
365            branches: vec![branch],
366            goals: Vec::new(),
367            tasks: Vec::new(),
368            execution: scriberig_session::SessionExecutionSelection::default(),
369        });
370        let activity_ids = session_model
371            .rows
372            .iter()
373            .map(|row| row.reference)
374            .collect::<Vec<_>>();
375        let mut state = crate::state::TuiState {
376            primary_view: crate::selection::PrimaryView::Session,
377            session: crate::inspector::LoadState::Ready(session_model),
378            ..crate::state::TuiState::default()
379        };
380        let simple = crate::render::primary_text(&state);
381        state.tool_invocation_visibility = crate::state::ToolInvocationVisibility::Detailed;
382        let detailed = crate::render::primary_text(&state);
383
384        assert!(simple.contains("Find documents"));
385        assert!(detailed.contains("[REDACTED]"));
386        assert_eq!(
387            activity_ids,
388            match &state.session {
389                crate::inspector::LoadState::Ready(model) => model
390                    .rows
391                    .iter()
392                    .map(|row| row.reference)
393                    .collect::<Vec<_>>(),
394                _ => unreachable!(),
395            }
396        );
397    }

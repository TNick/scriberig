79impl TuiState {
80    /// Reports whether this TUI session overrides one shared preference.
81    ///
82    /// - `control`: Presentation control to inspect.
83    ///
84    /// Whether the control has a local override.
85    /// # Returns
86    /// The operation result.
87    pub(crate) fn has_presentation_override(&self, control: PresentationControl) -> bool {
88        self.presentation_overrides.contains(&control)
89    }
90
91    /// Records a local override for one presentation control.
92    ///
93    /// - `control`: Presentation control that was changed locally.
94    pub(crate) fn set_presentation_override(&mut self, control: PresentationControl) {
95        self.presentation_overrides.insert(control);
96    }
97
98    /// Applies one local terminal event to presentation state.
99    /// - `event`: Argument named `event`.
100    ///
101    /// # Returns
102    /// The operation result.
103    pub fn apply_event(&mut self, event: &TuiEvent) -> Vec<Effect> {
104        match event {
105            TuiEvent::Key(key) => {
106                if let Some(action) = crate::keymap::action_for(self, key) {
107                    return reduce(self, action);
108                }
109            }
110            TuiEvent::Resize { width, height } => {
111                return reduce(
112                    self,
113                    Action::TerminalResized {
114                        width: *width,
115                        height: *height,
116                    },
117                );
118            }
119            TuiEvent::Shutdown => return reduce(self, Action::QuitRequested),
120            TuiEvent::Paste(text) => {
121                return reduce(self, Action::Paste(text.clone()));
122            }
123            TuiEvent::Application(event) => return reduce_application_event(self, event),
124            TuiEvent::Tick => {}
125        }
126        Vec::new()
127    }
128
129    /// Reports whether a text input currently owns keyboard focus.
130    #[must_use]
131    ///
132    /// # Returns
133    /// The operation result.
134    pub fn is_input_active(&self) -> bool {
135        matches!(self.overlay, Some(Overlay::Input(_) | Overlay::Composer(_)))
136            || (self.session_composer.is_some() && self.session_composer_focused)
137    }
138
139    /// Returns the current input text, when an input overlay is active.
140    #[must_use]
141    ///
142    /// # Returns
143    /// The operation result.
144    pub fn input_text(&self) -> Option<&str> {
145        match self.overlay.as_ref() {
146            Some(Overlay::Input(input)) => Some(&input.text),
147            Some(Overlay::Composer(composer)) => Some(&composer.text),
148            None => self
149                .session_composer
150                .as_ref()
151                .map(|composer| composer.text.as_str()),
152            _ => None,
153        }
154    }
155}

157/// Projects durable conversation state with active observations attached to branches.
158///
159/// - `state`: TUI state containing presentation-only active activity.
160/// - `session`: Exact durable session snapshot.
161///
162/// A tree projection that places live status at stable branch locations.
163/// # Returns
164/// The operation result.
165pub(crate) fn conversation_projection(
166    state: &TuiState,
167    session: &scriberig_session::Session,
168) -> crate::view::conversation::ConversationModel {
169    let mut conversation = crate::view::conversation::ConversationModel::from_session_with_expanded(
170        session,
171        &state.expanded_branches,
172    );
173    for activity in &state.transient_agent_activities {
174        let Ok(branch) = activity.branch.parse::<scriberig_core::EntityId>() else {
175            continue;
176        };
177        let summary = if activity.activity_kind
178            == scriberig_session::AgentActivityKind::ReasoningSummary
179            && state.reasoning_visibility == ReasoningVisibility::Hidden
180        {
181            "summary hidden".to_owned()
182        } else {
183            crate::inspector::sanitize(&activity.summary)
184        };
185        conversation.insert_transient_activity(
186            branch,
187            activity.key,
188            format!("Live {:?}: {}", activity.activity_kind, summary),
189        );
190    }
191    conversation

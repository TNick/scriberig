14/// Renders conversation rows with independently styled ancestry and reply target.
15///
16/// - `state`: Cached conversation and navigation state.
17/// - `available_width`: Width available for the rows.
18///
19/// Styled conversation rows with stable source identities controlling selection.
20/// # Returns
21/// The operation result.
22pub(super) fn conversation_text(state: &TuiState, available_width: usize) -> Text<'static> {
23    let Some(model) = state.conversation.as_ref() else {
24        return Text::from("No conversation yet. Type a message to start a session.");
25    };
26    let reply_anchor = match state.conversation_composer.target {
27        ConversationTarget::Existing { reply_anchor, .. } => Some(reply_anchor),
28        ConversationTarget::NewSession => None,
29    };
30    let rail_capacity = (available_width.saturating_sub(8) / 2).max(1);
31    let lines = model
32        .rows
33        .iter()
34        .flat_map(|row| {
35            let selected = state.conversation_cursor == Some(row.source);
36            let replying = reply_anchor.is_some_and(|anchor| row.reply_anchor == Some(anchor));
37            let visible = row.depth.min(rail_capacity);
38            let first = row.depth.saturating_sub(visible);
39            let rail_text = if first > 0 { "…" } else { "" };
40            let rails = (first..row.depth)
41                .map(|depth| Span::styled("▏ ", Style::default().fg(depth_color(depth))))
42                .collect::<Vec<_>>();
43            let content_width = available_width
44                .saturating_sub(2 + usize::from(first > 0) + visible.saturating_mul(2))
45                .max(1);
46            let labels = if row.kind == crate::view::conversation::ConversationRowKind::Message {
47                render_markdown(&row.label, content_width)
48            } else {
49                vec![ExternalText::new(&row.label)]
50            };
51            let indentation = if row.depth > visible {
52                "  ".repeat(visible)
53            } else {
54                String::new()
55            };
56            labels
57                .into_iter()
58                .enumerate()
59                .map(move |(line_index, label)| {
60                    let mut spans = vec![Span::raw(if selected && line_index == 0 {
61                        "> "
62                    } else {
63                        "  "
64                    })];
65                    if line_index == 0 {
66                        if !rail_text.is_empty() {
67                            spans.push(Span::raw(rail_text));
68                        }
69                        spans.extend(rails.clone());
70                    } else {
71                        spans.push(Span::raw("  ".repeat(visible + usize::from(first > 0))));
72                    }
73                    spans.push(Span::styled(
74                        format!("{indentation}{}", label.as_str()),
75                        if replying {
76                            Style::default().bg(Color::DarkGray).fg(Color::White)
77                        } else {
78                            Style::default()
79                        },
80                    ));
81                    Line::from(spans)
82                })
83        })
84        .collect::<Vec<_>>();
85    Text::from(lines)
86}

108    /// Verifies depth colors cycle through the finite ancestry palette.
109    #[test]
110    fn depth_palette_cycles_after_all_colors() {
111        assert_eq!(depth_color(0), Color::Indexed(39));
112        assert_eq!(depth_color(1), Color::Indexed(40));
113        assert_eq!(depth_color(6), Color::Indexed(39));
114    }

116    /// Verifies narrow rows keep a local rail and mark omitted ancestors.
117    #[test]
118    fn narrow_rows_compact_ancestors_with_an_ellipsis() {
119        let state = TuiState {
120            conversation: Some(ConversationModel {
121                rows: vec![ConversationRow {
122                    source: ConversationSource::Branch(EntityId::from_bytes([1; 16])),
123                    reply_anchor: None,
124                    depth: 3,
125                    label: "nested branch".to_owned(),
126                    kind: ConversationRowKind::Branch,
127                }],
128                coordinator_cursor: None,
129            }),
130            ..TuiState::default()
131        };
132
133        let text = conversation_text(&state, 10);
134        let rendered = text.lines[0]
135            .spans
136            .iter()
137            .map(|span| span.content.as_ref())
138            .collect::<String>();
139
140        assert!(rendered.contains('…'));
141        assert!(rendered.contains("▏ "));
142        assert!(rendered.contains("nested branch"));
143    }

145    /// Verifies the reply target uses a background separate from rail colors.
146    #[test]
147    fn reply_target_has_background_highlight() {
148        use crate::state::ConversationTarget;
149        let branch = EntityId::from_bytes([2; 16]);
150        let turn = EntityId::from_bytes([3; 16]);
151        let anchor = scriberig_session::ContinuationAnchor::AssistantOutput { branch, turn };
152        let source = ConversationSource::Activity(turn);
153        let mut state = TuiState::default();
154        state.conversation_composer.target = ConversationTarget::Existing {
155            session: EntityId::from_bytes([4; 16]),
156            reply_anchor: anchor,
157        };
158        state.conversation = Some(ConversationModel {
159            rows: vec![ConversationRow {
160                source,
161                reply_anchor: Some(anchor),
162                depth: 1,
163                label: "reply target".to_owned(),
164                kind: ConversationRowKind::Message,
165            }],
166            coordinator_cursor: None,
167        });
168
169        let text: Text<'static> = conversation_text(&state, 80);
170        let label = text.lines[0]
171            .spans
172            .iter()
173            .find(|span| span.content == "reply target")
174            .expect("reply label");
175
176        assert_eq!(label.style.bg, Some(Color::DarkGray));
177    }

179    /// Verifies message Markdown remains structured safe text in the TUI.
180    #[test]
181    fn message_markdown_uses_safe_terminal_rendering() {
182        let state = TuiState {
183            conversation: Some(ConversationModel {
184                rows: vec![ConversationRow {
185                    source: ConversationSource::Activity(EntityId::from_bytes([8; 16])),
186                    reply_anchor: None,
187                    depth: 0,
188                    label: "# heading\n\nunsafe \u{1b}[2J".to_owned(),
189                    kind: ConversationRowKind::Message,
190                }],
191                coordinator_cursor: None,
192            }),
193            ..TuiState::default()
194        };
195
196        let rendered = conversation_text(&state, 40);
197        let plain = rendered
198            .lines
199            .iter()
200            .map(ToString::to_string)
201            .collect::<Vec<_>>()
202            .join("\n");
203
204        assert!(plain.contains("heading"));
205        assert!(plain.contains("\\u{1b}[2J"));
206        assert!(!plain.contains('\u{1b}'));
207    }

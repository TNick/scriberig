1//! Parses C and C++ source into complete, serializable syntax trees.
2//!
3//! Canonical C and C++ state is stored through Scriberig's shared universal
4//! model; this crate does not create an independent source of truth.

12/// A parsed C or C++ source document.
13#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
14pub struct CDocument {
15    /// Adapter identifier selected from the source path.
16    pub adapter: String,
17    /// Complete syntax tree rooted at the translation unit.
18    pub tree: CNode,
19    /// Every syntax node in deterministic depth-first order.
20    pub nodes: Vec<CNodeRecord>,
21}
22
23/// One serializable C or C++ syntax node.
24#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
25pub struct CNodeRecord {
26    /// Stable path from the root node to this node.
27    pub path: String,
28    /// Grammar node kind.
29    pub kind: String,
30    /// Whether this is a named grammar node.
31    pub named: bool,
32    /// Whether the parser synthesized this node.
33    pub missing: bool,
34    /// Inclusive UTF-8 byte start offset.
35    pub start_byte: usize,
36    /// Exclusive UTF-8 byte end offset.
37    pub end_byte: usize,
38    /// Zero-based start line.
39    pub start_line: usize,
40    /// Zero-based start column in bytes.
41    pub start_column: usize,
42    /// Zero-based end line.
43    pub end_line: usize,
44    /// Zero-based end column in bytes.
45    pub end_column: usize,
46    /// Grammar field containing this node, when present.
47    pub field: Option<String>,
48    /// Source text for leaf nodes.
49    pub text: Option<String>,
50}

55    /// Grammar node kind.
56    pub kind: String,
57    /// Whether this is a named grammar node.

59    /// Whether the parser synthesized this node.

81/// A C or C++ parser failure with source-location diagnostics.
82#[derive(Clone, Debug, Eq, PartialEq)]
83pub struct CError {
84    /// Parser diagnostics in source order.
85    pub diagnostics: Vec<String>,
86}
87
88impl Display for CError {
89    /// Formats the first C-family parser diagnostic.
90    ///
91    /// # Parameters
92    /// - `self`: Parser failure to format.
93    /// - `formatter`: Formatting destination.
94    ///
95    /// # Returns
96    /// A formatting result.
97    ///
98    /// # Errors
99    /// Returns a formatting error when the destination rejects output.
100    fn fmt(&self, formatter: &mut Formatter<'_>) -> std::fmt::Result {
101        formatter.write_str("C-family syntax error")?;
102        if let Some(message) = self.diagnostics.first() {
103            write!(formatter, ": {message}")?;
104        }
105        Ok(())
106    }
107}
108
109impl std::error::Error for CError {}

111/// Parses C or C++ source based on its file extension.
112///
113/// # Parameters
114/// - `path`: Source path used to select the grammar.
115/// - `source`: UTF-8 C or C++ source text.
116///
117/// # Returns
118/// A complete serializable syntax tree and its individual nodes.
119///
120/// # Errors
121/// Returns all syntax-error diagnostics when the source is incomplete or invalid.
122pub fn parse(path: &Path, source: &str) -> Result<CDocument, CError> {
123    let adapter = adapter_for(path).ok_or_else(|| CError {
124        diagnostics: vec![format!("unsupported C-family path: {}", path.display())],
125    })?;
126    let mut parser = Parser::new();
127    let language = if adapter == "cpp" {
128        tree_sitter_cpp::LANGUAGE.into()
129    } else {
130        tree_sitter_c::LANGUAGE.into()
131    };
132    parser.set_language(&language).map_err(|error| CError {
133        diagnostics: vec![format!("failed to initialize {adapter} parser: {error}")],
134    })?;
135    let tree = parser.parse(source, None).ok_or_else(|| CError {
136        diagnostics: vec![format!("{adapter} parser returned no syntax tree")],
137    })?;
138    let mut diagnostics = Vec::new();
139    collect_errors(tree.root_node(), source, &mut diagnostics);
140    if !diagnostics.is_empty() {
141        return Err(CError { diagnostics });
142    }
143    let root = build_node(tree.root_node(), source, None);
144    let mut nodes = Vec::new();
145    flatten_node(&root, "", &mut nodes);
146    Ok(CDocument {
147        adapter: adapter.to_owned(),
148        tree: root,
149        nodes,
150    })
151}

153/// Selects the C-family adapter identifier for a source path.
154///
155/// # Parameters
156/// - `path`: Source path to classify.
157///
158/// # Returns
159/// The C or C++ adapter identifier, or none for an unsupported extension.
160fn adapter_for(path: &Path) -> Option<&'static str> {
161    match path.extension().and_then(|value| value.to_str()) {
162        Some("c" | "h") => Some("c"),
163        Some("cc" | "cpp" | "cxx" | "hpp" | "hh" | "hxx") => Some("cpp"),
164        _ => None,
165    }
166}

168/// Builds the serializable tree representation for one parser node.
169///
170/// # Parameters
171/// - `node`: Tree-sitter node to serialize.
172/// - `source`: Original source text.
173/// - `field`: Optional parent grammar field name.
174///
175/// # Returns
176/// A serializable node with recursively ordered children.
177fn build_node(node: Node<'_>, source: &str, field: Option<String>) -> CNode {
178    let mut cursor = node.walk();
179    let mut children = Vec::new();
180    if cursor.goto_first_child() {
181        loop {
182            let child = cursor.node();
183            let child_field = cursor.field_name().map(str::to_owned);
184            children.push(build_node(child, source, child_field));
185            if !cursor.goto_next_sibling() {
186                break;
187            }
188        }
189    }
190    let text = if node.child_count() == 0 {
191        Some(source[node.byte_range()].to_owned())
192    } else {
193        None
194    };
195    CNode {
196        kind: node.kind().to_owned(),
197        named: node.is_named(),
198        missing: node.is_missing(),
199        start_byte: node.start_byte(),
200        end_byte: node.end_byte(),
201        start_line: node.start_position().row,
202        start_column: node.start_position().column,
203        end_line: node.end_position().row,
204        end_column: node.end_position().column,
205        field,
206        text,
207        children,
208    }
209}

211/// Collects every syntax node in depth-first order.
212///
213/// # Parameters
214/// - `node`: Current serialized node.
215/// - `path`: Stable path to the node.
216/// - `nodes`: Destination flat node list.
217///
218/// # Returns
219/// Unit after the subtree is flattened.
220fn flatten_node(node: &CNode, path: &str, nodes: &mut Vec<CNodeRecord>) {
221    nodes.push(CNodeRecord {
222        path: path.to_owned(),
223        kind: node.kind.clone(),
224        named: node.named,
225        missing: node.missing,
226        start_byte: node.start_byte,
227        end_byte: node.end_byte,
228        start_line: node.start_line,
229        start_column: node.start_column,
230        end_line: node.end_line,
231        end_column: node.end_column,
232        field: node.field.clone(),
233        text: node.text.clone(),
234    });
235    for (index, child) in node.children.iter().enumerate() {
236        flatten_node(child, &format!("{path}/{index}"), nodes);
237    }
238}

240/// Collects parser error nodes with source coordinates.
241///
242/// # Parameters
243/// - `node`: Current parser node.
244/// - `source`: Original source text.
245/// - `diagnostics`: Destination diagnostic messages.
246///
247/// # Returns
248/// Unit after the subtree has been inspected.
249fn collect_errors(node: Node<'_>, source: &str, diagnostics: &mut Vec<String>) {
250    if node.is_error() || node.is_missing() {
251        let point = node.start_position();
252        let label = if node.is_missing() {
253            "missing syntax"
254        } else {
255            "syntax error"
256        };
257        let excerpt = source
258            .get(node.byte_range())
259            .filter(|text| !text.is_empty())
260            .unwrap_or("unexpected input");
261        diagnostics.push(format!(
262            "{label} at {}:{}: {excerpt}",
263            point.row + 1,
264            point.column + 1
265        ));
266    }
267    let mut cursor = node.walk();
268    for child in node.children(&mut cursor) {
269        collect_errors(child, source, diagnostics);
270    }
271}

277    /// Parses C declarations, comments, and preprocessor directives.
278    #[test]
279    fn parses_c_nodes() {
280        let source =
281            "#include <stdio.h>\n// greeting\nint greet(const char *name) { return 42; }\n";
282        let document = parse(Path::new("greet.c"), source).expect("C parses");
283        assert_eq!(document.adapter, "c");
284        assert!(
285            document
286                .nodes
287                .iter()
288                .any(|node| node.kind == "function_definition")
289        );
290        assert!(
291            document
292                .nodes
293                .iter()
294                .any(|node| node.kind == "preproc_include")
295        );
296        assert!(
297            document
298                .nodes
299                .iter()
300                .any(|node| node.text.as_deref() == Some("// greeting"))
301        );
302    }

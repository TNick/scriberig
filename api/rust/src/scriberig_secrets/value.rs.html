1//! Holds nonserializable zeroizing credential buffers.
2
3use crate::SecretError;
4use secrecy::{ExposeSecret, SecretString};
5
6/// Owns one protected credential without serialization or implicit cloning.
7///
8/// # Public members
9/// - `new`: Validates an owned credential.
10/// - `expose`: Borrows the credential at its use boundary.
11///
12/// # Private members
13/// - `0`: Zeroizing secret string.
14pub struct SecretValue(SecretString);
15
16impl SecretValue {
17    /// Takes ownership of a secret and validates the bearer-token input.
18    ///
19    /// - `value`: Secret string whose allocation will be zeroized on drop.
20    ///
21    /// A validated protected buffer.
22    ///
23    /// Returns `Invalid` for empty, oversized, or multiline input.
24    /// # Returns
25    /// The operation result.
26    /// # Errors
27    /// Returns an error when the operation fails.
28    pub fn new(value: String) -> Result<Self, SecretError> {
29        let protected = SecretString::from(value);
30        let text = protected.expose_secret();
31        if text.is_empty() || text.len() > 65_536 || text.contains(['\r', '\n', '\0']) {
32            return Err(SecretError::Invalid);
33        }
34        Ok(Self(protected))
35    }
36
37    /// Borrows the value explicitly at a storage or transport boundary.
38    ///
39    /// A borrowed secret that must never enter ordinary diagnostic output.
40    #[must_use]
41    /// # Returns
42    /// The operation result.
43    pub fn expose(&self) -> &str {
44        self.0.expose_secret()
45    }
46}
47
48impl std::fmt::Debug for SecretValue {
49    /// Formats a fixed redaction marker.
50    ///
51    /// - `formatter`: Destination formatter.
52    ///
53    /// Formatting success.
54    ///
55    /// Returns formatter failures.
56    /// # Returns
57    /// The operation result.
58    /// # Errors
59    /// Returns an error when the operation fails.
60    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
61        formatter.write_str("SecretValue([REDACTED])")
62    }
63}

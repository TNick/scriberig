1//! Orders metadata and OS-store mutations to fail with authorization disabled.
2
3use super::RegistryGuard;
4use crate::{Binding, SecretError, SecretRef, SecretStore, SecretValue, normalize_endpoint};
5
6impl RegistryGuard {
7    /// Creates or explicitly replaces a managed OS credential.
8    ///
9    /// - `store`: Native or deterministic test store.
10    /// - `name`: Managed alias.
11    /// - `value`: Protected input value.
12    /// - `replace`: Whether an existing item may be replaced.
13    ///
14    /// Unit after the store and metadata agree.
15    ///
16    /// Returns safe conflicts, backend errors, or partial-operation errors.
17    /// # Returns
18    /// The operation result.
19    /// # Errors
20    /// Returns an error when the operation fails.
21    pub fn set(
22        &mut self,
23        store: &dyn SecretStore,
24        name: &str,
25        value: &SecretValue,
26        replace: bool,
27    ) -> Result<(), SecretError> {
28        let reference = SecretRef::Keyring {
29            name: name.to_owned(),
30        };
31        reference.validate()?;
32        let exists = store.exists(name)?;
33        if (exists || self.registry.names.contains(name)) && !replace {
34            return Err(SecretError::Conflict);
35        }
36        let original = self.registry.policy.bindings.clone();
37        self.registry
38            .policy
39            .bindings
40            .retain(|binding| binding.credential != reference);
41        self.save()?;
42        store.set(name, value).map_err(|_| SecretError::Partial)?;
43        self.registry.names.insert(name.to_owned());
44        self.registry.policy.bindings = original;
45        self.save().map_err(|_| SecretError::Partial)
46    }
47
48    /// Removes authorization before deleting a managed credential.
49    ///
50    /// - `store`: Native or deterministic test store.
51    /// - `name`: Managed alias.
52    ///
53    /// Unit after idempotent deletion.
54    ///
55    /// Returns a safe metadata or partial backend failure.
56    /// # Returns
57    /// The operation result.
58    /// # Errors
59    /// Returns an error when the operation fails.
60    pub fn delete(&mut self, store: &dyn SecretStore, name: &str) -> Result<(), SecretError> {
61        let reference = SecretRef::Keyring {
62            name: name.to_owned(),
63        };
64        reference.validate()?;
65        self.registry
66            .policy
67            .bindings
68            .retain(|binding| binding.credential != reference);
69        self.registry.names.remove(name);
70        self.save()?;
71        match store.delete(name) {
72            Ok(()) | Err(SecretError::Missing) => Ok(()),
73            Err(_) => Err(SecretError::Partial),
74        }
75    }
76
77    /// Adds an exact destination/project grant without reading a credential.
78    ///
79    /// - `binding`: Explicitly requested local administrative grant.
80    ///
81    /// Unit after idempotently saving the grant.
82    ///
83    /// Returns safe validation or persistence failures.
84    /// # Returns
85    /// The operation result.
86    /// # Errors
87    /// Returns an error when the operation fails.
88    pub fn bind(&mut self, mut binding: Binding) -> Result<(), SecretError> {
89        binding.credential.validate()?;
90        if let SecretRef::Keyring { name } = &binding.credential {
91            if !self.registry.names.contains(name) {
92                return Err(SecretError::Missing);
93            }
94        }
95        binding.base_url = normalize_endpoint(&binding.base_url)?;
96        if let SecretRef::File { path } = &binding.credential {
97            self.registry.protected_files.insert(path.clone());
98        }
99        binding.project_root = binding
100            .project_root
101            .canonicalize()
102            .map_err(|_| SecretError::Invalid)?;
103        if binding.purpose != "model_provider" || !binding.project_root.is_dir() {
104            return Err(SecretError::Invalid);
105        }
106        if !self.registry.policy.bindings.contains(&binding) {
107            self.registry.policy.bindings.push(binding);
108        }
109        self.save()
110    }
111
112    /// Removes an exact destination/project grant.
113    ///
114    /// - `binding`: Exact binding to remove.
115    ///
116    /// Unit after idempotently saving the removal.
117    ///
118    /// Returns safe normalization or persistence failures.
119    /// # Returns
120    /// The operation result.
121    /// # Errors
122    /// Returns an error when the operation fails.
123    pub fn unbind(&mut self, mut binding: Binding) -> Result<(), SecretError> {
124        binding.base_url = normalize_endpoint(&binding.base_url)?;
125        binding.project_root = binding
126            .project_root
127            .canonicalize()
128            .map_err(|_| SecretError::Invalid)?;
129        self.registry
130            .policy
131            .bindings
132            .retain(|existing| existing != &binding);
133        self.save()
134    }
135}

1//! Defines secret-safe errors without backend diagnostic payloads.
2
3use std::fmt;
4
5/// Classifies failures without retaining secret or backend text.
6#[derive(Clone, Copy, Debug, Eq, PartialEq)]
7pub enum SecretError {
8    /// The referenced secret does not exist.
9    Missing,
10    /// The operating-system store requires unlocking.
11    Locked,
12    /// The configured backend cannot be reached.
13    Unavailable,
14    /// Policy or filesystem permissions deny the operation.
15    Denied,
16    /// This platform does not support the requested operation.
17    Unsupported,
18    /// The reference, value, URL, or registry document is invalid.
19    Invalid,
20    /// An existing item cannot be overwritten without explicit replacement.
21    Conflict,
22    /// A lifecycle operation partially completed; authorization is disabled.
23    Partial,
24}
25
26impl SecretError {
27    /// Returns a stable process exit status for this failure.
28    ///
29    /// A distinct local secret-command exit status.
30    #[must_use]
31    pub const fn exit_code(self) -> i32 {
32        match self {
33            Self::Missing => 20,
34            Self::Locked => 21,
35            Self::Unavailable => 22,
36            Self::Denied => 23,
37            Self::Unsupported => 24,
38            Self::Invalid => 25,
39            Self::Conflict => 26,
40            Self::Partial => 27,
41        }
42    }
43}
44
45impl fmt::Display for SecretError {
46    /// Writes a fixed safe diagnostic.
47    ///
48    /// - `formatter`: Destination formatter.
49    ///
50    /// Formatting success.
51    ///
52    /// Returns formatter failures.
53    /// # Returns
54    /// The operation result.
55    /// # Errors
56    /// Returns an error when the operation fails.
57    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
58        formatter.write_str(match self {
59            Self::Missing => "secret is missing",
60            Self::Locked => "secret store is locked",
61            Self::Unavailable => "secret store is unavailable",
62            Self::Denied => "secret access denied; register the exact destination and project binding",
63            Self::Unsupported => "secret backend is unsupported on this platform",
64            Self::Invalid => "invalid secret input, reference, destination, or policy",
65            Self::Conflict => "secret already exists; use explicit replacement",
66            Self::Partial => "secret operation partially completed; bindings disabled; inspect status before retrying",
67        })
68    }
69}
70
71impl std::error::Error for SecretError {}

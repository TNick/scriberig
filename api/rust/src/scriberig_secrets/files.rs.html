16/// Rejects symbolic links and Windows reparse points in every existing component.
17///
18/// - `path`: Absolute protected path to inspect.
19///
20/// Unit for an existing path without link components.
21///
22/// Returns a safe denial for missing, relative, or redirected paths.
23/// # Returns
24/// The operation result.
25/// # Errors
26/// Returns an error when the operation fails.
27pub fn reject_links(path: &Path) -> Result<(), SecretError> {
28    if !path.is_absolute() {
29        return Err(SecretError::Invalid);
30    }
31    let mut current = std::path::PathBuf::new();
32    for component in path.components() {
33        if matches!(component, std::path::Component::ParentDir) {
34            return Err(SecretError::Denied);
35        }
36        current.push(component);
37        if matches!(component, std::path::Component::Prefix(_)) {
38            continue;
39        }
40        let metadata = std::fs::symlink_metadata(&current).map_err(|_| SecretError::Denied)?;
41        if metadata.is_symlink() {
42            return Err(SecretError::Denied);
43        }
44        #[cfg(windows)]
45        {
46            use std::os::windows::fs::MetadataExt;
47            if metadata.file_attributes() & 0x400 != 0 {
48                return Err(SecretError::Denied);
49            }
50        }
51    }
52    Ok(())
53}

55/// Opens a regular protected file and validates its actual handle permissions.
56///
57/// - `path`: Absolute existing file path.
58///
59/// A verified open file handle.
60///
61/// Returns a safe missing, permission, or path validation failure.
62/// # Returns
63/// The operation result.
64/// # Errors
65/// Returns an error when the operation fails.
66pub fn open_private(path: &Path) -> Result<File, SecretError> {
67    match std::fs::symlink_metadata(path) {
68        Err(error) if error.kind() == std::io::ErrorKind::NotFound => {
69            return Err(SecretError::Missing);
70        }
71        Err(_) => return Err(SecretError::Denied),
72        Ok(_) => {}
73    }
74    reject_links(path)?;
75    let mut options = OpenOptions::new();
76    options.read(true);
77    platform::configure(&mut options);
78    let file = options.open(path).map_err(|_| SecretError::Denied)?;
79    if !file.metadata().map_err(|_| SecretError::Denied)?.is_file() {
80        return Err(SecretError::Denied);
81    }
82    #[cfg(windows)]
83    {
84        use std::os::windows::fs::MetadataExt;
85        if file
86            .metadata()
87            .map_err(|_| SecretError::Denied)?
88            .file_attributes()
89            & 0x400
90            != 0
91        {
92            return Err(SecretError::Denied);
93        }
94    }
95    platform::validate(&file)?;
96    Ok(file)
97}

99/// Reads a bounded mounted credential into a zeroizing value.
100///
101/// - `path`: Absolute protected file path.
102///
103/// A protected UTF-8 credential with one terminal line ending removed.
104///
105/// Returns a safe failure for unsafe files, invalid text, or invalid values.
106/// # Returns
107/// The operation result.
108/// # Errors
109/// Returns an error when the operation fails.
110pub fn read_secret_file(path: &Path) -> Result<SecretValue, SecretError> {
111    let file = open_private(path)?;
112    read_secret(file)
113}

115/// Reads secret input without permitting an unbounded allocation.
116///
117/// - `reader`: Stdin or a protected file reader.
118///
119/// A protected credential after removing one terminal line ending.
120///
121/// Returns `Invalid` for oversized, empty, multiline, or invalid UTF-8 input.
122/// # Returns
123/// The operation result.
124/// # Errors
125/// Returns an error when the operation fails.
126pub fn read_secret(reader: impl Read) -> Result<SecretValue, SecretError> {
127    let mut bytes = Zeroizing::new(Vec::new());
128    reader
129        .take(65_539)
130        .read_to_end(&mut bytes)
131        .map_err(|_| SecretError::Invalid)?;
132    let length = bytes.len();
133    if bytes.ends_with(b"\r\n") {
134        bytes.truncate(length - 2);
135    } else if bytes.ends_with(b"\n") {
136        bytes.truncate(length - 1);
137    }
138    let text = std::str::from_utf8(&bytes).map_err(|_| SecretError::Invalid)?;
139    SecretValue::new(text.to_owned())
140}

142/// Creates and protects an application-owned registry directory.
143///
144/// - `path`: Absolute registry directory path.
145///
146/// Unit once the directory has restrictive permissions.
147///
148/// Returns a safe permission or filesystem failure.
149/// # Returns
150/// The operation result.
151/// # Errors
152/// Returns an error when the operation fails.
153pub fn create_private_dir(path: &Path) -> Result<(), SecretError> {
154    if !path.exists() {
155        let parent = path.parent().ok_or(SecretError::Invalid)?;
156        std::fs::create_dir_all(parent).map_err(|_| SecretError::Denied)?;
157        reject_links(parent)?;
158        std::fs::create_dir(path).map_err(|_| SecretError::Denied)?;
159    }
160    reject_links(path)?;
161    platform::protect_directory(path)
162}

164/// Protects a newly created registry file before writing its contents.
165///
166/// - `file`: Application-created file handle.
167///
168/// Unit after restrictive permissions are installed and checked.

172/// The operation result.

175pub fn protect_new_file(file: &mut File) -> Result<(), SecretError> {

177    platform::validate(file)

180/// Creates a new protected file without overwriting an existing path.
181///
182/// - `path`: Absolute new file path inside a protected directory.
183///
184/// A readable and writable private file handle.
185///
186/// Returns a safe failure on collisions, unsafe parents, or permission errors.
187/// # Returns
188/// The operation result.
189/// # Errors
190/// Returns an error when the operation fails.
191pub fn create_private_file(path: &Path) -> Result<File, SecretError> {
192    reject_links(path.parent().ok_or(SecretError::Invalid)?)?;
193    let mut options = OpenOptions::new();
194    options.read(true).write(true).create_new(true);
195    platform::configure_creation(&mut options);
196    let mut file = options.open(path).map_err(|_| SecretError::Denied)?;
197    protect_new_file(&mut file)?;
198    Ok(file)
199}

201/// Opens an administrator-owned policy that ordinary users cannot modify.
202///
203/// - `path`: Absolute fixed system policy path.
204///
205/// A regular no-follow file with validated administrator write authority.
206///
207/// Returns `Denied` for unsafe owners, links, or permission grants.
208/// # Returns
209/// The operation result.
210/// # Errors
211/// Returns an error when the operation fails.
212pub fn open_system_policy(path: &Path) -> Result<File, SecretError> {
213    reject_links(path)?;
214    let mut options = OpenOptions::new();
215    options.read(true);
216    platform::configure(&mut options);
217    let file = options.open(path).map_err(|_| SecretError::Denied)?;
218    if !file.metadata().map_err(|_| SecretError::Denied)?.is_file() {
219        return Err(SecretError::Denied);
220    }
221    platform::validate_system(&file)?;
222    Ok(file)

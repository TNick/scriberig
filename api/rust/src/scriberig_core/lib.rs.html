27#[cfg(test)]
28mod tests {
29    use std::collections::BTreeMap;

31    use super::{
32        EntityId, EntityRevisionRef, KnowledgeClass, KnowledgeValue, Provenance, ProvenanceSource,

36    /// Checks nested values preserve list order and deterministic records.
37    #[test]
38    fn knowledge_values_preserve_structure() {
39        let mut record = BTreeMap::new();
40        record.insert("name".to_owned(), KnowledgeValue::String("item".to_owned()));
41        let value = KnowledgeValue::List(vec![
42            KnowledgeValue::UnsignedInteger(1),
43            KnowledgeValue::Record(record.clone()),
44        ]);
45        assert_eq!(
46            value,
47            KnowledgeValue::List(vec![
48                KnowledgeValue::UnsignedInteger(1),
49                KnowledgeValue::Record(record),
50            ])
51        );
52    }
53
54    /// Checks canonical and derived classes remain explicit vocabulary.
55    #[test]
56    fn knowledge_classes_are_distinct() {
57        assert_ne!(KnowledgeClass::Canonical, KnowledgeClass::Derived);
58    }
59
60    /// Checks provenance supports empty, multiple, and custom source inputs.
61    #[test]
62    fn provenance_preserves_exact_inputs_and_custom_sources() {
63        let empty = Provenance {
64            source: ProvenanceSource::User,
65            producer: None,
66            producer_version: None,
67            inputs: Vec::new(),
68            profile: None,
69            note: None,
70        };
71        assert_eq!(empty.inputs.len(), 0);
72        let populated = Provenance {
73            source: ProvenanceSource::Custom("review-bot".to_owned()),
74            producer: Some("bot-v1".to_owned()),
75            producer_version: None,
76            inputs: vec![
77                EntityRevisionRef {
78                    entity_id: EntityId::new([1; 16]),
79                    revision_id: RevisionId::new([2; 32]),
80                },
81                EntityRevisionRef {
82                    entity_id: EntityId::new([3; 16]),
83                    revision_id: RevisionId::new([4; 32]),
84                },
85            ],
86            profile: Some("review".to_owned()),
87            note: Some("derived from exact revisions".to_owned()),
88        };
89        assert_eq!(populated.inputs.len(), 2);
90        assert!(matches!(populated.source, ProvenanceSource::Custom(_)));
91    }
92
93    /// Checks records sort keys independently from insertion order.
94    #[test]
95    fn records_have_deterministic_key_order() {
96        let mut record = BTreeMap::new();
97        record.insert("zeta".to_owned(), KnowledgeValue::Null);
98        record.insert("alpha".to_owned(), KnowledgeValue::Bool(true));
99        let keys: Vec<_> = record.keys().cloned().collect();
100        assert_eq!(keys, vec!["alpha".to_owned(), "zeta".to_owned()]);
101    }
102
103    /// Checks semantic references stay in provenance instead of property values.
104    #[test]
105    fn semantic_references_use_typed_provenance_inputs() {
106        let reference = EntityRevisionRef {
107            entity_id: EntityId::new([5; 16]),
108            revision_id: RevisionId::new([6; 32]),
109        };
110        let property = KnowledgeValue::Record(BTreeMap::from([(
111            "label".to_owned(),
112            KnowledgeValue::String("related".to_owned()),
113        )]));
114        let provenance = Provenance {
115            source: ProvenanceSource::System,
116            producer: None,
117            producer_version: None,
118            inputs: vec![reference],
119            profile: None,
120            note: None,
121        };
122        assert_eq!(provenance.inputs, vec![reference]);
123        assert!(matches!(property, KnowledgeValue::Record(_)));
124    }

3// Import the core identifier used to version canonical bytes.
4use std::fmt::{self, Display, Formatter};

6/// Selects the canonical encoding rules used for a value.
7///
8/// # Public members
9/// - `new`: Constructs a version from its stable integer value.
10/// - `value`: Returns the encoded version number.
11///
12/// # Private members
13/// - `value`: Stores the version number.
14#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
15pub struct EncodingVersion {
16    value: u16,
17}
18
19// Provide version construction and inspection.
20impl EncodingVersion {
21    /// The initial canonical encoding version.
22    pub const V1: Self = Self { value: 1 };
23
24    /// Constructs an encoding version from its stable integer value.
25    ///
26    /// # Parameters
27    /// - `value`: Version number.
28    ///
29    /// # Returns
30    /// The version wrapper.
31    #[must_use]
32    pub const fn new(value: u16) -> Self {
33        Self { value }
34    }
35
36    /// Returns the stable integer value of this version.
37    ///
38    /// # Parameters
39    /// - `self`: Version to inspect.
40    ///
41    /// # Returns
42    /// The version number.
43    #[must_use]
44    pub const fn value(self) -> u16 {
45        self.value
46    }
47}
48
49// Format encoding versions without locale dependence.
50impl Display for EncodingVersion {
51    /// Formats the version as its decimal number.
52    ///
53    /// # Parameters
54    /// - `self`: Version to format.
55    /// - `formatter`: Destination formatter.
56    ///
57    /// # Returns
58    /// A formatting result.
59    ///
60    /// # Errors
61    /// Returns a formatting error if writing to the formatter fails.
62    fn fmt(&self, formatter: &mut Formatter<'_>) -> fmt::Result {
63        write!(formatter, "{}", self.value)
64    }
65}

67/// Appends values to a deterministic, length-delimited byte sequence.
68///
69/// # Public members
70/// - `new`: Creates an empty writer.
71/// - `with_capacity`: Creates a writer with reserved space.
72/// - `write_tag`: Writes a fixed domain tag.
73/// - `write_u16`: Writes an unsigned integer in big-endian order.
74/// - `write_u64`: Writes an unsigned integer in big-endian order.
75/// - `write_bytes`: Writes a length-prefixed byte string.
76/// - `write_str`: Writes a length-prefixed UTF-8 string.
77/// - `write_bool`: Writes a single canonical boolean byte.
78/// - `write_option`: Writes an option with explicit discriminant.
79/// - `write_sequence_len`: Writes a sequence element count.
80/// - `as_bytes`: Borrows the encoded bytes.
81/// - `into_bytes`: Consumes the writer and returns encoded bytes.
82///
83/// # Private members
84/// - `bytes`: Accumulates the canonical output.
85#[derive(Clone, Debug, Default, Eq, PartialEq)]
86pub struct CanonicalWriter {
87    bytes: Vec<u8>,
88}
89
90// Implement canonical primitive writing with explicit byte order.
91impl CanonicalWriter {
92    /// Creates an empty canonical writer.
93    ///
94    /// # Returns
95    /// An empty writer.
96    #[must_use]
97    pub const fn new() -> Self {
98        Self { bytes: Vec::new() }
99    }
100
101    /// Creates a writer with reserved output capacity.
102    ///
103    /// # Parameters
104    /// - `capacity`: Initial byte capacity.
105    ///
106    /// # Returns
107    /// An empty writer with reserved capacity.
108    #[must_use]
109    pub fn with_capacity(capacity: usize) -> Self {
110        Self {
111            bytes: Vec::with_capacity(capacity),
112        }
113    }
114
115    /// Writes a caller-defined fixed domain tag.
116    ///
117    /// # Parameters
118    /// - `tag`: Stable bytes identifying the value domain.
119    pub fn write_tag(&mut self, tag: &[u8]) {
120        self.bytes.extend_from_slice(tag);
121    }
122
123    /// Writes a 16-bit unsigned integer in big-endian order.
124    ///
125    /// # Parameters
126    /// - `value`: Integer to encode.
127    pub fn write_u16(&mut self, value: u16) {
128        self.bytes.extend_from_slice(&value.to_be_bytes());
129    }
130
131    /// Writes an 8-bit unsigned integer.
132    ///
133    /// # Parameters
134    /// - `value`: Integer to encode.
135    pub fn write_u8(&mut self, value: u8) {
136        self.bytes.push(value);
137    }
138
139    /// Writes a 32-bit unsigned integer in big-endian order.
140    ///
141    /// # Parameters
142    /// - `value`: Integer to encode.
143    pub fn write_u32(&mut self, value: u32) {
144        self.bytes.extend_from_slice(&value.to_be_bytes());
145    }
146
147    /// Writes a 64-bit unsigned integer in big-endian order.
148    ///
149    /// # Parameters
150    /// - `value`: Integer to encode.
151    pub fn write_u64(&mut self, value: u64) {
152        self.bytes.extend_from_slice(&value.to_be_bytes());
153    }
154
155    /// Writes a byte string preceded by its 64-bit length.
156    ///
157    /// # Parameters
158    /// - `value`: Bytes to encode.
159    pub fn write_bytes(&mut self, value: &[u8]) {
160        self.write_u64(value.len() as u64);
161        self.bytes.extend_from_slice(value);
162    }
163
164    /// Writes a UTF-8 string preceded by its byte length.
165    ///
166    /// # Parameters
167    /// - `value`: String to encode.
168    pub fn write_str(&mut self, value: &str) {
169        self.write_bytes(value.as_bytes());
170    }
171
172    /// Writes a boolean using zero for false and one for true.
173    ///
174    /// # Parameters
175    /// - `value`: Boolean to encode.
176    pub fn write_bool(&mut self, value: bool) {
177        self.bytes.push(u8::from(value));
178    }
179
180    /// Writes an optional value's presence discriminant.
181    ///
182    /// Call `encode` only for `Some`; the caller encodes its contents with this
183    /// writer after this method returns.
184    ///
185    /// # Parameters
186    /// - `is_some`: Whether the optional value is present.
187    pub fn write_option(&mut self, is_some: bool) {
188        self.bytes.push(u8::from(is_some));
189    }
190
191    /// Writes an ordered sequence's element count.
192    ///
193    /// # Parameters
194    /// - `length`: Number of sequence elements.
195    pub fn write_sequence_len(&mut self, length: usize) {
196        self.write_u64(length as u64);
197    }
198
199    /// Borrows the bytes written so far.
200    ///
201    /// # Returns
202    /// The current canonical byte sequence.
203    #[must_use]
204    pub fn as_bytes(&self) -> &[u8] {
205        &self.bytes
206    }
207
208    /// Consumes the writer and returns the encoded bytes.
209    ///
210    /// # Returns
211    /// The canonical byte sequence.
212    #[must_use]
213    pub fn into_bytes(self) -> Vec<u8> {
214        self.bytes
215    }
216}

218/// Encodes a value using explicit Scriberig canonical encoding rules.
219pub trait CanonicalEncode {
220    /// Appends this value's canonical representation to a writer.
221    ///
222    /// # Parameters
223    /// - `self`: Value to encode.
224    /// - `writer`: Output writer.
225    fn encode_canonical(&self, writer: &mut CanonicalWriter);
226}

228/// Hashes a complete revision envelope into an immutable revision identity.
229///
230/// The domain tag, encoding version, object type, and payload all participate
231/// in the digest so changes to any identity input produce a different value.
232///
233/// # Parameters
234/// - `version`: Canonical encoding version.
235/// - `object_type`: Semantic object type.
236/// - `payload`: Already canonicalized payload bytes.
237///
238/// # Returns
239/// The BLAKE3-derived revision identifier.
240#[must_use]
241pub fn revision_id(
242    version: EncodingVersion,
243    object_type: &crate::ObjectType,
244    payload: &[u8],
245) -> crate::RevisionId {
246    let mut writer = CanonicalWriter::new();
247    writer.write_tag(b"scriberig.revision.v1");
248    writer.write_u16(version.value());
249    writer.write_str(object_type.as_str());
250    writer.write_bytes(payload);
251    crate::RevisionId::from_bytes(*blake3::hash(writer.as_bytes()).as_bytes())
252}

254// Confirm primitive encodings and versions remain deterministic.
255#[cfg(test)]
256mod tests {
257    use super::{CanonicalEncode, CanonicalWriter, EncodingVersion, revision_id};
258    use crate::ObjectType;
259
260    /// Checks canonical primitive output is repeatable.
261    #[test]
262
263    fn canonical_writer_encodes_primitives_deterministically() {
264        let mut first = CanonicalWriter::new();
265        first.write_tag(b"test");
266        first.write_u16(1);
267        first.write_str("abc");
268        first.write_bool(true);
269
270        let mut second = CanonicalWriter::new();
271        second.write_tag(b"test");
272        second.write_u16(1);
273        second.write_str("abc");
274        second.write_bool(true);
275
276        assert_eq!(first.as_bytes(), second.as_bytes());
277        assert_eq!(
278            first.as_bytes(),
279            &[
280                116, 101, 115, 116, 0, 1, 0, 0, 0, 0, 0, 0, 0, 3, 97, 98, 99, 1
281            ]
282        );
283    }
284
285    /// Verifies revision identity includes every envelope component.
286    #[test]
287    fn revision_hash_is_deterministic_and_domain_separated() {
288        let object_type = ObjectType::new("markdown.paragraph").expect("valid type");
289        let first = revision_id(EncodingVersion::V1, &object_type, b"payload");
290        let second = revision_id(EncodingVersion::V1, &object_type, b"payload");
291        assert_eq!(first, second);
292        assert_ne!(
293            first,
294            revision_id(EncodingVersion::new(2), &object_type, b"payload")
295        );
296        let other_type = ObjectType::new("rust.function").expect("valid type");
297        assert_ne!(
298            first,
299            revision_id(EncodingVersion::V1, &other_type, b"payload")
300        );
301        assert_ne!(
302            first,
303            revision_id(EncodingVersion::V1, &object_type, b"other")
304        );
305    }

307    /// Checks independently constructed equivalent values encode identically.
308    #[test]
309
310    fn independent_values_encode_to_the_same_bytes() {
311        /// Holds the scalar used by this canonical encoding test.
312        #[derive(Clone, Copy)]
313        struct Sample(u64);
314
315        impl CanonicalEncode for Sample {
316            /// Encodes the sample's scalar value.
317            ///
318            /// # Parameters
319            /// - `self`: Sample to encode.
320            /// - `writer`: Canonical byte output.
321            fn encode_canonical(&self, writer: &mut CanonicalWriter) {
322                writer.write_u64(self.0);
323            }
324        }
325
326        let mut first = CanonicalWriter::new();
327        Sample(42).encode_canonical(&mut first);
328        let mut second = CanonicalWriter::new();
329        Sample(42).encode_canonical(&mut second);
330        assert_eq!(first.as_bytes(), second.as_bytes());
331    }

1//! Provides shared identifier formatting, parsing, and errors.
2
3// Import formatter primitives used by identifier helpers.
4use std::fmt::{self, Formatter};
5
6/// Reports malformed or invalid textual identifier values.
7///
8/// # Public members
9/// - `InvalidFormat`: Input text has the wrong prefix, length, or alphabet.
10/// - `EmptyName`: A root or object type name is empty or whitespace-only.
11///
12/// # Private members
13/// None.
14#[derive(Clone, Copy, Debug, Eq, PartialEq)]
15pub enum IdParseError {
16    /// The text does not match the identifier's required representation.
17    InvalidFormat,
18    /// The name must contain at least one non-whitespace character.
19    EmptyName,
20}
21
22// Format parse errors without depending on a backend or locale.
23impl fmt::Display for IdParseError {
24    /// Formats the parse error for diagnostics.
25    ///
26    /// - `self`: Error to format.
27    /// - `formatter`: Destination formatter.
28    ///
29    /// A formatting result.
30    ///
31    /// Returns a formatting error if writing to the formatter fails.
32    /// # Returns
33    /// The operation result.
34    /// # Errors
35    /// Returns an error when the operation fails.
36    fn fmt(&self, formatter: &mut Formatter<'_>) -> fmt::Result {
37        match self {
38            Self::InvalidFormat => formatter.write_str("invalid identifier format"),
39            Self::EmptyName => formatter.write_str("name must not be empty"),
40        }
41    }
42}
43
44// Make parse failures usable with standard error handling.
45impl std::error::Error for IdParseError {}
46
47/// Writes a type prefix and fixed-width lowercase hexadecimal bytes.
48///
49/// - `prefix`: Type-specific textual prefix.
50/// - `bytes`: Bytes to format.
51/// - `formatter`: Destination formatter.
52///
53/// A formatting result.
54///
55/// Returns a formatting error if writing to the formatter fails.
56/// # Returns
57/// The operation result.
58/// # Errors
59/// Returns an error when the operation fails.
60pub fn write_hex(prefix: &str, bytes: &[u8], formatter: &mut Formatter<'_>) -> fmt::Result {
61    formatter.write_str(prefix)?;
62    for byte in bytes {
63        write!(formatter, "{byte:02x}")?;
64    }
65    Ok(())
66}

68/// Parses exactly the expected number of prefixed hexadecimal bytes.
69///
70/// - `prefix`: Required type-specific prefix.
71/// - `text`: Text to parse.
72///
73/// A fixed-width byte array.
74///
75/// Returns `InvalidFormat` when the prefix, digit count, or alphabet is invalid.
76// Parse prefixed hexadecimal bytes without panicking.
77pub(super) fn parse_hex<const N: usize>(prefix: &str, text: &str) -> Result<[u8; N], IdParseError> {
78    let digits = text
79        .strip_prefix(prefix)
80        .ok_or(IdParseError::InvalidFormat)?;
81    if digits.len() != N * 2 || !digits.is_ascii() {
82        return Err(IdParseError::InvalidFormat);
83    }
84    let mut bytes = [0; N];
85    for (index, byte) in bytes.iter_mut().enumerate() {
86        let start = index * 2;
87        *byte = u8::from_str_radix(&digits[start..start + 2], 16)
88            .map_err(|_| IdParseError::InvalidFormat)?;
89    }
90    Ok(bytes)
91}

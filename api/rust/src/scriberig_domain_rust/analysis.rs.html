51/// Immutable derived Rust analysis output.
52#[derive(Clone, Debug, Eq, PartialEq)]
53pub struct RustAnalysis {

55    pub key: DerivationKey,

57    pub handle: DerivedResultHandle,

67    pub types: Vec<TypeFact>,

69    pub diagnostics: Vec<RustDiagnostic>,
70    /// Exact producer and input provenance.

74impl RustAnalysis {
75    /// Returns references targeting one canonical symbol.
76    ///
77    /// # Parameters
78    /// - `target`: Canonical symbol entity.
79    ///
80    /// # Returns
81    /// Matching references in deterministic source order.
82    #[must_use]
83    pub fn references_to(&self, target: EntityId) -> Vec<&SymbolReference> {
84        self.references
85            .iter()
86            .filter(|reference| reference.to == target)
87            .collect()
88    }
89
90    /// Returns canonical callers of one symbol.
91    ///
92    /// # Parameters
93    /// - `callee`: Canonical callee entity.
94    ///
95    /// # Returns
96    /// Caller entities in analysis order.
97    #[must_use]
98    pub fn callers_of(&self, callee: EntityId) -> Vec<EntityId> {
99        self.calls
100            .iter()
101            .filter(|call| call.callee == callee)
102            .map(|call| call.caller)
103            .collect()
104    }
105
106    /// Returns diagnostics affecting one canonical entity.
107    ///
108    /// # Parameters
109    /// - `entity`: Entity whose diagnostics are requested.
110    ///
111    /// # Returns
112    /// Matching derived diagnostics.
113    #[must_use]
114    pub fn diagnostics_for(&self, entity: EntityId) -> Vec<&RustDiagnostic> {
115        self.diagnostics
116            .iter()
117            .filter(|diagnostic| diagnostic.entity == Some(entity))
118            .collect()
119    }
120}

122/// Runs deterministic static analysis over canonical Rust source.
123///
124/// # Parameters
125/// - `source`: Canonical Rust source text.
126/// - `source_entity`: Canonical source entity identity.
127/// - `source_revision`: Exact canonical syntax revision.
128/// - `profile`: Compiler/profile inputs affecting semantics.
129///
130/// # Returns
131/// Immutable derived facts with exact dependency and provenance metadata.
132///
133/// # Errors
134/// Returns a structured parse failure. Canonical source is never changed.
135pub fn analyze(
136    source: &str,
137    source_entity: EntityId,
138    source_revision: RevisionId,
139    profile: &AnalysisProfile,
140) -> Result<RustAnalysis, RustAnalysisError> {
141    let file = syn::parse_file(source).map_err(|error| RustAnalysisError {
142        code: "parse".to_owned(),
143        message: error.to_string(),
144    })?;
145    let names: Vec<String> = file.items.iter().filter_map(function_name).collect();
146    let mapping = AnalysisMapping::for_functions(source_entity, names);
147    let mut references = Vec::new();
148    let mut calls = Vec::new();
149    let mut types = Vec::new();
150    for item in &file.items {
151        let Item::Fn(function) = item else { continue };
152        let Some(&caller) = mapping.symbols.get(function.sig.ident.to_string().as_str()) else {
153            continue;
154        };
155        if let ReturnType::Type(_, ty) = &function.sig.output {
156            types.push(TypeFact {
157                entity: caller,
158                type_text: type_text(ty),
159            });
160        }
161        let mut visitor = CallVisitor {
162            mapping: &mapping.symbols,
163            caller,
164            references: &mut references,
165            calls: &mut calls,
166        };
167        visitor.visit_block(&function.block);
168    }
169    let key = DerivationKey::new(
170        profile.derivation_profile(),
171        profile.dependencies(source_entity, source_revision),
172    );
173    let handle = DerivedResultHandle {
174        artifact_id: analysis_id(source_entity, source_revision, profile),
175    };
176    Ok(RustAnalysis {
177        key,
178        handle,
179        status: DerivedResultStatus::Succeeded,
180        mapping,
181        references,
182        calls,
183        types,
184        diagnostics: Vec::new(),
185        provenance: AnalysisProvenance {
186            producer: format!("rustc:{}", profile.toolchain),
187            profile: profile.clone(),
188            source_revision,
189        },
190    })
191}

193/// Extracts a function name from a top-level item.
194///
195/// # Parameters
196/// - `item`: Syntax item to inspect.
197///
198/// # Returns
199/// Function name when the item is a function.
200fn function_name(item: &Item) -> Option<String> {
201    match item {
202        Item::Fn(function) => Some(function.sig.ident.to_string()),
203        _ => None,
204    }
205}
206/// Renders a syntax type into a deterministic display representation.
207///
208/// # Parameters
209/// - `ty`: Syntax type to render.
210///
211/// # Returns
212/// Canonical token text for the type.
213fn type_text(ty: &Type) -> String {
214    ty.to_token_stream().to_string()
215}
216/// Derives an immutable result identity from all relevant analysis inputs.
217///
218/// # Parameters
219/// - `source`: Canonical source entity.
220/// - `revision`: Exact source revision.
221/// - `profile`: Analysis environment.
222///
223/// # Returns
224/// Content-derived result revision identity.
225fn analysis_id(source: EntityId, revision: RevisionId, profile: &AnalysisProfile) -> RevisionId {
226    let mut hasher = blake3::Hasher::new();
227    hasher.update(b"scriberig.rust.analysis.v1");
228    hasher.update(source.as_bytes());
229    hasher.update(revision.as_bytes());
230    hasher.update(profile.toolchain.as_bytes());
231    hasher.update(profile.configuration_id().as_bytes());
232    RevisionId::from_bytes(*hasher.finalize().as_bytes())
233}

235/// Visits one function body and resolves calls to known canonical functions.
236///
237/// # Private members
238/// - `mapping`: Source names to canonical IDs.
239/// - `caller`: Current canonical caller.
240/// - `references`: Collected symbol references.
241/// - `calls`: Collected caller/callee relations.
242struct CallVisitor<'a> {
243    mapping: &'a BTreeMap<String, EntityId>,
244    caller: EntityId,
245    references: &'a mut Vec<SymbolReference>,
246    calls: &'a mut Vec<CallRelation>,
247}
248impl<'ast> Visit<'ast> for CallVisitor<'_> {
249    /// Resolves a call expression when its path names a known function.
250    ///
251    /// # Parameters
252    /// - `call`: Call expression being visited.
253    fn visit_expr_call(&mut self, call: &'ast ExprCall) {
254        if let Expr::Path(path) = &*call.func {
255            if let Some(name) = path.path.get_ident().map(ToString::to_string) {
256                if let Some(&to) = self.mapping.get(&name) {
257                    self.references.push(SymbolReference {
258                        from: self.caller,
259                        to,
260                        name,
261                    });
262                    self.calls.push(CallRelation {
263                        caller: self.caller,
264                        callee: to,
265                    });
266                }
267            }
268        }
269        syn::visit::visit_expr_call(self, call);
270    }
271}

273/// Builds a failed derived result with an explicit diagnostic.
274///
275/// # Parameters
276/// - `error`: Structured analysis failure.
277/// - `source_entity`: Canonical source entity.
278/// - `source_revision`: Exact source revision.
279/// - `profile`: Analysis environment.
280///
281/// # Returns
282/// Failed analysis facts that preserve canonical identity and provenance.
283#[must_use]
284pub fn failed_analysis(
285    error: RustAnalysisError,
286    source_entity: EntityId,
287    source_revision: RevisionId,
288    profile: &AnalysisProfile,
289) -> RustAnalysis {
290    let key = DerivationKey::new(
291        profile.derivation_profile(),
292        profile.dependencies(source_entity, source_revision),
293    );
294    RustAnalysis {
295        key,
296        handle: DerivedResultHandle {
297            artifact_id: analysis_id(source_entity, source_revision, profile),
298        },
299        status: DerivedResultStatus::FailedDeterministic,
300        mapping: AnalysisMapping::for_functions(source_entity, Vec::<String>::new()),
301        references: Vec::new(),
302        calls: Vec::new(),
303        types: Vec::new(),
304        diagnostics: vec![RustDiagnostic {
305            severity: DiagnosticSeverity::Error,
306            code: Some(error.code),
307            message: error.message,
308            entity: Some(source_entity),
309        }],
310        provenance: AnalysisProvenance {
311            producer: format!("rustc:{}", profile.toolchain),
312            profile: profile.clone(),
313            source_revision,
314        },
315    }

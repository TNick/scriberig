1//! Extracts stable, human-readable Rust symbols for semantic navigation.
2
3use quote::ToTokens;
4use syn::Item;
5
6/// A named Rust symbol with a stable semantic key and source range.
7#[derive(Clone, Debug, Eq, PartialEq)]
8pub struct NamedRustSymbol {
9    /// Canonical semantic kind.
10    pub kind: String,
11    /// Human-readable declaration name.
12    pub name: String,
13    /// Hierarchical identity key within one source file.
14    pub key: String,
15    /// Approximate byte range covering the declaration name.
16    pub range: (usize, usize),
17}
18
19/// Extracts named modules, types, functions, and methods from Rust source.
20///
21/// # Parameters
22/// - `source`: Rust source text to parse.
23///
24/// # Returns
25/// Named symbols in source order.
26///
27/// # Errors
28/// Returns the parser diagnostic when the source is not valid Rust.
29pub fn named_symbols(source: &str) -> Result<Vec<NamedRustSymbol>, String> {
30    let file = syn::parse_file(source).map_err(|error| error.to_string())?;
31    let mut symbols = Vec::new();
32    collect_items(source, &file.items, "", &mut symbols);
33    Ok(symbols)
34}
35
36/// Walks a list of Rust items and records named declarations.
37///
38/// # Parameters
39/// - `source`: Source text used for ranges.
40/// - `items`: Rust items to walk.
41/// - `prefix`: Parent semantic key.
42/// - `symbols`: Output symbol collection.
43fn collect_items(source: &str, items: &[Item], prefix: &str, symbols: &mut Vec<NamedRustSymbol>) {
44    for item in items {
45        match item {
46            Item::Mod(module) => {
47                let name = module.ident.to_string();
48                let key = join_key(prefix, &name);
49                push_symbol(source, "rust.module", &name, &key, symbols);
50                if let Some((_, children)) = &module.content {
51                    collect_items(source, children, &key, symbols);
52                }
53            }
54            Item::Struct(value) => {
55                let name = value.ident.to_string();
56                let key = join_key(prefix, &name);
57                push_symbol(source, "rust.struct", &name, &key, symbols);
58            }
59            Item::Enum(value) => {
60                let name = value.ident.to_string();
61                let key = join_key(prefix, &name);
62                push_symbol(source, "rust.enum", &name, &key, symbols);
63            }
64            Item::Fn(value) => {
65                let name = value.sig.ident.to_string();
66                let key = join_key(prefix, &name);
67                push_symbol(source, "rust.function", &name, &key, symbols);
68            }
69            Item::Impl(value) => {
70                let owner = value.self_ty.to_token_stream().to_string();
71                for item in &value.items {
72                    if let syn::ImplItem::Fn(function) = item {
73                        let name = function.sig.ident.to_string();
74                        let key = format!("{}::{}", join_key(prefix, &owner), name);
75                        push_symbol(source, "rust.method", &name, &key, symbols);
76                    }
77                }
78            }
79            _ => {}
80        }
81    }
82}

84/// Appends a symbol with a deterministic name range.
85///
86/// # Parameters
87/// - `source`: Source text used for ranges.
88/// - `kind`: Semantic kind label.
89/// - `name`: Human-readable declaration name.
90/// - `key`: Hierarchical semantic key.
91/// - `symbols`: Output symbol collection.
92fn push_symbol(
93    source: &str,
94    kind: &str,
95    name: &str,
96    key: &str,
97    symbols: &mut Vec<NamedRustSymbol>,
98) {
99    let start = source.find(name).unwrap_or(0);
100    let key = unique_key(key, symbols);
101    symbols.push(NamedRustSymbol {
102        kind: kind.to_owned(),
103        name: name.to_owned(),
104        key,
105        range: (start, start.saturating_add(name.len())),
106    });
107}

109/// Makes a repeated semantic key unique within one source file.
110///
111/// # Parameters
112/// - `key`: Preferred semantic key.
113/// - `symbols`: Symbols already extracted from the source file.
114///
115/// # Returns
116/// The preferred key or a deterministic occurrence-suffixed variant.
117fn unique_key(key: &str, symbols: &[NamedRustSymbol]) -> String {
118    if !symbols.iter().any(|symbol| symbol.key == key) {
119        return key.to_owned();
120    }
121    let mut occurrence = 2;
122    loop {
123        let candidate = format!("{key}#{occurrence}");
124        if !symbols.iter().any(|symbol| symbol.key == candidate) {
125            return candidate;
126        }
127        occurrence += 1;
128    }
129}

131/// Joins one semantic path component.
132///
133/// # Parameters
134/// - `prefix`: Existing semantic path.
135/// - `name`: New path component.
136///
137/// # Returns
138/// The joined semantic path.
139fn join_key(prefix: &str, name: &str) -> String {
140    if prefix.is_empty() {
141        name.to_owned()
142    } else {
143        format!("{prefix}::{name}")
144    }
145}

147#[cfg(test)]
148mod tests {
149    use super::named_symbols;
150
151    /// Extracts the stable names used by the golden Rust fixture.
152    #[test]
153    fn extracts_nested_named_symbols() {
154        let source = "pub mod math { pub struct Calculator; impl Calculator { pub fn add(&self, left: i32, right: i32) -> i32 { left + right } } }";
155        let symbols = named_symbols(source).expect("valid Rust");
156        assert!(symbols.iter().any(|value| value.name == "math"));
157        assert!(symbols.iter().any(|value| value.name == "Calculator"));
158        assert!(symbols.iter().any(|value| value.name == "add"));
159    }
160
161    /// Disambiguates repeated semantic declarations without dropping either one.
162    #[test]
163    fn disambiguates_repeated_named_symbols() {
164        let source = "struct Calculator; impl Calculator { fn add(&self) {} } impl Calculator { fn add(&self) {} }";
165        let symbols = named_symbols(source).expect("valid Rust");
166        let methods = symbols
167            .iter()
168            .filter(|value| value.name == "add")
169            .collect::<Vec<_>>();
170        assert_eq!(methods.len(), 2);
171        assert_eq!(methods[0].key, "Calculator::add");
172        assert_eq!(methods[1].key, "Calculator::add#2");
173    }
174}
